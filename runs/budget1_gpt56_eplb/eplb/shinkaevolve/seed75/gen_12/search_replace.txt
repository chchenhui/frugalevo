<DESCRIPTION>
Add a bounded local-search refinement after the cardinality-constrained greedy packing pass. The existing LPT-style greedy assignment is fast but can leave a heavy and light pack with complementary items whose exchange significantly reduces the load range. For a small fixed number of iterations, the refinement considers only the current heaviest and lightest packs in every layer, evaluates all slot exchanges in a vectorized tensor operation, and applies the best exchange that transfers positive load without reversing the pack ordering.

This preserves the exact number of experts per pack and all output-map invariants. It improves both node group placement and GPU physical-expert placement because both stages use `balanced_packing`, while avoiding an expensive general combinatorial search.
</DESCRIPTION>

def balanced_packing(weight: torch.Tensor,
                     num_packs: int) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Pack n weighted objects to m packs, such that each bin contains exactly
    n/m objects and the weights of all packs are as balanced as possible.

    A greedy LPT assignment establishes a good initial layout. A few
    vectorized exchanges between the current heaviest and lightest packs then
    remove common greedy-placement errors without changing pack cardinality.

    Parameters:
        weight: [X, n], the weight of each item
        num_packs: number of packs

    Returns:
        pack_index: [X, n], the pack index of each item
        rank_in_pack: [X, n], the rank of the item in the pack
    """
    num_layers, num_groups = weight.shape
    assert num_groups % num_packs == 0
    groups_per_pack = num_groups // num_packs

    if groups_per_pack == 1:
        pack_index = torch.arange(weight.size(-1),
                                  dtype=torch.int64,
                                  device=weight.device).expand(weight.shape)
        rank_in_pack = torch.zeros_like(weight, dtype=torch.int64)
        return pack_index, rank_in_pack

    # Every layer follows the same greedy policy independently. Batch the
    # per-layer decisions to avoid Python loops over layers and tensor scalars.
    indices = weight.float().sort(-1, descending=True).indices
    pack_index = torch.empty_like(weight, dtype=torch.int64)
    rank_in_pack = torch.empty_like(pack_index)
    pack_weights = torch.zeros((num_layers, num_packs),
                               dtype=torch.float32,
                               device=weight.device)
    pack_items = torch.zeros((num_layers, num_packs),
                             dtype=torch.int64,
                             device=weight.device)
    unavailable = torch.finfo(pack_weights.dtype).max

    for position in range(num_groups):
        group = indices[:, position]
        pack = pack_weights.masked_fill(
            pack_items >= groups_per_pack, unavailable).min(dim=-1).indices
        current_rank = pack_items.gather(-1, pack.unsqueeze(-1)).squeeze(-1)

        pack_index.scatter_(-1, group.unsqueeze(-1), pack.unsqueeze(-1))
        rank_in_pack.scatter_(-1, group.unsqueeze(-1),
                              current_rank.unsqueeze(-1))
        group_weight = weight.gather(-1, group.unsqueeze(-1)).float()
        pack_weights.scatter_add_(-1, pack.unsqueeze(-1), group_weight)
        pack_items.scatter_add_(-1, pack.unsqueeze(-1),
                                torch.ones_like(pack.unsqueeze(-1)))

    # Represent each pack as fixed slots, making an exchange preserve the
    # required groups_per_pack cardinality automatically.
    slots = torch.arange(num_groups, dtype=torch.int64,
                         device=weight.device).expand(num_layers, -1)
    item_slot = pack_index * groups_per_pack + rank_in_pack
    slot_to_item = torch.empty_like(item_slot)
    slot_to_item.scatter_(1, item_slot, slots)

    # The best swap between the extreme packs transfers an amount close to
    # half their load difference. Four passes capture most useful exchanges
    # while keeping the refinement substantially cheaper than global search.
    rows = torch.arange(num_layers, dtype=torch.int64, device=weight.device)
    slot_rank = torch.arange(groups_per_pack, dtype=torch.int64,
                             device=weight.device)
    for _ in range(min(4, groups_per_pack)):
        heavy_pack = pack_weights.argmax(dim=-1)
        light_pack = pack_weights.argmin(dim=-1)
        difference = (pack_weights[rows, heavy_pack] -
                      pack_weights[rows, light_pack])

        heavy_slots = heavy_pack.unsqueeze(-1) * groups_per_pack + slot_rank
        light_slots = light_pack.unsqueeze(-1) * groups_per_pack + slot_rank
        heavy_items = slot_to_item.gather(1, heavy_slots)
        light_items = slot_to_item.gather(1, light_slots)
        heavy_weight = weight.gather(1, heavy_items).float()
        light_weight = weight.gather(1, light_items).float()

        transfer = heavy_weight.unsqueeze(-1) - light_weight.unsqueeze(-2)
        valid = ((transfer > 0) &
                 (transfer <= difference.view(-1, 1, 1)))
        cost = (difference.view(-1, 1, 1) - 2 * transfer).abs()
        cost.masked_fill_(~valid, unavailable)
        choice = cost.flatten(1).argmin(dim=-1)
        chosen_cost = cost.flatten(1).gather(1, choice.unsqueeze(-1)).squeeze(-1)
        improve = chosen_cost < unavailable

        heavy_rank = choice // groups_per_pack
        light_rank = choice.remainder(groups_per_pack)
        chosen_transfer = transfer[
            rows, heavy_rank, light_rank] * improve.float()
        chosen_heavy_slot = heavy_slots.gather(
            1, heavy_rank.unsqueeze(-1)).squeeze(-1)
        chosen_light_slot = light_slots.gather(
            1, light_rank.unsqueeze(-1)).squeeze(-1)
        chosen_heavy_item = slot_to_item.gather(
            1, chosen_heavy_slot.unsqueeze(-1)).squeeze(-1)
        chosen_light_item = slot_to_item.gather(
            1, chosen_light_slot.unsqueeze(-1)).squeeze(-1)

        slot_to_item.scatter_(1, chosen_heavy_slot.unsqueeze(-1),
                              torch.where(improve, chosen_light_item,
                                          chosen_heavy_item).unsqueeze(-1))
        slot_to_item.scatter_(1, chosen_light_slot.unsqueeze(-1),
                              torch.where(improve, chosen_heavy_item,
                                          chosen_light_item).unsqueeze(-1))
        pack_weights[rows, heavy_pack] -= chosen_transfer
        pack_weights[rows, light_pack] += chosen_transfer

    item_slot = torch.empty_like(slot_to_item)
    item_slot.scatter_(1, slot_to_item, slots)
    pack_index = item_slot // groups_per_pack
    rank_in_pack = item_slot.remainder(groups_per_pack)
    return pack_index, rank_in_pack
