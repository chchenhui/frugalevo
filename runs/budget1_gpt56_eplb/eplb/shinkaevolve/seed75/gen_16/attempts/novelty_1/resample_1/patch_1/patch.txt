+    """
+    Assign equally sized item sets to packs while greedily minimizing the

+    # EPLB planning operates on CPU. Converting once makes scalar heap updates
+    # inexpensive and avoids repeated tensor scalar synchronization.
+    cpu_weight = weight.float().cpu()
+    sorted_items = torch.argsort(cpu_weight, dim=-1, descending=True)

+    pack_index = torch.empty(
+        (num_layers, num_items), dtype=torch.int64, device="cpu"
+    )
+    rank_in_pack = torch.empty_like(pack_index)

+        # pack_id preserves deterministic preference for lower indexed packs.
+        available: list[tuple[float, int, int]] = [
+            (0.0, pack, 0) for pack in range(num_packs)
+        ]
+        heapq.heapify(available)
+
+        row_weight = cpu_weight[layer]
+        for item_tensor in sorted_items[layer]:
+            item = int(item_tensor)
+            load, pack, count = heapq.heappop(available)
+
+            pack_index[layer, item] = pack
+            rank_in_pack[layer, item] = count
+

+                    available, (load + float(row_weight[item]), pack, count)
+                )

+def _select_replica_candidates(
+    marginal_load: torch.Tensor,

+    """

+

+    equal marginal loads, prefer the source group with fewer previously added
+    replicas, then rely on argmax's stable lowest-index behavior.
+    """
