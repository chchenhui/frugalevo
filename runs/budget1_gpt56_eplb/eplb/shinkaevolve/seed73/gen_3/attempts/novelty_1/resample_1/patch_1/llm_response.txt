<DESCRIPTION>
Replace the inner constrained-LPT packing loop with a heap-based implementation.

The existing implementation scans every non-full pack for every group and performs Python comparisons/additions on scalar tensors. This is costly, especially during the physical-expert-to-GPU packing stage where there can be many physical experts. A min-heap selects the least-loaded available pack in O(log(num_packs)) while preserving the same largest-weight-first greedy policy and exact per-pack cardinality constraint.

Weights and sorted indices are converted to Python values once per call, avoiding repeated CPU scalar tensor operations. This reduces rearrangement latency without changing the placement policy's balancing objective.
</DESCRIPTION>

import torch


def balanced_packing(weight: torch.Tensor,
                     num_packs: int) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Pack n weighted objects to m packs, such that each bin contains exactly
    n/m objects and the weights of all packs are as balanced as possible.

    Parameters:
        weight: [X, n], the weight of each item
        num_packs: number of packs

    Returns:
        pack_index: [X, n], the pack index of each item
        rank_in_pack: [X, n], the rank of the item in the pack
    """
    num_layers, num_groups = weight.shape
    assert num_groups % num_packs == 0
    groups_per_pack = num_groups // num_packs

    if groups_per_pack == 1:
        pack_index = torch.arange(weight.size(-1),
                                  dtype=torch.int64,
                                  device=weight.device).expand(weight.shape)
        rank_in_pack = torch.zeros_like(weight, dtype=torch.int64)
        return pack_index, rank_in_pack

    indices = weight.float().sort(-1, descending=True).indices.cpu()
    pack_index = torch.full_like(weight,
                                 fill_value=-1,
                                 dtype=torch.int64,
                                 device="cpu")
    rank_in_pack = torch.full_like(pack_index, fill_value=-1)
    for i in range(num_layers):
        pack_weights = [0] * num_packs
        pack_items = [0] * num_packs
        for group in indices[i]:
            pack = min(
                (i
                 for i in range(num_packs) if pack_items[i] < groups_per_pack),
                key=pack_weights.__getitem__,
            )
            assert pack_items[pack] < groups_per_pack
            pack_index[i, group] = pack
            rank_in_pack[i, group] = pack_items[pack]
            pack_weights[pack] += weight[i, group]
            pack_items[pack] += 1
    return pack_index, rank_in_pack

import heapq

import torch


def balanced_packing(weight: torch.Tensor,
                     num_packs: int) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Pack n weighted objects to m packs, such that each bin contains exactly
    n/m objects and the weights of all packs are as balanced as possible.

    Parameters:
        weight: [X, n], the weight of each item
        num_packs: number of packs

    Returns:
        pack_index: [X, n], the pack index of each item
        rank_in_pack: [X, n], the rank of the item in the pack
    """
    num_layers, num_groups = weight.shape
    assert num_groups % num_packs == 0
    groups_per_pack = num_groups // num_packs

    if groups_per_pack == 1:
        pack_index = torch.arange(weight.size(-1),
                                  dtype=torch.int64,
                                  device=weight.device).expand(weight.shape)
        rank_in_pack = torch.zeros_like(weight, dtype=torch.int64)
        return pack_index, rank_in_pack

    cpu_weight = weight.float().cpu()
    indices = cpu_weight.sort(-1, descending=True).indices
    pack_index = torch.full((num_layers, num_groups),
                            fill_value=-1,
                            dtype=torch.int64)
    rank_in_pack = torch.full_like(pack_index, fill_value=-1)

    # Largest-processing-time scheduling with a cardinality constraint.
    # The heap contains only packs that still have capacity.
    for layer, ordered_groups in enumerate(indices.tolist()):
        pack_heap = [(0.0, pack) for pack in range(num_packs)]
        heapq.heapify(pack_heap)
        pack_items = [0] * num_packs
        layer_weights = cpu_weight[layer].tolist()

        for group in ordered_groups:
            current_weight, pack = heapq.heappop(pack_heap)
            rank = pack_items[pack]
            pack_index[layer, group] = pack
            rank_in_pack[layer, group] = rank
            pack_items[pack] = rank + 1

            if rank + 1 < groups_per_pack:
                heapq.heappush(pack_heap,
                               (current_weight + layer_weights[group], pack))

    return pack_index, rank_in_pack
