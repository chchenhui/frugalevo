```python
# EVOLVE-BLOCK-START
import logging
import math
import random
from dataclasses import dataclass
from typing import Any, Dict, List, Optional, Tuple

from skydiscover.config import DatabaseConfig
from skydiscover.search.base_database import Program, ProgramDatabase

logger = logging.getLogger(__name__)


@dataclass
class EvolvedProgram(Program):
    """Program for the evolved database."""


class EvolvedProgramDatabase(ProgramDatabase):
    """Adaptive elite search with novelty-aware parent and context selection."""

    def __init__(self, name: str, config: DatabaseConfig):
        super().__init__(name, config)
        self.random_state = random.Random(getattr(config, "random_seed", None))
        self.best_score_seen = float("-inf")
        self.meaningful_score = float("-inf")
        self.last_meaningful_iteration = -1
        self.last_seen_iteration = -1
        self.parent_uses: Dict[str, int] = {}
        self.context_uses: Dict[str, int] = {}
        self.parent_trials: Dict[str, int] = {}
        self.parent_wins: Dict[str, int] = {}

    def _score(self, program: EvolvedProgram) -> Optional[float]:
        value = program.metrics.get("combined_score")
        if isinstance(value, (int, float)):
            value = float(value)
            if math.isfinite(value):
                return value
        return None

    def _meaningful_improvement(self, score: float, reference: float) -> bool:
        if not math.isfinite(reference):
            return True
        return (score - reference) > 0.01 or score > reference * 1.01

    def add(
        self, program: EvolvedProgram, iteration: Optional[int] = None, **kwargs: Any
    ) -> str:
        self.programs[program.id] = program

        found_at = iteration
        if not isinstance(found_at, int):
            found_at = program.iteration_found if isinstance(program.iteration_found, int) else -1
        self.last_seen_iteration = max(self.last_seen_iteration, found_at)

        if iteration is not None:
            self.last_iteration = max(self.last_iteration, iteration)

        score = self._score(program)
        if score is not None:
            if score > self.best_score_seen:
                self.best_score_seen = score
            if self._meaningful_improvement(score, self.meaningful_score):
                self.meaningful_score = score
                self.last_meaningful_iteration = found_at

        if program.parent_id:
            parent_id = program.parent_id
            self.parent_uses[parent_id] = self.parent_uses.get(parent_id, 0) + 1
            self.parent_trials[parent_id] = self.parent_trials.get(parent_id, 0) + 1
            parent = self.get(parent_id)
            parent_score = self._score(parent) if parent is not None else None
            if (
                score is not None
                and parent_score is not None
                and self._meaningful_improvement(score, parent_score)
            ):
                self.parent_wins[parent_id] = self.parent_wins.get(parent_id, 0) + 1

        for context_id in program.other_context_ids:
            self.context_uses[context_id] = self.context_uses.get(context_id, 0) + 1

        if self.config.db_path:
            self._save_program(program)
        self._update_best_program(program)
        return program.id

    def sample(
        self, num_context_programs: Optional[int] = 4, **kwargs: Any
    ) -> Tuple[Dict[str, EvolvedProgram], Dict[str, List[EvolvedProgram]]]:
        scored = [(p, self._score(p)) for p in self.programs.values()]
        scored = [(p, s) for p, s in scored if s is not None]
        if not scored:
            candidates = list(self.programs.values())
            if not candidates:
                raise ValueError("No candidates available for sampling")
            parent = self.random_state.choice(candidates)
            return {"": parent}, {"": []}

        scored.sort(key=lambda item: item[1], reverse=True)
        elite_count = min(len(scored), max(4, min(10, len(scored) // 2 + 1)))
        elite = [p for p, _ in scored[:elite_count]]

        weights = []
        for rank, program in enumerate(elite):
            uses = self.parent_uses.get(program.id, 0)
            trials = self.parent_trials.get(program.id, 0)
            wins = self.parent_wins.get(program.id, 0)
            success = wins / trials if trials else 0.0
            weights.append(((elite_count - rank) ** 1.4) * (1.0 + success) / (1.0 + uses))

        parent = self.random_state.choices(elite, weights=weights, k=1)[0]

        stalled = (
            self.last_meaningful_iteration >= 0
            and self.last_seen_iteration - self.last_meaningful_iteration >= 10
        )
        phase = self.last_seen_iteration % 9 if self.last_seen_iteration >= 0 else -1
        if stalled and phase == 0:
            return {self.DIVERGE_LABEL: parent}, {"": []}
        if stalled and phase == 4:
            return {self.REFINE_LABEL: parent}, {"": []}

        wanted = max(0, num_context_programs or 0)
        available = [p for p, _ in scored if p.id != parent.id]
        contexts: List[EvolvedProgram] = []

        # Mostly show strong alternatives, while reserving one slot for a
        # nearby-but-different candidate that can expose a missed adjustment.
        strong = [p for p in elite if p.id != parent.id]
        while strong and len(contexts) < min(wanted, 3):
            weights = [1.0 / (1.0 + self.context_uses.get(p.id, 0)) for p in strong]
            choice = self.random_state.choices(strong, weights=weights, k=1)[0]
            contexts.append(choice)
            strong = [p for p in strong if p.id != choice.id]

        remaining = [p for p in available if p.id not in {c.id for c in contexts}]
        if remaining and len(contexts) < wanted:
            middle = remaining[min(2, len(remaining)) : min(12, len(remaining))]
            pool = middle or remaining
            self.random_state.shuffle(pool)
            contexts.append(pool[0])

        for program in remaining:
            if len(contexts) >= wanted:
                break
            if program.id not in {c.id for c in contexts}:
                contexts.append(program)

        return {"": parent}, {"": contexts[:wanted]}


# EVOLVE-BLOCK-END
```

Principles:
1. **Exploit without repeatedly mutating the same clone:** parents come from the high-scoring elite, but parent reuse is penalized and parents that previously produced meaningful gains receive a modest boost.
2. **React to genuine stagnation:** progress is tracked in `add()` using the required 1%/0.01 meaningful-improvement threshold. After a sustained plateau, occasional targeted divergence/refinement is used; otherwise contexts combine strong alternatives with a less-reused nearby candidate.
