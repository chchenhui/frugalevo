<DESCRIPTION>
Add several deterministic geometric initializers to complement purely Gaussian starts.  The two staggered hexagonal rings plus poles and the cube-plus-axes configuration are natural high-contact 14-point frameworks in three dimensions; they provide substantially better organized initial contact graphs than random clouds.  Small reproducible perturbations let the smooth optimization break unnecessary symmetry.  Also evaluate every initial configuration before its first update, so a good structured construction cannot be discarded by an initially unfavorable surrogate-gradient step.

    def structured_seed(which: int) -> np.ndarray:
        """High-contact deterministic frameworks, lightly symmetry-broken."""
        if which < 5:
            radius, height, twist, pole = (
                (0.946, 0.405, np.pi / 6.0, 1.000),
                (0.920, 0.385, np.pi / 6.0, 0.980),
                (0.975, 0.430, np.pi / 6.0, 1.030),
                (0.950, 0.415, 0.455, 1.000),
                (0.955, 0.400, 0.585, 1.000),
            )[which]
            angles = np.arange(6, dtype=float) * (np.pi / 3.0)
            lower = np.column_stack((
                radius * np.cos(angles), radius * np.sin(angles),
                -height * np.ones(6),
            ))
            upper = np.column_stack((
                radius * np.cos(angles + twist),
                radius * np.sin(angles + twist), height * np.ones(6),
            ))
            seed = np.vstack((
                lower, upper, [[0.0, 0.0, -pole], [0.0, 0.0, pole]]
            ))
        else:
            cube = np.array(
                [[a, b, c] for a in (-1.0, 1.0)
                 for b in (-1.0, 1.0) for c in (-1.0, 1.0)],
                dtype=float,
            )
            axes = np.array([
                [1.55, 0.0, 0.0], [-1.55, 0.0, 0.0],
                [0.0, 1.67, 0.0], [0.0, -1.67, 0.0],
                [0.0, 0.0, 1.60], [0.0, 0.0, -1.60],
            ])
            seed = np.vstack((cube, axes))
        return normalize(seed + 0.012 * rng.standard_normal((n, dim)))
