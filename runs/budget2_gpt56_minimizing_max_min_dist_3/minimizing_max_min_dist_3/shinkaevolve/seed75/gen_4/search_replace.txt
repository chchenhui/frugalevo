def min_max_dist_dim3_14() -> np.ndarray:
    """
    Return a symmetric 14-point packing with minimum distance sqrt(2)
    and diameter 2 * sqrt(2).
    """
    # The twelve permutations of (±1, ±1, 0) are cuboctahedron vertices.
    # Adding opposite axial points preserves the cuboctahedron's minimum
    # separation while keeping the same diameter.
    vertices = []
    for zero_axis in range(3):
        other_axes = [axis for axis in range(3) if axis != zero_axis]
        for sign_a in (-1.0, 1.0):
            for sign_b in (-1.0, 1.0):
                point = np.zeros(3)
                point[other_axes[0]] = sign_a
                point[other_axes[1]] = sign_b
                vertices.append(point)

    points = np.asarray(vertices + [
        np.array([0.0, 0.0, -np.sqrt(2.0)]),
        np.array([0.0, 0.0, np.sqrt(2.0)]),
    ])
    return points
