<DESCRIPTION>
This crossover retains the stronger unconstrained RMS-normalized search from the current program, while adding the accept-only active-contact polishing strategy from the higher-quality prior variant. It also replaces repeated full 14-by-14 distance-matrix gradient construction with direct computation over the 91 unique pairs, reducing allocation overhead and allowing the final exact min/max contact refinement to be used efficiently. The construction remains deterministic, returns exactly fourteen finite 3D points, and optimizes the evaluator’s squared distance ratio directly during polishing.

import numpy as np


def _normalize_rows(points: np.ndarray) -> np.ndarray:
    """Remove translation and fix RMS radius without imposing cosphericity."""
    centered = points - np.mean(points, axis=0, keepdims=True)
    scale = np.sqrt(np.mean(np.sum(centered * centered, axis=1)))
    return centered / max(float(scale), 1e-12)


def _pair_values(points: np.ndarray, upper: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """Return unique pair displacement vectors and their squared distances."""
    delta = points[upper[0]] - points[upper[1]]
    values = np.sum(delta * delta, axis=1)
    return delta, values


def _true_ratio_score(points: np.ndarray, upper: np.ndarray) -> float:
    """Evaluator objective: minimum squared distance divided by maximum."""
    _, values = _pair_values(points, upper)
    return float(np.min(values) / np.max(values))


def _cube_octahedron_seed() -> np.ndarray:
    """Cube vertices and axial vertices give a useful contact-graph prior."""
    cube = np.array(
        [[x, y, z] for x in (-1.0, 1.0)
                   for y in (-1.0, 1.0)
                   for z in (-1.0, 1.0)],
        dtype=float,
    )
    axes = np.vstack((np.eye(3), -np.eye(3))) * np.sqrt(3.0)
    return _normalize_rows(np.vstack((cube, axes)))


def _fibonacci_seed(count: int) -> np.ndarray:
    """Deterministic nearly uniform spherical seed."""
    index = np.arange(count, dtype=float)
    golden = np.pi * (3.0 - np.sqrt(5.0))
    z = 1.0 - 2.0 * (index + 0.5) / count
    radius = np.sqrt(np.maximum(0.0, 1.0 - z * z))
    angle = golden * index
    return np.column_stack((radius * np.cos(angle), radius * np.sin(angle), z))


def _remove_invariances(vector: np.ndarray, points: np.ndarray) -> np.ndarray:
    """Remove translation and global radial scale components."""
    vector = vector - np.mean(vector, axis=0, keepdims=True)
    radial = np.sum(vector * points) / max(float(np.sum(points * points)), 1e-12)
    return vector - radial * points


def _pair_force(
    points: np.ndarray,
    upper: np.ndarray,
    coefficients: np.ndarray,
) -> np.ndarray:
    """Accumulate logarithmic pairwise distance forces directly by pairs."""
    delta, values = _pair_values(points, upper)
    contribution = 2.0 * coefficients[:, None] * delta / np.maximum(
        values[:, None], 1e-12
    )
    force = np.zeros_like(points)
    np.add.at(force, upper[0], contribution)
    np.add.at(force, upper[1], -contribution)
    return force


def _smooth_ratio_gradient(
    points: np.ndarray,
    upper: np.ndarray,
    beta: float,
    active_mix: float,
) -> np.ndarray:
    """Gradient of a soft min/max logarithmic squared-distance ratio."""
    _, values = _pair_values(points, upper)
    values = np.maximum(values, 1e-12)
    logs = np.log(values)

    low_logits = -beta * logs
    low_logits -= np.max(low_logits)
    low_weights = np.exp(low_logits)
    low_weights /= np.sum(low_weights)

    high_logits = beta * logs
    high_logits -= np.max(high_logits)
    high_weights = np.exp(high_logits)
    high_weights /= np.sum(high_weights)

    coefficients = low_weights - high_weights

    if active_mix:
        minimum = float(np.min(values))
        maximum = float(np.max(values))
        close = values <= minimum * 1.014
        far = values >= maximum * 0.986
        active = np.zeros_like(values)
        active[close] = 1.0 / max(int(np.count_nonzero(close)), 1)
        active[far] -= 1.0 / max(int(np.count_nonzero(far)), 1)
        coefficients = (1.0 - active_mix) * coefficients + active_mix * active

    return _pair_force(points, upper, coefficients)


def _polish_contacts(
    points: np.ndarray,
    best_points: np.ndarray,
    best_score: float,
    upper: np.ndarray,
) -> tuple[np.ndarray, float]:
    """Accept-only refinement using the exact closest and diameter contacts."""
    velocity = np.zeros_like(points)

    for tolerance, base_step, iterations in (
        (0.030, 0.018, 90),
        (0.014, 0.010, 120),
        (0.006, 0.005, 150),
    ):
        for _ in range(iterations):
            _, values = _pair_values(points, upper)
            minimum = float(np.min(values))
            maximum = float(np.max(values))

            close = values <= minimum * (1.0 + tolerance)
            far = values >= maximum * (1.0 - tolerance)
            coefficients = np.zeros_like(values)
            coefficients[close] = 1.0 / max(int(np.count_nonzero(close)), 1)
            coefficients[far] -= 1.0 / max(int(np.count_nonzero(far)), 1)

            force = _remove_invariances(_pair_force(points, upper, coefficients), points)
            velocity = _remove_invariances(0.45 * velocity + force, points)
            norm = np.sqrt(np.mean(np.sum(velocity * velocity, axis=1)))
            if norm < 1e-14:
                break

            accepted = False
            for multiplier in (1.0, 0.5, 0.25, 0.125, 0.0625):
                trial = _normalize_rows(
                    points + base_step * multiplier * velocity / norm
                )
                score = _true_ratio_score(trial, upper)
                if score > best_score + 1e-13:
                    points = trial
                    best_points = trial.copy()
                    best_score = score
                    accepted = True
                    break

            if not accepted:
                velocity *= 0.25
                if np.sqrt(np.mean(np.sum(velocity * velocity, axis=1))) < 1e-10:
                    break

    return best_points, best_score


def _optimize_seed(
    initial_points: np.ndarray,
    upper: np.ndarray,
) -> tuple[np.ndarray, float]:
    """Staged smooth optimization followed by exact contact-set polishing."""
    points = initial_points.copy()
    velocity = np.zeros_like(points)
    best_points = points.copy()
    best_score = _true_ratio_score(points, upper)

    stages = (
        (7.0, 360, 0.040, 0.00),
        (16.0, 420, 0.026, 0.00),
        (38.0, 480, 0.016, 0.10),
        (90.0, 560, 0.009, 0.32),
        (180.0, 360, 0.005, 0.52),
    )

    for beta, iterations, initial_step, active_mix in stages:
        for iteration in range(iterations):
            gradient = _remove_invariances(
                _smooth_ratio_gradient(points, upper, beta, active_mix), points
            )
            velocity = _remove_invariances(0.72 * velocity + gradient, points)

            norm = np.sqrt(np.mean(np.sum(velocity * velocity, axis=1)))
            if norm > 3.0:
                velocity *= 3.0 / norm

            progress = iteration / max(iterations - 1, 1)
            step = initial_step * (1.0 - 0.70 * progress)
            points = _normalize_rows(points + step * velocity)

            if iteration % 12 == 0 or iteration == iterations - 1:
                score = _true_ratio_score(points, upper)
                if score > best_score:
                    best_score = score
                    best_points = points.copy()

    return _polish_contacts(best_points.copy(), best_points, best_score, upper)


def min_max_dist_dim3_14() -> np.ndarray:
    """Construct fourteen deterministic 3D points with a high distance ratio."""
    count = 14
    upper = np.triu_indices(count, k=1)
    rng = np.random.default_rng(917341)

    cube_axis = _cube_octahedron_seed()
    seeds = [cube_axis, _fibonacci_seed(count)]

    for magnitude in (0.018, 0.045, 0.085, 0.140):
        seeds.append(
            _normalize_rows(cube_axis + magnitude * rng.normal(size=(count, 3)))
        )

    for _ in range(12):
        seeds.append(_normalize_rows(rng.normal(size=(count, 3))))

    best_points = seeds[0]
    best_score = _true_ratio_score(best_points, upper)

    for seed in seeds:
        candidate, score = _optimize_seed(seed, upper)
        if score > best_score:
            best_points = candidate
            best_score = score

    return np.asarray(best_points, dtype=float)
