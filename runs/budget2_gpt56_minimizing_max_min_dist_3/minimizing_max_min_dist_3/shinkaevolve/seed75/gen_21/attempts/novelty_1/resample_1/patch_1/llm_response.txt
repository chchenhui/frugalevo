<DESCRIPTION>
Add a final deterministic constrained optimization phase using the exact packing formulation: maximize a common squared separation lower bound `r` subject to every pairwise squared distance lying in `[r, 1]`. The existing stochastic/contact-graph search is retained for global basin discovery, while SLSQP receives its best result as a high-quality feasible warm start. Anchoring one point at the origin removes translational degeneracy, and scaling the incumbent diameter to one makes the constraint system directly match the evaluator objective. Analytic constraint Jacobians make this polishing phase substantially more reliable and faster than finite-difference constrained optimization.
</DESCRIPTION>

    # Final constrained polish in the exact diameter-normalized formulation:
    # maximize r subject to r <= ||p_i-p_j||^2 <= 1 for every pair.  The
    # preceding search supplies a good contact graph; SLSQP then adjusts all
    # contacts simultaneously rather than accepting only one-point moves.
    try:
        from scipy.optimize import minimize

        qbest = distances_sq(best)
        diameter_pair = int(np.argmax(qbest))
        scale = np.sqrt(qbest[diameter_pair])
        anchored = best / scale - best[0] / scale
        y0 = anchored[1:].ravel()
        z0 = np.concatenate((y0, [0.999999 * np.min(qbest) / np.max(qbest)]))

        def unpack(z: np.ndarray) -> np.ndarray:
            p = np.zeros((n, 3), dtype=float)
            p[1:] = z[:-1].reshape(n - 1, 3)
            return p

        def constraint_values(z: np.ndarray) -> np.ndarray:
            p = unpack(z)
            difference = p[ii] - p[jj]
            q = np.einsum("ij,ij->i", difference, difference)
            return np.concatenate((q - z[-1], 1.0 - q))

        def constraint_jacobian(z: np.ndarray) -> np.ndarray:
            p = unpack(z)
            difference = p[ii] - p[jj]
            jac = np.zeros((2 * len(ii), 3 * (n - 1) + 1), dtype=float)
            rows = np.arange(len(ii))
            for coordinate in range(3):
                valid_i = ii != 0
                valid_j = jj != 0
                jac[rows[valid_i], 3 * (ii[valid_i] - 1) + coordinate] = (
                    2.0 * difference[valid_i, coordinate]
                )
                jac[rows[valid_j], 3 * (jj[valid_j] - 1) + coordinate] = (
                    -2.0 * difference[valid_j, coordinate]
                )
            jac[:len(ii), -1] = -1.0
            jac[len(ii):, :-1] = -jac[:len(ii), :-1]
            return jac

        result = minimize(
            lambda z: -z[-1],
            z0,
            jac=lambda z: np.r_[np.zeros(3 * (n - 1)), -1.0],
            constraints={"type": "ineq", "fun": constraint_values,
                         "jac": constraint_jacobian},
            bounds=[(None, None)] * (3 * (n - 1)) + [(0.0, 1.0)],
            method="SLSQP",
            options={"maxiter": 700, "ftol": 1.0e-12, "disp": False},
        )
        if result.success:
            polished = normalize(unpack(result.x))
            polished_value = ratio_sq(polished)
            if polished_value > best_value:
                best_value = polished_value
                best = polished
    except Exception:
        # The already computed exact-objective result remains valid if an
        # optional optimizer is unavailable in a restricted environment.
        pass

    return np.asarray(best, dtype=float)
