<DESCRIPTION>
Replace uniform soft active-pair averaging in the annealing stage with persistent-contact weighting. The nonsmooth diameter-packing objective is controlled by a small contact graph: pairs repeatedly near the current minimum distance should retain strong repulsion, while pairs repeatedly near the diameter should retain strong attraction. Transient pairs that only briefly enter a softmax tail otherwise dilute these forces.

The new update tracks exponentially averaged lower- and upper-band membership. It then reweights the existing differentiable log-distance forces by this persistence, while preserving a nonzero baseline so contacts can still change. The bands narrow as annealing progresses, allowing broad early exploration and sharper late contact-graph polishing. This directly targets stable min/max constraints without changing the exact SLSQP formulation or return format.
</DESCRIPTION>

    def anneal(seed: np.ndarray, steps: int = 520) -> np.ndarray:
        """
        Smooth log-diameter force flow.  Low-distance pairs receive
        repulsion and high-distance pairs receive attraction.  RMS
        normalization removes the irrelevant scale direction.
        """
        x = rms_normalize(seed)
        velocity = np.zeros_like(x)

        for it in range(steps):
            u = it / max(steps - 1, 1)
            sharp = 5.0 + 115.0 * u * u

            d = x[pi] - x[pj]
            q = np.einsum("ij,ij->i", d, d)
            q = np.maximum(q, 1e-10)
            z = np.log(q)

            a = -sharp * z
            a -= a.max()
            low = np.exp(a)
            low /= low.sum()

            b = sharp * z
            b -= b.max()
            high = np.exp(b)
            high /= high.sum()

            # Gradient of soft-min(log q) - soft-max(log q).
            force_pair = 2.0 * (low - high)[:, None] * d / q[:, None]
            force = np.zeros_like(x)
            np.add.at(force, pi, force_pair)
            np.add.at(force, pj, -force_pair)

            force -= force.mean(axis=0, keepdims=True)
            velocity = 0.67 * velocity + force
            velocity -= velocity.mean(axis=0, keepdims=True)

            limit = 3.0
            vn = np.linalg.norm(velocity, axis=1, keepdims=True)
            velocity *= np.minimum(1.0, limit / np.maximum(vn, 1e-12))

            step = 0.030 * (1.0 - 0.72 * u)
            x = rms_normalize(x + step * velocity)

        return diameter_normalize(x)

    def anneal(seed: np.ndarray, steps: int = 520) -> np.ndarray:
        """
        Smooth log-diameter force flow with persistent-contact weighting.

        The actual objective is governed by a small set of pairs attaining
        the minimum distance and diameter.  Exponentially averaging narrow
        contact-band membership prevents fleeting softmax-active pairs from
        diluting the forces on that emerging contact graph.
        """
        x = rms_normalize(seed)
        velocity = np.zeros_like(x)
        low_persistence = np.zeros(pairs, dtype=float)
        high_persistence = np.zeros(pairs, dtype=float)

        for it in range(steps):
            u = it / max(steps - 1, 1)
            sharp = 5.0 + 115.0 * u * u

            d = x[pi] - x[pj]
            q = np.einsum("ij,ij->i", d, d)
            q = np.maximum(q, 1e-10)
            z = np.log(q)

            a = -sharp * z
            a -= a.max()
            low = np.exp(a)
            low /= low.sum()

            b = sharp * z
            b -= b.max()
            high = np.exp(b)
            high /= high.sum()

            qlo = float(q.min())
            qhi = float(q.max())
            tau = 0.30 - 0.22 * u
            lower_band = np.exp(
                -np.minimum(q / qlo - 1.0, 30.0) / max(tau, 0.04)
            )
            upper_band = np.exp(
                -np.minimum(qhi / q - 1.0, 30.0) / max(tau, 0.04)
            )
            memory = 0.82 + 0.13 * u
            low_persistence = memory * low_persistence + (1.0 - memory) * lower_band
            high_persistence = memory * high_persistence + (1.0 - memory) * upper_band

            # Preserve a baseline soft force so that the contact graph can
            # still evolve, but emphasize pairs that repeatedly remain active.
            low *= 0.20 + low_persistence
            low /= low.sum()
            high *= 0.20 + high_persistence
            high /= high.sum()
