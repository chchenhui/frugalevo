+            except Exception:
+                pass

+            g1 = self._global_order(ser, weighted=True)

+        try:

+            pass

+            candidates.append(self._conditional_orders(ser, depth_limit))
+        except Exception:
+            pass

+        if not candidates:

+        best_orders, best_score = None, -1.0

+            try:
+                s = self._score(ser, orders)
+            except Exception:

+            if s > best_score:

+        if best_orders is None:

+        col_names = list(work.columns)
