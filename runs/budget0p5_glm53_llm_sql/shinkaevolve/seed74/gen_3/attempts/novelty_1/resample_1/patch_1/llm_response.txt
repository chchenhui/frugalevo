import pandas as pd
import numpy as np
from solver import Algorithm
from typing import Tuple, List


class Evolved(Algorithm):
    """
    Character-Trie-aware column reordering.

    Strategy: build a few cheap candidate global column orders, score each by
    the exact ideal serial character-Trie reuse (sum of adjacent longest-common
    prefixes over the sorted serialized row strings), and return the best.
    All original cells, rows and columns are preserved exactly.
    """

    # ---------------- serialization helpers ----------------

    @staticmethod
    def _cell_str(v) -> str:
        try:
            if v is None or (isinstance(v, float) and np.isnan(v)):
                return ""
        except Exception:
            pass
        try:
            if pd.isna(v):
                return ""
        except Exception:
            pass
        return str(v)

    def _col_str_arrays(self, df: pd.DataFrame, cols: List[str]):
        """Per-column lists of serialized cell strings (missing -> '')."""
        arrays = {}
        for c in cols:
            s = df[c]
            arrays[c] = [self._cell_str(v) for v in s.tolist()]
        return arrays

    @staticmethod
    def _lcp(a: str, b: str) -> int:
        n = min(len(a), len(b))
        if n == 0:
            return 0
        if a[:n] == b[:n]:
            return n
        # binary search on prefix equality (C-speed slicing comparisons)
        lo, hi = 0, n
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if a[:mid] == b[:mid]:
                lo = mid
            else:
                hi = mid - 1
        return lo

    def _trie_score(self, row_strings: List[str]) -> int:
        if len(row_strings) <= 1:
            return 0
        ss = sorted(set(row_strings)) if len(set(row_strings)) < len(row_strings) else sorted(row_strings)
        total = 0
        prev = ss[0]
        # fully credit duplicated rows once via set; handle separately
        if len(ss) < len(row_strings):
            # duplicates share the entire string with their twin
            counts = {}
            for r in row_strings:
                counts[r] = counts.get(r, 0) + 1
            dup_extra = sum((cnt - 1) * len(r) for r, cnt in counts.items())
        else:
            dup_extra = 0
        for cur in ss[1:]:
            total += self._lcp(prev, cur)
            prev = cur
        return total + dup_extra

    def _serialize_rows(self, n_rows: int, cols: List[str], col_arrays) -> List[str]:
        out = []
        for i in range(n_rows):
            parts = [col_arrays[c][i] for c in cols]
            out.append("".join(parts))
        return out

    # ---------------- candidate order constructions ----------------

    def _candidate_orders(self, df: pd.DataFrame, cols: List[str], col_arrays) -> List[List[str]]:
        n = len(df)
        if n == 0:
            return [list(cols)]

        candidates = [list(cols)]  # 1. original order

        # 2. length-weighted pair repetition: sum len(v) * cnt * (cnt-1), desc
        scores = {}
        for c in cols:
            vals = col_arrays[c]
            counts = {}
            for v in vals:
                counts[v] = counts.get(v, 0) + 1
            s = sum(len(v) * k * (k - 1) for v, k in counts.items())
            scores[c] = s
        order2 = sorted(cols, key=lambda c: (-scores[c], cols.index(c), str(c)))
        candidates.append(order2)

        # 3. distinct-count ascending (low-cardinality first), len-weighted tiebreak
        distinct = {}
        for c in cols:
            distinct[c] = len(set(col_arrays[c]))
        order3 = sorted(cols, key=lambda c: (distinct[c], -scores[c], cols.index(c), str(c)))
        candidates.append(order3)

        return candidates

    # ---------------- public API ----------------

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        # Work on a copy so the input is never mutated.
        out_df = df.copy()

        # Honor column merges using the base-class semantics.
        if col_merge:
            for group in col_merge:
                group_cols = [c for c in out_df.columns if c in group]
                if len(group_cols) > 1:
                    try:
                        out_df = self.merging_columns(out_df, group_cols, prepended=False)
                    except Exception:
                        # Fallback: keep columns but place them adjacent in order.
                        pass

        cols = list(out_df.columns)
        n_rows = len(out_df)

        if len(cols) == 0 or n_rows == 0:
            empty_orders: List[List[str]] = [list(cols) for _ in range(n_rows)]
            return out_df, empty_orders

        col_arrays = self._col_str_arrays(out_df, cols)
        candidates = self._candidate_orders(out_df, cols, col_arrays)

        best_order = candidates[0]
        best_score = -1
        for order in candidates:
            rows = self._serialize_rows(n_rows, order, col_arrays)
            sc = self._trie_score(rows)
            if sc > best_score:
                best_score = sc
                best_order = order

        result_df = out_df.loc[:, best_order].copy()
        # keep object dtype for mixed-type frames to avoid unwanted coercion
        try:
            if result_df.dtypes.apply(lambda d: d == object).any():
                result_df = result_df.astype(object)
        except Exception:
            pass

        column_orderings: List[List[str]] = [list(best_order) for _ in range(n_rows)]

        # Safety: shape and cell preservation (same multiset of values per column).
        assert result_df.shape == out_df.shape
        return result_df, column_orderings
