+def _lcp(a: str, b: str) -> int:
+    """Longest common prefix length via binary search (C-speed slices)."""
+    if a == b:
+        return len(a)
+    n = min(len(a), len(b))
+    lo, hi = 0, n
+    while lo < hi:
+        mid = (lo + hi + 1) // 2
+        if a[:mid] == b[:mid]:
+            lo = mid
+        else:
+            hi = mid - 1
+    return lo

+        orders = [list(order_fn(i)) for i in range(n)]
+        for i in range(n):
+            assert sorted(orders[i]) == sorted(cols)
+
+        # Sort rows by serialized string (trie reuse is order-independent,
+        # but this yields a deterministic, cache-friendly output).
+        rowstr = ["".join(ser[c][i] for c in orders[i]) for i in range(n)]
+        perm = sorted(range(n), key=lambda i: rowstr[i])
+
+        # Materialize per-row permutations into an object-dtype frame,
+        # copying raw values verbatim.
+        arr = np.empty((n, len(cols)), dtype=object)
+        for r, i in enumerate(perm):
+            ordr = orders[i]
+            for j, c in enumerate(ordr):
+                arr[r, j] = raw[c][i]
+        out = pd.DataFrame(arr, columns=cols)
+
+        column_orderings = [orders[i] for i in perm]
+
+        assert out.shape == (n, len(cols))
+        return out, column_orderings
