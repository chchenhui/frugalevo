def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    # Backfill unknown costs with a large finite penalty so shortest-path
    # computations never crash on None and unknown links are deprioritized.
    for u, v, d in h.edges(data=True):
        if d.get("cost") is None:
            h[u][v]["cost"] = 1e6
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    # Pre-compute single-source shortest paths once (avoids repeated Dijkstra runs)
    dist, paths = nx.single_source_dijkstra(h, src, weight="cost")

    for dst in dsts:
        # Collect candidate paths: the shortest path plus a few cheap
        # alternatives, to spread load across parallel routes/networks.
        candidates = []
        if paths.get(dst) is not None:
            candidates.append(paths[dst])
        try:
            count = 0
            for cand in nx.shortest_simple_paths(h, src, dst, weight="cost"):
                if cand not in candidates:
                    candidates.append(cand)
                count += 1
                if count >= max(3, num_partitions):
                    break
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            pass

        if not candidates:
            continue

        # Assign each partition a path, rotating through candidates for
        # load balancing across parallel routes/networks.
        for j in range(bc_topology.num_partitions):
            path = candidates[j % len(candidates)]
            for i in range(0, len(path) - 1):
                s, t = path[i], path[i + 1]
                bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology
