<DESCRIPTION>
The current algorithm sends every partition of the data along the exact same shortest path (computed once per dst), so all partitions redundantly traverse the same links, driving up cost and congestion. A better strategy: for each destination, compute multiple edge-disjoint low-cost paths (using short path enumeration with a cost cap) and assign different partitions to different paths. This spreads load across networks/links, avoids redundant transfers over the same expensive edges, and exploits multi-network topologies. Additionally, paths should be sorted by cost so the cheapest paths carry partitions first.

def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    for dst in dsts:
        # Enumerate candidate simple paths sorted by cost; assign
        # partitions round-robin across distinct low-cost paths so
        # partitions spread over disjoint/multi-network links instead of
        # all redundantly traversing the single cheapest path.
        try:
            candidate_paths = list(
                nx.shortest_simple_paths(h, src, dst, weight="cost")
            )
        except nx.NetworkXNoPath:
            # Fallback: unreachable destination, leave partitions unset
            continue

        # Compute per-path cost, keep a diverse set (deduplicate identical
        # edge sets), sort by total cost ascending.
        scored = []
        seen_edge_sets = set()
        for path in candidate_paths[: num_partitions * 5]:
            edge_set = frozenset(zip(path[:-1], path[1:]))
            if edge_set in seen_edge_sets:
                continue
            seen_edge_sets.add(edge_set)
            total_cost = 0.0
            valid = True
            for i in range(len(path) - 1):
                s, t = path[i], path[i + 1]
                edge_cost = G[s][t].get("cost")
                if edge_cost is None:
                    valid = False
                    break
                total_cost += edge_cost
            if valid:
                scored.append((total_cost, path))
        scored.sort(key=lambda x: x[0])

        if not scored:
            # No valid-cost path found; fall back to dijkstra on 'cost'
            try:
                path = nx.dijkstra_path(h, src, dst, weight="cost")
            except nx.NetworkXNoPath:
                continue
            for j in range(bc_topology.num_partitions):
                for i in range(len(path) - 1):
                    s, t = path[i], path[i + 1]
                    bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])
            continue

        # Round-robin partitions over the cheapest distinct paths so load
        # is balanced across links and parallel networks.
        for j in range(bc_topology.num_partitions):
            _, path = scored[j % len(scored)]
            for i in range(len(path) - 1):
                s, t = path[i], path[i + 1]
                bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology
