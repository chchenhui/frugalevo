<DESCRIPTION>
The current algorithm routes every partition of every destination along the single cheapest Dijkstra path. This creates heavy congestion on shared links and ignores alternative low-cost routes. Improvement: for each (dst, partition) pair, pick from the k-shortest simple paths (by cost), cycling across them so partitions are spread over disjoint/diverse routes. This reduces redundant transfers on the same links, balances load across networks, and exploits multi-network topologies to lower total cost. Since cost likely depends on congestion/flow on edges, spreading partitions across parallel paths lowers per-edge flow and total cost.

def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    # Maintain a per-edge congestion count to guide path selection away
    # from already-loaded links (multi-network load balancing).
    edge_load = {}
    num_paths = max(2, num_partitions)

    for dst in dsts:
        # Generate multiple low-cost simple paths (Yen's algorithm on cost).
        try:
            paths = list(
                nx.shortest_simple_paths(h, src, dst, weight="cost")
            )
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            paths = []
        paths = paths[:num_paths]

        if not paths:
            continue

        for j in range(bc_topology.num_partitions):
            # Score each candidate path by base cost plus congestion penalty.
            best_path, best_score = None, None
            for cand in paths:
                score = 0.0
                for i in range(len(cand) - 1):
                    s, t = cand[i], cand[i + 1]
                    score += h[s][t]["cost"]
                    score += edge_load.get((s, t), 0) * h[s][t]["cost"]
                if best_score is None or score < best_score:
                    best_path, best_score = cand, score

            for i in range(len(best_path) - 1):
                s, t = best_path[i], best_path[i + 1]
                edge_load[(s, t)] = edge_load.get((s, t), 0) + 1
                bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology
