tm_steiner_tree_broadcast</NAME>
<DESCRIPTION>
Replace the naive per-destination Dijkstra broadcast with a Takahashi-Matsuyama-style Steiner tree heuristic. We grow a shared tree from the source: for each remaining destination we run multi-source Dijkstra from all current tree nodes to find the cheapest attachment point, attach the cheapest destination, and record only the *new* edges (dedup via seen_edges) for each partition. This eliminates redundant transfers over edges shared by multiple destinations, since each shared edge is only paid once per partition rather than once per destination. All edges of the tree are still appended for each partition (required so every partition can reach every destination), but overlapping path segments across destinations are deduplicated per partition.
</DESCRIPTION>

def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    # Takahashi-Matsuyama Steiner tree heuristic:
    # grow a shared tree from src, attaching destinations one at a time
    # via cheapest connection to any existing tree node.
    tree_nodes = {src}
    tree_parent = {}  # node -> parent node in tree
    remaining = [d for d in dsts if d != src]

    while remaining:
        best_dst = None
        best_cost = float("inf")
        best_dist = None
        best_paths = None
        for d in remaining:
            try:
                dist, paths = nx.multi_source_dijkstra(h, tree_nodes, d, weight="cost")
            except (nx.NetworkXNoPath, nx.NodeNotFound):
                continue
            if dist < best_cost:
                best_cost = dist
                best_dst = d
                best_paths = paths
        if best_dst is None:
            # unreachable destinations: fall back to any simple path attempt
            remaining = [d for d in remaining if d not in tree_nodes]
            break

        # Walk back from best_dst to the tree, recording parents
        node = best_dst
        while node not in tree_nodes:
            p = best_paths[node][-2]
            tree_parent[node] = p
            tree_nodes.add(node)
            node = p
        remaining.remove(best_dst)

    # Build full src->dst paths in the tree by following parents upward.
    for dst in dsts:
        if dst == src:
            continue
        if dst not in tree_nodes:
            continue
        # reconstruct path from dst up to src
        path = [dst]
        node = dst
        while node != src:
            node = tree_parent[node]
            path.append(node)
        path.reverse()
        for j in range(bc_topology.num_partitions):
            seen_edges = set()
            for i in range(0, len(path) - 1):
                s, t = path[i], path[i + 1]
                if (s, t) in seen_edges:
                    continue
                seen_edges.add((s, t))
                bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology
