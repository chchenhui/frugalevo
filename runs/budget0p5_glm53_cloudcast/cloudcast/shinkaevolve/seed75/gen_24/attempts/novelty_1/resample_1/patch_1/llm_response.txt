<DESCRIPTION>
The Steiner-tree multicast approach achieved the best cost (759.09) by deduplicating shared transfers, while the PathOracle/TreeCostModel approach contributed the idea of marginal-cost greedy assignment with a shared-edge accounting model. The crossover builds the Steiner tree once (as in the best program), then refines it with the shared-edge greedy: for each destination we consider the tree path plus up to k alternative shortest simple paths, and select the one with the lowest marginal cost given edges already committed (used edges are free). Destinations are processed in order of tree-path cost so the cheapest, most-shareable routes commit first. Each partition reuses the final per-destination routes, keeping shared edges transferred only once while allowing a destination to take a cheap alternate path when it beats its tree path. This preserves the exact BroadCastTopology API and output format.

# EVOLVE-BLOCK-START
import networkx as nx
import json
import os
import pandas as pd
from typing import Dict, List
from collections import defaultdict


def prepare_graph(G: nx.DiGraph, src: str) -> nx.DiGraph:
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    return h


# ---------------------------------------------------------------------------
# Stage 1: Steiner-style multicast tree construction (shared, built once)
# ---------------------------------------------------------------------------
def build_multicast_tree(h: nx.DiGraph, src: str, dsts: List[str]) -> Dict:
    rh = h.reverse(copy=True)
    tree_parent = {src: None}
    in_tree = {src}
    remaining = set(d for d in dsts if d in h)

    while remaining:
        best_dst, best_dist, best_path = None, float("inf"), None
        for d in remaining:
            if d not in rh:
                continue
            try:
                d_dist, d_paths = nx.single_source_dijkstra(rh, d, weight="cost")
            except (nx.NetworkXNoPath, nx.NodeNotFound):
                continue
            target, tdist = None, float("inf")
            for node in in_tree:
                if node in d_dist and d_dist[node] < tdist:
                    target, tdist = node, d_dist[node]
            if target is not None and tdist < best_dist:
                best_dst, best_dist, best_path = d, tdist, d_paths[target]

        if best_dst is None:
            d = remaining.pop()
            if nx.has_path(h, src, d):
                p = nx.dijkstra_path(h, src, d, weight="cost")
                for i in range(len(p) - 1):
                    if p[i + 1] not in in_tree and p[i + 1] != src:
                        tree_parent[p[i + 1]] = p[i]
                        in_tree.add(p[i + 1])
            continue

        remaining.discard(best_dst)
        for i in range(len(best_path) - 1):
            child, parent = best_path[i], best_path[i + 1]
            if child not in in_tree and child != src:
                tree_parent[child] = parent
                in_tree.add(child)
        in_tree.add(best_dst)
        if best_dst not in tree_parent and len(best_path) >= 2:
            tree_parent[best_dst] = best_path[1]

    return tree_parent


def dst_path_from_tree(tree_parent: Dict, src: str, dst: str):
    chain = []
    cur = dst
    guard = 0
    while cur != src and guard <= len(tree_parent) + 1:
        par = tree_parent.get(cur)
        if par is None:
            return None
        chain.append((par, cur))
        cur = par
        guard += 1
    if cur != src:
        return None
    chain.reverse()
    return chain


# ---------------------------------------------------------------------------
# Stage 2: candidate generation (from PathOracle approach, k alternatives)
# ---------------------------------------------------------------------------
def candidate_paths(h: nx.DiGraph, src: str, dst: str, k: int = 3) -> List[List[str]]:
    try:
        gen = nx.shortest_simple_paths(h, src, dst, weight="cost")
        paths = []
        for i, p in enumerate(gen):
            if i >= k:
                break
            paths.append(p)
        return paths
    except (nx.NetworkXNoPath, nx.NodeNotFound):
        return []


def path_to_edges(p: List[str]) -> List[tuple]:
    return [(p[i], p[i + 1]) for i in range(len(p) - 1)]


# ---------------------------------------------------------------------------
# Stage 3: shared-edge greedy refinement with load-aware selection
# ---------------------------------------------------------------------------
def select_routes(h: nx.DiGraph, src: str, dsts: List[str], num_partitions: int,
                   k: int = 3) -> Dict[str, List[tuple]]:
    tree_parent = build_multicast_tree(h, src, dsts)

    def edge_cost(s, t):
        return h[s][t].get("cost") or 0.0

    routes: Dict[str, List[tuple]] = {}
    used_edges = set()
    edge_load = defaultdict(int)

    # tree paths as a baseline
    tree_paths = {}
    for dst in dsts:
        chain = dst_path_from_tree(tree_parent, src, dst)
        if chain is not None:
            tree_paths[dst] = chain

    # process destinations with a known tree route first, cheapest first
    # (maximizes sharing), then unreachable-from-tree ones as fallback
    ordered = []
    for dst in sorted(tree_paths, key=lambda d: sum(edge_cost(s, t) for s, t in tree_paths[d])):
        ordered.append(dst)
    ordered.extend(d for d in dsts if d not in tree_paths)

    for dst in ordered:
        cands = []
        if dst in tree_paths:
            cands.append(tree_paths[dst])
        tp = path_to_edges  # alias
        for p in candidate_paths(h, src, dst, k=k):
            e = tp(p)
            if e and e not in cands:
                cands.append(e)
        if not cands:
            routes[dst] = []
            continue

        def marginal(edges):
            return sum(0.0 if e in used_edges else edge_cost(*e) for e in edges)

        # tie-break on lower total load to balance networks
        def load(edges):
            return sum(edge_load[e] for e in edges)

        scored = sorted(cands, key=lambda e: (marginal(e), load(e)))
        chosen = scored[0]
        for e in chosen:
            used_edges.add(e)
            edge_load[e] += num_partitions
        routes[dst] = chosen

    return routes


# ---------------------------------------------------------------------------
# Stage 4: topology writer (same API/output as original)
# ---------------------------------------------------------------------------
def search_algorithm(src, dsts, G, num_partitions):
    h = prepare_graph(G, src)
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    routes = select_routes(h, src, dsts, num_partitions)

    for dst in dsts:
        chain = routes.get(dst)
        if not chain:
            # fallback: direct shortest path (guarded)
            try:
                if nx.has_path(h, src, dst):
                    p = nx.dijkstra_path(h, src, dst, weight="cost")
                    chain = [(p[i], p[i + 1]) for i in range(len(p) - 1)]
                else:
                    continue
            except (nx.NetworkXNoPath, nx.NodeNotFound):
                continue
        for j in range(num_partitions):
            for s, t in chain:
                bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology


class SingleDstPath(Dict):
    partition: int
    edges: List[List]  # [[src, dst, edge data]]


class BroadCastTopology:
    def __init__(self, src: str, dsts: List[str], num_partitions: int = 4, paths: Dict[str, SingleDstPath] = None):
        self.src = src  # single str
        self.dsts = dsts  # list of strs
        self.num_partitions = num_partitions

        # dict(dst) --> dict(partition) --> list(nx.edges)
        if paths is not None:
            self.paths = paths
        else:
            self.paths = {dst: {str(i): None for i in range(num_partitions)} for dst in dsts}

    def get_paths(self):
        print(f"now the set path is: {self.paths}")
        return self.paths

    def set_num_partitions(self, num_partitions: int):
        self.num_partitions = num_partitions

    def set_dst_partition_paths(self, dst: str, partition: int, paths: List[List]):
        partition = str(partition)
        self.paths[dst][partition] = paths

    def append_dst_partition_path(self, dst: str, partition: int, path: List):
        partition = str(partition)
        if self.paths[dst][partition] is None:
            self.paths[dst][partition] = []
        self.paths[dst][partition].append(path)


def make_nx_graph(cost_path=None, throughput_path=None, num_vms=1):
    """
    Default graph with capacity constraints and cost info
    nodes: regions, edges: links
    per edge:
        throughput: max tput achievable (gbps)
        cost: $/GB
        flow: actual flow (gbps), must be < throughput, default = 0
    """
    current_dir = os.path.dirname(os.path.abspath(__file__))

    if cost_path is None:
        cost = pd.read_csv(os.path.join(current_dir, "profiles/cost.csv"))
    else:
        cost = pd.read_csv(cost_path)

    if throughput_path is None:
        throughput = pd.read_csv(os.path.join(current_dir, "profiles/throughput.csv"))
    else:
        throughput = pd.read_csv(throughput_path)

    G = nx.DiGraph()
    for _, row in throughput.iterrows():
        if row["src_region"] == row["dst_region"]:
            continue
        G.add_edge(row["src_region"], row["dst_region"], cost=None, throughput=num_vms * row["throughput_sent"] / 1e9)

    for _, row in cost.iterrows():
        if row["src"] in G and row["dest"] in G[row["src"]]:
            G[row["src"]][row["dest"]]["cost"] = row["cost"]

    no_cost_pairs = []
    for edge in G.edges.data():
        src, dst = edge[0], edge[1]
        if edge[-1]["cost"] is None:
            no_cost_pairs.append((src, dst))
    print("Unable to get costs for: ", no_cost_pairs)

    return G

# Helper functions that won't be evolved
def create_broadcast_topology(src: str, dsts: List[str], num_partitions: int = 4):
    """Create a broadcast topology instance"""
    return BroadCastTopology(src, dsts, num_partitions)

def run_search_algorithm(src: str, dsts: List[str], G, num_partitions: int):
    """Run the search algorithm and return the topology"""
    return search_algorithm(src, dsts, G, num_partitions)
