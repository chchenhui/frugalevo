<DESCRIPTION>
The current program uses a single Dijkstra shortest path per destination, repeated for all partitions. This can be suboptimal when another destination's cheapest path is slightly more expensive but avoids a shared trunk that the first path already saturates, or when parallel paths allow cheaper per-partition routing. I propose enumerating a bounded set of k-shortest simple paths (k=4) per destination using a generator (never materializing all paths), then greedily assigning each partition to the cheapest candidate path under a congestion penalty that accumulates as partitions are assigned. This reduces redundant transfers by spreading partitions across parallel paths, balances load across networks, and exploits multi-network topologies to minimize total cost, while staying within the Gen 9 safety bound (4-8 candidates preserves 100% success).

def search_algorithm(src, dsts, G, num_partitions):
    # Parameters
    K_PATHS = 4              # bounded number of candidate paths per destination
    CONGESTION_WEIGHT = 0.5  # penalty per concurrent load on an edge

    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    def edge_cost(s, t):
        base = h[s][t].get("cost")
        if base is None:
            base = 1e9
        return base

    def path_cost(path, edge_load):
        c = 0.0
        for i in range(len(path) - 1):
            c += edge_cost(path[i], path[i + 1])
            c += CONGESTION_WEIGHT * edge_load.get((path[i], path[i + 1]), 0)
        return c

    for dst in dsts:
        if dst == src:
            continue
        # Bounded k-shortest simple paths via generator (never materialize all)
        candidates = []
        try:
            count = 0
            for path in nx.shortest_simple_paths(h, src, dst, weight="cost"):
                candidates.append(path)
                count += 1
                if count >= K_PATHS:
                    break
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            candidates = []

        if not candidates:
            # Fallback: single Dijkstra path
            try:
                candidates = [nx.dijkstra_path(h, src, dst, weight="cost")]
            except (nx.NetworkXNoPath, nx.NodeNotFound):
                continue

        # Track congestion for this destination's partitions so
        # successive partitions get spread across parallel paths.
        edge_load = {}
        for j in range(num_partitions):
            best_c, best_p = None, None
            for p in candidates:
                c = path_cost(p, edge_load)
                if best_c is None or c < best_c:
                    best_c, best_p = c, p
            for i in range(len(best_p) - 1):
                key = (best_p[i], best_p[i + 1])
                edge_load[key] = edge_load.get(key, 0) + 1
                s, t = best_p[i], best_p[i + 1]
                bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology
