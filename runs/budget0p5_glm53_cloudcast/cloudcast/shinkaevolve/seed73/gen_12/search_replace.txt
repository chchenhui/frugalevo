<DESCRIPTION>
The current algorithm routes every (dst, partition) pair independently along the shortest path, ignoring that broadcast traffic to multiple destinations can share a multicast tree. Since all partitions must reach every destination, sending them along a shared Steiner tree means each edge carries each partition only once, instead of duplicating traffic on overlapping shortest paths to different destinations. I implement the classic distance-network (metric closure + MST) 2-approximation for the Steiner tree over terminals = {src} ∪ dsts, then extract the unique tree path from src to each dst. Edges shared by multiple destinations are traversed only once per partition, eliminating redundant transfers and lowering total cost substantially.

def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    # Steiner tree approximation (distance-network heuristic):
    # terminals = src + dsts. Build metric closure over terminals using
    # shortest-path distances, take the MST, then union the expanded
    # shortest paths to form a shared multicast tree. Shared edges carry
    # each partition only once, avoiding redundant transfers.
    terminals = [src] + list(dsts)
    dist = {}
    pred = {}
    for t in terminals:
        d, p = nx.dijkstra_path(h, t, weight="cost", return_predecessors=True) \
            if False else nx.single_source_dijkstra(h, t, weight="cost")
        dist[t] = d
        pred[t] = p

    metric = nx.Graph()
    for a in terminals:
        for b in terminals:
            if a != b and b in dist[a] and dist[a][b] < float("inf"):
                metric.add_edge(a, b, weight=dist[a][b])
    mst = nx.minimum_spanning_tree(metric, weight="weight")

    # Union of shortest paths induced by MST edges
    tree = nx.DiGraph()
    for a, b in mst.edges():
        # recover path from a to b using dijkstra on h
        path = nx.dijkstra_path(h, a, b, weight="cost")
        for i in range(len(path) - 1):
            s, t = path[i], path[i + 1]
            if tree.has_edge(s, t):
                if h[s][t]["cost"] < tree[s][t]["cost"]:
                    tree.remove_edge(s, t)
                    tree.add_edge(s, t, cost=h[s][t]["cost"], data=G[s][t])
            else:
                tree.add_edge(s, t, cost=h[s][t]["cost"], data=G[s][t])
            # also add reverse direction so the tree stays connected
            if not tree.has_edge(t, s):
                if h.has_edge(t, s):
                    tree.add_edge(t, s, cost=h[t][s]["cost"], data=G[t][s])
                else:
                    tree.add_edge(t, s, cost=float("inf"), data=G[s][t])

    # Direct the tree away from src: BFS from src, each dst reached by
    # its unique tree path.
    for dst in dsts:
        try:
            path = nx.shortest_path(tree, src, dst, weight="cost")
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            path = nx.dijkstra_path(h, src, dst, weight="cost")
        for i in range(0, len(path) - 1):
            s, t = path[i], path[i + 1]
            edge_data = G[s][t] if G.has_edge(s, t) else tree[s][t]["data"]
            for j in range(bc_topology.num_partitions):
                bc_topology.append_dst_partition_path(dst, j, [s, t, edge_data])

    return bc_topology
