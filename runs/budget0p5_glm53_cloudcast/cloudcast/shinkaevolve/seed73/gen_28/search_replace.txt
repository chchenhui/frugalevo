Implement the untested Gen 15 idea correctly: for each destination, evaluate up to 6 bounded simple path candidates and score them with a discount for edges already used by previously routed destinations (trunk consolidation). Select the candidate with the best discounted score, but only adopt it if its true (undiscounted) cost is ≤ the Dijkstra baseline cost for that destination — guaranteeing we are never worse than the current 1035.14 baseline, while potentially reducing cost when a slightly longer path shares cheap trunk edges with earlier destinations. The discount alpha is set to 0.3 within the recommended range.

def search_algorithm(src, dsts, G, num_partitions):
    router = Router(G, src, dsts)
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    def true_cost(path):
        total = 0.0
        for i in range(len(path) - 1):
            c = G[path[i]][path[i + 1]].get("cost")
            if c is None:
                c = DEFAULT_FALLBACK_COST
            total += c
        return total

    used_edges = set()
    alpha = 0.3

    for dst in dsts:
        # baseline: plain Dijkstra on true cost
        try:
            baseline_path = router.builder.shortest_path(router.h, src, dst)
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            continue
        baseline_cost = true_cost(baseline_path)

        # gather up to 6 bounded candidate simple paths
        candidates = []
        try:
            gen = nx.shortest_simple_paths(router.h, src, dst,
                                           weight=router.builder._weight)
            for i, p in enumerate(gen):
                candidates.append(p)
                if i + 1 >= 6:
                    break
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            candidates = []
        if baseline_path not in candidates:
            candidates.append(baseline_path)

        # score with trunk-reuse discount; pick cheapest discounted score
        best_path = None
        best_score = None
        for p in candidates:
            score = true_cost(p)
            for i in range(len(p) - 1):
                if (p[i], p[i + 1]) in used_edges:
                    score -= alpha * (G[p[i]][p[i + 1]].get("cost") or 0.0)
            if best_score is None or score < best_score:
                best_score = score
                best_path = p

        # adopt only if true cost is never worse than baseline
        if best_path is not None and true_cost(best_path) <= baseline_cost:
            chosen = best_path
        else:
            chosen = baseline_path

        for i in range(len(chosen) - 1):
            used_edges.add((chosen[i], chosen[i + 1]))

        for i in range(len(chosen) - 1):
            s, t = chosen[i], chosen[i + 1]
            edge_data = G[s][t]
            for j in range(num_partitions):
                bc_topology.append_dst_partition_path(dst, j, [s, t, edge_data])

    return bc_topology
