shared_edge_reuse_multicast

<DESCRIPTION>
Treat the broadcast as a multicast (Steiner-tree-like) problem rather than independent per-destination shortest paths. Total cost depends on the union of edges used, so if a later destination can reach an already-delivered intermediate node cheaply and then reuse an already-paid edge prefix, the overall cost drops. Concretely: route destinations sequentially; after each destination is routed, mark its edges as "paid". For subsequent destinations, run Dijkstra on a reduced-cost graph where paid edges cost ~0 marginal (small epsilon to prefer shorter reuse), so routing strongly prefers reusing existing broadcast edges. This creates a shared multicast tree instead of parallel per-destination paths, reducing redundant transfers and total cost.

    def __init__(self, src, dsts, G, num_partitions):
        self.src = src
        self.dsts = dsts
        self.G = G
        self.num_partitions = num_partitions

    def plan(self):
        topo = BroadCastTopology(self.src, self.dsts, self.num_partitions)
        router = CandidateRouter(self.G)
        router.prepare(self.src)

        # Multicast-style routing: once an edge is used by any destination,
        # its marginal cost is (near) zero for later destinations, so they
        # reuse the existing broadcast tree instead of paying for parallel
        # redundant transfers.
        paid_edges = set()
        # Tiny epsilon > 0 so Dijkstra still prefers shorter reuse chains.
        REUSE_EPSILON = 1e-9

        # Reduced-cost view of the graph for subsequent destinations.
        reduced = router.h.copy()
        for u, v in reduced.edges():
            reduced[u][v]["cost"] = _edge_cost(self.G, u, v)

        for dst in self.dsts:
            if dst == self.src:
                continue
            try:
                path = router.route(dst)
            except (nx.NetworkXNoPath, nx.NodeNotFound):
                continue
            for i in range(len(path) - 1):
                s, t = path[i], path[i + 1]
                for j in range(self.num_partitions):
                    topo.append_dst_partition_path(dst, j, [s, t, self.G[s][t]])
                if (s, t) not in paid_edges:
                    paid_edges.add((s, t))
                    reduced[s][t]["cost"] = REUSE_EPSILON
            # Re-prepare the router so future routes use reduced costs
            # while candidate validation still uses true edge costs.
            router._cache = {}
            router.baseline_fn = lambda d, r=reduced: nx.dijkstra_path(r, router.src, d, weight="cost")
            router.route_fn_dst = dst

        return topo
