<DESCRIPTION>
The current algorithm computes a shortest path independently for each destination, ignoring that many destinations may share the same low-cost egress links. This causes redundant expensive transfers on parallel/identical routes. A better strategy: process destinations in order, and once an edge has been used by a previous destination's path, discount its effective cost (near zero) for subsequent route searches. This encourages later destinations to reuse already-paid-for links, producing a Steiner-tree-like broadcast structure that substantially reduces total transfer cost, while still guaranteeing each destination remains reachable. We use per-destination copies of the graph weights so the original graph G is untouched, and all num_partitions per destination are still populated exactly as before so the topology format stays compatible.

def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    # ensure every edge has a finite positive cost for pathfinding
    for s, t, data in h.edges(data=True):
        if data.get("cost") is None:
            h[s][t]["cost"] = 1e9
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    # Steiner-tree-like strategy: once an edge is used by one destination,
    # discount it for subsequent destinations so shared (already paid for)
    # links are reused instead of paying for new parallel transfers.
    used_edges = set()
    # process destinations in order of original shortest-path cost so the
    # most expensive destinations lock in their cheapest routes first
    dst_order = sorted(
        dsts,
        key=lambda d: nx.shortest_path_length(h, src, d, weight="cost"),
    )

    for dst in dst_order:
        # build a temp graph with discounted weights on already-used edges
        w = {}
        for s, t, data in h.edges(data=True):
            w[(s, t)] = 0.001 if (s, t) in used_edges else data["cost"]
        h_discounted = h.copy()
        for (s, t), weight in w.items():
            h_discounted[s][t]["cost"] = weight

        path = nx.dijkstra_path(h_discounted, src, dst, weight="cost")
        for i in range(0, len(path) - 1):
            s, t = path[i], path[i + 1]
            used_edges.add((s, t))
            for j in range(bc_topology.num_partitions):
                bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology
