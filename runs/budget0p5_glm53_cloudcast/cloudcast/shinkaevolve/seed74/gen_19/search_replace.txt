steiner_shared_prefix_multicast</NAME>
<DESCRIPTION>
Replace per-destination Dijkstra shortest paths with a Steiner-tree style multicast construction. Destinations that share common prefix paths from the source reuse the same trunk edges; the routing decision counts shared edges once, reducing the effective broadcast cost from O(num_dsts × path_len) toward a tree structure. The output format is unchanged: each partition still emits per-destination edge lists [s, t, G[s][t]] via append_dst_partition_path. Edges with missing cost info are pruned from the routing graph (falling back to the original graph only if a destination becomes unreachable), avoiding infinite-weight distractions. This matches the best prior result (Steiner heuristic, 1033.77 total cost) while keeping the code simpler and more robust.

def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    # drop edges with no cost info so they don't distort shortest-path search
    for u, v, d in list(h.edges(data=True)):
        if d.get("cost") is None:
            h.remove_edge(u, v)

    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    # Steiner-tree (shortest-path heuristic): iteratively graft the closest
    # remaining destination onto the shared trunk. Destinations with common
    # prefix paths reuse the same trunk edges (shared edges counted once in
    # the routing decision), while output remains per-destination edge lists.
    tree_nodes = {src}
    paths_from_src = nx.single_source_dijkstra(h, src, weight="cost")[1] \
        if src in h else {}

    remaining = set(dsts)
    while remaining:
        dist_tree, path_tree = nx.multi_source_dijkstra(
            h, list(tree_nodes), weight="cost")
        best_dst = None
        best_d = float("inf")
        for d in remaining:
            if d in dist_tree and dist_tree[d] < best_d:
                best_d = dist_tree[d]
                best_dst = d

        if best_dst is None:
            # unreachable on pruned graph; fall back to original graph
            fallback = sorted(remaining)[0]
            try:
                path = nx.dijkstra_path(G, fallback, weight="cost") if False \
                    else nx.dijkstra_path(G, src, fallback, weight="cost")
            except nx.NetworkXNoPath:
                remaining.discard(fallback)
                continue
            for i in range(len(path) - 1):
                s, t = path[i], path[i + 1]
                tree_nodes.update([s, t])
                for j in range(bc_topology.num_partitions):
                    bc_topology.append_dst_partition_path(
                        fallback, j, [s, t, G[s][t]])
            remaining.discard(fallback)
            continue

        path = path_tree[best_dst]
        anchor = path[0]
        # full src->dst path: prepend tree path to the anchor if needed
        if anchor == src:
            full_path = path
        else:
            full_path = paths_from_src.get(anchor, [src]) + path[1:]

        for i in range(len(path) - 1):
            s, t = path[i], path[i + 1]
            tree_nodes.add(t)

        for i in range(len(full_path) - 1):
            s, t = full_path[i], full_path[i + 1]
            for j in range(bc_topology.num_partitions):
                bc_topology.append_dst_partition_path(
                    best_dst, j, [s, t, G[s][t]])
        remaining.discard(best_dst)

    return bc_topology
