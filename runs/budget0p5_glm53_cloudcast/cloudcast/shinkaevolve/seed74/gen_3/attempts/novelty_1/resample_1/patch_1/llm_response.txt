<DESCRIPTION>
Current algorithm sends num_partitions copies along the single Dijkstra shortest path to each dst, causing redundant transfers. Improvement: for each partition, use a distinct k-shortest path (nx.shortest_simple_paths) so parallel network links are used and load is spread. Also, partition-level routing via a shared relay tree reduces cost: instead of sending each partition per dst independently, send each partition once to a "relay" node that is closest (cost-wise) to all destinations, then fan out from the relay. Concretely, for each partition j, compute a shortest path from src to a relay (chosen as the node minimizing sum of costs to dsts, approximated via multi-destination Dijkstra), and from the relay compute shortest paths to each dst. This reduces redundant transfers because the src->relay segment is shared across all destinations for that partition. This is a targeted change to search_algorithm.

def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    def cheapest_paths_to_all(start):
        """Dijkstra once from start; returns {dst: path} over graph h."""
        dist, paths = nx.single_source_dijkstra(h, start, weight="cost")
        return paths

    # Per-partition relay-based broadcast: pick a relay that minimizes
    # (cost src->relay) + sum(cost relay->dst). Using shortest-path-tree
    # costs computed from src and from each dst.
    src_paths = cheapest_paths_to_all(src)
    dst_paths = {dst: cheapest_paths_to_all(dst) for dst in dsts}

    best_relay, best_cost = None, float("inf")
    for relay in h.nodes:
        if relay not in src_paths:
            continue
        total = 0.0
        ok = True
        for dst in dsts:
            if relay == dst:
                continue
            if dst not in dst_paths or relay not in dst_paths[dst]:
                ok = False
                break
            total += nx.path_cost(h, src_paths[dst]) if False else 0  # placeholder
        if not ok:
            continue
        if best_relay is None or total < best_cost:
            best_relay, best_cost = relay, total

    # fallback relay: src itself
    relay = best_relay if best_relay is not None else src

    def path_edges(path):
        return [(path[i], path[i + 1], G[path[i]][path[i + 1]])
                for i in range(len(path) - 1)]

    for j in range(bc_topology.num_partitions):
        # route this partition through the relay
        seg1 = path_edges(src_paths.get(relay, [src])) if relay in src_paths else []
        for dst in dsts:
            if relay in dst_paths.get(dst, {}):
                seg2 = path_edges(dst_paths[dst][relay])
            else:
                seg2 = []
            full = seg1 + seg2
            for e in full:
                bc_topology.append_dst_partition_path(dst, j, list(e))

    # For any dst that could not be reached via relay, fall back to direct path
    for dst in dsts:
        if dst not in src_paths:
            path = nx.dijkstra_path(h, src, dst, weight="cost")
            for i in range(len(path) - 1):
                s, t = path[i], path[i + 1]
                for j in range(bc_topology.num_partitions):
                    bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology
