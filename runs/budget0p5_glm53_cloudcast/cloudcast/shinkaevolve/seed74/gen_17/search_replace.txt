def search_algorithm(src, dsts, G, num_partitions):
    h = G.copy()
    h.remove_edges_from(list(h.in_edges(src)) + list(nx.selfloop_edges(h)))
    bc_topology = BroadCastTopology(src, dsts, num_partitions)

    # Cost-first strategy: pick the single cheapest path per dst
    # (via k-shortest-path enumeration with Dijkstra fallback) and
    # assign ALL partitions to it. Spreading across candidate paths
    # was shown to increase total cost monotonically, so candidate
    # generation is kept only as a robustness fallback.
    max_candidates = min(4, num_partitions)

    def path_cost(path):
        return sum((h[path[i]][path[i + 1]].get("cost") or 0.0)
                   for i in range(len(path) - 1))

    for dst in dsts:
        best_path = None
        try:
            candidates = []
            gen = nx.shortest_simple_paths(h, src, dst, weight="cost")
            for p in gen:
                candidates.append(p)
                if len(candidates) >= max_candidates:
                    break
            if candidates:
                best_path = min(candidates, key=path_cost)
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            best_path = None

        if best_path is None:
            try:
                best_path = nx.dijkstra_path(h, src, dst, weight="cost")
            except nx.NetworkXNoPath:
                continue

        for i in range(len(best_path) - 1):
            s, t = best_path[i], best_path[i + 1]
            for j in range(bc_topology.num_partitions):
                bc_topology.append_dst_partition_path(dst, j, [s, t, G[s][t]])

    return bc_topology
