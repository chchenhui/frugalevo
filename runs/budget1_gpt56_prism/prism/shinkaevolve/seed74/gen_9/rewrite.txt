GPU_MEM_SIZE = 80  # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """
    Place models on GPUs while minimizing the maximum KV cache pressure.

    The placement is built in two phases:
      1. Binary-search a transformed KVPR capacity and construct a feasible
         bin-packing assignment.
      2. Locally rebalance the resulting assignment using memory-safe moves
         and swaps that improve the GPU pressure distribution.
    """
    EPS = 1e-10

    if gpu_num <= 0:
        raise ValueError("gpu_num must be positive")

    if not models:
        return {gpu: [] for gpu in range(gpu_num)}

    sizes = [float(model.model_size) for model in models]
    weights = [float(model.req_rate) / float(model.slo) for model in models]
    item_count = len(models)

    if any(size > GPU_MEM_SIZE + EPS for size in sizes):
        bad_size = max(sizes)
        raise ValueError(
            f"Unable to place model of size {bad_size} GB: "
            f"it exceeds GPU memory of {GPU_MEM_SIZE} GB."
        )

    def build_placement(assignment):
        placement = {gpu: [] for gpu in range(gpu_num)}
        for item, gpu in enumerate(assignment):
            placement[gpu].append(models[item])
        return placement

    def memory_assignment():
        """Best-fit decreasing assignment used as a reliable fallback."""
        remaining = [float(GPU_MEM_SIZE)] * gpu_num
        assignment = [-1] * item_count

        order = sorted(
            range(item_count),
            key=lambda i: (-sizes[i], -weights[i])
        )

        for item in order:
            choices = [
                gpu for gpu in range(gpu_num)
                if remaining[gpu] + EPS >= sizes[item]
            ]
            if not choices:
                return None

            gpu = min(
                choices,
                key=lambda g: (remaining[g] - sizes[item], g)
            )
            assignment[item] = gpu
            remaining[gpu] -= sizes[item]

        return assignment

    def transformed_assignment(pressure):
        """
        Pack items using the equivalent transformed capacity condition:

            weight + pressure * size <= pressure * GPU_MEM_SIZE
        """
        capacity = pressure * GPU_MEM_SIZE
        costs = [
            weights[item] + pressure * sizes[item]
            for item in range(item_count)
        ]

        if any(cost > capacity + EPS for cost in costs):
            return None

        orders = (
            sorted(
                range(item_count),
                key=lambda i: (-costs[i], -sizes[i], -weights[i])
            ),
            sorted(
                range(item_count),
                key=lambda i: (-costs[i], -weights[i], -sizes[i])
            ),
            sorted(
                range(item_count),
                key=lambda i: (-sizes[i], -costs[i], -weights[i])
            ),
            sorted(
                range(item_count),
                key=lambda i: (-weights[i], -costs[i], -sizes[i])
            ),
        )

        for order in orders:
            remaining = [capacity] * gpu_num
            assignment = [-1] * item_count
            feasible = True

            for item in order:
                candidates = [
                    gpu for gpu in range(gpu_num)
                    if remaining[gpu] + EPS >= costs[item]
                ]

                if not candidates:
                    feasible = False
                    break

                # Best-fit leaves the smallest transformed residual capacity.
                gpu = min(
                    candidates,
                    key=lambda g: (
                        remaining[g] - costs[item],
                        g
                    )
                )

                assignment[item] = gpu
                remaining[gpu] -= costs[item]

            if feasible:
                return assignment

        return None

    def pressure_vector(used, load):
        """Return a descending pressure vector for lexicographic comparison."""
        values = []

        for gpu in range(gpu_num):
            free_memory = GPU_MEM_SIZE - used[gpu]

            if free_memory <= EPS:
                value = float("inf") if load[gpu] > EPS else 0.0
            else:
                value = load[gpu] / free_memory

            values.append(value)

        return tuple(sorted(values, reverse=True))

    def rebalance(assignment):
        """
        Improve an assignment with pressure-reducing moves and swaps.

        A lexicographic pressure vector is used so equal maximum-pressure
        placements can still improve their next-most-loaded GPUs.
        """
        members = [[] for _ in range(gpu_num)]
        used = [0.0] * gpu_num
        load = [0.0] * gpu_num

        for item, gpu in enumerate(assignment):
            members[gpu].append(item)
            used[gpu] += sizes[item]
            load[gpu] += weights[item]

        current_score = pressure_vector(used, load)
        max_rounds = min(80, max(12, item_count * 3))

        for _ in range(max_rounds):
            current_max = current_score[0]
            overloaded = [
                gpu for gpu in range(gpu_num)
                if abs(
                    (float("inf") if GPU_MEM_SIZE - used[gpu] <= EPS and load[gpu] > EPS
                     else (load[gpu] / max(EPS, GPU_MEM_SIZE - used[gpu])))
                    - current_max
                ) <= 1e-9
                or (
                    current_max != float("inf")
                    and load[gpu] / max(EPS, GPU_MEM_SIZE - used[gpu])
                    >= current_max - 1e-10
                )
            ]

            best_score = current_score
            best_action = None

            for source in overloaded:
                for item in list(members[source]):
                    size = sizes[item]
                    weight = weights[item]

                    # Try moving an item away from a most-loaded GPU.
                    for target in range(gpu_num):
                        if target == source:
                            continue
                        if used[target] + size > GPU_MEM_SIZE + EPS:
                            continue

                        test_used = used[:]
                        test_load = load[:]
                        test_used[source] -= size
                        test_load[source] -= weight
                        test_used[target] += size
                        test_load[target] += weight

                        score = pressure_vector(test_used, test_load)
                        if score < best_score:
                            best_score = score
                            best_action = ("move", item, source, target)

                    # Try swapping the item with an item on another GPU.
                    for target in range(gpu_num):
                        if target == source:
                            continue

                        for other in members[target]:
                            new_source_used = used[source] - size + sizes[other]
                            new_target_used = used[target] - sizes[other] + size

                            if (
                                new_source_used > GPU_MEM_SIZE + EPS
                                or new_target_used > GPU_MEM_SIZE + EPS
                            ):
                                continue

                            test_used = used[:]
                            test_load = load[:]

                            test_used[source] = new_source_used
                            test_used[target] = new_target_used
                            test_load[source] = (
                                load[source] - weight + weights[other]
                            )
                            test_load[target] = (
                                load[target] - weights[other] + weight
                            )

                            score = pressure_vector(test_used, test_load)
                            if score < best_score:
                                best_score = score
                                best_action = (
                                    "swap",
                                    item,
                                    other,
                                    source,
                                    target,
                                )

            if best_action is None:
                break

            if best_action[0] == "move":
                _, item, source, target = best_action

                members[source].remove(item)
                members[target].append(item)
                assignment[item] = target

                used[source] -= sizes[item]
                load[source] -= weights[item]
                used[target] += sizes[item]
                load[target] += weights[item]

            else:
                _, item, other, source, target = best_action

                members[source].remove(item)
                members[target].remove(other)
                members[source].append(other)
                members[target].append(item)

                assignment[item] = target
                assignment[other] = source

                used[source] += sizes[other] - sizes[item]
                used[target] += sizes[item] - sizes[other]
                load[source] += weights[other] - weights[item]
                load[target] += weights[item] - weights[other]

            current_score = best_score

        return assignment

    total_size = sum(sizes)
    total_weight = sum(weights)

    if total_size > gpu_num * GPU_MEM_SIZE + EPS:
        raise ValueError("Unable to place all models on available GPUs.")

    # A model using an entire GPU with nonzero load has unavoidable infinite
    # pressure. Return a memory-feasible placement without searching infinity.
    if any(
        size >= GPU_MEM_SIZE - EPS and weight > EPS
        for size, weight in zip(sizes, weights)
    ):
        assignment = memory_assignment()
        if assignment is None:
            raise ValueError("Unable to place all models on available GPUs.")
        return build_placement(assignment)

    total_free = gpu_num * GPU_MEM_SIZE - total_size
    lower = 0.0

    if total_weight > EPS and total_free > EPS:
        lower = total_weight / total_free

    for size, weight in zip(sizes, weights):
        free_memory = GPU_MEM_SIZE - size
        if weight > EPS and free_memory > EPS:
            lower = max(lower, weight / free_memory)

    upper = max(1.0, lower * 1.25 + 1e-9)
    assignment = transformed_assignment(upper)

    while assignment is None:
        upper *= 2.0
        if upper > 1e15:
            assignment = memory_assignment()
            if assignment is None:
                raise ValueError("Unable to place all models on available GPUs.")
            break
        assignment = transformed_assignment(upper)

    left = lower
    right = upper
    best_assignment = assignment

    for _ in range(42):
        middle = (left + right) / 2.0
        candidate = transformed_assignment(middle)

        if candidate is None:
            left = middle
        else:
            right = middle
            best_assignment = candidate

    best_assignment = rebalance(best_assignment)
    return build_placement(best_assignment)

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float
    import numpy as np

    test_cases = generate_test_gpu_models()
    all_kvpr = []

    for gpu_num, gpu_models in test_cases:
        results = compute_model_placement(gpu_num, gpu_models)
        max_kvpr = calculate_kvcache_pressure(results)
        all_kvpr.append(safe_float(max_kvpr))

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr

    print(f"Max KVPR: {avg_kvpr:.3f}")
