GPU_MEM_SIZE = 80  # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """
    Place models while minimizing the maximum GPU KV cache pressure.

    For a candidate maximum pressure T:

        load / (GPU_MEM_SIZE - used_memory) <= T

    is equivalent to:

        load + T * used_memory <= T * GPU_MEM_SIZE.

    Thus every model has transformed packing cost:

        req_rate / slo + T * model_size

    and each GPU has capacity T * GPU_MEM_SIZE.  The algorithm binary-searches
    T and solves each candidate using several best-fit bin-packing variants.
    """
    EPS = 1e-10

    if gpu_num <= 0:
        raise ValueError("gpu_num must be positive")

    n = len(models)
    if n == 0:
        return {gpu_id: [] for gpu_id in range(gpu_num)}

    sizes = [float(m.model_size) for m in models]
    weights = [float(m.req_rate) / float(m.slo) for m in models]

    for size in sizes:
        if size > GPU_MEM_SIZE + EPS:
            raise ValueError(
                f"Unable to place model of size {size} GB: "
                f"it exceeds GPU memory of {GPU_MEM_SIZE} GB."
            )

    def memory_packing():
        """A reliable ordinary-memory placement used for degenerate cases."""
        remaining = [float(GPU_MEM_SIZE)] * gpu_num
        assignment = [-1] * n

        for item in sorted(range(n), key=lambda i: (-sizes[i], -weights[i])):
            candidates = [
                gpu for gpu in range(gpu_num)
                if remaining[gpu] + EPS >= sizes[item]
            ]
            if not candidates:
                return None

            # Best fit preserves room for later large models.
            gpu = min(candidates, key=lambda g: (remaining[g] - sizes[item], g))
            assignment[item] = gpu
            remaining[gpu] -= sizes[item]

        return assignment

    # A positive-weight model filling an entire GPU necessarily has infinite
    # KVPR. Return a feasible memory placement rather than searching infinity.
    if any(size >= GPU_MEM_SIZE - EPS and weight > EPS
           for size, weight in zip(sizes, weights)):
        assignment = memory_packing()
        if assignment is None:
            raise ValueError("Unable to place all models on available GPUs.")
        placement = {gpu: [] for gpu in range(gpu_num)}
        for item, gpu in enumerate(assignment):
            placement[gpu].append(models[item])
        return placement

    def try_pack(pressure):
        """
        Test transformed bin-packing feasibility at a pressure bound.
        Several orderings prevent one greedy ordering from dominating results.
        """
        capacity = pressure * GPU_MEM_SIZE
        costs = [weights[i] + pressure * sizes[i] for i in range(n)]

        if any(cost > capacity + EPS for cost in costs):
            return None

        orders = [
            sorted(range(n), key=lambda i: (-costs[i], -sizes[i], -weights[i])),
            sorted(range(n), key=lambda i: (-costs[i], -weights[i], -sizes[i])),
            sorted(range(n), key=lambda i: (-sizes[i], -costs[i])),
            sorted(range(n), key=lambda i: (-weights[i], -costs[i])),
        ]

        for order in orders:
            remaining = [capacity] * gpu_num
            assignment = [-1] * n
            feasible = True

            for item in order:
                fitting = [
                    gpu for gpu in range(gpu_num)
                    if remaining[gpu] + EPS >= costs[item]
                ]

                if not fitting:
                    feasible = False
                    break

                # Best-fit decreasing: use the tightest viable bin.
                gpu = min(
                    fitting,
                    key=lambda g: (remaining[g] - costs[item], g)
                )
                assignment[item] = gpu
                remaining[gpu] -= costs[item]

            if feasible:
                return assignment

        return None

    total_size = sum(sizes)
    total_weight = sum(weights)

    # Analytical lower bounds from global and single-GPU constraints.
    lower = 0.0
    total_free = gpu_num * GPU_MEM_SIZE - total_size
    if total_weight > EPS and total_free > EPS:
        lower = max(lower, total_weight / total_free)

    for size, weight in zip(sizes, weights):
        free = GPU_MEM_SIZE - size
        if weight > EPS and free > EPS:
            lower = max(lower, weight / free)

    # Locate a feasible upper bound.  A modest positive seed also handles
    # zero-rate workloads cleanly.
    upper = max(1.0, lower * 1.25 + 1e-9)
    best_assignment = try_pack(upper)

    while best_assignment is None:
        upper *= 2.0
        if upper > 1e15:
            fallback = memory_packing()
            if fallback is None:
                raise ValueError(
                    "Unable to place all models on available GPUs under "
                    "the memory constraints."
                )
            best_assignment = fallback
            break
        best_assignment = try_pack(upper)

    # Binary search the lowest transformed-packing pressure discovered.
    # Keeping the latest feasible assignment guarantees a valid output.
    left = lower
    right = upper

    for _ in range(42):
        middle = (left + right) / 2.0
        candidate = try_pack(middle)

        if candidate is not None:
            right = middle
            best_assignment = candidate
        else:
            left = middle

    placement = {gpu_id: [] for gpu_id in range(gpu_num)}
    for model_id, gpu_id in enumerate(best_assignment):
        placement[gpu_id].append(models[model_id])

    return placement

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float
    import numpy as np

    test_cases = generate_test_gpu_models()
    all_kvpr = []

    for i, (gpu_num, gpu_models) in enumerate(test_cases):
        results = compute_model_placement(gpu_num, gpu_models)
        max_kvpr = calculate_kvcache_pressure(results)
        all_kvpr.append(safe_float(max_kvpr))

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr

    print(f"Max KVPR: {avg_kvpr:.3f}")
