<DESCRIPTION>
The prior approach used four greedy orderings followed by up to 40 local relocation/swap iterations, but it directly optimized only the current greedy peak pressure. The new implementation treats a candidate maximum KVPR as a feasibility threshold. For threshold `T`, a GPU accepts a model only when both memory and `load <= T * remaining_memory` remain feasible. It uses six deterministic model orderings and three bin-selection modes, then binary-searches the lowest feasible threshold for 38 iterations.

A successful threshold packing immediately measures its actual maximum KVPR and tightens the binary-search upper bound to that measured value. This objective-aligned tightening avoids retaining slack from a feasible greedy assignment. A final 60-iteration relocation/swap improvement phase further improves the best threshold-generated placement.
</DESCRIPTION>

GPU_MEM_SIZE = 80  # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """
    Place models while minimizing the maximum KV cache pressure ratio:

        sum(req_rate / slo) / remaining_gpu_memory

    The algorithm uses threshold feasibility packing, adaptive upper-bound
    tightening, multiple deterministic construction orders, and local search.
    """
    EPS = 1e-10
    BINARY_SEARCH_STEPS = 38
    LOCAL_SEARCH_STEPS = 60

    if gpu_num <= 0:
        if models:
            raise ValueError("Unable to place models because gpu_num must be positive.")
        return {}

    def load_of(model):
        return model.req_rate / model.slo

    def pressure(load, remaining):
        if remaining <= EPS:
            return float("inf") if load > EPS else 0.0
        return load / remaining

    def score(loads, remaining):
        return max(pressure(loads[g], remaining[g]) for g in range(gpu_num))

    for model in models:
        if model.model_size > GPU_MEM_SIZE + EPS:
            raise ValueError(
                "Unable to place all models within GPU memory. "
                f"Each GPU has {GPU_MEM_SIZE} GB."
            )

    if sum(model.model_size for model in models) > gpu_num * GPU_MEM_SIZE + EPS:
        raise ValueError(
            "Unable to place all models within GPU memory. "
            f"Each GPU has {GPU_MEM_SIZE} GB."
        )

    # Different orders address load-dominated, memory-dominated, and mixed cases.
    orderings = [
        sorted(models, key=lambda m: (load_of(m), m.model_size), reverse=True),
        sorted(models, key=lambda m: (m.model_size, load_of(m)), reverse=True),
        sorted(
            models,
            key=lambda m: (
                load_of(m) / max(GPU_MEM_SIZE - m.model_size, EPS),
                m.model_size,
            ),
            reverse=True,
        ),
        sorted(models, key=lambda m: (load_of(m) * m.model_size, load_of(m)), reverse=True),
        sorted(
            models,
            key=lambda m: (
                load_of(m) / max(m.model_size, EPS),
                load_of(m),
                m.model_size,
            ),
            reverse=True,
        ),
        sorted(
            models,
            key=lambda m: (
                load_of(m) + m.model_size,
                load_of(m),
                m.model_size,
            ),
            reverse=True,
        ),
    ]

    def construct_unbounded(ordered, mode):
        """Create an initial memory-feasible placement without a KVPR limit."""
        placement = {g: [] for g in range(gpu_num)}
        remaining = [float(GPU_MEM_SIZE)] * gpu_num
        loads = [0.0] * gpu_num

        for model in ordered:
            size = model.model_size
            load = load_of(model)
            choice = None
            choice_key = None

            for g in range(gpu_num):
                if size > remaining[g] + EPS:
                    continue

                new_remaining = remaining[g] - size
                new_load = loads[g] + load
                new_pressure = pressure(new_load, new_remaining)

                if mode == 0:
                    key = (new_pressure, -new_remaining, g)
                elif mode == 1:
                    key = (new_remaining, new_pressure, g)
                else:
                    key = (new_pressure, new_remaining, g)

                if choice_key is None or key < choice_key:
                    choice_key = key
                    choice = g

            if choice is None:
                return None

            placement[choice].append(model)
            remaining[choice] -= size
            loads[choice] += load

        return placement, remaining, loads

    best = None
    best_score = float("inf")

    # Obtain a reliable feasible upper bound before threshold search.
    for order_index, ordered in enumerate(orderings):
        candidate = construct_unbounded(ordered, order_index % 3)
        if candidate is not None:
            candidate_score = score(candidate[2], candidate[1])
            if candidate_score < best_score:
                best = candidate
                best_score = candidate_score

    if best is None:
        raise ValueError(
            "Unable to place all models within GPU memory. "
            f"Each GPU has {GPU_MEM_SIZE} GB."
        )

    def threshold_pack(limit, ordered, mode):
        """
        Pack under a maximum KVPR limit. For a bin after adding a model:

            new_load / new_remaining <= limit

        is equivalent to:

            new_load <= limit * new_remaining.
        """
        placement = {g: [] for g in range(gpu_num)}
        remaining = [float(GPU_MEM_SIZE)] * gpu_num
        loads = [0.0] * gpu_num

        for model in ordered:
            size = model.model_size
            load = load_of(model)
            chosen_gpu = None
            chosen_key = None

            for g in range(gpu_num):
                if size > remaining[g] + EPS:
                    continue

                new_remaining = remaining[g] - size
                new_load = loads[g] + load
                slack = limit * new_remaining - new_load

                if slack < -EPS:
                    continue

                new_pressure = pressure(new_load, new_remaining)

                # Several deterministic policies reduce greedy packing bias.
                if mode == 0:
                    # Fill the tightest still-feasible pressure slot.
                    key = (max(0.0, slack), new_remaining, g)
                elif mode == 1:
                    # Balance pressure first, preserving memory where possible.
                    key = (new_pressure, max(0.0, slack), -new_remaining, g)
                else:
                    # Memory best-fit while retaining threshold feasibility.
                    key = (new_remaining, max(0.0, slack), new_pressure, g)

                if chosen_key is None or key < chosen_key:
                    chosen_key = key
                    chosen_gpu = g

            if chosen_gpu is None:
                return None

            placement[chosen_gpu].append(model)
            remaining[chosen_gpu] -= size
            loads[chosen_gpu] += load

        return placement, remaining, loads

    def find_threshold_candidate(limit):
        candidate_best = None
        candidate_score = float("inf")

        for order_index, ordered in enumerate(orderings):
            for mode in range(3):
                candidate = threshold_pack(limit, ordered, (order_index + mode) % 3)
                if candidate is None:
                    continue

                candidate_score_value = score(candidate[2], candidate[1])
                if candidate_score_value < candidate_score:
                    candidate_score = candidate_score_value
                    candidate_best = candidate

        return candidate_best, candidate_score

    # Binary search for the smallest threshold accepted by the packing heuristic.
    left = 0.0
    right = best_score

    for _ in range(BINARY_SEARCH_STEPS):
        if right - left <= EPS * max(1.0, right):
            break

        middle = (left + right) / 2.0
        candidate, actual_score = find_threshold_candidate(middle)

        if candidate is None:
            left = middle
            continue

        if actual_score < best_score - EPS:
            best = candidate
            best_score = actual_score

        # A feasible placement can have real pressure below the tested limit.
        # Tightening to that actual value is objective-aligned and removes slack.
        tightened = min(middle, actual_score)
        if tightened < right - EPS:
            right = tightened

            # Immediately try the measured bound before the next binary step.
            tighter_candidate, tighter_score = find_threshold_candidate(right)
            if tighter_candidate is not None:
                if tighter_score < best_score - EPS:
                    best = tighter_candidate
                    best_score = tighter_score
                right = min(right, tighter_score)
        else:
            right = middle

    def improve(placement, remaining, loads):
        """Best-improving feasible relocation and swap local search."""
        for _ in range(LOCAL_SEARCH_STEPS):
            current = score(loads, remaining)
            action = None
            action_score = current

            for source in range(gpu_num):
                for model in list(placement[source]):
                    size = model.model_size
                    load = load_of(model)

                    for target in range(gpu_num):
                        if source == target or size > remaining[target] + EPS:
                            continue

                        test_loads = list(loads)
                        test_remaining = list(remaining)
                        test_loads[source] -= load
                        test_remaining[source] += size
                        test_loads[target] += load
                        test_remaining[target] -= size
                        value = score(test_loads, test_remaining)

                        if value < action_score - EPS:
                            action_score = value
                            action = ("move", source, target, model)

            for left_gpu in range(gpu_num):
                for right_gpu in range(left_gpu + 1, gpu_num):
                    for left_model in list(placement[left_gpu]):
                        for right_model in list(placement[right_gpu]):
                            left_size = left_model.model_size
                            right_size = right_model.model_size

                            new_left_remaining = (
                                remaining[left_gpu] + left_size - right_size
                            )
                            new_right_remaining = (
                                remaining[right_gpu] + right_size - left_size
                            )

                            if new_left_remaining < -EPS or new_right_remaining < -EPS:
                                continue

                            left_load = load_of(left_model)
                            right_load = load_of(right_model)

                            test_loads = list(loads)
                            test_remaining = list(remaining)
                            test_loads[left_gpu] += right_load - left_load
                            test_loads[right_gpu] += left_load - right_load
                            test_remaining[left_gpu] = new_left_remaining
                            test_remaining[right_gpu] = new_right_remaining
                            value = score(test_loads, test_remaining)

                            if value < action_score - EPS:
                                action_score = value
                                action = (
                                    "swap",
                                    left_gpu,
                                    right_gpu,
                                    left_model,
                                    right_model,
                                )

            if action is None:
                break

            if action[0] == "move":
                _, source, target, model = action
                placement[source].remove(model)
                placement[target].append(model)
                remaining[source] += model.model_size
                remaining[target] -= model.model_size
                loads[source] -= load_of(model)
                loads[target] += load_of(model)
            else:
                _, left_gpu, right_gpu, left_model, right_model = action
                placement[left_gpu].remove(left_model)
                placement[right_gpu].remove(right_model)
                placement[left_gpu].append(right_model)
                placement[right_gpu].append(left_model)

                remaining[left_gpu] += left_model.model_size - right_model.model_size
                remaining[right_gpu] += right_model.model_size - left_model.model_size
                loads[left_gpu] += load_of(right_model) - load_of(left_model)
                loads[right_gpu] += load_of(left_model) - load_of(right_model)

        return placement, remaining, loads

    placement, remaining, loads = best
    placement, remaining, loads = improve(placement, remaining, loads)
    return placement

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float
    import numpy as np

    test_cases = generate_test_gpu_models()
    all_kvpr = []

    for gpu_num, gpu_models in test_cases:
        results = compute_model_placement(gpu_num, gpu_models)
        max_kvpr = calculate_kvcache_pressure(results)
        all_kvpr.append(safe_float(max_kvpr))

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr

    print(f"Max KVPR: {avg_kvpr:.3f}")
