GPU_MEM_SIZE = 80  # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """
    Place models on GPUs while minimizing maximum KV cache pressure:

        KVPR = sum(req_rate / slo) / remaining_memory

    For a candidate threshold T, KVPR <= T is equivalent to:

        load + T * used_memory <= T * GPU_MEM_SIZE

    Thus each model has transformed size:

        req_rate / slo + T * model_size

    and each GPU has transformed capacity T * GPU_MEM_SIZE.
    """
    EPS = 1e-10
    THRESHOLD_ITERATIONS = 22
    LOCAL_SEARCH_PASSES = 28
    TAIL_EXACT_MODELS = 7
    MAX_TAIL_NODES = 40000

    if gpu_num <= 0:
        raise ValueError("gpu_num must be positive")

    count = len(models)
    if count == 0:
        return {gpu_id: [] for gpu_id in range(gpu_num)}

    sizes = [float(model.model_size) for model in models]
    weights = [float(model.req_rate) / float(model.slo) for model in models]
    ids = list(range(count))

    for size in sizes:
        if size > GPU_MEM_SIZE + EPS:
            raise ValueError(
                f"Unable to place model of size {size} GB: "
                f"it exceeds GPU memory of {GPU_MEM_SIZE} GB."
            )

    def kvpr(load, remaining):
        if remaining <= EPS:
            return float("inf") if load > EPS else 0.0
        return load / remaining

    def score(loads, remaining):
        pressures = sorted(
            (kvpr(loads[gpu], remaining[gpu]) for gpu in range(gpu_num)),
            reverse=True,
        )
        return (
            pressures[0],
            pressures[1] if gpu_num > 1 else 0.0,
            pressures[2] if gpu_num > 2 else 0.0,
            -min(remaining),
        )

    def make_state(assignment):
        loads = [0.0] * gpu_num
        remaining = [float(GPU_MEM_SIZE)] * gpu_num
        for item, gpu in enumerate(assignment):
            loads[gpu] += weights[item]
            remaining[gpu] -= sizes[item]
        return loads, remaining

    def real_greedy(order, pack_memory):
        """Greedy construction directly minimizing current maximum KVPR."""
        assignment = [-1] * count
        loads = [0.0] * gpu_num
        remaining = [float(GPU_MEM_SIZE)] * gpu_num

        for item in order:
            best_gpu = None
            best_key = None

            for gpu in range(gpu_num):
                if sizes[item] > remaining[gpu] + EPS:
                    continue

                new_load = loads[gpu] + weights[item]
                new_remaining = remaining[gpu] - sizes[item]

                maximum = kvpr(new_load, new_remaining)
                for other in range(gpu_num):
                    if other != gpu:
                        maximum = max(maximum, kvpr(loads[other], remaining[other]))

                memory_tie = new_remaining if pack_memory else -new_remaining
                key = (
                    maximum,
                    kvpr(new_load, new_remaining),
                    memory_tie,
                    gpu,
                )

                if best_key is None or key < best_key:
                    best_key = key
                    best_gpu = gpu

            if best_gpu is None:
                return None

            assignment[item] = best_gpu
            loads[best_gpu] += weights[item]
            remaining[best_gpu] -= sizes[item]

        return assignment, loads, remaining

    def memory_bfd(order):
        """
        Physical-memory best-fit construction with exact completion for the
        final few items.  This is primarily a robust fallback for finding an
        initial feasible placement.
        """
        assignment = [-1] * count
        loads = [0.0] * gpu_num
        remaining = [float(GPU_MEM_SIZE)] * gpu_num
        split = max(0, count - TAIL_EXACT_MODELS)

        for position in range(split):
            item = order[position]
            choices = [
                gpu for gpu in range(gpu_num)
                if sizes[item] <= remaining[gpu] + EPS
            ]
            if not choices:
                return None

            gpu = min(
                choices,
                key=lambda g: (remaining[g] - sizes[item], g),
            )
            assignment[item] = gpu
            loads[gpu] += weights[item]
            remaining[gpu] -= sizes[item]

        tail = order[split:]
        nodes = [0]

        def complete(position):
            if position == len(tail):
                return True

            if nodes[0] >= MAX_TAIL_NODES:
                return False
            nodes[0] += 1

            item = tail[position]
            tried = set()
            candidates = []

            for gpu in range(gpu_num):
                if sizes[item] <= remaining[gpu] + EPS:
                    signature = round(remaining[gpu], 9)
                    if signature not in tried:
                        tried.add(signature)
                        candidates.append(gpu)

            candidates.sort(key=lambda g: (remaining[g] - sizes[item], g))

            for gpu in candidates:
                assignment[item] = gpu
                loads[gpu] += weights[item]
                remaining[gpu] -= sizes[item]

                if complete(position + 1):
                    return True

                remaining[gpu] += sizes[item]
                loads[gpu] -= weights[item]
                assignment[item] = -1

            return False

        if not complete(0):
            return None

        return assignment, loads, remaining

    def threshold_pack(order, threshold):
        """
        Best-fit decreasing under transformed capacity.  The final
        TAIL_EXACT_MODELS items are completed by bounded recursion so a greedy
        prefix failure does not incorrectly reject a feasible threshold.
        """
        if threshold <= EPS:
            return None

        capacity = threshold * GPU_MEM_SIZE
        transformed = [
            weights[item] + threshold * sizes[item]
            for item in range(count)
        ]

        assignment = [-1] * count
        loads = [0.0] * gpu_num
        remaining = [float(GPU_MEM_SIZE)] * gpu_num
        used_transformed = [0.0] * gpu_num
        split = max(0, count - TAIL_EXACT_MODELS)

        for position in range(split):
            item = order[position]
            item_value = transformed[item]
            choices = []

            for gpu in range(gpu_num):
                if (
                    sizes[item] <= remaining[gpu] + EPS
                    and used_transformed[gpu] + item_value <= capacity + EPS
                ):
                    choices.append(gpu)

            if not choices:
                return None

            # Best fit leaves the least transformed capacity and prevents
            # fragmentation of the threshold-constrained bins.
            gpu = min(
                choices,
                key=lambda g: (
                    capacity - used_transformed[g] - item_value,
                    remaining[g] - sizes[item],
                    g,
                ),
            )

            assignment[item] = gpu
            loads[gpu] += weights[item]
            remaining[gpu] -= sizes[item]
            used_transformed[gpu] += item_value

        tail = order[split:]
        nodes = [0]

        def complete(position):
            if position == len(tail):
                return True

            if nodes[0] >= MAX_TAIL_NODES:
                return False
            nodes[0] += 1

            item = tail[position]
            item_value = transformed[item]
            candidates = []
            seen = set()

            for gpu in range(gpu_num):
                if (
                    sizes[item] <= remaining[gpu] + EPS
                    and used_transformed[gpu] + item_value <= capacity + EPS
                ):
                    # GPUs with the same remaining capacities are symmetric.
                    signature = (
                        round(capacity - used_transformed[gpu], 9),
                        round(remaining[gpu], 9),
                    )
                    if signature not in seen:
                        seen.add(signature)
                        candidates.append(gpu)

            candidates.sort(
                key=lambda g: (
                    capacity - used_transformed[g] - item_value,
                    remaining[g] - sizes[item],
                    g,
                )
            )

            for gpu in candidates:
                assignment[item] = gpu
                loads[gpu] += weights[item]
                remaining[gpu] -= sizes[item]
                used_transformed[gpu] += item_value

                if complete(position + 1):
                    return True

                used_transformed[gpu] -= item_value
                remaining[gpu] += sizes[item]
                loads[gpu] -= weights[item]
                assignment[item] = -1

            return False

        if not complete(0):
            return None

        return assignment, loads, remaining

    base_orders = [
        sorted(ids, key=lambda i: (-sizes[i], -weights[i])),
        sorted(ids, key=lambda i: (-weights[i], -sizes[i])),
        sorted(
            ids,
            key=lambda i: (
                -(weights[i] / max(GPU_MEM_SIZE - sizes[i], EPS)),
                -sizes[i],
            ),
        ),
        sorted(
            ids,
            key=lambda i: (
                -(weights[i] / max(sizes[i], EPS)),
                -sizes[i],
            ),
        ),
        sorted(ids, key=lambda i: (-(weights[i] * sizes[i]), -sizes[i])),
    ]

    best = None
    best_value = None

    # Initial real-KVPR greedy solutions.
    for order in base_orders:
        for pack_memory in (False, True):
            candidate = real_greedy(order, pack_memory)
            if candidate is not None:
                value = score(candidate[1], candidate[2])
                if best_value is None or value < best_value:
                    best = candidate
                    best_value = value

    # Robust physical packing fallback when pressure-greedy construction
    # fragments memory.
    for order in base_orders[:2]:
        candidate = memory_bfd(order)
        if candidate is not None:
            value = score(candidate[1], candidate[2])
            if best_value is None or value < best_value:
                best = candidate
                best_value = value

    if best is None:
        raise ValueError(
            "Unable to place all models on available GPUs. "
            "The models do not fit under the memory constraints."
        )

    upper = best_value[0]
    lower = 0.0

    # Pressure-feasibility search.  Every accepted construction is concrete,
    # and therefore its measured maximum pressure is a valid upper bound.
    if upper != float("inf") and upper > EPS:
        for _ in range(THRESHOLD_ITERATIONS):
            threshold = (lower + upper) / 2.0

            transformed_orders = [
                sorted(
                    ids,
                    key=lambda i: (
                        -(weights[i] + threshold * sizes[i]),
                        -sizes[i],
                    ),
                ),
                sorted(
                    ids,
                    key=lambda i: (
                        -(weights[i] + threshold * sizes[i]),
                        -weights[i],
                    ),
                ),
                sorted(
                    ids,
                    key=lambda i: (
                        -(weights[i] + threshold * sizes[i]),
                        -(weights[i] / max(sizes[i], EPS)),
                    ),
                ),
            ]

            threshold_best = None
            threshold_value = None

            for order in transformed_orders:
                candidate = threshold_pack(order, threshold)
                if candidate is None:
                    continue

                value = score(candidate[1], candidate[2])
                if threshold_value is None or value < threshold_value:
                    threshold_best = candidate
                    threshold_value = value

            if threshold_best is None:
                lower = threshold
            else:
                upper = threshold
                if threshold_value < best_value:
                    best = threshold_best
                    best_value = threshold_value

    assignment, loads, remaining = best

    # Final steepest-descent refinement over single moves and pairwise swaps.
    for _ in range(LOCAL_SEARCH_PASSES):
        current = score(loads, remaining)
        candidate_value = current
        operation = None

        for item in range(count):
            source = assignment[item]

            for target in range(gpu_num):
                if target == source or sizes[item] > remaining[target] + EPS:
                    continue

                loads[source] -= weights[item]
                remaining[source] += sizes[item]
                loads[target] += weights[item]
                remaining[target] -= sizes[item]

                value = score(loads, remaining)

                loads[source] += weights[item]
                remaining[source] -= sizes[item]
                loads[target] -= weights[item]
                remaining[target] += sizes[item]

                if value < candidate_value:
                    candidate_value = value
                    operation = ("move", item, target)

        for first in range(count):
            first_gpu = assignment[first]

            for second in range(first + 1, count):
                second_gpu = assignment[second]
                if first_gpu == second_gpu:
                    continue

                new_first_remaining = (
                    remaining[first_gpu] + sizes[first] - sizes[second]
                )
                new_second_remaining = (
                    remaining[second_gpu] + sizes[second] - sizes[first]
                )

                if new_first_remaining < -EPS or new_second_remaining < -EPS:
                    continue

                old_first_remaining = remaining[first_gpu]
                old_second_remaining = remaining[second_gpu]

                loads[first_gpu] += weights[second] - weights[first]
                loads[second_gpu] += weights[first] - weights[second]
                remaining[first_gpu] = new_first_remaining
                remaining[second_gpu] = new_second_remaining

                value = score(loads, remaining)

                loads[first_gpu] += weights[first] - weights[second]
                loads[second_gpu] += weights[second] - weights[first]
                remaining[first_gpu] = old_first_remaining
                remaining[second_gpu] = old_second_remaining

                if value < candidate_value:
                    candidate_value = value
                    operation = ("swap", first, second)

        if operation is None:
            break

        if operation[0] == "move":
            _, item, target = operation
            source = assignment[item]

            assignment[item] = target
            loads[source] -= weights[item]
            remaining[source] += sizes[item]
            loads[target] += weights[item]
            remaining[target] -= sizes[item]
        else:
            _, first, second = operation
            first_gpu = assignment[first]
            second_gpu = assignment[second]

            assignment[first], assignment[second] = second_gpu, first_gpu

            loads[first_gpu] += weights[second] - weights[first]
            loads[second_gpu] += weights[first] - weights[second]
            remaining[first_gpu] += sizes[first] - sizes[second]
            remaining[second_gpu] += sizes[second] - sizes[first]

    placement = {gpu_id: [] for gpu_id in range(gpu_num)}
    for item, gpu in enumerate(assignment):
        placement[gpu].append(models[item])

    return placement

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float
    import numpy as np

    test_cases = generate_test_gpu_models()
    all_kvpr = []

    for gpu_num, gpu_models in test_cases:
        results = compute_model_placement(gpu_num, gpu_models)
        all_kvpr.append(safe_float(calculate_kvcache_pressure(results)))

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr

    print(f"Max KVPR: {avg_kvpr:.3f}")
