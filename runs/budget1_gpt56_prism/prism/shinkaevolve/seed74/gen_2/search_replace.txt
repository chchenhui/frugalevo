<DESCRIPTION>
Replace the current-state greedy rule with a projected minimax rule: each candidate placement is scored by the maximum KVPR that would exist after placing the model, rather than by the GPU's KVPR before placement. This directly aligns every greedy decision with the objective.

Add a lightweight relocation improvement pass after the initial placement. It repeatedly moves a model from an overloaded GPU when doing so strictly reduces the global maximum KVPR while preserving memory feasibility. This corrects unfavorable early greedy choices without requiring an expensive exhaustive search over all placements.
</DESCRIPTION>

    Compute a model placement that minimizes the maximum KVPR across all GPUs.

    Args:
        gpu_num: Number of GPUs
        models: List of models to place

        A placement of models to GPUs
    """

    sorted_models = sorted(models, key=lambda m: (m.req_rate / m.slo), reverse=True)

    # 2) Initialize per-GPU states
    placement = {gpu_id: [] for gpu_id in range(gpu_num)}
    shared_kv = [GPU_MEM_SIZE for _ in range(gpu_num)]  # remaining memory per GPU
    weighted_req_rate = [0.0 for _ in range(gpu_num)]   # sum of r_j / s_j per GPU

    # 3) Assign each model to the GPU that minimizes current KVPR while fitting in memory
    for model in sorted_models:
        best_idx = None
        best_ratio = float('inf')

        for gpu_id in range(gpu_num):
            if model.model_size <= shared_kv[gpu_id] and shared_kv[gpu_id] > 0:
                current_ratio = weighted_req_rate[gpu_id] / shared_kv[gpu_id]
                if current_ratio < best_ratio:
                    best_ratio = current_ratio
                    best_idx = gpu_id

        # Failure: if no GPU can fit, raise an error instead of overcommitting
        if best_idx is None:
            raise ValueError(
                f"Unable to place model of size {model.model_size} GB on any GPU. "
                f"Remaining per-GPU memory: {shared_kv}"
            )

        placement[best_idx].append(model)
        weighted_req_rate[best_idx] += model.req_rate / model.slo
        shared_kv[best_idx] -= model.model_size

    return placement

def compute_model_placement(gpu_num, models):
    """
    Compute a memory-feasible placement minimizing the maximum KV cache pressure.
    """

    def kvpr(load, remaining):
        if remaining <= 0:
            return float("inf") if load > 0 else 0.0
        return load / remaining

    # Place the most pressure-intensive models first.  For each placement,
    # evaluate the resulting global maximum KVPR, not the pre-placement KVPR.
    sorted_models = sorted(
        models,
        key=lambda m: (m.req_rate / m.slo, m.model_size),
        reverse=True,
    )
    placement = {gpu_id: [] for gpu_id in range(gpu_num)}
    remaining = [GPU_MEM_SIZE for _ in range(gpu_num)]
    load = [0.0 for _ in range(gpu_num)]

    for model in sorted_models:
        model_load = model.req_rate / model.slo
        best_gpu = None
        best_max_kvpr = float("inf")

        for gpu_id in range(gpu_num):
            if model.model_size > remaining[gpu_id]:
                continue

            projected = [
                kvpr(
                    load[i] + (model_load if i == gpu_id else 0.0),
                    remaining[i] - (model.model_size if i == gpu_id else 0.0),
                )
                for i in range(gpu_num)
            ]
            projected_max = max(projected)

            if projected_max < best_max_kvpr:
                best_max_kvpr = projected_max
                best_gpu = gpu_id

        if best_gpu is None:
            raise ValueError(
                f"Unable to place model of size {model.model_size} GB on any GPU. "
                f"Remaining per-GPU memory: {remaining}"
            )

        placement[best_gpu].append(model)
        load[best_gpu] += model_load
        remaining[best_gpu] -= model.model_size

    # Relocate models while a single feasible move improves the global maximum.
    # This inexpensive local repair fixes imbalances introduced by greedy choices.
    while True:
        current_max = max(kvpr(load[i], remaining[i]) for i in range(gpu_num))
        best_move = None
        best_max_kvpr = current_max

        for source in range(gpu_num):
            for model in placement[source]:
                model_load = model.req_rate / model.slo
                for target in range(gpu_num):
                    if source == target or model.model_size > remaining[target]:
                        continue

                    candidate_max = 0.0
                    for gpu_id in range(gpu_num):
                        if gpu_id == source:
                            value = kvpr(
                                load[gpu_id] - model_load,
                                remaining[gpu_id] + model.model_size,
                            )
                        elif gpu_id == target:
                            value = kvpr(
                                load[gpu_id] + model_load,
                                remaining[gpu_id] - model.model_size,
                            )
                        else:
                            value = kvpr(load[gpu_id], remaining[gpu_id])
                        candidate_max = max(candidate_max, value)

                    if candidate_max < best_max_kvpr - 1e-12:
                        best_max_kvpr = candidate_max
                        best_move = (source, target, model, model_load)

        if best_move is None:
            break

        source, target, model, model_load = best_move
        placement[source].remove(model)
        placement[target].append(model)
        load[source] -= model_load
        load[target] += model_load
        remaining[source] += model.model_size
        remaining[target] -= model.model_size

    return placement
