<DESCRIPTION>
This version improves placement quality through deterministic multi-start construction and stronger local search.

It first tries several memory-aware ordering strategies, retaining the feasible greedy placement with the lowest maximum KVPR. It then applies best-improvement relocations and swaps rather than accepting the first available improvement. If strict local search stalls, it permits up to two controlled non-worsening plateau moves that redistribute memory capacity, then resumes strict search. This helps escape fragmentation-related local minima while preserving feasibility and avoiding uncontrolled cycles.

GPU_MEM_SIZE = 80 # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """Place models to minimize the maximum KV cache pressure ratio."""

    if gpu_num <= 0:
        if models:
            raise ValueError("Unable to place models without GPUs.")
        return {}

    def pressure(model):
        return model.req_rate / model.slo

    def ratio(load, remaining):
        if remaining > 0:
            return load / remaining
        return float("inf") if load > 0 else 0.0

    def objective(loads, remaining):
        return max(ratio(loads[gpu], remaining[gpu]) for gpu in range(gpu_num))

    def build_greedy(ordered_models):
        placement = {gpu: [] for gpu in range(gpu_num)}
        loads = [0.0] * gpu_num
        remaining = [GPU_MEM_SIZE] * gpu_num

        for model in ordered_models:
            model_pressure = pressure(model)
            best_gpu = None
            best_score = None

            current_ratios = [ratio(loads[gpu], remaining[gpu]) for gpu in range(gpu_num)]

            for gpu in range(gpu_num):
                if model.model_size > remaining[gpu]:
                    continue

                new_load = loads[gpu] + model_pressure
                new_remaining = remaining[gpu] - model.model_size
                new_ratio = ratio(new_load, new_remaining)

                candidate_max = new_ratio
                for other in range(gpu_num):
                    if other != gpu and current_ratios[other] > candidate_max:
                        candidate_max = current_ratios[other]

                score = (candidate_max, new_ratio, -new_remaining)
                if best_score is None or score < best_score:
                    best_score = score
                    best_gpu = gpu

            if best_gpu is None:
                return None

            placement[best_gpu].append(model)
            loads[best_gpu] += model_pressure
            remaining[best_gpu] -= model.model_size

        return placement, loads, remaining

    def candidate_objective(loads, remaining, left, right=None,
                            left_load=None, left_remaining=None,
                            right_load=None, right_remaining=None):
        value = 0.0
        for gpu in range(gpu_num):
            if gpu == left:
                current = ratio(left_load, left_remaining)
            elif gpu == right:
                current = ratio(right_load, right_remaining)
            else:
                current = ratio(loads[gpu], remaining[gpu])
            if current > value:
                value = current
        return value

    def improve(placement, loads, remaining):
        """Apply best strict relocation/swap improvements."""
        epsilon = 1e-12

        while True:
            current_value = objective(loads, remaining)
            best_value = current_value
            best_action = None

            for source in range(gpu_num):
                for source_index, model in enumerate(placement[source]):
                    model_pressure = pressure(model)

                    for target in range(gpu_num):
                        if source == target or model.model_size > remaining[target]:
                            continue

                        source_load = loads[source] - model_pressure
                        source_remaining = remaining[source] + model.model_size
                        target_load = loads[target] + model_pressure
                        target_remaining = remaining[target] - model.model_size

                        value = candidate_objective(
                            loads, remaining, source, target,
                            source_load, source_remaining,
                            target_load, target_remaining,
                        )

                        if value < best_value - epsilon:
                            best_value = value
                            best_action = (
                                "move", source, source_index, target,
                                source_load, source_remaining,
                                target_load, target_remaining,
                            )

            for left in range(gpu_num):
                for right in range(left + 1, gpu_num):
                    for left_index, left_model in enumerate(placement[left]):
                        left_pressure = pressure(left_model)

                        for right_index, right_model in enumerate(placement[right]):
                            right_pressure = pressure(right_model)
                            left_remaining = (
                                remaining[left]
                                + left_model.model_size
                                - right_model.model_size
                            )
                            right_remaining = (
                                remaining[right]
                                + right_model.model_size
                                - left_model.model_size
                            )

                            if left_remaining < 0 or right_remaining < 0:
                                continue

                            left_load = loads[left] - left_pressure + right_pressure
                            right_load = loads[right] - right_pressure + left_pressure

                            value = candidate_objective(
                                loads, remaining, left, right,
                                left_load, left_remaining,
                                right_load, right_remaining,
                            )

                            if value < best_value - epsilon:
                                best_value = value
                                best_action = (
                                    "swap", left, left_index, right, right_index,
                                    left_load, left_remaining,
                                    right_load, right_remaining,
                                )

            if best_action is None:
                break

            if best_action[0] == "move":
                _, source, source_index, target, source_load, source_remaining, target_load, target_remaining = best_action
                model = placement[source].pop(source_index)
                placement[target].append(model)
                loads[source] = source_load
                remaining[source] = source_remaining
                loads[target] = target_load
                remaining[target] = target_remaining
            else:
                _, left, left_index, right, right_index, left_load, left_remaining, right_load, right_remaining = best_action
                placement[left][left_index], placement[right][right_index] = (
                    placement[right][right_index],
                    placement[left][left_index],
                )
                loads[left] = left_load
                remaining[left] = left_remaining
                loads[right] = right_load
                remaining[right] = right_remaining

        return placement, loads, remaining

    def plateau_move(placement, loads, remaining):
        """
        Make one feasible non-worsening move or swap.  The tie-breaker favors
        lower secondary pressure, which can expose a later strict improvement.
        """
        epsilon = 1e-12
        current_value = objective(loads, remaining)
        best_score = None
        best_action = None

        def secondary_score(test_loads, test_remaining):
            values = sorted(
                ratio(test_loads[gpu], test_remaining[gpu])
                for gpu in range(gpu_num)
            )
            return sum(value * value for value in values)

        for source in range(gpu_num):
            for source_index, model in enumerate(placement[source]):
                model_pressure = pressure(model)

                for target in range(gpu_num):
                    if source == target or model.model_size > remaining[target]:
                        continue

                    test_loads = loads[:]
                    test_remaining = remaining[:]
                    test_loads[source] -= model_pressure
                    test_remaining[source] += model.model_size
                    test_loads[target] += model_pressure
                    test_remaining[target] -= model.model_size

                    value = objective(test_loads, test_remaining)
                    if value <= current_value + epsilon:
                        score = (value, secondary_score(test_loads, test_remaining))
                        if best_score is None or score < best_score:
                            best_score = score
                            best_action = (
                                "move", source, source_index, target,
                                test_loads, test_remaining,
                            )

        for left in range(gpu_num):
            for right in range(left + 1, gpu_num):
                for left_index, left_model in enumerate(placement[left]):
                    for right_index, right_model in enumerate(placement[right]):
                        left_remaining = (
                            remaining[left]
                            + left_model.model_size
                            - right_model.model_size
                        )
                        right_remaining = (
                            remaining[right]
                            + right_model.model_size
                            - left_model.model_size
                        )

                        if left_remaining < 0 or right_remaining < 0:
                            continue

                        test_loads = loads[:]
                        test_remaining = remaining[:]
                        test_loads[left] += pressure(right_model) - pressure(left_model)
                        test_loads[right] += pressure(left_model) - pressure(right_model)
                        test_remaining[left] = left_remaining
                        test_remaining[right] = right_remaining

                        value = objective(test_loads, test_remaining)
                        if value <= current_value + epsilon:
                            score = (value, secondary_score(test_loads, test_remaining))
                            if best_score is None or score < best_score:
                                best_score = score
                                best_action = (
                                    "swap", left, left_index, right, right_index,
                                    test_loads, test_remaining,
                                )

        if best_action is None:
            return None

        if best_action[0] == "move":
            _, source, source_index, target, loads, remaining = best_action
            model = placement[source].pop(source_index)
            placement[target].append(model)
        else:
            _, left, left_index, right, right_index, loads, remaining = best_action
            placement[left][left_index], placement[right][right_index] = (
                placement[right][right_index],
                placement[left][left_index],
            )

        return placement, loads, remaining

    ordered_sets = [
        sorted(models, key=lambda m: (pressure(m), m.model_size), reverse=True),
        sorted(models, key=lambda m: (m.model_size, pressure(m)), reverse=True),
        sorted(
            models,
            key=lambda m: (pressure(m) / max(m.model_size, 1e-12), pressure(m)),
            reverse=True,
        ),
        sorted(
            models,
            key=lambda m: (pressure(m) * m.model_size, m.model_size),
            reverse=True,
        ),
    ]

    best_result = None
    best_value = float("inf")

    for ordered_models in ordered_sets:
        result = build_greedy(ordered_models)
        if result is None:
            continue

        placement, loads, remaining = improve(*result)
        value = objective(loads, remaining)

        if value < best_value:
            best_value = value
            best_result = (placement, loads, remaining)

    if best_result is None:
        largest = max(models, key=lambda model: model.model_size)
        raise ValueError(
            f"Unable to place model of size {largest.model_size} GB on any GPU."
        )

    placement, loads, remaining = best_result

    # Controlled plateau exploration can escape memory-fragmentation local minima.
    for _ in range(2):
        moved = plateau_move(placement, loads, remaining)
        if moved is None:
            break

        placement, loads, remaining = improve(*moved)
        value = objective(loads, remaining)

        if value < best_value:
            best_value = value
            best_result = (placement, loads, remaining)

    return best_result[0]

# EVOLVE-BLOCK-END
