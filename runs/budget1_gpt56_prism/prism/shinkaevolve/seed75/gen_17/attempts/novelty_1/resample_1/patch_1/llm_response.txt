<DESCRIPTION>
Replace the single pressure-first construction with several deterministic greedy starts and retain the best complete placement before local improvement. Model placement is sensitive to ordering: pressure-first distributes load well but can fragment memory, while size-first and hard-to-place-first preserve feasible regions for large models. Each start still chooses the GPU with the minimum resulting global maximum KVPR. Selecting the best completed start improves the local search starting point without changing the output format or permitting memory overcommitment.
</DESCRIPTION>

    model_list = list(models)

    def demand(model):
        return model.req_rate / model.slo

    def construct(order):
        trial_placement = {gpu_id: [] for gpu_id in range(gpu_num)}
        trial_remaining = [GPU_MEM_SIZE for _ in range(gpu_num)]
        trial_pressure = [0.0 for _ in range(gpu_num)]

        for model in order:
            weight = demand(model)
            candidates = [
                gpu_id for gpu_id in range(gpu_num)
                if model.model_size < trial_remaining[gpu_id]
            ]
            if not candidates:
                return None

            best_idx = min(
                candidates,
                key=lambda gpu_id: (
                    max(
                        (trial_pressure[other] + weight) /
                        (trial_remaining[other] - model.model_size)
                        if other == gpu_id else
                        trial_pressure[other] / trial_remaining[other]
                        for other in range(gpu_num)
                    ),
                    -(trial_remaining[gpu_id] - model.model_size),
                    gpu_id,
                ),
            )
            trial_placement[best_idx].append(model)
            trial_pressure[best_idx] += weight
            trial_remaining[best_idx] -= model.model_size

        objective = max(
            trial_pressure[gpu_id] / trial_remaining[gpu_id]
            for gpu_id in range(gpu_num)
        )
        return objective, trial_placement, trial_remaining, trial_pressure
