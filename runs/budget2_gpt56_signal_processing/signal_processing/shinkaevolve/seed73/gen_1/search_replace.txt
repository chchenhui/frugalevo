<DESCRIPTION>
Replace the fixed exponentially weighted average with a causal, robust adaptive alpha-beta tracker. The tracker explicitly models both level and local velocity, which reduces lag relative to averaging while preserving sustained trends through one-step prediction. Innovation clipping prevents isolated volatile samples from causing false directional reversals, and a velocity deadband suppresses small noise-driven slope sign changes. The window size is retained as a warm-up/output-alignment parameter and controls the adaptation time constant, preserving the existing API and output length expected by downstream metric alignment.

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using weighted moving average.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Create weights that emphasize recent samples
    weights = np.exp(np.linspace(-2, 0, window_size))
    weights = weights / np.sum(weights)

    for i in range(output_length):
        window = x[i : i + window_size]

        # Weighted moving average with exponential weights
        y[i] = np.sum(window * weights)

    return y

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Causal robust alpha-beta tracker for volatile non-stationary signals.

    A level and local velocity are estimated recursively.  Prediction from the
    velocity reduces the lag of a moving average, while clipped innovations and
    a velocity deadband avoid noise-induced reversals.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Warm-up length and adaptation time scale

    Returns:
        y: Filtered output aligned with x[window_size - 1:]
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        raise ValueError("Input signal must contain only finite values")

    # A robust initial innovation scale avoids allowing the first large
    # measurement excursion to dominate the tracker.
    initial_differences = np.diff(x[:window_size])
    noise_scale = 1.4826 * np.median(np.abs(initial_differences - np.median(initial_differences)))
    noise_scale = max(noise_scale, np.finfo(float).eps)

    level = x[0]
    velocity = 0.0
    filtered = np.empty(len(x), dtype=float)
    filtered[0] = level

    # Larger windows favour smoothness; the innovation term still raises the
    # gain quickly when there is persistent evidence of a real movement.
    adaptation = 2.0 / (window_size + 1.0)
    base_alpha = 0.08 + 0.10 * adaptation
    max_alpha = 0.42
    velocity_gain = 0.035 + 0.06 * adaptation
    scale_rate = adaptation * 0.18

    for i in range(1, len(x)):
        prediction = level + velocity
        innovation = x[i] - prediction

        # Estimate local volatility from innovations.  This follows changing
        # noise conditions without requiring an additional full window pass.
        noise_scale = (1.0 - scale_rate) * noise_scale + scale_rate * abs(innovation)
        noise_scale = max(noise_scale, np.finfo(float).eps)

        normalized_innovation = abs(innovation) / noise_scale
        alpha = base_alpha + (max_alpha - base_alpha) * np.clip(
            (normalized_innovation - 0.8) / 2.5, 0.0, 1.0
        )

        # Robust gating limits impulsive measurements but leaves genuine,
        # sustained transitions responsive because their predicted residuals
        # continue to update the state over consecutive samples.
        bounded_innovation = np.clip(innovation, -2.75 * noise_scale, 2.75 * noise_scale)
        level = prediction + alpha * bounded_innovation
        velocity = velocity + velocity_gain * alpha * bounded_innovation

        # Eliminate tiny alternating velocity estimates that are characteristic
        # of measurement noise and are heavily penalized as false reversals.
        velocity_deadband = 0.055 * noise_scale
        if abs(velocity) < velocity_deadband:
            velocity = 0.0

        filtered[i] = level

    return filtered[window_size - 1:]
