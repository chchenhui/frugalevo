"""
Robust adaptive real-time signal filtering for volatile non-stationary series.

The filter is causal: each returned value corresponds to the most recent sample
in its associated sliding window.  It uses a robust, adaptive constant-velocity
Kalman tracker rather than a delayed moving average.
"""
import numpy as np


def _robust_scale(values, floor=1e-8):
    """MAD-based standard deviation estimate, stable for short windows."""
    values = np.asarray(values, dtype=float)
    if values.size == 0:
        return floor
    median = np.median(values)
    scale = 1.4826 * np.median(np.abs(values - median))
    return max(float(scale), floor)


def _robust_kalman_filter(x, window_size):
    """
    Causal adaptive level-and-slope tracker.

    A local MAD estimate of first differences controls observation uncertainty;
    a local MAD estimate of slope changes controls allowed state acceleration.
    This separates smooth trend motion from short-lived measurement noise.
    """
    x = np.asarray(x, dtype=float)
    n = len(x)
    y_full = np.empty(n, dtype=float)

    # Preserve finite operation even if an input contains accidental non-finite
    # samples, using the last valid observation as a conservative substitute.
    finite = np.isfinite(x)
    if not np.any(finite):
        return np.zeros(n - window_size + 1, dtype=float)
    first_valid = float(x[np.flatnonzero(finite)[0]])
    z_values = np.where(finite, x, first_valid)

    # State: [level, slope]. Initialize with the first local direction.
    level = float(z_values[0])
    slope = float(z_values[1] - z_values[0]) if n > 1 else 0.0

    # Broad initial uncertainty lets the tracker lock onto the signal quickly.
    p00, p01, p11 = 1.0, 0.0, 1.0
    previous_stable_slope = slope
    y_full[0] = level

    for t in range(1, n):
        start = max(0, t - window_size + 1)
        local = z_values[start:t + 1]
        diffs = np.diff(local)

        # Difference noise contains noise from two samples; divide by sqrt(2).
        diff_scale = _robust_scale(diffs)
        measurement_sigma = max(diff_scale / np.sqrt(2.0), 1e-6)
        r = measurement_sigma * measurement_sigma

        # Robust acceleration estimate permits real changes in trend but does
        # not inject high-frequency observation jitter into the slope state.
        if diffs.size >= 2:
            acceleration_sigma = _robust_scale(np.diff(diffs))
        else:
            acceleration_sigma = measurement_sigma
        q = max((0.18 * measurement_sigma) ** 2,
                (0.55 * acceleration_sigma) ** 2,
                1e-10)

        # Constant-velocity prediction, with continuous white-acceleration Q.
        predicted_level = level + slope
        predicted_slope = slope
        pp00 = p00 + 2.0 * p01 + p11 + 0.25 * q
        pp01 = p01 + p11 + 0.5 * q
        pp11 = p11 + q

        # Huber innovation gating: an impulse cannot initiate a false reversal.
        innovation = z_values[t] - predicted_level
        gate = 3.5 * measurement_sigma + 0.20 * abs(predicted_slope)
        clipped_innovation = np.clip(innovation, -gate, gate)

        innovation_variance = max(pp00 + r, 1e-12)
        k0 = pp00 / innovation_variance
        k1 = pp01 / innovation_variance

        level = predicted_level + k0 * clipped_innovation
        candidate_slope = predicted_slope + k1 * clipped_innovation

        # Joseph-equivalent scalar covariance update, symmetrized implicitly.
        p00 = max((1.0 - k0) * pp00, 1e-12)
        p01 = (1.0 - k0) * pp01
        p11 = max(pp11 - k1 * pp01, 1e-12)

        # Trend hysteresis only acts on weak reversals. Strong evidence is left
        # untouched, retaining responsiveness at genuine turning points.
        reversal_threshold = 0.30 * measurement_sigma + 0.12 * acceleration_sigma
        if (candidate_slope * previous_stable_slope < 0.0 and
                abs(candidate_slope) < reversal_threshold):
            candidate_slope = 0.65 * previous_stable_slope
        elif abs(candidate_slope) >= reversal_threshold:
            previous_stable_slope = candidate_slope

        slope = candidate_slope
        y_full[t] = level

    # Match the original API's window-end alignment and output length.
    return y_full[window_size - 1:]


def adaptive_filter(x, window_size=20):
    """
    Robust adaptive causal filter.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Sliding-window size used for local adaptation.

    Returns:
        Filtered output with length len(x) - window_size + 1.
    """
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )
    if window_size < 1:
        raise ValueError("window_size must be >= 1")
    return _robust_kalman_filter(x, window_size)


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Adaptive robust trend-preserving filter.

    This public function retains the original interface while using the same
    low-lag robust state-space tracker as the primary enhanced implementation.
    """
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )
    if window_size < 1:
        raise ValueError("window_size must be >= 1")
    return _robust_kalman_filter(x, window_size)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Apply the selected filtering algorithm.

    Args:
        input_signal: Input time series data.
        window_size: Local adaptation window size.
        algorithm_type: "basic" or "enhanced".

    Returns:
        Filtered signal.
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal.
        noise_level: Standard deviation of noise to add.
        seed: Random seed for reproducibility.

    Returns:
        Tuple of (noisy_signal, clean_signal).
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000,
                          noise_level=0.3, window_size=20):
    """
    Run signal processing and return results using the original output schema.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(
            signal_length, noise_level
        )
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1 else 0
        )
        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0 else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
