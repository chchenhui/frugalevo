def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using weighted moving average.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Create weights that emphasize recent samples
    weights = np.exp(np.linspace(-2, 0, window_size))
    weights = weights / np.sum(weights)

    for i in range(output_length):
        window = x[i : i + window_size]

        # Weighted moving average with exponential weights
        y[i] = np.sum(window * weights)

    return y

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Causal robust adaptive alpha-beta filter for noisy non-stationary signals.

    The level/velocity state predicts current motion, avoiding the fixed delay of a
    moving average.  Measurement innovations are robustly limited using an adaptive
    noise scale so isolated spikes do not create false trend reversals.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the adaptation horizon

    Returns:
        y: Filtered output signal, aligned with input samples [window_size - 1:]
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be one-dimensional")
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    # A robust initial scale makes adaptation insensitive to startup outliers.
    initial_differences = np.diff(x[:window_size])
    noise_scale = np.median(np.abs(initial_differences - np.median(initial_differences)))
    noise_scale = max(1.4826 * noise_scale, np.finfo(float).eps)

    # Longer windows favor smoother baseline behavior while retaining prediction.
    base_alpha = 2.0 / (window_size + 3.0)
    base_beta = base_alpha * base_alpha * 0.35
    scale_rate = 2.0 / (window_size + 1.0)

    level = x[0]
    velocity = 0.0
    filtered = np.empty(len(x), dtype=float)
    filtered[0] = level

    for i in range(1, len(x)):
        prediction = level + velocity
        innovation = x[i] - prediction

        # Update robust residual scale before limiting an individual innovation.
        absolute_innovation = abs(innovation)
        noise_scale = (1.0 - scale_rate) * noise_scale + scale_rate * absolute_innovation
        noise_scale = max(noise_scale, np.finfo(float).eps)
        limited_innovation = np.clip(innovation, -3.5 * noise_scale, 3.5 * noise_scale)

        # Raise gains only for motion that is large relative to local noise.
        motion_ratio = abs(velocity) / noise_scale
        responsiveness = motion_ratio / (motion_ratio + 1.5)
        alpha = base_alpha + (0.62 - base_alpha) * responsiveness
        beta = base_beta + (0.18 - base_beta) * responsiveness

        level = prediction + alpha * limited_innovation
        velocity = (1.0 - beta) * velocity + beta * limited_innovation
        filtered[i] = level

    # Preserve the original API's endpoint alignment and output length.
    return filtered[window_size - 1:]
