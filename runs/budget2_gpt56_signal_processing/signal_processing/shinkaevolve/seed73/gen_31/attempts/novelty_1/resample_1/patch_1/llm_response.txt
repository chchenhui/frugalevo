<DESCRIPTION>
The current approach uses a Kalman level/slope tracker with a short two-sample reversal confirmation. Its primary weakness is excessive directional switching: slope changes increased to 189.4 and false reversals to 151 despite good lag (0.31) and accuracy (0.91).

This revision replaces covariance-heavy updates with a robust multi-scale endpoint-regression alpha-beta tracker. Key changes are: (1) a recency-weighted robust regression observation rather than direct noisy measurement updates, (2) short- and long-window slope consensus for trend evidence, (3) a two-state flat/trend controller with hysteresis thresholds of 1.35 entry, 0.70 exit, 2 entry windows, and 3 exit windows, (4) lower alpha/beta gains in flat conditions (`0.24`, `0.035`) and higher gains in sustained trends (`0.52`, `0.15`), and (5) three-confirmation reversals in flat state versus two in trend state. This should reduce noise-driven reversals and slope changes while retaining rapid tracking when repeated slope evidence indicates a genuine move. Robust Huber weighting and innovation clipping reduce impulse sensitivity without adding non-causal delay.

"""
Real-Time Adaptive Signal Processing Algorithm for Non-Stationary Time Series.

The enhanced filter is a causal, robust, multi-scale level-and-velocity tracker.
Each output is aligned with the newest sample in its sliding window.
"""
import numpy as np


def adaptive_filter(x, window_size=20):
    """
    Basic sliding-window moving-average filter.

    Args:
        x: Input one-dimensional signal.
        window_size: Number of samples in each window.

    Returns:
        Filtered signal of length len(x) - window_size + 1.
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be a one-dimensional array")
    if window_size < 1:
        raise ValueError("window_size must be at least 1")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )

    csum = np.concatenate(([0.0], np.cumsum(x)))
    return (csum[window_size:] - csum[:-window_size]) / float(window_size)


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Causal robust multi-scale endpoint tracker.

    A robust recency-weighted local linear fit provides a denoised endpoint and
    slope observation. A two-state hysteretic controller selects conservative
    gains in noise-dominated regions and responsive gains during persistent
    trends. Direction reversals require repeated slope evidence.

    Args:
        x: Input one-dimensional signal.
        window_size: Trailing window used for noise and trend estimation.

    Returns:
        Filtered signal with length len(x) - window_size + 1.
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be a one-dimensional array")
    if window_size < 2:
        raise ValueError("window_size must be at least 2")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )

    finite = np.isfinite(x)
    if not np.any(finite):
        return np.zeros(len(x) - window_size + 1, dtype=float)

    first_valid = float(x[np.flatnonzero(finite)[0]])
    z = np.where(finite, x, first_valid)
    output_length = len(z) - window_size + 1
    y = np.empty(output_length, dtype=float)

    # Long-window recency weighting.  The endpoint has time coordinate zero.
    t_long = np.arange(window_size, dtype=float) - (window_size - 1)
    base_long = np.exp(2.6 * t_long / max(window_size - 1, 1))

    # The short slope detects real changes earlier, while the long slope is
    # more stable and suppresses isolated noisy directional indications.
    short_size = max(4, window_size // 2)
    t_short = np.arange(short_size, dtype=float) - (short_size - 1)
    base_short = np.exp(1.8 * t_short / max(short_size - 1, 1))

    def robust_endpoint_fit(values, time, base_weights):
        """Two-pass weighted Huber linear fit evaluated at the newest sample."""
        weights = base_weights
        intercept = float(values[-1])
        slope = 0.0
        residual = values - intercept

        for _ in range(2):
            sw = np.sum(weights)
            mt = np.sum(weights * time) / sw
            mv = np.sum(weights * values) / sw
            dt = time - mt
            denominator = np.sum(weights * dt * dt)

            if denominator <= 1e-12:
                slope = 0.0
            else:
                slope = np.sum(weights * dt * (values - mv)) / denominator
            intercept = mv - slope * mt
            residual = values - (intercept + slope * time)

            mad = np.median(np.abs(residual - np.median(residual)))
            scale = max(1.4826 * mad, 1e-6)
            huber = np.minimum(1.0, 2.35 * scale / np.maximum(np.abs(residual), 1e-12))
            weights = base_weights * huber

        scale = max(
            1.4826 * np.median(np.abs(residual - np.median(residual))),
            1e-6,
        )
        effective_denominator = max(np.sum(weights * (time - mt) ** 2), 1e-12)
        slope_std = scale / np.sqrt(effective_denominator)
        return float(intercept), float(slope), float(scale), float(slope_std)

    initial_window = z[:window_size]
    initial_level, initial_slope, initial_scale, _ = robust_endpoint_fit(
        initial_window, t_long, base_long
    )

    level = initial_level
    velocity = initial_slope
    published = None
    confirmed_direction = 0
    pending_direction = 0
    pending_count = 0

    # Trend/flat controller parameters.
    trend_state = False
    trend_count = 0
    flat_count = 0
    trend_entry_evidence = 1.35
    trend_exit_evidence = 0.70
    trend_entry_windows = 2
    trend_exit_windows = 3

    for out_index in range(output_length):
        end = out_index + window_size - 1
        long_window = z[end - window_size + 1:end + 1]
        local_level, long_slope, residual_scale, long_slope_std = robust_endpoint_fit(
            long_window, t_long, base_long
        )

        short_window = z[end - short_size + 1:end + 1]
        _, short_slope, short_scale, short_slope_std = robust_endpoint_fit(
            short_window, t_short, base_short
        )

        # Agreement between scales increases confidence.  If they disagree,
        # retain the stable long estimate and avoid impulsive mode changes.
        if long_slope * short_slope > 0.0:
            consensus_slope = 0.58 * short_slope + 0.42 * long_slope
            consensus_std = 0.58 * short_slope_std + 0.42 * long_slope_std
        else:
            consensus_slope = 0.78 * long_slope + 0.22 * short_slope
            consensus_std = long_slope_std + 0.35 * short_slope_std

        evidence = abs(consensus_slope) / max(consensus_std, 1e-8)

        if not trend_state:
            trend_count = trend_count + 1 if evidence >= trend_entry_evidence else 0
            if trend_count >= trend_entry_windows:
                trend_state = True
                trend_count = 0
                flat_count = 0
        else:
            flat_count = flat_count + 1 if evidence <= trend_exit_evidence else 0
            if flat_count >= trend_exit_windows:
                trend_state = False
                flat_count = 0
                trend_count = 0

        if out_index == 0:
            level = local_level
            velocity = consensus_slope
        else:
            predicted = level + velocity
            innovation = local_level - predicted
            noise_scale = max(residual_scale, short_scale, 0.20 * initial_scale, 1e-6)

            # Trend state prioritizes low lag; flat state emphasizes smoothness.
            if trend_state:
                alpha = 0.52
                beta = 0.15
                innovation_limit = 3.6 * noise_scale + 0.35 * abs(velocity)
                accel_limit = 0.55 * abs(velocity) + 1.10 * noise_scale
                reversal_required = 2
                deadband_factor = 0.09
            else:
                alpha = 0.24
                beta = 0.035
                innovation_limit = 2.7 * noise_scale + 0.20 * abs(velocity)
                accel_limit = 0.22 * abs(velocity) + 0.42 * noise_scale
                reversal_required = 3
                deadband_factor = 0.16

            innovation = np.clip(innovation, -innovation_limit, innovation_limit)
            level = predicted + alpha * innovation

            observed_step = level - published
            target_velocity = (
                0.52 * consensus_slope
                + 0.28 * long_slope
                + 0.20 * observed_step
            )
            velocity_delta = np.clip(
                target_velocity - velocity,
                -accel_limit,
                accel_limit,
            )
            candidate_velocity = velocity + beta * velocity_delta

            velocity_deadband = max(
                0.17 * noise_scale,
                0.10 * abs(velocity),
                1e-8,
            )
            old_direction = 1 if velocity > velocity_deadband else -1 if velocity < -velocity_deadband else 0
            new_direction = (
                1 if candidate_velocity > velocity_deadband
                else -1 if candidate_velocity < -velocity_deadband
                else 0
            )
            slope_direction = (
                1 if consensus_slope > 0.55 * velocity_deadband
                else -1 if consensus_slope < -0.55 * velocity_deadband
                else 0
            )

            # Do not immediately let an uncertain velocity estimate reverse.
            if (
                old_direction != 0
                and new_direction == -old_direction
                and slope_direction == new_direction
            ):
                if pending_direction == new_direction:
                    pending_count += 1
                else:
                    pending_direction = new_direction
                    pending_count = 1

                if pending_count < reversal_required:
                    candidate_velocity = 0.55 * velocity
                else:
                    pending_direction = 0
                    pending_count = 0
            elif slope_direction == old_direction or new_direction == old_direction:
                pending_direction = 0
                pending_count = 0

            velocity = candidate_velocity

        if published is None:
            published = level
            if abs(velocity) > 1e-10:
                confirmed_direction = 1 if velocity > 0 else -1
        else:
            change = level - published
            noise_scale = max(residual_scale, short_scale, 0.20 * initial_scale, 1e-6)
            deadband = max(deadband_factor * noise_scale, 1e-8)

            if abs(change) <= deadband:
                candidate = published
            else:
                direction = 1 if change > 0.0 else -1

                if confirmed_direction == 0 or direction == confirmed_direction:
                    confirmed_direction = direction
                    pending_direction = 0
                    pending_count = 0
                    candidate = level
                else:
                    # Publication hysteresis is distinct from state estimation:
                    # the state can prepare for a turn before it is displayed.
                    if pending_direction == direction:
                        pending_count += 1
                    else:
                        pending_direction = direction
                        pending_count = 1

                    if pending_count >= reversal_required:
                        confirmed_direction = direction
                        pending_direction = 0
                        pending_count = 0
                        candidate = level
                    else:
                        candidate = published

            published = candidate

        y[out_index] = published

    return y


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Apply either the basic or enhanced filtering algorithm.
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate a synthetic non-stationary signal and its clean reference.
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend
    clean_signal += np.cumsum(np.random.randn(length) * 0.05)

    noisy_signal = clean_signal + np.random.normal(0, noise_level, length)
    return noisy_signal, clean_signal


def run_signal_processing(
    noisy_signal=None,
    signal_length=1000,
    noise_level=0.3,
    window_size=20,
):
    """
    Process a provided signal or a generated test signal.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1
            else 0
        )
        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0
            else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
