<DESCRIPTION>
Replaces the single local smoother with a structured multiscale pipeline: three endpoint-aligned weighted regressions produce causal level, slope, and residual-noise estimates; confidence-driven blending selects faster recent weighting during reliable trends and gentler weighting during noisy/flat intervals. A robust adaptive alpha-beta tracking stage then applies innovation clipping, acceleration limiting, and direction hysteresis to reduce false reversals while retaining low-lag trend tracking. Shared validation and regression helpers improve maintainability while preserving all original public inputs and outputs.

"""
Real-Time Adaptive Signal Processing Algorithm for Non-Stationary Time Series.

The enhanced path uses a causal, endpoint-aligned multiscale regression bank
followed by a robust adaptive alpha-beta tracker.  Every output corresponds to
the newest sample in its input window, preserving the original API alignment.
"""
import numpy as np


def _validate_signal(x, window_size):
    """Convert and validate common signal-processing inputs."""
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be a 1D array")
    if window_size < 1:
        raise ValueError("window_size must be >= 1")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )
    return x


def adaptive_filter(x, window_size=20):
    """
    Efficient sliding-window moving average baseline.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window (W samples)

    Returns:
        y: Filtered output signal with length = len(x) - window_size + 1
    """
    x = _validate_signal(x, window_size)
    cumulative = np.concatenate(([0.0], np.cumsum(x, dtype=float)))
    return (cumulative[window_size:] - cumulative[:-window_size]) / float(window_size)


def _endpoint_regression(x, window_size, decay_strength):
    """
    Calculate a causal weighted linear fit for every complete window.

    The fit coordinate ends at zero, so the returned level is the fitted value
    at the newest sample.  This avoids the half-window phase delay of a normal
    moving average.
    """
    if window_size == 1:
        level = x.copy()
        return level, np.zeros_like(level), np.full_like(level, 1e-8)

    time = np.arange(-(window_size - 1), 1, dtype=float)
    weights = np.exp(decay_strength * time / float(window_size - 1))

    sw = np.sum(weights)
    st = np.sum(weights * time)
    stt = np.sum(weights * time * time)
    determinant = max(sw * stt - st * st, 1e-12)

    # Reversed kernels make each valid convolution use chronological window
    # weights from oldest to newest sample.
    kernel = weights[::-1]
    sy = np.convolve(x, kernel, mode="valid")
    sty = np.convolve(x, (weights * time)[::-1], mode="valid")
    syy = np.convolve(x * x, kernel, mode="valid")

    slope = (sw * sty - st * sy) / determinant
    level = (stt * sy - st * sty) / determinant

    # Weighted residual variance, computed from regression normal equations.
    fitted_energy = level * sy + slope * sty
    residual_var = np.maximum((syy - fitted_energy) / sw, 1e-10)
    return level, slope, residual_var


def _multiscale_measurements(x, window_size):
    """
    Produce confidence-adaptive measurements from slow, medium, and fast fits.

    Gentle weighting is preferred when directional evidence is weak.  A more
    recent weighting is progressively selected only when its slope exceeds the
    local residual-noise floor, reducing both lag and noise-induced reversals.
    """
    slow_level, slow_slope, slow_var = _endpoint_regression(x, window_size, 1.5)
    mid_level, mid_slope, mid_var = _endpoint_regression(x, window_size, 3.0)
    fast_level, fast_slope, fast_var = _endpoint_regression(x, window_size, 5.0)

    scale = np.sqrt(np.maximum(mid_var, 1e-10))
    slope_confidence = np.abs(mid_slope) / (scale + 1e-12)

    # Fast estimates are valuable only for coherent movement.  The slow fit
    # receives additional weight in low-confidence, noise-dominated windows.
    fast_weight = 0.08 + 0.52 * np.tanh(0.85 * slope_confidence)
    slow_weight = 0.34 * np.exp(-1.15 * slope_confidence)
    mid_weight = np.maximum(1.0 - fast_weight - slow_weight, 0.15)

    total = slow_weight + mid_weight + fast_weight
    slow_weight /= total
    mid_weight /= total
    fast_weight /= total

    level = slow_weight * slow_level + mid_weight * mid_level + fast_weight * fast_level
    slope = slow_weight * slow_slope + mid_weight * mid_slope + fast_weight * fast_slope
    variance = (
        slow_weight * slow_var
        + mid_weight * mid_var
        + fast_weight * fast_var
    )

    return level, slope, np.maximum(variance, 1e-10), slope_confidence


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Robust multiscale causal regression and adaptive trend tracker.

    A vectorized bank of endpoint-aligned exponentially weighted regressions
    supplies level, slope, and local noise measurements.  A sequential
    alpha-beta tracker follows persistent trends predictively, clips isolated
    innovations, and applies hysteretic direction control to suppress false
    reversals without imposing moving-average phase delay.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Sliding-window length used for local context and output
            alignment.

    Returns:
        Filtered signal aligned with x[window_size - 1:].
    """
    x = _validate_signal(x, window_size)

    measured_level, measured_slope, measured_var, confidence = (
        _multiscale_measurements(x, window_size)
    )
    count = len(measured_level)
    output = np.empty(count, dtype=float)

    level = float(measured_level[0])
    velocity = float(measured_slope[0])
    noise_var = float(measured_var[0])
    output[0] = level

    for index in range(1, count):
        local_scale = np.sqrt(max(noise_var, measured_var[index], 1e-10))
        predicted_level = level + velocity
        innovation = measured_level[index] - predicted_level

        # Robust innovation treatment prevents an endpoint outlier from
        # introducing a temporary direction change.
        innovation_limit = 3.2 * local_scale + 0.15 * abs(measured_slope[index])
        bounded_innovation = np.clip(
            innovation, -innovation_limit, innovation_limit
        )

        trend_confidence = min(float(confidence[index]), 2.5) / 2.5
        transition = min(
            abs(innovation) / (2.0 * local_scale + 1e-12),
            1.0,
        )

        # High-confidence trends use a high alpha for near-zero lag. Flat or
        # uncertain intervals retain more smoothing.
        alpha = 0.58 + 0.25 * trend_confidence + 0.11 * transition
        level = predicted_level + min(alpha, 0.94) * bounded_innovation

        desired_velocity = (
            0.58 * velocity
            + 0.27 * measured_slope[index]
            + (0.12 + 0.15 * trend_confidence) * bounded_innovation
        )

        # Bounded acceleration rejects the rapid velocity alternation typical
        # of high-frequency measurement noise.
        acceleration_limit = (
            0.13 * local_scale
            + 0.32 * abs(measured_slope[index])
            + 0.015 * abs(velocity)
        )
        velocity_change = np.clip(
            desired_velocity - velocity,
            -acceleration_limit,
            acceleration_limit,
        )
        candidate_velocity = velocity + velocity_change

        # Require stronger evidence to reverse an established direction than
        # to continue it. Genuine sustained reversals still pass quickly as
        # the endpoint regression slope and innovation agree.
        reversal_floor = 0.045 * local_scale + 0.12 * abs(measured_slope[index])
        if (
            velocity * candidate_velocity < 0.0
            and abs(candidate_velocity) < reversal_floor
            and np.sign(measured_slope[index]) != np.sign(candidate_velocity)
        ):
            candidate_velocity = 0.0

        deadband = 0.018 * local_scale
        velocity = 0.0 if abs(candidate_velocity) < deadband else candidate_velocity

        squared_innovation = bounded_innovation * bounded_innovation
        noise_var = max(
            0.94 * noise_var + 0.04 * squared_innovation + 0.02 * measured_var[index],
            1e-10,
        )
        output[index] = level

    return output


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Main signal processing function that applies the selected algorithm.

    Args:
        input_signal: Input time series data
        window_size: Window size for processing
        algorithm_type: Type of algorithm to use ("basic" or "enhanced")

    Returns:
        Filtered signal
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the generated signal
        noise_level: Standard deviation of additive noise
        seed: Random seed for reproducibility

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise
    return noisy_signal, clean_signal


def run_signal_processing(
    noisy_signal=None,
    signal_length=1000,
    noise_level=0.3,
    window_size=20,
):
    """
    Run the signal processing algorithm on either supplied or generated data.

    Returns:
        Dictionary containing filtered output and available quality metrics.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(
            signal_length, noise_level
        )
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1
            else 0
        )
        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0
            else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
