"""
Causal multi-scale robust local-polynomial trend filtering.

The estimator uses endpoint-local polynomial regression rather than recursive
state estimation.  A short robust fit preserves fast local movement, a longer
robust fit rejects noise, and a slope-coherence controller prevents isolated
noise samples from creating false directional reversals.
"""
import numpy as np


def _mad_scale(values, fallback=1e-6):
    """Robust standard-deviation estimate."""
    values = np.asarray(values, dtype=float)
    values = values[np.isfinite(values)]

    if values.size == 0:
        return float(fallback)

    center = np.median(values)
    scale = 1.4826 * np.median(np.abs(values - center))

    if not np.isfinite(scale) or scale < 1e-10:
        scale = np.std(values)

    if not np.isfinite(scale) or scale < 1e-10:
        scale = fallback

    return float(max(scale, 1e-10))


def _robust_endpoint_fit(samples, recency_strength=2.0):
    """
    Robust weighted local polynomial fit evaluated at the newest sample.

    Returns:
        endpoint estimate, per-sample slope, residual noise scale
    """
    samples = np.asarray(samples, dtype=float)
    n = len(samples)
    valid = np.isfinite(samples)

    if not np.any(valid):
        return np.nan, 0.0, 1e-6

    values = samples[valid]

    if n == 1 or values.size == 1:
        return float(values[-1]), 0.0, 1e-6

    # Time coordinates are normalized so polynomial conditioning is stable.
    span = float(max(n - 1, 1))
    time = (np.arange(n, dtype=float) - (n - 1)) / span
    time = time[valid]

    # Linear fit for two observations; quadratic endpoint model otherwise.
    degree = 2 if values.size >= 3 else 1
    design = np.vander(time, degree + 1, increasing=True)

    # Endpoint-weighted regression has no group delay while still allowing
    # older samples to reject measurement noise.
    base_weights = np.exp(recency_strength * time)
    weights = base_weights.copy()

    coefficients = np.zeros(degree + 1, dtype=float)
    residual_scale = _mad_scale(np.diff(values), fallback=1e-5) / np.sqrt(2.0)

    # A small bounded IRLS loop provides Huber robustness at low latency.
    for _ in range(3):
        root_weights = np.sqrt(np.maximum(weights, 1e-12))
        weighted_design = design * root_weights[:, None]
        weighted_values = values * root_weights

        try:
            coefficients = np.linalg.lstsq(
                weighted_design, weighted_values, rcond=None
            )[0]
        except np.linalg.LinAlgError:
            coefficients[0] = values[-1]
            coefficients[1:] = 0.0
            break

        residuals = values - design.dot(coefficients)
        residual_scale = _mad_scale(residuals, fallback=residual_scale)

        huber_limit = max(1.5 * residual_scale, 1e-8)
        robust_weights = np.minimum(
            1.0, huber_limit / np.maximum(np.abs(residuals), 1e-12)
        )
        weights = base_weights * robust_weights

    endpoint = float(coefficients[0])
    slope = float(coefficients[1] / span) if degree >= 1 else 0.0
    return endpoint, slope, residual_scale


def adaptive_filter(x, window_size=20):
    """
    Causal multi-scale robust endpoint trend filter.

    Args:
        x: Input signal as a one-dimensional real-valued array.
        window_size: Number of historical observations required before output.

    Returns:
        Filtered endpoint estimates with length len(x) - window_size + 1.
    """
    x = np.asarray(x, dtype=float)

    if x.ndim != 1:
        raise ValueError("Input signal must be a 1D array")
    if window_size < 1:
        raise ValueError("window_size must be >= 1")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )

    n = len(x)
    output_length = n - window_size + 1

    if not np.any(np.isfinite(x)):
        return np.full(output_length, np.nan)

    # The short scale retains maneuver responsiveness.  The long scale provides
    # a stable trend and robust local noise estimate.
    short_size = max(3, min(window_size, window_size // 2 + 1))
    estimates = np.empty(output_length, dtype=float)

    previous_output = np.nan
    previous_direction = 0.0
    pending_direction = 0.0
    pending_count = 0
    running_noise = None

    for output_index, endpoint_index in enumerate(range(window_size - 1, n)):
        long_window = x[endpoint_index - window_size + 1 : endpoint_index + 1]
        short_window = x[endpoint_index - short_size + 1 : endpoint_index + 1]

        long_level, long_slope, long_noise = _robust_endpoint_fit(
            long_window, recency_strength=1.35
        )
        short_level, short_slope, short_noise = _robust_endpoint_fit(
            short_window, recency_strength=2.35
        )

        if not np.isfinite(long_level):
            candidate = short_level
            local_noise = short_noise
        elif not np.isfinite(short_level):
            candidate = long_level
            local_noise = long_noise
        else:
            local_noise = max(long_noise, short_noise, 1e-8)

            # Coherent short/long slopes indicate a genuine move.  Conflicting
            # slopes usually occur near noise excursions, so favor the stable
            # long-scale estimate in that case.
            slope_product = short_slope * long_slope
            coherent = slope_product > 0.0
            slope_strength = abs(short_slope) / max(local_noise, 1e-8)

            if coherent:
                short_weight = 0.58 + 0.22 * min(1.0, slope_strength / 1.5)
            else:
                short_weight = 0.25

            # A large agreement between estimates supports rapid adaptation;
            # otherwise retain more of the low-noise long trend.
            separation = abs(short_level - long_level) / max(local_noise, 1e-8)
            if coherent and separation > 1.5:
                short_weight = min(0.87, short_weight + 0.08)

            candidate = short_weight * short_level + (1.0 - short_weight) * long_level

        if not np.isfinite(candidate):
            candidate = previous_output if np.isfinite(previous_output) else 0.0

        if running_noise is None:
            running_noise = local_noise
        else:
            # Slow adaptation prevents short maneuvers from being classified as
            # permanent noise changes.
            running_noise = 0.92 * running_noise + 0.08 * local_noise

        if not np.isfinite(previous_output):
            filtered = candidate
            delta = 0.0
        else:
            delta = candidate - previous_output
            reversal_threshold = max(0.16 * running_noise, 1e-8)
            direction = np.sign(delta) if abs(delta) >= reversal_threshold else 0.0

            is_reversal = (
                direction != 0.0
                and previous_direction != 0.0
                and direction != previous_direction
            )

            if is_reversal:
                if direction == pending_direction:
                    pending_count += 1
                else:
                    pending_direction = direction
                    pending_count = 1

                # Require confirmation for weak turns. Strong moves bypass most
                # of the hold so real turning points retain low lag.
                strong_turn = abs(delta) > 2.4 * reversal_threshold

                if pending_count >= 2 or strong_turn:
                    filtered = 0.78 * candidate + 0.22 * previous_output
                    previous_direction = direction
                    pending_direction = 0.0
                    pending_count = 0
                else:
                    # Partial motion avoids a visible flat plateau while making
                    # a one-sample noise reversal too small to count as a turn.
                    filtered = previous_output + 0.22 * delta
            else:
                pending_direction = 0.0
                pending_count = 0

                # Mild deadband removes sub-noise zig-zag without suppressing
                # meaningful high-frequency components.
                if direction == 0.0:
                    filtered = previous_output + 0.18 * delta
                else:
                    filtered = candidate
                    previous_direction = direction

        estimates[output_index] = filtered
        previous_output = filtered

    return estimates


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """Backward-compatible enhanced public filtering entry point."""
    return adaptive_filter(x, window_size)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Apply the selected filtering algorithm.

    Args:
        input_signal: Input time-series data.
        window_size: Required trailing history.
        algorithm_type: "basic" or "enhanced".

    Returns:
        Filtered signal with length len(input_signal) - window_size + 1.
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal
        noise_level: Standard deviation of noise to add
        seed: Random seed for reproducibility

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000, noise_level=0.3, window_size=20):
    """
    Run the signal processing algorithm on a test signal.

    Args:
        noisy_signal: Optional input signal to filter.
        signal_length: Generated signal length when noisy_signal is omitted.
        noise_level: Generated signal noise level.
        window_size: Processing window length.

    Returns:
        Dictionary containing filtered data and available metrics.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1
            else 0
        )

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0
            else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
