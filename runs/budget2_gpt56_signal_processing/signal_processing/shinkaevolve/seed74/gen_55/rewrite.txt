"""
Robust multi-scale causal trend tracker for volatile non-stationary signals.

The enhanced filter combines robust full-window and recent-window endpoint
regressions with a confidence-adaptive alpha-beta state observer.  A dedicated
reversal confirmation stage reduces noise-induced slope changes while retaining
a fast bypass for large, coherent trend reversals.
"""
import numpy as np


def adaptive_filter(x, window_size=20):
    """
    Endpoint-aligned trailing moving average baseline.

    Args:
        x: One-dimensional real-valued input signal.
        window_size: Trailing averaging window.

    Returns:
        Filtered signal of length len(x) - window_size + 1.
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be one-dimensional")
    if window_size < 1:
        raise ValueError("window_size must be at least 1")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )
    if not np.all(np.isfinite(x)):
        raise ValueError("Input signal must contain only finite values")

    cumulative = np.concatenate(([0.0], np.cumsum(x)))
    return (cumulative[window_size:] - cumulative[:-window_size]) / window_size


def _regression_operators(size, decay):
    """Create exponentially weighted local-linear regression operators."""
    time = np.arange(size, dtype=float) - float(size - 1)
    weights = np.exp(np.linspace(-decay, 0.0, size))
    weights /= np.sum(weights)

    mean_time = np.dot(weights, time)
    centered_time = time - mean_time
    variance = max(np.dot(weights, centered_time * centered_time), 1e-12)
    return time, weights, mean_time, centered_time, variance


def _weighted_endpoint_fit(values, time, weights):
    """Fit a weighted line and return endpoint level, slope, and residuals."""
    weight_sum = max(np.sum(weights), 1e-12)
    mean_time = np.dot(weights, time) / weight_sum
    mean_value = np.dot(weights, values) / weight_sum
    centered_time = time - mean_time
    variance = max(np.dot(weights, centered_time * centered_time), 1e-12)

    slope = np.dot(weights * centered_time, values - mean_value) / variance
    level = mean_value - slope * mean_time
    residuals = values - (level + slope * time)
    return level, slope, residuals


def _robust_endpoint_fit(values, time, base_weights, presentation_direction=0.0):
    """
    One Huber reweighting pass plus endpoint contradiction protection.

    The newest sample receives extra downweighting only when it is an extreme
    residual that opposes both the fitted local slope and established output
    direction.  This avoids treating genuine accelerating turns as outliers.
    """
    level, slope, residuals = _weighted_endpoint_fit(values, time, base_weights)

    residual_center = np.median(residuals)
    scale = max(
        1.4826 * np.median(np.abs(residuals - residual_center)),
        1e-8,
    )

    # Single robust residual-reweighting pass.
    huber_limit = 2.30 * scale
    huber = np.minimum(
        1.0,
        huber_limit / np.maximum(np.abs(residuals - residual_center), 1e-8),
    )
    robust_weights = base_weights * huber

    # Isolated newest endpoint spikes are particularly harmful because an
    # endpoint observer otherwise interprets them as immediate acceleration.
    newest_residual = residuals[-1] - residual_center
    newest_opposes_slope = newest_residual * slope < 0.0
    newest_opposes_direction = (
        presentation_direction != 0.0
        and newest_residual * presentation_direction < 0.0
    )
    if (
        abs(newest_residual) > 3.15 * scale
        and newest_opposes_slope
        and newest_opposes_direction
    ):
        robust_weights[-1] *= 0.22

    # The penultimate point is reduced only for a paired contradictory spike;
    # normal recent samples retain authority for low-lag tracking.
    if len(values) >= 6:
        penultimate_residual = residuals[-2] - residual_center
        if (
            abs(penultimate_residual) > 3.35 * scale
            and penultimate_residual * slope < 0.0
            and presentation_direction != 0.0
            and penultimate_residual * presentation_direction < 0.0
        ):
            robust_weights[-2] *= 0.52

    level, slope, robust_residuals = _weighted_endpoint_fit(
        values, time, robust_weights
    )
    robust_center = np.median(robust_residuals)
    robust_scale = max(
        1.4826 * np.median(np.abs(robust_residuals - robust_center)),
        1e-8,
    )
    return level, slope, robust_scale


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Causal robust multi-scale endpoint trend tracker.

    Args:
        x: One-dimensional real-valued input signal.
        window_size: Required trailing history length, at least 3.

    Returns:
        Endpoint-aligned filtered signal with length len(x)-window_size+1.
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be one-dimensional")
    if window_size < 3:
        raise ValueError("window_size must be at least 3")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )
    if not np.all(np.isfinite(x)):
        raise ValueError("Input signal must contain only finite values")

    output_length = len(x) - window_size + 1
    y = np.empty(output_length, dtype=float)

    # Long horizon is intentionally less recency-biased than the old 2.85
    # configuration. This stabilizes the consensus slope and reduces chatter.
    full_time, full_weights, _, _, _ = _regression_operators(window_size, 2.05)

    # Recent horizon supplies turn responsiveness without allowing a single
    # shortest-scale slope estimate to directly alter the output state.
    recent_size = min(window_size, max(6, int(round(0.45 * window_size))))
    recent_time, recent_weights, _, _, _ = _regression_operators(
        recent_size, 1.05
    )

    position = 0.0
    velocity = 0.0
    confirmed_direction = 0.0
    pending_direction = 0.0
    pending_count = 0

    for i in range(output_length):
        window = x[i:i + window_size]
        recent = window[-recent_size:]

        full_level, full_slope, full_noise = _robust_endpoint_fit(
            window, full_time, full_weights, confirmed_direction
        )
        recent_level, recent_slope, recent_noise = _robust_endpoint_fit(
            recent, recent_time, recent_weights, confirmed_direction
        )

        noise_scale = max(
            min(full_noise, recent_noise),
            0.20 * max(full_noise, recent_noise),
            1e-8,
        )
        slope_floor = max(0.060 * noise_scale, 1e-10)

        full_direction = (
            np.sign(full_slope) if abs(full_slope) > slope_floor else 0.0
        )
        recent_direction = (
            np.sign(recent_slope) if abs(recent_slope) > slope_floor else 0.0
        )
        slopes_agree = (
            full_direction != 0.0
            and full_direction == recent_direction
        )

        slope_ratio = min(abs(full_slope), abs(recent_slope)) / max(
            abs(full_slope), abs(recent_slope), slope_floor
        )
        agreement = slope_ratio if slopes_agree else 0.0

        # Endpoint disagreement detects curvature/model staleness. Short-horizon
        # authority is bounded to retain the noise suppression of the long fit.
        endpoint_gap = abs(recent_level - full_level) / noise_scale
        curvature = endpoint_gap / (1.0 + endpoint_gap)
        short_weight = 0.25 + 0.36 * curvature
        if slopes_agree:
            short_weight *= 0.82
        short_weight = float(np.clip(short_weight, 0.22, 0.57))

        measurement = (
            (1.0 - short_weight) * full_level + short_weight * recent_level
        )
        slope_measurement = 0.68 * full_slope + 0.32 * recent_slope

        if i == 0:
            position = measurement
            velocity = slope_measurement
            confirmed_direction = (
                np.sign(velocity) if abs(velocity) > slope_floor else 0.0
            )
            y[i] = position
            continue

        prediction = position + velocity
        innovation = measurement - prediction
        activity = min(abs(innovation) / (2.35 * noise_scale), 1.0)
        bounded_innovation = np.clip(
            innovation,
            -3.0 * noise_scale,
            3.0 * noise_scale,
        )

        # Conservative ordinary gains reduce output slope chatter. Coherent
        # multi-scale motion and curvature progressively recover responsiveness.
        alpha = 0.15 + 0.31 * activity + 0.13 * agreement * activity
        alpha += 0.08 * curvature * activity
        alpha = float(np.clip(alpha, 0.15, 0.62))

        velocity_blend = 0.13 + 0.19 * agreement + 0.10 * activity
        velocity_blend += 0.07 * curvature * activity
        velocity_blend = float(np.clip(velocity_blend, 0.12, 0.47))

        candidate_velocity = (
            (1.0 - velocity_blend) * velocity
            + velocity_blend * slope_measurement
            + 0.040 * bounded_innovation
        )
        candidate_direction = (
            np.sign(candidate_velocity)
            if abs(candidate_velocity) > 0.075 * noise_scale
            else 0.0
        )

        is_reversal = (
            confirmed_direction != 0.0
            and candidate_direction != 0.0
            and candidate_direction != confirmed_direction
        )
        turn_evidence = (
            is_reversal
            and recent_direction == candidate_direction
            and abs(recent_slope) > 0.095 * noise_scale
            and abs(innovation) > 0.85 * noise_scale
        )
        decisive_turn = (
            turn_evidence
            and abs(innovation) > 2.45 * noise_scale
            and abs(recent_slope) > 0.17 * noise_scale
        )

        if is_reversal:
            if turn_evidence and pending_direction == candidate_direction:
                pending_count += 1
            elif turn_evidence:
                pending_direction = candidate_direction
                pending_count = 1
            else:
                pending_direction = 0.0
                pending_count = 0

            if pending_count < 2 and not decisive_turn:
                # Keep a small directional correction but prevent one noisy
                # endpoint estimate from visibly reversing the presentation.
                candidate_velocity = 0.88 * velocity
                alpha = min(alpha, 0.31)
            else:
                confirmed_direction = candidate_direction
                pending_direction = 0.0
                pending_count = 0
                candidate_velocity = 0.30 * velocity + 0.70 * candidate_velocity
                alpha = max(alpha, 0.40)
        else:
            pending_direction = 0.0
            pending_count = 0
            if candidate_direction != 0.0:
                confirmed_direction = candidate_direction

        position = prediction + alpha * bounded_innovation
        velocity = candidate_velocity
        y[i] = position

    return y


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Apply either the trailing-average baseline or enhanced causal tracker.

    Args:
        input_signal: Input one-dimensional time series.
        window_size: Size of trailing history.
        algorithm_type: "basic" or "enhanced".

    Returns:
        Filtered endpoint-aligned signal.
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)
