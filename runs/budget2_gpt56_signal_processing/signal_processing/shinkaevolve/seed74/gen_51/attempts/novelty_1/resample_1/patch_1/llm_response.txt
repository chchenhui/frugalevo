        normalized = abs(raw_innovation) / max(noise, 1e-8)
        maneuver = np.clip((normalized - 0.8) / 2.5, 0.0, 1.0)

        # A short endpoint slope is deliberately used only to control internal
        # Kalman adaptation, never blended into the visible output. Agreement
        # across short, long, and state horizons is strong evidence of genuine
        # motion; short-horizon disagreement is typical endpoint noise.
        def endpoint_slope(length):
            start = max(0, index - length + 1)
            values = x[start:index + 1]
            time = np.arange(len(values), dtype=float)
            centered_time = time - np.mean(time)
            denominator = np.dot(centered_time, centered_time)
            if denominator <= 1e-12:
                return 0.0
            return float(
                np.dot(centered_time, values - np.mean(values)) / denominator
            )
