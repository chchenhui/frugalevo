Replace the fixed exponential moving average with a causal, robust local-linear estimator followed by an adaptive alpha-beta tracking stage. The local regression evaluates the trend at the newest sample, avoiding the window-centering lag of a moving average. Huber reweighting suppresses impulsive samples, while the innovation-dependent gain increases during real directional movement and decreases during stationary noisy periods. A small slope dead-band prevents noise-scale sign changes from becoming false reversals without freezing meaningful trends. The output contract remains unchanged: one filtered value per complete window, aligned to each window's newest sample.

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Causal robust local-trend filter with adaptive predictive tracking.

    Each output is aligned with the newest sample in its window.  A robust
    exponentially weighted line fit provides a low-lag observation, then an
    alpha-beta style predictor rejects innovations consistent with local noise
    while responding quickly to sustained trend changes.
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be a 1D array")
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")
    if window_size < 2:
        raise ValueError("window_size must be at least 2")

    output_length = len(x) - window_size + 1
    y = np.empty(output_length)

    # Time is measured backwards from the current sample, so the fitted
    # intercept is the estimate at the newest point and has no centering lag.
    time = np.arange(1 - window_size, 1, dtype=float)
    base_weights = np.exp(time / max(window_size * 0.35, 1.0))
    base_weights /= base_weights.sum()

    level = 0.0
    velocity = 0.0
    initialized = False

    for i in range(output_length):
        window = x[i : i + window_size]

        # Initial weighted local-linear fit.
        weight_sum = base_weights.sum()
        time_mean = np.dot(base_weights, time) / weight_sum
        value_mean = np.dot(base_weights, window) / weight_sum
        centered_time = time - time_mean
        denominator = np.dot(base_weights, centered_time * centered_time)
        slope = np.dot(base_weights, centered_time * (window - value_mean)) / denominator
        observation = value_mean - slope * time_mean

        # One Huber reweighting pass makes the endpoint estimate insensitive to
        # isolated spikes, a common source of false directional reversals.
        residuals = window - (observation + slope * time)
        scale = 1.4826 * np.median(np.abs(residuals - np.median(residuals))) + 1e-8
        huber = np.minimum(1.0, 2.5 * scale / (np.abs(residuals) + 1e-12))
        weights = base_weights * huber
        weight_sum = weights.sum()
        time_mean = np.dot(weights, time) / weight_sum
        value_mean = np.dot(weights, window) / weight_sum
        centered_time = time - time_mean
        denominator = np.dot(weights, centered_time * centered_time)
        slope = np.dot(weights, centered_time * (window - value_mean)) / denominator
        observation = value_mean - slope * time_mean

        if not initialized:
            level = observation
            velocity = slope
            initialized = True
            y[i] = level
            continue

        prediction = level + velocity
        innovation = observation - prediction

        # Gain rises for genuine movement but remains bounded to avoid tracking
        # individual noisy samples.  The local residual scale adapts it to
        # non-stationary noise levels.
        dynamic_scale = abs(slope) + 0.35 * abs(innovation)
        gain = dynamic_scale / (dynamic_scale + 1.5 * scale + 1e-12)
        gain = np.clip(gain, 0.12, 0.78)

        level = prediction + gain * innovation
        proposed_velocity = 0.72 * velocity + 0.28 * slope + 0.18 * gain * innovation

        # Do not accept a tiny velocity sign flip unless it exceeds the local
        # noise-derived dead-band; this specifically penalizes false reversals.
        dead_band = 0.20 * scale
        if velocity * proposed_velocity < 0 and abs(proposed_velocity) < dead_band:
            proposed_velocity = 0.0
        velocity = proposed_velocity
        y[i] = level

    return y
