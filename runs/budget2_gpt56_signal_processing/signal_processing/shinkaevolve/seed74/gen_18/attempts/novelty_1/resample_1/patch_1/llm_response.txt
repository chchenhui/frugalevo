<DESCRIPTION>
The current implementation uses an adaptive alpha-beta tracker with alpha values of 0.24–0.52, beta values of 0.035–0.105, a 3-sigma innovation clip, and only a small `0.12 * noise_scale` slope deadband. This produces good correlation and lag, but the gain can become high even when recent directional evidence is inconsistent, resulting in excessive slope changes and false reversals.

The new implementation combines a robust causal alpha-beta tracker with a short-horizon/full-window trend-consistency gate. It estimates slopes using weighted local linear regression over both the full window and a recent subwindow. High gains are enabled only when both slope estimates agree in direction and have comparable magnitude. Otherwise, level gain is capped near 0.38 and velocity gain near 0.045. Reversals additionally require two consecutive significant, coherent counter-trend observations. A rolling MAD estimate based on first differences provides stable noise-scale bootstrap and robust innovation clipping. This reduces noise-induced direction changes while preserving responsive tracking during genuine trend transitions.

"""
Real-Time Adaptive Signal Processing Algorithm for Non-Stationary Time Series.

This version uses a causal robust alpha-beta trend tracker with:
  * rolling MAD noise estimation from first differences,
  * weighted full-window and recent-window slope estimates,
  * trend-consistency gated gains,
  * innovation clipping for impulse rejection, and
  * two-sample reversal confirmation.

The output remains aligned with input samples starting at window_size - 1.
"""
import numpy as np


def adaptive_filter(x, window_size=20):
    """
    Simple moving-average baseline.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Sliding-window length

    Returns:
        Filtered output with length len(x) - window_size + 1.
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be one-dimensional")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )

    kernel = np.ones(window_size, dtype=float) / window_size
    return np.convolve(x, kernel, mode="valid")


def _weighted_endpoint_slope(values):
    """
    Causal weighted least-squares slope estimate.

    Recent samples receive larger weights. The returned slope is per sample.
    """
    n = len(values)
    if n < 2:
        return 0.0

    t = np.arange(n, dtype=float)
    # Moderate exponential recency weighting; avoids the high variance caused
    # by relying only on the final one or two observations.
    weights = np.exp(np.linspace(-1.4, 0.0, n))
    weight_sum = np.sum(weights)

    t_mean = np.sum(weights * t) / weight_sum
    v_mean = np.sum(weights * values) / weight_sum

    centered_t = t - t_mean
    denominator = np.sum(weights * centered_t * centered_t)
    if denominator <= 1e-12:
        return 0.0

    return np.sum(weights * centered_t * (values - v_mean)) / denominator


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Robust trend-consistent causal alpha-beta filter.

    Key parameters:
        alpha_low=0.20, alpha_incoherent=0.38, alpha_high=0.56
            Level gains. High gain is permitted only for coherent trends.
        beta_low=0.012, beta_incoherent=0.045, beta_high=0.105
            Slope adaptation gains.
        innovation_clip=3.25 * noise_scale
            Robust outlier limit.
        reversal_confirmations=2
            Number of coherent counter-trend observations required before
            permitting a velocity sign reversal.
        deadband=0.16 * noise_scale
            Suppresses insignificant velocity fluctuations.
        recent_window=max(5, window_size // 3)
            Short horizon used with the full window to verify trend direction.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Rolling analysis/noise-estimation window

    Returns:
        Filtered samples aligned with input samples window_size - 1 onward.
    """
    x = np.asarray(x, dtype=float)

    if x.ndim != 1:
        raise ValueError("Input signal must be one-dimensional")
    if window_size < 2:
        raise ValueError("window_size must be at least 2")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )

    n = len(x)
    estimates = np.empty(n, dtype=float)

    # Parameters tuned to prefer smooth directional behavior unless independent
    # short and long horizon trend measurements support a fast update.
    alpha_low = 0.20
    alpha_incoherent = 0.38
    alpha_high = 0.56

    beta_low = 0.012
    beta_incoherent = 0.045
    beta_high = 0.105

    clip_multiplier = 3.25
    deadband_multiplier = 0.16
    reversal_confirmations = 2
    recent_window = max(5, min(10, window_size // 3))

    level = float(x[0])
    slope = 0.0
    pending_direction = 0.0
    pending_count = 0
    estimates[0] = level

    differences = np.diff(x)

    for i in range(1, n):
        history_start = max(0, i - window_size + 1)
        history = x[history_start : i + 1]

        # Difference MAD avoids the near-zero residual-scale startup problem
        # of the previous tracker. Dividing by sqrt(2) converts difference
        # variation approximately back to sample-domain noise variation.
        diff_start = max(0, i - window_size)
        recent_diffs = differences[diff_start:i]

        if len(recent_diffs) >= 3:
            diff_median = np.median(recent_diffs)
            diff_mad = np.median(np.abs(recent_diffs - diff_median))
            noise_scale = 1.4826 * diff_mad / np.sqrt(2.0)
        else:
            noise_scale = np.std(history) if len(history) > 1 else 0.0

        # A small scale floor handles locally constant signals safely without
        # forcing a large gain in quiet regions.
        local_amplitude = np.median(np.abs(history - np.median(history)))
        noise_scale = max(noise_scale, 0.03 * local_amplitude, 1e-6)

        full_slope = _weighted_endpoint_slope(history)
        short_history = history[-min(recent_window, len(history)) :]
        recent_slope = _weighted_endpoint_slope(short_history)

        # Trend consistency is the central gain gate. Slopes must agree in
        # sign and neither estimate can be dramatically smaller than the other.
        full_direction = np.sign(full_slope)
        recent_direction = np.sign(recent_slope)
        slope_floor = 0.055 * noise_scale

        both_significant = (
            abs(full_slope) > slope_floor and abs(recent_slope) > slope_floor
        )
        same_direction = (
            full_direction != 0.0 and full_direction == recent_direction
        )

        slope_ratio = min(abs(full_slope), abs(recent_slope)) / (
            max(abs(full_slope), abs(recent_slope)) + 1e-12
        )
        coherent_trend = both_significant and same_direction and slope_ratio >= 0.35

        predicted_level = level + slope
        innovation = x[i] - predicted_level
        normalized_innovation = abs(innovation) / noise_scale
        clipped_innovation = np.clip(
            innovation,
            -clip_multiplier * noise_scale,
            clip_multiplier * noise_scale,
        )

        # Strong innovation is useful only if trend evidence is coherent.
        # Incoherent evidence receives a capped gain, reducing noise chasing.
        if coherent_trend:
            strength = min(1.0, normalized_innovation / 2.5)
            alpha = alpha_incoherent + (alpha_high - alpha_incoherent) * strength
            beta = beta_incoherent + (beta_high - beta_incoherent) * strength
            reference_slope = 0.55 * recent_slope + 0.45 * full_slope
        else:
            strength = min(1.0, normalized_innovation / 3.5)
            alpha = alpha_low + (alpha_incoherent - alpha_low) * strength
            beta = beta_low + (beta_incoherent - beta_low) * strength
            reference_slope = full_slope

        level = predicted_level + alpha * clipped_innovation

        # Blend model-derived slope and innovation-derived velocity update.
        innovation_slope = slope + beta * clipped_innovation
        proposed_slope = 0.62 * innovation_slope + 0.38 * reference_slope
        slope_deadband = deadband_multiplier * noise_scale

        old_direction = np.sign(slope)
        new_direction = np.sign(proposed_slope)
        is_reversal = (
            old_direction != 0.0
            and new_direction != 0.0
            and new_direction != old_direction
            and abs(proposed_slope) >= slope_deadband
        )

        # Counter-trend motion must be independently supported by both local
        # regressions and persist for two samples before changing direction.
        reversal_evidence = (
            is_reversal
            and coherent_trend
            and np.sign(clipped_innovation) == new_direction
        )

        if reversal_evidence:
            if new_direction == pending_direction:
                pending_count += 1
            else:
                pending_direction = new_direction
                pending_count = 1

            if pending_count >= reversal_confirmations:
                # A confirmed turn changes velocity promptly enough to avoid
                # excessive lag, but does not inject a full noisy slope jump.
                slope = 0.42 * slope + 0.58 * proposed_slope
                pending_direction = 0.0
                pending_count = 0
            else:
                # Preserve existing direction during one-sample counter noise.
                slope *= 0.985
        else:
            pending_direction = 0.0
            pending_count = 0

            if abs(proposed_slope) < slope_deadband:
                proposed_slope = 0.0

            # More smoothing when trend estimates disagree; faster velocity
            # tracking when both horizons verify the same direction.
            slope_blend = 0.34 if coherent_trend else 0.16
            slope = (1.0 - slope_blend) * slope + slope_blend * proposed_slope

        estimates[i] = level

    return estimates[window_size - 1 :]


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Main signal processing function.

    Args:
        input_signal: Input time series data
        window_size: Processing window size
        algorithm_type: "basic" or "enhanced"

    Returns:
        Filtered signal.
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Number of samples
        noise_level: Additive noise standard deviation
        seed: Random seed

    Returns:
        Tuple of (noisy_signal, clean_signal).
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(
    noisy_signal=None,
    signal_length=1000,
    noise_level=0.3,
    window_size=20,
):
    """
    Run the adaptive signal processing algorithm.

    Args:
        noisy_signal: Optional externally supplied signal
        signal_length: Generated signal length when noisy_signal is None
        noise_level: Generated signal noise level
        window_size: Filter window size

    Returns:
        Dictionary containing filtered data and metrics.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1
            else 0
        )

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0
            else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
