<DESCRIPTION>
This design replaces the state-space Kalman recursion with a causal, multi-scale robust endpoint-regression filter. It draws on the prior programs’ useful low-lag principle—emphasizing the newest observations—but avoids a single adaptive tracker whose innovation response can create excessive slope reversals.

For each completed window, the algorithm fits two robust local linear models: a short-horizon model for responsiveness and a longer-horizon model for noise suppression. Their endpoint estimates are fused according to slope agreement: when both horizons indicate the same direction, the short model receives more weight; when they disagree, the longer model stabilizes the estimate. A MAD/Huber reweighting pass limits impulses and outliers. Finally, an adaptive reversal deadband rejects small direction changes that are below the locally estimated residual noise level, reducing false reversals without introducing moving-average phase delay. Outputs remain causally aligned with the newest sample in each window and retain the original public API and output lengths.

"""
Causal multi-scale robust endpoint regression for volatile non-stationary signals.

The filter emits one estimate for each complete input window.  Each estimate is
aligned with the newest observation in its window, preserving the original API
and alignment convention while minimizing phase delay.
"""
import numpy as np


def _robust_scale(values, fallback=1e-6):
    """Return a finite MAD-based scale estimate."""
    values = np.asarray(values, dtype=float)
    values = values[np.isfinite(values)]

    if values.size == 0:
        return float(fallback)

    center = np.median(values)
    scale = 1.4826 * np.median(np.abs(values - center))

    if not np.isfinite(scale) or scale < 1e-10:
        scale = np.std(values)

    if not np.isfinite(scale) or scale < 1e-10:
        scale = max(float(fallback), 1e-6)

    return float(scale)


def _robust_endpoint_fit(window, decay, fallback_scale):
    """
    Causal exponentially weighted robust local-linear endpoint fit.

    Returns:
        endpoint estimate, local slope, residual scale
    """
    window = np.asarray(window, dtype=float)
    length = len(window)
    valid = np.isfinite(window)

    if not np.any(valid):
        return np.nan, 0.0, float(fallback_scale)

    if np.count_nonzero(valid) == 1:
        value = float(window[valid][0])
        return value, 0.0, float(fallback_scale)

    # Time is referenced to the current endpoint, so intercept is directly the
    # zero-delay estimate for the newest sample in the window.
    t_full = np.arange(length, dtype=float) - float(length - 1)
    t = t_full[valid]
    z = window[valid]

    tau = max(float(decay), 1.0)
    weights = np.exp(t / tau)
    weights /= max(np.sum(weights), 1e-12)

    def solve(current_weights):
        sw = np.sum(current_weights)
        st = np.sum(current_weights * t)
        stt = np.sum(current_weights * t * t)
        sz = np.sum(current_weights * z)
        stz = np.sum(current_weights * t * z)

        determinant = sw * stt - st * st
        if determinant <= 1e-14:
            return float(np.average(z, weights=current_weights)), 0.0

        intercept = (stt * sz - st * stz) / determinant
        slope = (sw * stz - st * sz) / determinant
        return float(intercept), float(slope)

    intercept, slope = solve(weights)

    # One Huber IRLS pass captures most outlier resistance at very low cost.
    residual = z - (intercept + slope * t)
    scale = _robust_scale(residual, fallback=fallback_scale)
    huber_limit = max(1.75 * scale, 1e-10)
    huber_weights = np.minimum(1.0, huber_limit / np.maximum(np.abs(residual), 1e-12))

    intercept, slope = solve(weights * huber_weights)
    residual = z - (intercept + slope * t)
    scale = _robust_scale(residual, fallback=fallback_scale)

    return intercept, slope, scale


def adaptive_filter(x, window_size=20):
    """
    Robust multi-scale causal endpoint filter.

    Args:
        x: Input one-dimensional real-valued signal.
        window_size: Required history before emitting the first output.

    Returns:
        Filtered endpoint estimates of length len(x) - window_size + 1.
    """
    x = np.asarray(x, dtype=float)

    if x.ndim != 1:
        raise ValueError("Input signal must be a 1D array")
    if window_size < 1:
        raise ValueError("window_size must be >= 1")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )

    n = len(x)
    output_length = n - window_size + 1

    if not np.any(np.isfinite(x)):
        return np.full(output_length, np.nan)

    if window_size == 1:
        return x.copy()

    finite_x = x[np.isfinite(x)]
    if finite_x.size > 2:
        global_noise = _robust_scale(np.diff(finite_x), fallback=1e-5) / np.sqrt(2.0)
    else:
        global_noise = _robust_scale(finite_x, fallback=1e-5)

    global_noise = max(float(global_noise), 1e-6)

    # The shorter model preserves fast local dynamics.  The long model supplies
    # a stable consensus whenever short-scale behavior is ambiguous.
    short_size = max(4, min(window_size, int(np.ceil(window_size * 0.52))))
    short_decay = max(1.5, 0.34 * short_size)
    long_decay = max(2.0, 0.42 * window_size)

    y = np.empty(output_length, dtype=float)
    previous = np.nan
    accepted_direction = 0.0

    for out_index, endpoint in enumerate(range(window_size - 1, n)):
        long_window = x[endpoint - window_size + 1:endpoint + 1]
        short_window = x[endpoint - short_size + 1:endpoint + 1]

        long_level, long_slope, long_scale = _robust_endpoint_fit(
            long_window, long_decay, global_noise
        )
        short_level, short_slope, short_scale = _robust_endpoint_fit(
            short_window, short_decay, global_noise
        )

        if not np.isfinite(long_level):
            candidate = short_level
            candidate_slope = short_slope
            local_scale = short_scale
        elif not np.isfinite(short_level):
            candidate = long_level
            candidate_slope = long_slope
            local_scale = long_scale
        else:
            # Agreement between independent horizons is evidence of genuine
            # motion.  Disagreement shifts weight toward the quieter long fit.
            slope_reference = max(
                0.35 * (abs(short_slope) + abs(long_slope)),
                0.12 * global_noise,
                1e-9,
            )
            agreement = np.tanh(
                (short_slope * long_slope) / (slope_reference * slope_reference)
            )

            # 0.70 on agreement keeps endpoint response high; 0.34 on conflict
            # suppresses isolated short-window noise excursions.
            short_weight = 0.52 + 0.18 * agreement
            short_weight = float(np.clip(short_weight, 0.34, 0.70))

            candidate = short_weight * short_level + (1.0 - short_weight) * long_level
            candidate_slope = (
                short_weight * short_slope + (1.0 - short_weight) * long_slope
            )
            local_scale = max(0.5 * (short_scale + long_scale), 0.25 * global_noise)

        if not np.isfinite(candidate):
            candidate = previous if np.isfinite(previous) else 0.0

        # Small predictive endpoint correction offsets the attenuation of a
        # local linear fit, without requiring future samples.
        candidate += 0.12 * candidate_slope

        if not np.isfinite(previous):
            filtered = candidate
        else:
            delta = candidate - previous
            delta_sign = np.sign(delta)

            # Noise-adaptive hysteresis removes tiny opposite moves while
            # allowing sufficiently large, persistent real reversals through.
            deadband = max(0.11 * local_scale, 0.018 * global_noise)

            if (
                accepted_direction != 0.0
                and delta_sign != 0.0
                and delta_sign != accepted_direction
                and abs(delta) < deadband
            ):
                filtered = previous
            else:
                filtered = candidate

            actual_delta = filtered - previous
            if abs(actual_delta) >= deadband:
                accepted_direction = np.sign(actual_delta)

        y[out_index] = filtered
        previous = filtered

    return y


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """Backward-compatible trend-preserving public entry point."""
    return adaptive_filter(x, window_size)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Apply the selected signal-processing algorithm.

    Args:
        input_signal: Input time series data.
        window_size: Sliding history size.
        algorithm_type: "basic" or "enhanced".

    Returns:
        Filtered signal with length len(input_signal) - window_size + 1.
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal
        noise_level: Standard deviation of noise to add
        seed: Random seed for reproducibility

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(
    noisy_signal=None,
    signal_length=1000,
    noise_level=0.3,
    window_size=20,
):
    """
    Run the signal processing algorithm on generated or supplied data.

    Returns:
        Dictionary containing filtered data and, for generated data, metrics.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1
            else 0
        )

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0
            else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
