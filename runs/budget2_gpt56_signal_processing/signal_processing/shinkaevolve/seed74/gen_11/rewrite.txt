"""
Causal robust multi-scale LOESS filtering for volatile non-stationary signals.

The estimator uses robust endpoint polynomial fits rather than a recursive Kalman
state.  Short and long causal local models are blended according to local motion,
then a noise-scaled directional hysteresis removes insignificant reversals.
"""
import numpy as np


def _robust_scale(values, fallback=1.0):
    """Return a stable MAD noise-scale estimate."""
    values = np.asarray(values, dtype=float)
    values = values[np.isfinite(values)]

    if values.size == 0:
        return float(fallback)

    median = np.median(values)
    mad = np.median(np.abs(values - median))
    scale = 1.4826 * mad

    if not np.isfinite(scale) or scale < 1e-10:
        scale = np.std(values)

    if not np.isfinite(scale) or scale < 1e-10:
        scale = max(abs(median) * 0.05, fallback, 1e-6)

    return float(scale)


def _endpoint_polynomial(window, degree=2, robust_iterations=2):
    """
    Robust causal local-polynomial fit evaluated at the newest sample.

    The coordinate system has u=0 at the endpoint, so the intercept is the
    filtered endpoint estimate and the linear coefficient is its local slope.
    """
    values = np.asarray(window, dtype=float)
    valid = np.isfinite(values)

    if not np.any(valid):
        return np.nan, 0.0, np.nan

    values = values[valid]
    count = values.size

    if count == 1:
        return float(values[0]), 0.0, 0.0

    effective_degree = min(degree, count - 1)
    # Oldest sample is negative and newest sample has coordinate zero.
    u = np.arange(count, dtype=float) - float(count - 1)
    scale = max(float(count - 1), 1.0)
    z = u / scale

    design = np.column_stack([z ** order for order in range(effective_degree + 1)])

    # Tricube endpoint kernel: recent samples are emphasized while all samples
    # retain a small floor to prevent an ill-conditioned fit.
    distance = np.abs(z)
    kernel = (1.0 - np.minimum(distance, 1.0) ** 3) ** 3
    kernel = 0.08 + 0.92 * kernel

    weights = kernel.copy()
    coefficients = np.zeros(effective_degree + 1, dtype=float)

    for _ in range(robust_iterations + 1):
        sqrt_w = np.sqrt(np.maximum(weights, 1e-12))
        weighted_design = design * sqrt_w[:, None]
        weighted_values = values * sqrt_w

        try:
            coefficients = np.linalg.lstsq(
                weighted_design, weighted_values, rcond=None
            )[0]
        except np.linalg.LinAlgError:
            coefficients[0] = np.average(values, weights=weights)
            coefficients[1:] = 0.0
            break

        residuals = values - design @ coefficients
        residual_scale = _robust_scale(
            residuals, fallback=max(np.std(values) * 0.15, 1e-6)
        )

        # Tukey bisquare weights completely suppress large isolated impulses.
        cutoff = 4.685 * max(residual_scale, 1e-8)
        ratio = residuals / cutoff
        robust = np.where(np.abs(ratio) < 1.0, (1.0 - ratio * ratio) ** 2, 0.03)
        weights = kernel * np.maximum(robust, 0.03)

    fitted = design @ coefficients
    residual_scale = _robust_scale(
        values - fitted, fallback=max(np.std(values) * 0.15, 1e-6)
    )

    # Convert normalized-coordinate slope back to samples/sample.
    slope = coefficients[1] / scale if effective_degree >= 1 else 0.0
    return float(coefficients[0]), float(slope), float(residual_scale)


def adaptive_filter(x, window_size=20):
    """
    Causal robust multi-scale endpoint smoother.

    Args:
        x: Input one-dimensional real-valued signal.
        window_size: Required trailing history before producing output.

    Returns:
        Endpoint-aligned filtered values with length len(x)-window_size+1.
    """
    x = np.asarray(x, dtype=float)

    if x.ndim != 1:
        raise ValueError("Input signal must be a 1D array")
    if window_size < 1:
        raise ValueError("window_size must be >= 1")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )

    n = len(x)
    if not np.any(np.isfinite(x)):
        return np.full(n - window_size + 1, np.nan)

    # A short quadratic model follows bends; the full-window model supplies a
    # lower-variance consensus estimate in quiet portions of the signal.
    short_span = min(window_size, max(5, int(round(0.45 * window_size))))
    if window_size <= 5:
        short_span = window_size

    estimates = np.empty(n, dtype=float)
    previous_level = np.nan
    committed_direction = 0.0
    pending_direction = 0.0
    pending_count = 0

    finite_values = x[np.isfinite(x)]
    global_scale = _robust_scale(
        np.diff(finite_values) if finite_values.size > 2 else finite_values,
        fallback=1e-5,
    )

    for t in range(n):
        start_long = max(0, t - window_size + 1)
        start_short = max(0, t - short_span + 1)

        long_level, long_slope, long_noise = _endpoint_polynomial(
            x[start_long:t + 1], degree=2
        )
        short_level, short_slope, short_noise = _endpoint_polynomial(
            x[start_short:t + 1], degree=2
        )

        if not np.isfinite(short_level):
            candidate = previous_level if np.isfinite(previous_level) else long_level
            candidate_slope = 0.0
            local_noise = global_scale
        elif not np.isfinite(long_level):
            candidate = short_level
            candidate_slope = short_slope
            local_noise = short_noise
        else:
            local_noise = max(
                min(short_noise, long_noise) if np.isfinite(short_noise + long_noise)
                else global_scale,
                global_scale * 0.18,
                1e-7,
            )

            # Agreement means a stable region, where the longer fit is safer.
            # Large disagreement or strong local slope/curvature favors the
            # short endpoint model and prevents moving-average-like lag.
            disagreement = abs(short_level - long_level) / local_noise
            slope_activity = abs(short_slope - long_slope) / local_noise
            activity = max(disagreement, 1.5 * slope_activity)

            short_weight = 0.38 + 0.52 * (activity / (1.0 + activity))
            short_weight = float(np.clip(short_weight, 0.38, 0.90))

            candidate = short_weight * short_level + (1.0 - short_weight) * long_level
            candidate_slope = (
                short_weight * short_slope + (1.0 - short_weight) * long_slope
            )

        if not np.isfinite(previous_level):
            level = candidate
            committed_direction = np.sign(candidate_slope)
        else:
            movement = candidate - previous_level
            # Small endpoint movements are predominantly residual measurement
            # noise.  The deadband is intentionally much smaller than a real
            # maneuver, preserving responsiveness at actual turning points.
            deadband = max(0.075 * local_noise, 0.012 * global_scale, 1e-8)
            proposed_direction = np.sign(movement) if abs(movement) > deadband else 0.0

            if proposed_direction == 0.0:
                level = previous_level
            elif committed_direction == 0.0 or proposed_direction == committed_direction:
                level = candidate
                committed_direction = proposed_direction
                pending_direction = 0.0
                pending_count = 0
            else:
                # A direction change requires either two corroborating endpoint
                # estimates or a clearly significant displacement. This removes
                # one-sample false turns without imposing a fixed filter delay.
                if proposed_direction == pending_direction:
                    pending_count += 1
                else:
                    pending_direction = proposed_direction
                    pending_count = 1

                decisive_move = abs(movement) > 0.42 * local_noise
                if pending_count >= 2 or decisive_move:
                    level = candidate
                    committed_direction = proposed_direction
                    pending_direction = 0.0
                    pending_count = 0
                else:
                    level = previous_level

        estimates[t] = level
        previous_level = level

    return estimates[window_size - 1:]


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """Backward-compatible name for the robust multiscale trend filter."""
    return adaptive_filter(x, window_size)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Apply the selected filtering algorithm.

    Args:
        input_signal: Input time series data.
        window_size: Size of required trailing history.
        algorithm_type: "basic" or "enhanced".

    Returns:
        Filtered endpoint-aligned signal.
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal
        noise_level: Standard deviation of noise to add
        seed: Random seed for reproducibility

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000, noise_level=0.3, window_size=20):
    """
    Run the signal processing algorithm on a supplied or generated signal.

    Returns:
        Dictionary containing filtered values and, when available, metrics.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1 else 0
        )

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0 else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
