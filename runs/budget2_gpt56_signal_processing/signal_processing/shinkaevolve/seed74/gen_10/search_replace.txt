<DESCRIPTION>
Replace the lagging exponential moving average with a robust, weighted local-linear endpoint estimator. Each trailing window estimates its local slope using exponentially weighted regression, clips large residuals with a Huber-style reweighting pass, and applies only a bounded fraction of the inferred trend correction to the weighted level estimate. This substantially reduces moving-average phase lag while retaining smoothing. The robust residual pass prevents isolated spikes from biasing the slope and causing false reversals; the final light causal continuity blend further suppresses short one-sample directional changes without adding meaningful latency.

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Robust causal local-trend filter with bounded lag compensation.

    A weighted local linear model estimates the current endpoint rather than
    merely averaging historical values.  Robust reweighting prevents impulsive
    observations from creating noise-induced slope reversals.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the trailing processing window

    Returns:
        y: Filtered output aligned with x[window_size - 1:]
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be a 1D array")
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")
    if not np.all(np.isfinite(x)):
        raise ValueError("Input signal must contain only finite values")

    output_length = len(x) - window_size + 1
    y = np.empty(output_length)

    # Time is measured relative to the newest sample, making the fitted
    # intercept directly equal to the endpoint estimate.
    time = np.arange(window_size, dtype=float) - (window_size - 1)
    base_weights = np.exp(np.linspace(-2.4, 0.0, window_size))
    base_weights /= base_weights.sum()

    previous_value = None
    previous_slope = 0.0

    for i in range(output_length):
        window = x[i : i + window_size]

        def weighted_line(weights):
            weight_sum = weights.sum()
            mean_t = np.dot(weights, time) / weight_sum
            mean_x = np.dot(weights, window) / weight_sum
            centered_t = time - mean_t
            denominator = np.dot(weights, centered_t * centered_t)
            slope = np.dot(weights, centered_t * (window - mean_x)) / max(denominator, 1e-12)
            intercept = mean_x - slope * mean_t
            return intercept, slope

        level, slope = weighted_line(base_weights)

        # One inexpensive Huber reweighting pass makes trend estimation stable
        # under transient spikes and locally changing noise variance.
        residual = window - (level + slope * time)
        scale = 1.4826 * np.median(np.abs(residual - np.median(residual))) + 1e-8
        robust_weights = base_weights * np.minimum(1.0, 2.5 * scale / np.maximum(np.abs(residual), 1e-8))
        level, slope = weighted_line(robust_weights)

        weighted_level = np.dot(base_weights, window)
        effective_age = -np.dot(base_weights, time)

        # A bounded correction removes most filter lag while retaining the
        # lower variance of the weighted level estimate.
        correction_limit = 3.0 * scale
        trend_correction = np.clip(slope * effective_age, -correction_limit, correction_limit)
        candidate = weighted_level + 0.72 * trend_correction

        if previous_value is None:
            y[i] = candidate
        else:
            # Predictive continuity smooths jitter but follows sustained turns.
            predicted = previous_value + previous_slope
            y[i] = 0.78 * candidate + 0.22 * predicted
            previous_slope = 0.65 * previous_slope + 0.35 * (y[i] - previous_value)

        previous_value = y[i]

    return y
