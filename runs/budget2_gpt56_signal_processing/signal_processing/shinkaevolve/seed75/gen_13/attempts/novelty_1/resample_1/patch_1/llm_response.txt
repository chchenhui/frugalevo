<DESCRIPTION>
This replaces the Kalman estimator with a causal multi-scale robust local-polynomial trend filter. For every trailing window, it fits robust weighted polynomial models at short and long time scales, then adaptively blends them according to detected local curvature and disagreement. Endpoint polynomial evaluation provides near-zero additional phase delay, robust Tukey weighting rejects spikes, and scale fusion reduces noise-induced slope reversals without flattening genuine rapid changes. The implementation preserves the original function signatures, output length, alignment, and return structures.

"""
Causal Multi-Scale Robust Local Trend Filter for non-stationary time series.

The enhanced filter uses robust endpoint local-polynomial regression rather than
a recursive Kalman state estimator.  Each output is estimated from a trailing
window using two time scales:

* A long-scale robust quadratic fit rejects noise and suppresses false reversals.
* A short-scale robust quadratic fit preserves genuine local acceleration.
* Adaptive fusion selects more short-scale content only when the local dynamics
  are coherent and statistically significant.

All estimates are evaluated at the newest sample in the window, preserving the
original trailing-window output convention with negligible phase delay.
"""
import numpy as np


def _validate_signal(x, window_size):
    """Validate a one-dimensional finite-valued signal."""
    x = np.asarray(x, dtype=float)

    if x.ndim != 1:
        raise ValueError("Input signal must be a 1D array of real-valued samples")
    if not isinstance(window_size, (int, np.integer)) or window_size < 1:
        raise ValueError("window_size must be a positive integer")
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )
    if not np.all(np.isfinite(x)):
        raise ValueError("Input signal must contain only finite real-valued samples")

    return x


def adaptive_filter(x, window_size=20):
    """
    Efficient moving-average baseline retaining the original API.

    Args:
        x: Input signal.
        window_size: Trailing window size.

    Returns:
        A length len(x) - window_size + 1 filtered signal.
    """
    x = _validate_signal(x, window_size)
    cumulative = np.concatenate(([0.0], np.cumsum(x, dtype=float)))
    return (cumulative[window_size:] - cumulative[:-window_size]) / float(window_size)


def _robust_endpoint_fit(values, degree, decay):
    """
    Robust weighted polynomial estimate at the final sample of ``values``.

    Time is normalized to [-1, 0], so the fitted intercept is the endpoint
    estimate. Exponential recency weighting makes the estimate causal and
    responsive, while two Tukey-biweight reweighting passes suppress impulses.
    """
    m = len(values)

    if m == 1:
        return float(values[0]), 0.0, 0.0

    degree = min(degree, m - 1)
    time = np.linspace(-1.0, 0.0, m)
    design = np.vander(time, N=degree + 1, increasing=True)

    # Moderately recency-weighted kernel. The oldest samples remain useful for
    # reliable slope/curvature estimation instead of being discarded entirely.
    base_weights = np.exp(decay * time)
    weights = base_weights.copy()

    coefficients = np.zeros(degree + 1, dtype=float)

    for _ in range(3):
        root_weights = np.sqrt(np.maximum(weights, 1e-12))
        weighted_design = design * root_weights[:, None]
        weighted_values = values * root_weights

        coefficients, _, _, _ = np.linalg.lstsq(
            weighted_design, weighted_values, rcond=None
        )
        residual = values - design @ coefficients

        # MAD residual scale is robust to isolated spikes and abrupt samples.
        median_residual = np.median(residual)
        scale = 1.4826 * np.median(np.abs(residual - median_residual))
        scale = max(scale, 1e-8)

        # Tukey's bounded influence function completely rejects strong outliers.
        u = residual / (4.685 * scale)
        robust_weights = np.square(np.maximum(0.0, 1.0 - u * u))
        weights = base_weights * np.maximum(robust_weights, 1e-5)

    endpoint = float(coefficients[0])
    slope = float(coefficients[1]) if degree >= 1 else 0.0
    curvature = float(2.0 * coefficients[2]) if degree >= 2 else 0.0
    return endpoint, slope, curvature


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Causal robust multi-scale endpoint polynomial filter.

    The output remains aligned with the newest element in each trailing window:
    output index i estimates input index i + window_size - 1.

    Long and short local regressions are fused using local curvature and
    cross-scale disagreement. Smooth regions preferentially use the stable
    long-scale trend; sustained high-curvature regions receive more short-scale
    contribution, avoiding the phase lag of a moving average.

    Args:
        x: Input signal (1D array of real-valued samples).
        window_size: Size of each trailing analysis window.

    Returns:
        Filtered signal with length len(x) - window_size + 1.
    """
    x = _validate_signal(x, window_size)
    n = len(x)

    if window_size == 1:
        return x.copy()

    output_length = n - window_size + 1
    y = np.empty(output_length, dtype=float)

    # The short scale retains local oscillatory dynamics; the long scale
    # supplies the reversal-resistant low-noise trend estimate.
    short_size = max(4, int(np.ceil(0.55 * window_size)))
    short_size = min(short_size, window_size)

    # Estimate global innovation scale from first differences. This scale is
    # only used to normalize the adaptive fusion and is robust to nonstationary
    # impulses through the MAD estimator.
    differences = np.diff(x)
    difference_median = np.median(differences)
    difference_mad = np.median(np.abs(differences - difference_median))
    innovation_scale = max(1e-7, 1.4826 * difference_mad / np.sqrt(2.0))

    previous_output = None
    previous_slope = 0.0

    for out_index, end in enumerate(range(window_size - 1, n)):
        long_window = x[end - window_size + 1 : end + 1]
        short_window = long_window[-short_size:]

        long_level, long_slope, long_curvature = _robust_endpoint_fit(
            long_window, degree=2, decay=1.7
        )
        short_level, short_slope, short_curvature = _robust_endpoint_fit(
            short_window, degree=2, decay=1.25
        )

        # A persistent curvature signal identifies authentic changing dynamics.
        # Scale disagreement alone is not sufficient, since disagreement during
        # low-SNR periods is usually noise and should favor the long estimate.
        curvature_strength = (
            abs(short_curvature) * max(short_size - 1, 1) / innovation_scale
        )
        slope_strength = abs(short_slope) / innovation_scale
        disagreement = abs(short_level - long_level) / innovation_scale

        dynamic_weight = curvature_strength / (1.5 + curvature_strength)
        dynamic_weight *= 0.55 + 0.45 * (
            slope_strength / (1.0 + slope_strength)
        )

        # Large cross-scale disagreement without coherent slope is likely a
        # transient disturbance, so attenuate short-scale influence.
        disagreement_penalty = 1.0 / (1.0 + 0.18 * disagreement)
        short_weight = 0.18 + 0.67 * dynamic_weight * disagreement_penalty
        short_weight = float(np.clip(short_weight, 0.14, 0.72))

        estimate = short_weight * short_level + (1.0 - short_weight) * long_level

        # Directional hysteresis is applied only to tiny, statistically weak
        # output changes. It removes jitter reversals while allowing meaningful
        # turns and all high-curvature motion to pass unchanged.
        if previous_output is not None:
            proposed_slope = estimate - previous_output
            deadband = 0.075 * innovation_scale

            reversal_is_tiny = (
                proposed_slope * previous_slope < 0.0
                and abs(proposed_slope) < deadband
                and curvature_strength < 0.8
            )

            if reversal_is_tiny:
                estimate = previous_output

            actual_slope = estimate - previous_output
            if abs(actual_slope) >= 0.35 * deadband:
                previous_slope = actual_slope

        y[out_index] = estimate
        previous_output = estimate

    return y


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Apply the requested filtering algorithm.

    Args:
        input_signal: Input time series.
        window_size: Trailing window size.
        algorithm_type: "basic" for moving average, otherwise enhanced mode.

    Returns:
        Filtered signal.
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal.
        noise_level: Standard deviation of noise to add.
        seed: Random seed for reproducibility.

    Returns:
        Tuple of (noisy_signal, clean_signal).
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(
    noisy_signal=None,
    signal_length=1000,
    noise_level=0.3,
    window_size=20,
):
    """
    Run filtering and return output and evaluation metrics.

    Args:
        noisy_signal: Optional externally supplied noisy signal.
        signal_length: Generated signal length when noisy_signal is None.
        noise_level: Generated noise standard deviation.
        window_size: Filtering window size.

    Returns:
        Dictionary containing filtered output and, where available, metrics.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1
            else 0
        )

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0
            else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
