<DESCRIPTION>
Replace the fixed exponentially weighted moving average with a causal adaptive constant-velocity Kalman filter. Unlike a trailing average, the filter explicitly estimates both signal level and local slope, which substantially reduces phase lag while retaining genuine trend changes. The measurement-noise estimate is updated robustly from innovations, and the process noise is temporarily increased for large innovations so real regime changes are tracked quickly. Innovation clipping limits isolated noise spikes from producing false directional reversals. The output remains aligned to the newest sample in each sliding window and preserves the existing output-length contract.

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using weighted moving average.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Create weights that emphasize recent samples
    weights = np.exp(np.linspace(-2, 0, window_size))
    weights = weights / np.sum(weights)

    for i in range(output_length):
        window = x[i : i + window_size]

        # Weighted moving average with exponential weights
        y[i] = np.sum(window * weights)

    return y

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Causal adaptive level-and-slope Kalman filter.

    The state contains the current signal level and its local slope.  This avoids
    the fixed lag of a moving average while adaptively rejecting measurement noise.
    Output sample i corresponds to input sample i + window_size - 1.
    """
    x = np.asarray(x, dtype=float)
    if x.ndim != 1:
        raise ValueError("Input signal must be a 1D array")
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")
    if not np.all(np.isfinite(x)):
        raise ValueError("Input signal must contain only finite values")

    # Estimate the short-term measurement-noise scale robustly.  Differencing
    # prevents a slowly varying trend from being interpreted as observation noise.
    initial_window = x[:window_size]
    differences = np.diff(initial_window)
    mad = np.median(np.abs(differences - np.median(differences)))
    noise_std = max(mad / (0.6745 * np.sqrt(2.0)), np.std(differences) / np.sqrt(2.0), 1e-6)
    measurement_var = noise_std * noise_std

    # State is [level, slope].  A smaller base acceleration variance produces
    # smooth trajectories; innovation-driven adaptation restores responsiveness
    # at genuine turns or abrupt changes.
    state = np.array([x[0], 0.0], dtype=float)
    covariance = np.array(
        [[measurement_var, 0.0], [0.0, measurement_var * 0.1]],
        dtype=float,
    )
    transition = np.array([[1.0, 1.0], [0.0, 1.0]])
    filtered = np.empty(len(x), dtype=float)
    filtered[0] = state[0]

    for index in range(1, len(x)):
        predicted_state = transition @ state
        predicted_covariance = transition @ covariance @ transition.T

        innovation = x[index] - predicted_state[0]
        innovation_scale = np.sqrt(max(predicted_covariance[0, 0] + measurement_var, 1e-12))

        # Isolated spikes should not reverse the estimated trend.  Sustained
        # changes still pass through because later samples retain the same sign.
        bounded_innovation = np.clip(innovation, -3.5 * innovation_scale, 3.5 * innovation_scale)

        # Raise process noise only when the observation is unusually inconsistent
        # with the current local trend, reducing lag during real dynamics.
        normalized_innovation = abs(innovation) / max(innovation_scale, 1e-12)
        acceleration_var = measurement_var * 0.0025 * (
            1.0 + min(8.0, max(0.0, normalized_innovation - 1.5) ** 2)
        )
        process_covariance = acceleration_var * np.array(
            [[0.25, 0.5], [0.5, 1.0]],
            dtype=float,
        )
        predicted_covariance += process_covariance

        innovation_variance = predicted_covariance[0, 0] + measurement_var
        gain = predicted_covariance[:, 0] / innovation_variance
        state = predicted_state + gain * bounded_innovation
        covariance = predicted_covariance - np.outer(gain, predicted_covariance[0, :])
        covariance = 0.5 * (covariance + covariance.T)

        # Robust residual adaptation follows changing noise levels without letting
        # a single transition make the filter permanently sluggish.
        residual_var = min(
            bounded_innovation * bounded_innovation,
            9.0 * measurement_var,
        )
        measurement_var = max(0.97 * measurement_var + 0.03 * residual_var, 1e-12)
        filtered[index] = state[0]

    return filtered[window_size - 1 :]
