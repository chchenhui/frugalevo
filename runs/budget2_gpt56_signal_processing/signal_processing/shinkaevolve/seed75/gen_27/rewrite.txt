"""
Dual-scale robust adaptive signal filter for volatile non-stationary series.

The enhanced mode combines:
  * fast and slow robust MAD noise estimation on first differences,
  * causal alpha-beta level/slope tracking,
  * bounded innovation clipping,
  * adaptive gain scheduling,
  * low-lag prediction lead, and
  * volatility-scaled reversal hysteresis.

All enhanced outputs are causal estimates aligned with the final sample of each
trailing window, preserving the original API and output length.
"""
from collections import deque
import numpy as np


class _DualScaleTrendTracker:
    """
    Causal robust level-and-slope tracker.

    Parameters are intentionally bounded to balance four objectives:
      - lower alpha/beta gains in quiet regions reduce slope changes,
      - gain expansion during sustained innovations limits lag,
      - first-difference MAD avoids treating slow trends as noise,
      - output hysteresis rejects sub-noise directional reversals.
    """

    def __init__(
        self,
        window_size,
        fast_span=8,
        noise_decay=0.965,
        fast_weight=0.62,
        base_alpha=0.105,
        max_alpha=0.500,
        base_beta=0.0035,
        max_beta=0.090,
        prediction_lead=0.28,
        reversal_band=0.30,
        immediate_band=0.95,
        reversal_confirmation=2,
    ):
        self.window_size = max(3, int(window_size))
        self.fast_span = max(4, min(int(fast_span), self.window_size))
        self.noise_decay = noise_decay
        self.fast_weight = fast_weight

        self.base_alpha = base_alpha
        self.max_alpha = max_alpha
        self.base_beta = base_beta
        self.max_beta = max_beta
        self.prediction_lead = prediction_lead

        self.reversal_band = reversal_band
        self.immediate_band = immediate_band
        self.reversal_confirmation = reversal_confirmation

        self.level = None
        self.slope = 0.0
        self.last_sample = None
        self.noise_scale = None

        self.differences = deque(maxlen=self.window_size)
        self.last_output = None
        self.output_direction = 0.0
        self.pending_direction = 0.0
        self.pending_count = 0

    @staticmethod
    def _robust_difference_scale(values):
        """
        Estimate observation-noise standard deviation from first differences.

        For independent observation noise, std(diff) = sqrt(2) * std(noise).
        MAD is used so impulses and genuine discontinuities have limited impact.
        """
        if len(values) < 3:
            return None

        d = np.asarray(values, dtype=float)
        center = np.median(d)
        mad = np.median(np.abs(d - center))
        mad_scale = mad / (0.67448975 * np.sqrt(2.0))

        # A lightly winsorized RMS is useful when MAD is exactly zero because
        # of quantized signals, while retaining robustness to isolated spikes.
        clipped = np.clip(d - center, -3.5 * max(mad / 0.67448975, 1e-8),
                          3.5 * max(mad / 0.67448975, 1e-8))
        rms_scale = np.sqrt(np.mean(clipped * clipped)) / np.sqrt(2.0)

        if mad_scale < 1e-8:
            return max(rms_scale, 1e-6)
        return max(mad_scale, 0.45 * rms_scale, 1e-6)

    def _update_noise_scale(self):
        """Blend short and window-length MAD scales with bounded adaptation."""
        slow_scale = self._robust_difference_scale(self.differences)

        if len(self.differences) >= self.fast_span:
            recent = list(self.differences)[-self.fast_span:]
            fast_scale = self._robust_difference_scale(recent)
        else:
            fast_scale = slow_scale

        if fast_scale is None and slow_scale is None:
            return

        if fast_scale is None:
            target = slow_scale
        elif slow_scale is None:
            target = fast_scale
        else:
            # Restrict disagreement so a noisy short segment cannot abruptly
            # inflate gain thresholds, but can still respond to real volatility.
            fast_scale = np.clip(fast_scale, 0.45 * slow_scale, 2.75 * slow_scale)
            target = self.fast_weight * fast_scale + (1.0 - self.fast_weight) * slow_scale

        if self.noise_scale is None:
            self.noise_scale = max(target, 1e-6)
        else:
            self.noise_scale = max(
                1e-6,
                self.noise_decay * self.noise_scale
                + (1.0 - self.noise_decay) * target,
            )

    def update(self, sample):
        """Consume one finite observation and return its causal filtered value."""
        sample = float(sample)

        if self.level is None:
            self.level = sample
            self.last_sample = sample
            self.last_output = sample
            self.noise_scale = max(abs(sample) * 0.025, 0.03, 1e-6)
            return sample

        difference = sample - self.last_sample
        self.last_sample = sample
        self.differences.append(difference)
        self._update_noise_scale()

        sigma = max(self.noise_scale, 1e-6)
        predicted_level = self.level + self.slope
        raw_innovation = sample - predicted_level

        # Clip only extreme impulses.  Clipping is deliberately wider than the
        # reversal gate: genuine fast changes should reach the state estimator.
        clip_limit = max(3.8 * sigma, 1e-6)
        innovation = np.clip(raw_innovation, -clip_limit, clip_limit)
        normalized = abs(innovation) / sigma

        # Smooth saturating gain schedule.  Normal samples use modest gains;
        # sustained movement gets a fast response without unstable overshoot.
        response = normalized / (normalized + 2.35)
        alpha = self.base_alpha + (self.max_alpha - self.base_alpha) * response
        beta = self.base_beta + (self.max_beta - self.base_beta) * (response ** 2)

        self.level = predicted_level + alpha * innovation
        self.slope = (1.0 - beta) * self.slope + beta * innovation

        # Conservative prediction lead compensates causal estimator delay.
        lead_limit = max(1.25 * sigma, 0.75 * abs(self.slope), 1e-8)
        lead = np.clip(self.prediction_lead * self.slope, -lead_limit, lead_limit)
        candidate = self.level + lead

        delta = candidate - self.last_output
        direction = np.sign(delta)

        if direction == 0.0:
            return self.last_output

        small_band = max(self.reversal_band * sigma, 1e-7)
        large_band = max(self.immediate_band * sigma, 1e-7)

        if self.output_direction == 0.0 or direction == self.output_direction:
            self.output_direction = direction
            self.pending_direction = 0.0
            self.pending_count = 0
            self.last_output = candidate

        elif abs(delta) >= large_band:
            # A large move is likely a genuine turn and should not incur delay.
            self.output_direction = direction
            self.pending_direction = 0.0
            self.pending_count = 0
            self.last_output = candidate

        elif abs(delta) <= small_band:
            # Suppress sub-noise counter-moves, reducing false reversals.
            self.pending_direction = 0.0
            self.pending_count = 0

        else:
            if direction == self.pending_direction:
                self.pending_count += 1
            else:
                self.pending_direction = direction
                self.pending_count = 1

            if self.pending_count >= self.reversal_confirmation:
                self.output_direction = direction
                self.pending_direction = 0.0
                self.pending_count = 0
                self.last_output = candidate

        return self.last_output


class _SignalProcessingPipeline:
    """Validation and execution coordinator preserving the public API."""

    def __init__(self, window_size, mode):
        self.window_size = int(window_size)
        self.mode = mode

    @staticmethod
    def _as_signal_array(x):
        signal = np.asarray(x, dtype=float)
        if signal.ndim != 1:
            raise ValueError("Input signal must be a 1D array of real-valued samples")
        return signal

    def validate(self, x):
        signal = self._as_signal_array(x)
        if self.window_size <= 0:
            raise ValueError("window_size must be a positive integer")
        if len(signal) < self.window_size:
            raise ValueError(
                f"Input signal length ({len(signal)}) must be >= "
                f"window_size ({self.window_size})"
            )
        return signal

    def run_basic(self, signal):
        """Vectorized trailing moving average with original output alignment."""
        cumulative = np.concatenate(([0.0], np.cumsum(signal, dtype=float)))
        return (
            cumulative[self.window_size:] - cumulative[:-self.window_size]
        ) / self.window_size

    def run_enhanced(self, signal):
        """
        Run the causal dual-scale tracker.

        Non-finite values retain the most recent valid measurement, matching the
        tolerant behavior of the previous enhanced implementation.
        """
        tracker = _DualScaleTrendTracker(self.window_size)
        estimates = np.empty(signal.size, dtype=float)
        last_valid = 0.0

        for index, value in enumerate(signal):
            if np.isfinite(value):
                last_valid = float(value)
            estimates[index] = tracker.update(last_valid)

        return estimates[self.window_size - 1:]

    def run(self, x):
        signal = self.validate(x)
        if self.mode == "enhanced":
            return self.run_enhanced(signal)
        return self.run_basic(signal)


def adaptive_filter(x, window_size=20):
    """
    Basic trailing moving-average filter.

    Returns an array of length len(x) - window_size + 1.
    """
    return _SignalProcessingPipeline(window_size, "basic").run(x)


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Dual-scale robust low-lag adaptive trend-preserving filter.

    Returns an array of length len(x) - window_size + 1, aligned to the final
    sample in each trailing input window.
    """
    return _SignalProcessingPipeline(window_size, "enhanced").run(x)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Apply either the basic trailing average or enhanced adaptive filter.

    Args:
        input_signal: One-dimensional real-valued time series.
        window_size: Trailing-window size and enhanced noise-estimation span.
        algorithm_type: "enhanced" selects the adaptive tracker; all other
            values select the basic moving average.

    Returns:
        Filtered output with length len(input_signal) - window_size + 1.
    """
    mode = "enhanced" if algorithm_type == "enhanced" else "basic"
    return _SignalProcessingPipeline(window_size, mode).run(input_signal)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """Generate a reproducible volatile non-stationary test signal."""
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    clean_signal += 0.1 * t * np.sin(0.2 * t)
    clean_signal += np.cumsum(np.random.randn(length) * 0.05)

    noisy_signal = clean_signal + np.random.normal(0, noise_level, length)
    return noisy_signal, clean_signal


def run_signal_processing(
    noisy_signal=None,
    signal_length=1000,
    noise_level=0.3,
    window_size=20,
):
    """Run filtering and return compatible diagnostic results."""
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(
            signal_length, noise_level
        )
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1]
            if min_length > 1
            else 0
        )
        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before
            if noise_before > 0
            else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }

    if len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }

    return {
        "filtered_signal": [],
        "clean_signal": [],
        "noisy_signal": [],
        "correlation": 0,
        "noise_reduction": 0,
        "signal_length": 0,
    }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
