def compute_model_placement(gpu_num, models):
    """
    Compute a model placement that minimizes the maximum KVPR across all GPUs.

    Strategy:
      1) Greedy: place models (sorted by pressure density r/s descending) onto
         the GPU that yields the smallest resulting KVPR while fitting memory.
      2) Local search: repeatedly pick the GPU with maximum KVPR and try to
         move one of its models to another GPU if that strictly lowers the
         overall maximum KVPR. Repeat until no improvement.
    """

    def safe_ratio(W, used_mem):
        free = GPU_MEM_SIZE - used_mem
        if free <= 1e-9:
            return float('inf')
        return W / free

    # Sort by pressure density descending
    sorted_models = sorted(models, key=lambda m: (m.req_rate / m.slo), reverse=True)

    placement = {i: [] for i in range(gpu_num)}
    used = [0.0] * gpu_num       # used memory per GPU
    W = [0.0] * gpu_num          # sum of r/s per GPU

    # --- Phase 1: greedy best-fit by resulting KVPR ---
    for m in sorted_models:
        best_i, best_val = None, float('inf')
        for i in range(gpu_num):
            if m.model_size <= GPU_MEM_SIZE - used[i]:
                val = safe_ratio(W[i] + m.req_rate / m.slo, used[i] + m.model_size)
                if val < best_val:
                    best_val, best_i = val, i
        if best_i is None:
            raise ValueError(
                f"Unable to place model of size {m.model_size} GB on any GPU."
            )
        placement[best_i].append(m)
        used[best_i] += m.model_size
        W[best_i] += m.req_rate / m.slo

    # --- Phase 2: local search (move model off the hottest GPU) ---
    def kvpr(i):
        return safe_ratio(W[i], used[i])

    improved = True
    guard = 0
    while improved and guard < 10000:
        improved = False
        guard += 1
        max_i = max(range(gpu_num), key=kvpr)
        max_val = kvpr(max_i)
        if max_val == float('inf') and used[max_i] >= GPU_MEM_SIZE - 1e-9:
            # GPU completely full of weights: no free memory, try swaps instead
            for m in placement[max_i]:
                for j in range(gpu_num):
                    if j == max_i:
                        continue
                    for m2 in placement[j]:
                        # swap sizes compatible?
                        if (used[max_i] - m.model_size + m2.model_size) <= GPU_MEM_SIZE + 1e-9 and \
                           (used[j] - m2.model_size + m.model_size) <= GPU_MEM_SIZE + 1e-9:
                            nW1 = W[max_i] - m.req_rate / m.slo + m2.req_rate / m2.slo
                            nW2 = W[j] - m2.req_rate / m.slo + m.req_rate / m.slo
                            n1 = safe_ratio(nW1, used[max_i] - m.model_size + m2.model_size)
                            n2 = safe_ratio(nW2, used[j] - m2.model_size + m.model_size)
                            nmax = max(max(n1, n2), max(kvpr(k) for k in range(gpu_num) if k not in (max_i, j)))
                            if nmax < max_val:
                                placement[max_i].remove(m)
                                placement[j].remove(m2)
                                placement[max_i].append(m2)
                                placement[j].append(m)
                                W[max_i], W[j] = nW1, nW2
                                used[max_i] = used[max_i] - m.model_size + m2.model_size
                                used[j] = used[j] - m2.model_size + m.model_size
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break
            continue

        # try moves
        for m in list(placement[max_i]):
            for j in range(gpu_num):
                if j == max_i:
                    continue
                if m.model_size <= GPU_MEM_SIZE - used[j]:
                    n1 = safe_ratio(W[max_i] - m.req_rate / m.slo,
                                    used[max_i] - m.model_size)
                    n2 = safe_ratio(W[j] + m.req_rate / m.slo,
                                    used[j] + m.model_size)
                    others = max(kvpr(k) for k in range(gpu_num) if k not in (max_i, j)) if gpu_num > 2 else -1.0
                    nmax = max(n1, n2, others)
                    if nmax < max_val - 1e-12:
                        placement[max_i].remove(m)
                        placement[j].append(m)
                        W[max_i] -= m.req_rate / m.slo
                        used[max_i] -= m.model_size
                        W[j] += m.req_rate / m.slo
                        used[j] += m.model_size
                        improved = True
                        break
            if improved:
                break

    return placement

def compute_model_placement(gpu_num, models):
    """
    Compute a model placement that minimizes the maximum KVPR across all GPUs.

    Strategy: binary-search-on-threshold packing (first-fit with KVPR
    feasibility check), wrapped in a randomized Las Vegas restart loop.
    Each restart uses a slightly perturbed model order (small random
    jitter on the density sort key), and we keep the assignment whose
    final max KVPR is smallest.
    """
    import random

    def pack(order):
        """First-fit pack models in `order` into at most gpu_num GPUs with
        per-GPU KVPR <= lam. Returns placement dict or None."""

        def try_pack(lam):
            rem = [GPU_MEM_SIZE] * gpu_num
            W = [0.0] * gpu_num
            assign = {i: [] for i in range(gpu_num)}
            used = 0
            for m in order:
                w = m.req_rate / m.slo
                s = m.model_size
                best_i, best_val = None, None
                for i in range(used):
                    if s <= rem[i]:
                        val = (W[i] + w) / (rem[i] - s) if (rem[i] - s) > 0 else float('inf')
                        if val <= lam + 1e-12 and (best_val is None or val < best_val):
                            best_val, best_i = val, i
                if best_i is None and used < gpu_num and s <= GPU_MEM_SIZE:
                    val = w / (GPU_MEM_SIZE - s) if (GPU_MEM_SIZE - s) > 0 else float('inf')
                    if val <= lam + 1e-12:
                        best_i, best_val = used, val
                if best_i is None:
                    return None
                rem[best_i] -= s
                W[best_i] += w
                assign[best_i].append(m)
                if best_i == used:
                    used += 1
            return assign

        # Binary search smallest feasible lambda
        lo, hi = 0.0, 1e9
        best = try_pack(hi)
        if best is None:
            return None
        for _ in range(60):
            mid = (lo + hi) / 2
            res = try_pack(mid)
            if res is not None:
                best = res
                hi = mid
            else:
                lo = mid
        return best

    def kvpr_of(assign):
        vals = []
        for i in range(gpu_num):
            used_mem = sum(m.model_size for m in assign[i])
            free = GPU_MEM_SIZE - used_mem
            if free <= 1e-9:
                vals.append(float('inf'))
            else:
                vals.append(sum(m.req_rate / m.slo for m in assign[i]) / free)
        return max(vals)

    # Base order: pressure density descending
    base = sorted(models, key=lambda m: (m.req_rate / m.slo), reverse=True)

    best_assign = None
    best_score = float('inf')

    for restart in range(25):
        if restart == 0:
            order = base
        else:
            # random jitter on the density key to shuffle near-equal models
            order = sorted(models,
                           key=lambda m: (m.req_rate / m.slo) * (1.0 + random.uniform(-0.02, 0.02)),
                           reverse=True)
        assign = pack(order)
        if assign is None:
            continue
        score = kvpr_of(assign)
        if score < best_score:
            best_score = score
            best_assign = assign

    if best_assign is not None:
        return best_assign

    # Fallback: memory-only first-fit (no KVPR constraint)
    rem = [GPU_MEM_SIZE] * gpu_num
    assign = {i: [] for i in range(gpu_num)}
    for m in base:
        placed = False
        for i in range(gpu_num):
            if m.model_size <= rem[i]:
                rem[i] -= m.model_size
                assign[i].append(m)
                placed = True
                break
        if not placed:
            raise ValueError(
                f"Unable to place model of size {m.model_size} GB on any GPU."
            )
    return assign
