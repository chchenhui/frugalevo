<DESCRIPTION>
Cross-over of the best-performing binary-search-on-threshold placement (score 25.69) with the local-search refinement from the greedy program (score 22.52). The binary search finds a near-optimal max-KVPR threshold and packs via first-fit-decreasing; then a local-search phase moves models off the hottest GPU when it strictly lowers the global max KVPR, with safe division to avoid float division by zero. The final placement is rebuilt cleanly to guarantee a well-formed output dict.
</DESCRIPTION>

GPU_MEM_SIZE = 80 # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """
    Compute a model placement minimizing the maximum KVPR across all GPUs
    using binary-search-on-threshold packing plus local-search refinement.
    """

    def safe_ratio(W, used_mem):
        free = GPU_MEM_SIZE - used_mem
        if free <= 1e-9:
            return float('inf')
        return W / free

    sorted_models = sorted(models, key=lambda m: (m.req_rate / m.slo), reverse=True)

    def pack(lam, record=False):
        """Pack all models into at most gpu_num GPUs with KVPR <= lam.
        If record, return (placement, rem, W); else return feasibility."""
        placement = {i: [] for i in range(gpu_num)}
        rem = [GPU_MEM_SIZE] * gpu_num
        W = [0.0] * gpu_num
        gidx = 0  # number of opened GPUs
        for m in sorted_models:
            w = m.req_rate / m.slo
            s = m.model_size
            placed = False
            # first-fit among opened GPUs
            for i in range(gidx):
                if s <= rem[i] and (W[i] + w) <= lam * (rem[i] - s) + 1e-12:
                    rem[i] -= s
                    W[i] += w
                    placement[i].append(m)
                    placed = True
                    break
            if not placed:
                # try best-fit among all opened GPUs (relaxed order) if lam is None-ish
                best_i, best_val = None, None
                for i in range(gidx):
                    if s <= rem[i]:
                        val = (W[i] + w) / (rem[i] - s) if (rem[i] - s) > 1e-12 else float('inf')
                        if best_val is None or val < best_val:
                            best_val, best_i = val, i
                if best_i is not None and (lam is None or best_val <= lam + 1e-12):
                    rem[best_i] -= s
                    W[best_i] += w
                    placement[best_i].append(m)
                    placed = True
                elif not placed:
                    # open a new GPU
                    if gidx < gpu_num and s <= GPU_MEM_SIZE:
                        val = w / (GPU_MEM_SIZE - s) if (GPU_MEM_SIZE - s) > 1e-12 else float('inf')
                        if lam is None or val <= lam + 1e-12:
                            rem[gidx] = GPU_MEM_SIZE - s
                            W[gidx] = w
                            placement[gidx].append(m)
                            gidx += 1
                            placed = True
            if not placed:
                return None
        if record:
            return placement, rem, W
        return True

    # Binary search on lam
    lo, hi = 0.0, 1e9
    if pack(hi) is None:
        hi = None  # infeasible even at huge threshold -> memory-only fallback
    else:
        for _ in range(60):
            mid = (lo + hi) / 2
            if pack(mid) is not None:
                hi = mid
            else:
                lo = mid

    result = pack(hi, record=True)
    if result is None:
        # Fallback: memory-only best-fit packing
        placement = {i: [] for i in range(gpu_num)}
        used = [0.0] * gpu_num
        W = [0.0] * gpu_num
        for m in sorted_models:
            best_i, best_val = None, float('inf')
            for i in range(gpu_num):
                if m.model_size <= GPU_MEM_SIZE - used[i]:
                    val = safe_ratio(W[i] + m.req_rate / m.slo, used[i] + m.model_size)
                    if val < best_val:
                        best_val, best_i = val, i
            if best_i is None:
                raise ValueError(
                    f"Unable to place model of size {m.model_size} GB on any GPU."
                )
            placement[best_i].append(m)
            used[best_i] += m.model_size
            W[best_i] += m.req_rate / m.slo
        return placement

    placement, rem, W = result
    used = [GPU_MEM_SIZE - r for r in rem]

    # --- Local search refinement: move models off the hottest GPU ---
    def kvpr(i):
        return safe_ratio(W[i], used[i])

    improved = True
    guard = 0
    while improved and guard < 5000:
        improved = False
        guard += 1
        max_i = max(range(gpu_num), key=kvpr)
        max_val = kvpr(max_i)
        if max_val == float('inf'):
            break
        for m in list(placement[max_i]):
            w = m.req_rate / m.slo
            s = m.model_size
            for j in range(gpu_num):
                if j == max_i:
                    continue
                if s <= GPU_MEM_SIZE - used[j]:
                    n1 = safe_ratio(W[max_i] - w, used[max_i] - s)
                    n2 = safe_ratio(W[j] + w, used[j] + s)
                    if gpu_num > 2:
                        others = max(kvpr(k) for k in range(gpu_num) if k not in (max_i, j))
                        nmax = max(n1, n2, others)
                    else:
                        nmax = max(n1, n2)
                    if nmax < max_val - 1e-12:
                        placement[max_i].remove(m)
                        placement[j].append(m)
                        W[max_i] -= w
                        used[max_i] -= s
                        W[j] += w
                        used[j] += s
                        improved = True
                        break
            if improved:
                break

    return placement

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    # Test the algorithm

    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float
    import numpy as np

    test_cases = generate_test_gpu_models()
    all_kvpr = []
    for i, (gpu_num, gpu_models) in enumerate(test_cases):

        results = compute_model_placement(gpu_num, gpu_models)
        max_kvpr = calculate_kvcache_pressure(results)
        all_kvpr.append(safe_float(max_kvpr))

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr


    print(f"Max KVPR: {avg_kvpr:.3f}")
