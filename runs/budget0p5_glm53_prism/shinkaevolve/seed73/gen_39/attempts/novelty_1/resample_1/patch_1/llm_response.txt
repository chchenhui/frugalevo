regret_multistart_ils
</DESCRIPTION>
Keep the existing GPUState/refine/ILS framework but add: (1) regret-based greedy candidates that place each model on the GPU with the best-vs-second-best KVPR differential, (2) a refine variant that also considers moving models off the second-hottest GPU, and (3) slightly longer ILS with multiple seeds. All memory checks guard `remaining_mem <= 0` with `float('inf')`.
</CODE>
```python
GPU_MEM_SIZE = 80 # GB

# EVOLVE-BLOCK-START

import random

def compute_model_placement(gpu_num, models):
    """
    Compute a model placement that minimizes the maximum KVPR across all GPUs.
    Approach: multi-seed greedy construction (incl. regret ordering) + move/swap
    local search + iterated local search.
    """
    rng = random.Random(12345)

    class GPUState:
        __slots__ = ("models", "remaining_mem", "load")
        def __init__(self):
            self.models = []
            self.remaining_mem = GPU_MEM_SIZE
            self.load = 0.0

        def kvpr(self):
            if self.remaining_mem <= 0:
                return float('inf')
            return self.load / self.remaining_mem

    def clone_gpus(gpus):
        new = []
        for g in gpus:
            ng = GPUState()
            ng.models = list(g.models)
            ng.remaining_mem = g.remaining_mem
            ng.load = g.load
            new.append(ng)
        return new

    def build_placement(sorted_models, use_post_kvpr):
        gpus = [GPUState() for _ in range(gpu_num)]
        for model in sorted_models:
            req_per_slo = model.req_rate / model.slo
            best_idx, best_val = None, None
            for i, g in enumerate(gpus):
                if model.model_size >= g.remaining_mem:
                    continue
                if use_post_kvpr:
                    val = (g.load + req_per_slo) / (g.remaining_mem - model.model_size)
                else:
                    val = g.kvpr()
                if best_val is None or val < best_val:
                    best_val, best_idx = val, i
            if best_idx is None:
                return None
            gpus[best_idx].models.append(model)
            gpus[best_idx].remaining_mem -= model.model_size
            gpus[best_idx].load += req_per_slo
        return gpus

    def build_placement_regret(sorted_models):
        # Regret greedy: at each step, choose the model whose best-vs-second-best
        # KVPR differential is largest, place it on its best GPU.
        remaining = list(sorted_models)
        gpus = [GPUState() for _ in range(gpu_num)]
        while remaining:
            best_pick = None   # (regret, model, gpu_idx)
            for model in remaining:
                req_per_slo = model.req_rate / model.slo
                vals = []
                for i, g in enumerate(gpus):
                    if model.model_size >= g.remaining_mem:
                        continue
                    vals.append(((g.load + req_per_slo) / (g.remaining_mem - model.model_size), i))
                if not vals:
                    return None
                vals.sort()
                regret = vals[1][0] - vals[0][0] if len(vals) > 1 else float('inf')
                if best_pick is None or regret > best_pick[0]:
                    best_pick = (regret, model, vals[0][1])
            regret, model, idx = best_pick
            gpus[idx].models.append(model)
            gpus[idx].remaining_mem -= model.model_size
            gpus[idx].load += model.req_rate / model.slo
            remaining.remove(model)
        return gpus

    def max_kvpr(gpus):
        return max(g.kvpr() for g in gpus)

    def refine(gpus, rounds=200):
        for _ in range(rounds):
            kvs = [g.kvpr() for g in gpus]
            # consider moves off the hottest or second-hottest GPU
            order = sorted(range(len(gpus)), key=lambda i: -kvs[i])
            improved = False
            for hot in order[:2]:
                if improved:
                    break
                for idx in range(len(gpus[hot].models) - 1, -1, -1):
                    model = gpus[hot].models[idx]
                    rps = model.req_rate / model.slo
                    for j, g in enumerate(gpus):
                        if j == hot:
                            continue
                        if model.model_size >= g.remaining_mem:
                            continue
                        if gpus[hot].remaining_mem + model.model_size <= 0:
                            continue
                        new_hot = (gpus[hot].load - rps) / (gpus[hot].remaining_mem + model.model_size)
                        new_other = (g.load + rps) / (g.remaining_mem - model.model_size)
                        new_kvs = list(kvs)
                        new_kvs[hot] = new_hot
                        new_kvs[j] = new_other
                        if max(new_kvs) < kvs[hot] - 1e-12:
                            gpus[hot].models.pop(idx)
                            gpus[hot].remaining_mem += model.model_size
                            gpus[hot].load -= rps
                            g.models.append(model)
                            g.remaining_mem -= model.model_size
                            g.load += rps
                            improved = True
                            break
                    if improved:
                        break
            if not improved:
                # swaps between hottest GPU and others
                hot = order[0]
                for idx in range(len(gpus[hot].models)):
                    m_hot = gpus[hot].models[idx]
                    rps_hot = m_hot.req_rate / m_hot.slo
                    for j, g in enumerate(gpus):
                        if j == hot:
                            continue
                        for k in range(len(g.models)):
                            m_other = g.models[k]
                            rps_other = m_other.req_rate / m_other.slo
                            if gpus[hot].remaining_mem + m_hot.model_size - m_other.model_size <= 0:
                                continue
                            if g.remaining_mem + m_other.model_size - m_hot.model_size <= 0:
                                continue
                            new_hot = (gpus[hot].load - rps_hot + rps_other) / (
                                gpus[hot].remaining_mem + m_hot.model_size - m_other.model_size)
                            new_other = (g.load - rps_other + rps_hot) / (
                                g.remaining_mem + m_other.model_size - m_hot.model_size)
                            new_kvs = list(kvs)
                            new_kvs[hot] = new_hot
                            new_kvs[j] = new_other
                            if max(new_kvs) < kvs[hot] - 1e-12:
                                gpus[hot].models[idx] = m_other
                                g.models[k] = m_hot
                                gpus[hot].remaining_mem += m_hot.model_size - m_other.model_size
                                gpus[hot].load += rps_other - rps_hot
                                g.remaining_mem += m_other.model_size - m_hot.model_size
                                g.load += rps_hot - rps_other
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
            if not improved:
                break
        return gpus

    key_rs = lambda m: m.req_rate / m.slo
    key_size = lambda m: m.model_size
    key_density = lambda m: key_rs(m) / m.model_size

    candidates = []
    orders = [
        sorted(models, key=key_rs, reverse=True),
        sorted(models, key=key_size, reverse=True),
        sorted(models, key=key_density, reverse=True),
        sorted(models, key=key_size, reverse=False),
    ]
    for o in orders:
        candidates.append((o, True))
        candidates.append((o, False))

    refined = []
    for sorted_models, use_post in candidates:
        gpus = build_placement(sorted_models, use_post)
        if gpus is not None:
            gpus = refine(gpus)
            refined.append((max_kvpr(gpus), gpus))
    for o in orders:
        gpus = build_placement_regret(o)
        if gpus is not None:
            gpus = refine(gpus)
            refined.append((max_kvpr(gpus), gpus))

    if not refined:
        raise ValueError("Unable to place all models within GPU memory.")

    refined.sort(key=lambda t: t[0])
    best_score = refined[0][0]
    best_gpus = refined[0][1]

    # Iterated local search from each of the top-3 refined candidates
    for start_score, start_gpus in refined[:3]:
        cur_gpus = start_gpus
        cur_score = start_score
        ils_rng = random.Random(777)
        for _ in range(200):
            cand = clone_gpus(cur_gpus)
            ok = True
            for _step in range(ils_rng.randint(1, 3)):
                srcs = [i for i in range(gpu_num) if cand[i].models]
                if not srcs:
                    ok = False
                    break
                src = ils_rng.choice(srcs)
                m_idx = ils_rng.randrange(len(cand[src].models))
                model = cand[src].models.pop(m_idx)
                cand[src].remaining_mem += model.model_size
                cand[src].load -= model.req_rate / model.slo
                dsts = [i for i in range(gpu_num) if i != src and model.model_size < cand[i].remaining_mem]
                if not dsts:
                    cand[src].models.insert(m_idx, model)
                    cand[src].remaining_mem -= model.model_size
                    cand[src].load += model.req_rate / model.slo
                    break
                dst = ils_rng.choice(dsts)
                cand[dst].models.append(model)
                cand[dst].remaining_mem -= model.model_size
                cand[dst].load += model.req_rate / model.slo
            if not ok:
                break
            cand = refine(cand)
            score = max_kvpr(cand)
            if score < cur_score - 1e-12:
                cur_score = score
                cur_gpus = cand
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_gpus = cur_gpus

    return {gpu_id: best_gpus[gpu_id].models for gpu_id in range(gpu_num)}

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    # Test the algorithm

    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float
    import numpy as np

    test_cases = generate_test_gpu_models()
    all_kvpr = []
    for i, (gpu_num, gpu_models) in enumerate(test_cases):

        results = compute_model_placement(gpu_num, gpu_models)
        max_kvpr = calculate_kvcache_pressure(results)
        all_kvpr.append(safe_float(max_kvpr))

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr


    print(f"Max KVPR: {avg_kvpr:.3f}")
```
