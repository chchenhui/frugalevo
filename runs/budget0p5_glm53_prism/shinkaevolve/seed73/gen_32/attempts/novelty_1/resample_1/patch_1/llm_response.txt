multistart_ils_placement</NAME>
<DESCRIPTION>
Combine the best greedy variant (minimize post-placement KVPR, tie-break by more remaining memory) with a candidate-pool approach: build several diverse greedy initial solutions (different sort orders and tie-breaking rules), run local search (move + swap + 3-cycle) from each candidate, and keep the best result. This hedges against a single greedy ordering landing in a poor local basin, as recommended. Runtime is tiny so we can afford multiple restarts. Local-search moves are shared via helper functions to keep the code compact.</DESCRIPTION>

GPU_MEM_SIZE = 80 # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """
    Compute a model placement that minimizes the maximum KVPR across all GPUs.
    Multi-start greedy + local search (move/swap/3-cycle).
    """

    def w_of(m):
        return m.req_rate / m.slo

    def build_greedy(order, mode):
        placement = {g: [] for g in range(gpu_num)}
        shared_kv = [GPU_MEM_SIZE] * gpu_num
        wr = [0.0] * gpu_num
        for model in order:
            best_idx, best_ratio, best_mem = None, float('inf'), -1.0
            for g in range(gpu_num):
                rem = shared_kv[g] - model.model_size
                if rem <= 0:
                    continue
                if mode == 0:
                    r = (wr[g] + w_of(model)) / rem
                else:
                    r = wr[g] / shared_kv[g] if shared_kv[g] > 0 else float('inf')
                if r < best_ratio - 1e-12 or (abs(r - best_ratio) <= 1e-9 and rem > best_mem):
                    best_ratio, best_mem, best_idx = r, rem, g
            if best_idx is None:
                return None
            placement[best_idx].append(model)
            wr[best_idx] += w_of(model)
            shared_kv[best_idx] -= model.model_size
        return placement

    def kvprs(placement):
        out = []
        for g in range(gpu_num):
            mem_left = GPU_MEM_SIZE - sum(m.model_size for m in placement[g])
            wr = sum(w_of(m) for m in placement[g])
            out.append(float('inf') if mem_left <= 0 else wr / mem_left)
        return out

    def local_search(placement):
        kvs = kvprs(placement)
        for _round in range(25):
            best_max = max(kvs)
            improved = False
            # single-model move from a hottest GPU
            hot = [g for g in range(gpu_num) if kvs[g] >= best_max - 1e-12]
            for src in hot:
                if improved:
                    break
                for pos in range(len(placement[src])):
                    if improved:
                        break
                    model = placement[src][pos]
                    for dst in range(gpu_num):
                        if dst == src:
                            continue
                        mem_left_dst = GPU_MEM_SIZE - sum(m.model_size for m in placement[dst]) - model.model_size
                        if mem_left_dst <= 0:
                            continue
                        # simulate
                        placement[src].pop(pos)
                        placement[dst].append(model)
                        new_kvs = kvprs(placement)
                        if max(new_kvs) < best_max - 1e-12:
                            kvs = new_kvs
                            improved = True
                            break
                        placement[dst].pop()
                        placement[src].insert(pos, model)
                if improved:
                    break
            if improved:
                continue
            # pairwise swap
            for src in range(gpu_num):
                for dst in range(src + 1, gpu_num):
                    if improved:
                        break
                    for p1 in range(len(placement[src])):
                        for p2 in range(len(placement[dst])):
                            m1, m2 = placement[src][p1], placement[dst][p2]
                            if (GPU_MEM_SIZE - sum(m.model_size for m in placement[src]) - m2.model_size + m1.model_size <= 0 or
                                GPU_MEM_SIZE - sum(m.model_size for m in placement[dst]) - m1.model_size + m2.model_size <= 0):
                                continue
                            placement[src][p1], placement[dst][p2] = m2, m1
                            new_kvs = kvprs(placement)
                            if max(new_kvs) < best_max - 1e-12:
                                kvs = new_kvs
                                improved = True
                                break
                            placement[src][p1], placement[dst][p2] = m1, m2
                        if improved:
                            break
                    if improved:
                        break
            if improved:
                continue
            break
        return placement

    # candidate pool: diverse orders and greedy modes
    orders = [
        sorted(models, key=lambda m: -w_of(m)),
        sorted(models, key=lambda m: -m.model_size),
        sorted(models, key=lambda m: m.model_size),
        sorted(models, key=lambda m: -(w_of(m) / m.model_size)),
        sorted(models, key=lambda m: w_of(m)),
    ]
    candidates = []
    for order in orders:
        for mode in (0, 1):
            p = build_greedy(order, mode)
            if p is not None:
                candidates.append(p)

    best_placement, best_score = None, float('inf')
    for cand in candidates:
        p = {g: list(cand[g]) for g in range(gpu_num)}
        p = local_search(p)
        score = max(kvprs(p))
        if score < best_score:
            best_score, best_placement = score, p

    if best_placement is None:
        raise ValueError("Unable to place models on any GPU.")

    return best_placement

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    # Test the algorithm

    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float

    test_cases = generate_test_gpu_models()
    all_kvpr = []
    for i, (gpu_num, gpu_models) in enumerate(test_cases):

        results = compute_model_placement(gpu_num, gpu_models)
        max_kvpr = calculate_kvcache_pressure(results)
        all_kvpr.append(safe_float(max_kvpr))

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr)


    print(f"Max KVPR: {avg_kvpr:.3f}")
