def compute_model_placement(gpu_num, models):
    """
    Compute a model placement that minimizes the maximum KVPR across all GPUs.

    Args:
        gpu_num: Number of GPUs
        models: List of models to place

    Returns:
        A placement of models to GPUs
    """
    if gpu_num <= 0:
        raise ValueError("gpu_num must be positive")

    # 1) Sort models by r_j / s_j in descending order
    sorted_models = sorted(models, key=lambda m: (m.req_rate / m.slo), reverse=True)

    best_placement = None
    best_score = float('inf')
    last_error = None

    # 2) Multi-restart with seed-perturbed randomized tie-breaking; keep the best
    for seed in range(12):
        placement, score = _run_attempt(gpu_num, sorted_models, seed)
        if placement is None:
            last_error = ValueError(
                "Unable to place all models on the available GPUs."
            )
            continue
        if score < best_score:
            best_score = score
            best_placement = placement

    if best_placement is None:
        raise last_error if last_error is not None else ValueError("Placement failed.")

    return best_placement
