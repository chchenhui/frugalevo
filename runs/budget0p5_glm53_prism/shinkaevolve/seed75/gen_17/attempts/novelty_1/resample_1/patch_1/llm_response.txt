multi_order_greedy_local_search</NAME>
<DESCRIPTION>
The current program uses a single greedy ordering (by req_rate/slo) plus local search. The key parameters are the model ordering key and the local search neighborhood. I propose running the greedy + local-search pipeline with multiple ordering keys — req_rate/slo, model_size, density req_rate/(slo*model_size), and the composite (weight, size) — then returning the placement with the lowest resulting max KVPR. This is essentially free computationally and directly lowers the max KVPR objective, as different orderings favor different test instances.</DESCRIPTION>

GPU_MEM_SIZE = 80 # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """
    Compute a model placement that minimizes the maximum KVPR across all GPUs.

    Args:
        gpu_num: Number of GPUs
        models: List of models to place

    Returns:
        A placement of models to GPUs
    """

    def kvpr_of(load, used_mem):
        denom = GPU_MEM_SIZE - used_mem
        if denom <= 0:
            return float('inf') if load > 0 else 0.0
        return load / denom

    def run_pipeline(order_key):
        sorted_models = sorted(models, key=order_key, reverse=True)

        placement = {g: [] for g in range(gpu_num)}
        free_mem = [GPU_MEM_SIZE] * gpu_num
        load = [0.0] * gpu_num

        for model in sorted_models:
            w = model.req_rate / model.slo
            best_idx, best_kvpr = None, float('inf')
            for g in range(gpu_num):
                if model.model_size <= free_mem[g]:
                    k = kvpr_of(load[g] + w,
                                GPU_MEM_SIZE - free_mem[g] + model.model_size)
                    if k < best_kvpr:
                        best_kvpr, best_idx = k, g
            if best_idx is None:
                # fallback: last-resort exact-fill search
                for g in range(gpu_num):
                    if model.model_size <= free_mem[g]:
                        best_idx = g
                        break
            if best_idx is None:
                return None, float('inf')
            placement[best_idx].append(model)
            load[best_idx] += w
            free_mem[best_idx] -= model.model_size

        # local search: move single models to reduce max KVPR
        for _ in range(40):
            kvprs = [kvpr_of(load[g], GPU_MEM_SIZE - free_mem[g])
                     for g in range(gpu_num)]
            hot = max(range(gpu_num), key=lambda g: kvprs[g])
            hot_kvpr = kvprs[hot]
            others = [kvprs[g] for g in range(gpu_num) if g != hot]
            base_others = max(others) if others else 0.0

            best_gain, best_move = 0.0, None
            for model in placement[hot]:
                w = model.req_rate / model.slo
                new_used = GPU_MEM_SIZE - free_mem[hot] - model.model_size
                src_kvpr = kvpr_of(load[hot] - w, new_used) if new_used > 0 else 0.0
                for g in range(gpu_num):
                    if g == hot or model.model_size > free_mem[g]:
                        continue
                    dst_kvpr = kvpr_of(load[g] + w,
                                       GPU_MEM_SIZE - free_mem[g] + model.model_size)
                    new_max = max(base_others, src_kvpr, dst_kvpr)
                    gain = hot_kvpr - new_max
                    if gain > best_gain + 1e-12:
                        best_gain, best_move = gain, (model, g)

            if best_move is None:
                break
            model, dst = best_move
            w = model.req_rate / model.slo
            placement[hot].remove(model)
            placement[dst].append(model)
            load[hot] -= w
            free_mem[hot] += model.model_size
            load[dst] += w
            free_mem[dst] -= model.model_size

        kvprs = [kvpr_of(load[g], GPU_MEM_SIZE - free_mem[g])
                 for g in range(gpu_num)]
        return placement, max(kvprs)

    if gpu_num <= 0:
        raise ValueError("gpu_num must be positive")

    orderings = [
        lambda m: m.req_rate / m.slo,
        lambda m: m.model_size,
        lambda m: (m.req_rate / m.slo) / max(m.model_size, 1e-9),
        lambda m: (m.req_rate / m.slo, m.model_size),
    ]

    best_placement, best_max = None, float('inf')
    for key in orderings:
        placement, mx = run_pipeline(key)
        if placement is not None and mx < best_max:
            best_max, best_placement = mx, placement

    if best_placement is None:
        raise ValueError("Unable to place all models on the given GPUs.")

    return best_placement

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    # Test the algorithm

    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float

    test_cases = generate_test_gpu_models()
    all_kvpr = []

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr


    print(f"Max KVPR: {avg_kvpr:.3f}")
