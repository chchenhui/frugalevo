GPU_MEM_SIZE = 80 # GB

# EVOLVE-BLOCK-START

def compute_model_placement(gpu_num, models):
    """
    Compute a model placement minimizing the maximum KVPR across GPUs.
    Multi-ordering best-fit greedy + move + swap local search.
    """
    if gpu_num <= 0:
        raise ValueError("gpu_num must be positive")

    def kvpr_of(load, free):
        denom = GPU_MEM_SIZE - free
        if denom <= 0:
            return float('inf') if load > 0 else 0.0
        return load / denom

    def evaluate(placement):
        loads = [sum(m.req_rate / m.slo for m in placement[g]) for g in placement]
        frees = [GPU_MEM_SIZE - sum(m.model_size for m in placement[g]) for g in placement]
        vals = [kvpr_of(loads[g], frees[g]) for g in placement]
        return max(vals) if vals else 0.0

    def run_pipeline(order_key):
        placement = {g: [] for g in range(gpu_num)}
        free_mem = [GPU_MEM_SIZE] * gpu_num
        load = [0.0] * gpu_num

        def kvpr(g):
            return kvpr_of(load[g], free_mem[g])

        def max_kvpr(exclude=None):
            vals = [kvpr(g) for g in range(gpu_num) if g != exclude]
            return max(vals) if vals else 0.0

        sorted_models = sorted(models, key=order_key, reverse=True)

        # best-fit greedy
        for model in sorted_models:
            best_g, best_val = None, float('inf')
            for g in range(gpu_num):
                if model.model_size > free_mem[g]:
                    continue
                denom = GPU_MEM_SIZE - (free_mem[g] - model.model_size)
                if denom <= 0:
                    continue
                val = (load[g] + model.req_rate / model.slo) / denom
                if val < best_val:
                    best_val, best_g = val, g
            if best_g is None:
                raise ValueError(
                    f"Unable to place model of size {model.model_size} GB on any GPU. "
                    f"Free memory per GPU: {free_mem}"
                )
            placement[best_g].append(model)
            free_mem[best_g] -= model.model_size
            load[best_g] += model.req_rate / model.slo

        # ---- move phase ----
        improved = True
        it = 0
        while improved and it < 100:
            improved = False
            it += 1
            worst = max(range(gpu_num), key=lambda g: kvpr(g))
            cur_max = max_kvpr()
            for m_idx in range(len(placement[worst])):
                model = placement[worst][m_idx]
                for g in range(gpu_num):
                    if g == worst or model.model_size > free_mem[g]:
                        continue
                    placement[worst].pop(m_idx)
                    free_mem[worst] += model.model_size
                    load[worst] -= model.req_rate / model.slo
                    free_mem[g] -= model.model_size
                    load[g] += model.req_rate / model.slo
                    if max_kvpr() < cur_max - 1e-12:
                        placement[g].append(model)
                        improved = True
                        break
                    free_mem[g] += model.model_size
                    load[g] -= model.req_rate / model.slo
                    free_mem[worst] -= model.model_size
                    load[worst] += model.req_rate / model.slo
                    placement[worst].insert(m_idx, model)
                else:
                    continue
                break

        # ---- swap phase ----
        improved = True
        it = 0
        while improved and it < 100:
            improved = False
            it += 1
            worst = max(range(gpu_num), key=lambda g: kvpr(g))
            cur_max = max_kvpr()
            done = False
            for m_idx in range(len(placement[worst])):
                model = placement[worst][m_idx]
                for g in range(gpu_num):
                    if g == worst:
                        continue
                    for o_idx in range(len(placement[g])):
                        other = placement[g][o_idx]
                        # check fit
                        if other.model_size > free_mem[worst] + model.model_size:
                            continue
                        # simulate swap
                        placement[worst][m_idx] = other
                        placement[g][o_idx] = model
                        free_mem[worst] += model.model_size - other.model_size
                        load[worst] += (other.req_rate / other.slo) - (model.req_rate / model.slo)
                        free_mem[g] += other.model_size - model.model_size
                        load[g] += (model.req_rate / model.slo) - (other.req_rate / other.slo)
                        if max_kvpr() < cur_max - 1e-12:
                            improved = True
                            done = True
                            break
                        # revert
                        placement[worst][m_idx] = model
                        placement[g][o_idx] = other
                        free_mem[worst] -= model.model_size - other.model_size
                        load[worst] -= (other.req_rate / other.slo) - (model.req_rate / model.slo)
                        free_mem[g] -= other.model_size - model.model_size
                        load[g] -= (model.req_rate / model.slo) - (other.req_rate / other.slo)
                    if done:
                        break
                if done:
                    break

        return placement

    orderings = [
        lambda m: (m.req_rate / m.slo, -m.model_size),
        lambda m: (m.req_rate / m.slo),
        lambda m: (m.model_size,),
        lambda m: (m.req_rate / (m.slo * m.model_size),),
    ]

    best_placement = None
    best_val = float('inf')
    for key in orderings:
        try:
            p = run_pipeline(key)
            v = evaluate(p)
        except ValueError:
            continue
        if v < best_val:
            best_val = v
            best_placement = p

    if best_placement is None:
        # fall back to default ordering so error surfaces if truly infeasible
        best_placement = run_pipeline(lambda m: (m.req_rate / m.slo, -m.model_size))

    return best_placement

# EVOLVE-BLOCK-END


if __name__ == "__main__":
    # Test the algorithm

    from evaluator import generate_test_gpu_models
    from evaluator import calculate_kvcache_pressure
    from evaluator import safe_float
    import numpy as np

    test_cases = generate_test_gpu_models()
    all_kvpr = []
    for i, (gpu_num, gpu_models) in enumerate(test_cases):

        results = compute_model_placement(gpu_num, gpu_models)
        max_kvpr = calculate_kvcache_pressure(results)
        all_kvpr.append(safe_float(max_kvpr))

    avg_kvpr = np.mean(all_kvpr)
    if avg_kvpr != 0:
        avg_kvpr = 1.0 / avg_kvpr


    print(f"Max KVPR: {avg_kvpr:.3f}")
