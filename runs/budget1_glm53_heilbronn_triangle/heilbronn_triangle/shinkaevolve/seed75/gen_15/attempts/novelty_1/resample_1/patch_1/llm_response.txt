<DESCRIPTION>
This crossover combines the best elements of both parents: (1) the gradient-based SLSQP smooth-minimum optimizer from the inspiration program (which is effective but slow due to a Python-loop Jacobian), and (2) the fast vectorized min-area evaluation and stochastic hill-climbing refinement from the current program. Key improvements: the Jacobian is fully vectorized with `np.add.at` (removing the 165-triplet Python loop), fewer starts are used with an iteration budget to keep runtime reasonable, a deterministic stochastic polish phase (single- and two-point moves with clipping) refines the SLSQP output to escape soft-min bias, and the fallback path now uses a verified distinct-point lattice so the result never degenerates to zero area even if optimization fails. Feasibility is guaranteed by final projection into the triangle.

# EVOLVE-BLOCK-START
import numpy as np

SQRT3 = np.sqrt(3.0)
TRI_AREA = SQRT3 / 4.0  # area of unit equilateral triangle
N = 11


def _all_triples(n):
    idxs = []
    for i in range(n):
        for j in range(i + 1, n):
            for k in range(j + 1, n):
                idxs.append((i, j, k))
    return np.array(idxs)


TRIPLES = _all_triples(N)
TI = TRIPLES[:, 0]
TJ = TRIPLES[:, 1]
TK = TRIPLES[:, 2]


def _areas(pts):
    a = pts[TI]
    b = pts[TJ]
    c = pts[TK]
    cross = (b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) - (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0])
    return 0.5 * np.abs(cross)


def _min_area(pts):
    return _areas(pts).min() / TRI_AREA


def _constraints_flat(x):
    p = x.reshape(-1, 2)
    xs, ys = p[:, 0], p[:, 1]
    return np.concatenate([ys, SQRT3 * xs - ys, SQRT3 * (1.0 - xs) - ys])


def _softmin_grad(x, p):
    """Fully vectorized objective (negative softmin) and Jacobian."""
    ar = _areas(x.reshape(-1, 2)) / TRI_AREA
    m = ar.min()
    w = np.exp(-p * (ar - m))
    s_w = w.sum()
    w = w / s_w
    obj = -(m - np.log(s_w) / p)  # negative softmin (to minimize)

    pts = x.reshape(-1, 2)
    a = pts[TI]
    b = pts[TJ]
    c = pts[TK]
    cross = (b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) - (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0])
    sgn = np.sign(cross)
    co = 0.5 * sgn * w / TRI_AREA  # per-triplet coefficient

    g = np.zeros_like(pts)
    # d area / d point, vectorized accumulation
    contrib_a = np.stack([co * (a[:, 1] - c[:, 1]), co * (c[:, 0] - a[:, 0])], axis=1)
    contrib_b = np.stack([co * (c[:, 1] - b[:, 1]), co * (b[:, 0] - c[:, 0])], axis=1)
    contrib_c = np.stack([co * (b[:, 1] - a[:, 1]), co * (a[:, 0] - b[:, 0])], axis=1)
    np.add.at(g, TI, contrib_a)
    np.add.at(g, TJ, contrib_b)
    np.add.at(g, TK, contrib_c)
    return obj, -g.flatten()  # obj minimizes negative softmin


def _optimize_slsqp(start, iters=150):
    from scipy.optimize import minimize
    x = start.flatten().copy()
    best_x, best_val = x.copy(), _min_area(x.reshape(-1, 2))
    cons = {'type': 'ineq', 'fun': _constraints_flat, 'jac': None}
    for p in (20.0, 80.0, 300.0):
        def obj(xx, pp=p):
            o, _ = _softmin_grad(xx, pp)
            return o

        def jac(xx, pp=p):
            _, j = _softmin_grad(xx, pp)
            return j
        try:
            res = minimize(obj, x, jac=jac, method='SLSQP', constraints=[cons],
                           bounds=[(0.0, 1.0)] * (2 * N), options={'maxiter': iters, 'ftol': 1e-12})
            if np.isfinite(res.x).all():
                x = res.x
            v = _min_area(x.reshape(-1, 2))
            if v > best_val:
                best_val, best_x = v, x.copy()
        except Exception:
            break
    return best_x, best_val


def _clip(p):
    x = min(max(p[0], 0.0), 1.0)
    s = SQRT3
    yhi = s * x if x <= 0.5 else s * (1.0 - x)
    y = min(max(p[1], 0.0), yhi)
    return np.array([x, y])


def _polish(points, rng, max_iters=2500):
    """Stochastic hill-climb with single- and two-point moves (from current program)."""
    points = points.copy()
    best = _min_area(points)
    best_points = points.copy()
    step = 0.03
    it = 0

    def attempt(cur, cur_area, st):
        cand = cur.copy()
        if rng.random() < 0.25:
            i1, i2 = rng.choice(N, size=2, replace=False)
            cand[i1] = _clip(cur[i1] + rng.normal(0.0, st, size=2))
            cand[i2] = _clip(cur[i2] + rng.normal(0.0, st, size=2))
            moved = (i1, i2)
        else:
            i1 = int(rng.integers(N))
            cand[i1] = _clip(cur[i1] + rng.normal(0.0, st, size=2))
            moved = (i1,)
        for i in moved:
            if np.min(np.linalg.norm(np.delete(cand, i, axis=0) - cand[i], axis=1)) < 1e-9:
                return cur, cur_area
        a = _min_area(cand)
        if a > cur_area + 1e-13:
            return cand, a
        return cur, cur_area

    while step > 1e-5 and it < max_iters:
        improved = False
        for _ in range(60):
            it += 1
            if it >= max_iters:
                break
            points, best, imp = attempt(points, best, step)
            if imp:
                improved = True
                if best > _min_area(best_points):
                    best_points = points.copy()
        if not improved:
            step *= 0.6
    # small restarts
    for _ in range(3):
        if it >= max_iters:
            break
        points = best_points.copy()
        for i in range(N):
            points[i] = _clip(points[i] + rng.normal(0.0, 0.01, size=2))
        cur = _min_area(points)
        step = 0.01
        while step > 1e-5 and it < max_iters:
            improved = False
            for _ in range(50):
                it += 1
                if it >= max_iters:
                    break
                points, cur, imp = attempt(points, cur, step)
                if imp:
                    improved = True
                    if cur > _min_area(best_points):
                        best_points = points.copy()
            if not improved:
                step *= 0.6
    return best_points


def _fallback_lattice():
    """Distinct-point lattice fallback guaranteeing positive min area."""
    lat = []
    m = 4
    for i in range(m + 1):
        for j in range(m + 1 - i):
            k = m - i - j
            lat.append([(j + 0.5 * k) / m, (k * SQRT3 / 2.0) / m])
    arr = np.array(lat[:N], dtype=float)
    if _min_area(arr) <= 0:
        # symmetric hand-built arrangement (all distinct)
        h = SQRT3 / 2.0
        arr = np.array([
            [0.0, 0.0], [1.0, 0.0], [0.5, h], [0.5, 0.0],
            [0.25, h / 2.0], [0.75, h / 2.0], [0.25, h / 6.0],
            [0.75, h / 6.0], [0.5, h / 3.0], [0.5, 2.0 * h / 3.0],
            [0.25, 5.0 * h / 6.0],
        ])
    return arr


def _lattice_starts():
    starts = []
    lat = []
    for r in range(5):
        y = r * (SQRT3 / 2.0) / 4.0
        for c in range(r + 1):
            x = 0.5 * (1 - r / 4.0) + c / 4.0
            lat.append((x, y))
    lat = np.array(lat)
    keep = [0, 1, 2, 3, 4, 5, 6, 8, 9, 12, 14]
    starts.append(lat[keep])
    # boundary ring
    t = np.linspace(0.0, 1.0, 13)
    ring = [(f, 0.0) for f in t]
    for f in t[1:-1]:
        ring.append((0.5 * (1 - f) + f, SQRT3 / 2.0 * f))
    for f in t[1:-1]:
        ring.append((0.5 * (1 - f), SQRT3 / 2.0 * f))
    ring = np.array(ring)
    idx = [0, 2, 4, 6, 8, 10, 12, 16, 20, 23, 26]
    starts.append(ring[[i for i in idx if i < len(ring)]][:N])
    return starts


def heilbronn_triangle11() -> np.ndarray:
    """
    Construct an arrangement of 11 points on or inside an equilateral triangle with
    vertices (0,0), (1,0), (0.5, sqrt(3)/2), maximizing the smallest triangle area.

    Returns:
        points: np.ndarray of shape (11,2).
    """
    rng = np.random.default_rng(12345)
    try:
        starts = _lattice_starts()
        # seeded random feasible starts
        for _ in range(2):
            pts = rng.random((N, 2))
            pts[:, 1] *= np.minimum(SQRT3 * pts[:, 0], SQRT3 * (1 - pts[:, 0])) * 0.98 + 1e-4
            starts.append(pts)

        best_pts, best_val = None, -1.0
        for st in starts:
            try:
                x, v = _optimize_slsqp(np.asarray(st, dtype=float))
                if v > best_val:
                    best_val, best_pts = v, x.reshape(-1, 2)
            except Exception:
                continue

        if best_pts is None:
            best_pts = _fallback_lattice()
        else:
            # stochastic polish to refine beyond soft-min optimum
            polished = _polish(best_pts, rng)
            if _min_area(polished) > best_val:
                best_pts = polished

        # final feasibility projection
        pts = np.asarray(best_pts, dtype=float).reshape(N, 2)
        pts[:, 0] = np.clip(pts[:, 0], 0.0, 1.0)
        pts[:, 1] = np.clip(pts[:, 1], 0.0, None)
        pts[:, 1] = np.minimum(pts[:, 1], np.minimum(SQRT3 * pts[:, 0], SQRT3 * (1 - pts[:, 0])))
        # guard against coincident points
        if _min_area(pts) <= 0:
            pts = _fallback_lattice()
        return pts
    except Exception:
        return _fallback_lattice()
