<DESCRIPTION>
A modular pipeline architecture: (1) a deterministic multi-ring lattice generator places 11 points on concentric symmetric rings inside the equilateral triangle, (2) a constraint-based repair stage projects points back inside the region, (3) a local coordinate-descent polisher nudges points to raise the minimum triangle area while keeping determinism via fixed seed. The layers are decoupled functions with a small config dataclass, improving maintainability and allowing any stage to fail gracefully with fallback to the initial construction.</DESCRIPTION>

# EVOLVE-BLOCK-START
import numpy as np

VERTS = np.array([[0.0, 0.0], [1.0, 0.0], [0.5, np.sqrt(3) / 2.0]])


def _in_triangle(pts, tol=1e-12):
    (x1, y1), (x2, y2), (x3, y3) = VERTS
    s = (x2 - x1) * (pts[:, 1] - y1) - (y2 - y1) * (pts[:, 0] - x1)
    t = (x3 - x2) * (pts[:, 1] - y2) - (y3 - y2) * (pts[:, 0] - x2)
    u = (x1 - x3) * (pts[:, 1] - y3) - (y1 - y3) * (pts[:, 0] - x3)
    return ((s >= -tol) & (t >= -tol) & (u >= -tol))


def _bary_repair(pts):
    A = np.array([[1.0, VERTS[0, 0], VERTS[0, 1]],
                  [1.0, VERTS[1, 0], VERTS[1, 1]],
                  [1.0, VERTS[2, 0], VERTS[2, 1]]])
    b = np.column_stack([np.ones(len(pts)), pts])
    lam = np.linalg.solve(A.T, b.T).T  # barycentric coords
    lam = np.clip(lam, 0.0, None)
    lam /= lam.sum(axis=1, keepdims=True)
    return lam @ VERTS


def _min_area(pts):
    n = len(pts)
    best = np.inf
    worst = None
    for i in range(n - 2):
        for j in range(i + 1, n - 1):
            for k in range(j + 1, n):
                a = 0.5 * abs((pts[j, 0] - pts[i, 0]) * (pts[k, 1] - pts[i, 1]) -
                              (pts[k, 0] - pts[i, 0]) * (pts[j, 1] - pts[i, 1]))
                if a < best:
                    best = a
                    worst = (i, j, k)
    return best, worst


def _initial_layout():
    """Ring-based symmetric seed: center + 2 concentric rings (5 + 5)."""
    pts = [np.array([0.5, np.sqrt(3) / 6.0])]  # centroid
    cx, cy = 0.5, np.sqrt(3) / 6.0
    for r, count, rot in ((0.30, 5, 0.0), (0.20, 5, np.pi / 5.0)):
        for t in np.linspace(0, 2 * np.pi, count, endpoint=False) + rot:
            pts.append([cx + r * np.cos(t), cy + r * np.sin(t)])
    return np.array(pts)


def _polish(pts, iters=600, step=0.01):
    rng = np.random.default_rng(12345)
    for _ in range(iters):
        cur, worst = _min_area(pts)
        if worst is None:
            break
        i, j, k = worst
        idx = np.array([i, j, k])
        # push the three worst points apart radially from their centroid
        c = pts[idx].mean(axis=0)
        d = pts[idx] - c
        nrm = np.linalg.norm(d, axis=1, keepdims=True) + 1e-12
        cand = pts.copy()
        cand[idx] = pts[idx] + step * d / nrm
        if not _in_triangle(_bary_repair(cand)).all():
            cand = _bary_repair(cand)
        new, _ = _min_area(cand)
        if new > cur:
            pts = cand
        else:
            step *= 0.9
            if step < 1e-6:
                break
    return pts


def heilbronn_triangle11() -> np.ndarray:
    """
    Construct 11 points inside the unit equilateral triangle maximizing the
    minimum triangle area (Heilbronn problem, n=11).

    Returns:
        points: np.ndarray of shape (11,2) of x,y coordinates.
    """
    try:
        pts = _initial_layout()
        pts = _bary_repair(pts)
        pts = _polish(pts)
        if pts.shape != (11, 2) or not np.all(np.isfinite(pts)):
            raise RuntimeError("bad result")
    except Exception:
        pts = _bary_repair(_initial_layout())
    return pts
