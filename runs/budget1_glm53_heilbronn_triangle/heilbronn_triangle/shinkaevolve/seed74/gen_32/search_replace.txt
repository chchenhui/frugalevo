    def bottleneck_points(self, b, m=3):
        a = _areas(_to_xy(b), self.tri)
        thresh = a.min() * 1.5 + 1e-12
        tight = a <= thresh
        cnt = self.member[:, tight].sum(axis=1)
        return np.argsort(-cnt)[:m], cnt

    def weighted_pick(self, cnt, k):
        """Pick k distinct points with probability proportional to cnt."""
        w = cnt.astype(float) + 1e-12
        chosen = []
        for _ in range(k):
            p = w / w.sum()
            idx = int(self.rng.choice(len(w), p=p))
            chosen.append(idx)
            w[idx] = 0.0
            if w.sum() <= 0:
                break
        return chosen
