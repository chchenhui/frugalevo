def _seed_golden(seed=0):
    """Low-discrepancy seed from fractional parts of k*phi, k*phi^2."""
    phi = 0.5 * (1.0 + np.sqrt(5.0))
    k = np.arange(11) + 1 + seed
    b = np.empty((11, 2))
    b[:, 0] = (k * phi) % 1.0
    b[:, 1] = (k * phi * phi) % 1.0
    b[0] = [0.0, 0.0]
    b[1] = [1.0, 0.0]
    b[2] = [0.0, 1.0]
    return _clamp_simplex(b)


def _seed_repulsion(seed=0, iters=50):
    """Seed from vectorized inverse-square repulsion inside the simplex."""
    rng = np.random.default_rng(seed)
    b = rng.random((11, 2)) * 0.9
    b[0] = [0.0, 0.0]
    b[1] = [1.0, 0.0]
    b[2] = [0.0, 1.0]
    b = _clamp_simplex(b)
    for _ in range(iters):
        d = b[:, None, :] - b[None, :, :]
        dist2 = (d ** 2).sum(axis=2) + 1e-6
        np.fill_diagonal(dist2, 1.0)
        force = (d / (dist2 ** 1.5)[:, :, None]).sum(axis=1)
        b = _clamp_simplex(b + 0.01 * force / np.abs(force).max())
        b[0] = [0.0, 0.0]
        b[1] = [1.0, 0.0]
        b[2] = [0.0, 1.0]
    return _clamp_simplex(b)
