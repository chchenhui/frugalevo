<DESCRIPTION>
The current greedy jitter search plateaus at min_area 0.01 (score 0.39). The bottleneck is that greedy single-point moves cannot coordinate all 11 points simultaneously. A much better approach for this kind of max-min problem is to optimize a smooth surrogate of the min: the softmin log-sum-exp over all 165 triangle doubled-areas, whose gradient w.r.t. the barycentric coordinates is computable in closed form via vectorized numpy. Running Adam on this smooth objective pushes area values into near-degeneracy (many triangles near the min), then we harden the temperature (lower softmin sharpness) and finish with the existing local search as a polish step. This directly attacks the plateau and is the classic way to escape greedy local optima in Heilbronn-style packing problems.</DESCRIPTION>

def _area_gradients(b):
    """Doubled areas for all triplets and gradient d(cross)/d(b) for each point."""
    xy = _to_xy(b)
    T = _TRIPLETS
    p = xy[T[:, 0]]; q = xy[T[:, 1]]; r = xy[T[:, 2]]
    dqx = q[:, 0] - p[:, 0]; dqy = q[:, 1] - p[:, 1]
    drx = r[:, 0] - p[:, 0]; dry = r[:, 1] - p[:, 1]
    cross = np.abs(dqx * dry - dqy * drx)
    return xy, cross, (dqx, dqy, drx, dry)


def _smooth_opt(b0, iters=300, tau0=0.02, seed=0):
    """Adam on softmin log-sum-exp of doubled triangle areas."""
    b = _clamp_simplex(b0.copy())
    m = np.zeros_like(b); v = np.zeros_like(b)
    lr = 0.004
    for it in range(iters):
        tau = max(0.002, tau0 * np.exp(-it / 250.0))
        xy, cross, (dqx, dqy, drx, dry) = _area_gradients(b)
        # softmin weights: exp(-cross/tau), normalized
        w = np.exp(-(cross - cross.min()) / tau)
        w /= w.sum()
        # gradient of cross (signed) w.r.t. each of p,q,r (in xy), then chain to barycentric
        # d(dqx*dry - dqy*drx)/dp = (dry - drx*?) -> dp = ( -dry, drx )
        sign = np.where(dqx * dry - dqy * drx >= 0, 1.0, -1.0) * w
        gxy = np.zeros_like(b)  # reuse shape (11,2) as xy gradient container
        gxy_p = np.column_stack([-dry, drx]) * sign[:, None]
        gxy_q = np.column_stack([dry, -dqy * 0.0 - dqx * 0.0 + dry * 0.0])  # placeholder
        # correct q gradient: d/dq = (dry, -drx)
        gxy_q = np.column_stack([dry, -drx]) * sign[:, None]
        gxy_r = np.column_stack([-dqy, dqx]) * sign[:, None]
        grad = np.zeros((11, 2))
        np.add.at(grad, _TRIPLETS[:, 0], gxy_p)
        np.add.at(grad, _TRIPLETS[:, 1], gxy_q)
        np.add.at(grad, _TRIPLETS[:, 2], gxy_r)
        # chain rule xy -> barycentric: x = u + 0.5v, y = (sqrt3/2) v
        gb = np.column_stack([grad[:, 0], 0.5 * grad[:, 0] + 0.5 * _SQRT3 * grad[:, 1]])
        # Adam update (ascent)
        t = it + 1
        m = 0.9 * m + 0.1 * gb
        v = 0.999 * v + 0.001 * gb * gb
        mh = m / (1 - 0.9 ** t); vh = v / (1 - 0.999 ** t)
        b = b + lr * mh / (np.sqrt(vh) + 1e-8)
        b = _clamp_simplex(b)
    return b
