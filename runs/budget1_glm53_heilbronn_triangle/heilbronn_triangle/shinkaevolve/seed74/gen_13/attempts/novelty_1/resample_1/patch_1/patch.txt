+def _clip_to_triangle(pts: np.ndarray) -> np.ndarray:
+    """Clip points into the equilateral triangle (0,0),(1,0),(0.5,sqrt(3)/2)."""
+    s = np.sqrt(3.0)
+    p = pts.copy()
+    p[:, 0] = np.clip(p[:, 0], 0.0, 1.0)
+    p[:, 1] = np.clip(p[:, 1], 0.0, s / 2.0)
+    # upper edges: y <= sqrt(3)*x  and  y <= sqrt(3)*(1-x)
+    p[:, 1] = np.minimum(p[:, 1], s * p[:, 0])
+    p[:, 1] = np.minimum(p[:, 1], s * (1.0 - p[:, 0]))
+    return p
