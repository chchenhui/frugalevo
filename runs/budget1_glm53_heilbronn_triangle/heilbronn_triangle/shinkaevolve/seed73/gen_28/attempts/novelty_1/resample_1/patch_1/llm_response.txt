<DESCRIPTION>
Expands the multi-start to 8 diverse seeds (symmetric base, its perturbations, hexagonal-lattice-in-triangle seed, concentric "rings" seed, and several fixed-seed Dirichlet samples), each refined with a two-phase hill climb: a soft global-perturbation phase followed by a targeted hard phase that pushes vertices of the minimal triangle apart. Uses a ~8s time budget split across seeds, vectorized min-area over all 165 triplets, and barycentric folding for feasibility. More starts plus cheap vectorized evaluation should lift the normalized min-area well above the current 0.01.
</DESCRIPTION>

# EVOLVE-BLOCK-START
import time
import numpy as np
from itertools import combinations

_TRI_AREA = np.sqrt(3.0) / 4.0
_TRIPLETS = np.array(list(combinations(range(11), 3)), dtype=int)

_A = np.array([0.0, 0.0])
_B = np.array([1.0, 0.0])
_C = np.array([0.5, np.sqrt(3.0) / 2.0])
_VERTS = np.stack([_A, _B, _C])

_V0 = _B - _A
_V1 = _C - _A
_D00 = _V0 @ _V0
_D01 = _V0 @ _V1
_D11 = _V1 @ _V1
_DEN = _D00 * _D11 - _D01 * _D01


def _barycentric_fold(pts):
    pts = np.atleast_2d(pts)
    v2 = pts - _A
    d20 = v2 @ _V0
    d21 = v2 @ _V1
    b1 = (_D11 * d20 - _D01 * d21) / _DEN
    b2 = (_D00 * d21 - _D01 * d20) / _DEN
    b0 = 1.0 - b1 - b2
    b = np.stack([b0, b1, b2], axis=1)
    b = np.maximum(b, 0.0)
    b /= b.sum(axis=1, keepdims=True)
    return b @ _VERTS


def _min_area_and_triplet(pts):
    p = pts[_TRIPLETS]
    a = np.abs((p[:, 1, 0] - p[:, 0, 0]) * (p[:, 2, 1] - p[:, 0, 1]) -
               (p[:, 1, 1] - p[:, 0, 1]) * (p[:, 2, 0] - p[:, 0, 0]))
    m = a.argmin()
    return a[m] / (2.0 * _TRI_AREA), m


def _make_seeds(rng):
    n = 11
    h = np.sqrt(3.0) / 2.0
    base = np.array([
        [0.0, 0.0], [1.0, 0.0], [0.5, h],
        [0.25, h / 2.0], [0.75, h / 2.0], [0.5, 0.0],
        [0.125, h / 4.0], [0.875, h / 4.0],
        [0.375, 3.0 * h / 4.0], [0.625, 3.0 * h / 4.0],
        [0.5, h / 4.0],
    ])
    seeds = [base.copy()]
    # perturbations of the symmetric layout
    for s in (0.015, 0.04):
        seeds.append(_barycentric_fold(base + rng.normal(0.0, s, base.shape)))
    # hexagonal-lattice-like seed: pick 11 spread points from a fine lattice
    lin = np.linspace(0.05, 0.95, 9)
    cand = []
    for i, x in enumerate(lin):
        for y in np.linspace(0.02, h * 0.98, 8):
            cand.append((x, y))
    cand = np.array(cand)
    # farthest-point sampling of 11 points
    chosen = [cand[np.argmax(cand[:, 1] * 0 + cand[:, 0])]]
    d = np.linalg.norm(cand - chosen[0], axis=1)
    for _ in range(10):
        i = d.argmax()
        chosen.append(cand[i])
        d = np.minimum(d, np.linalg.norm(cand - cand[i], axis=1))
    seeds.append(_barycentric_fold(np.array(chosen)))
    # rings seed: concentric scaled triangles
    ring = []
    for scale in (1.0, 0.62, 0.3):
        for v in _VERTS:
            ring.append(_A + (v - _A) * scale)
    ring.append([0.5, h * 0.31])
    ring.append([0.5, h * 0.65])
    seeds.append(_barycentric_fold(np.array(ring) + rng.normal(0.0, 0.01, (11, 2))))
    # Dirichlet samples
    for alpha in (2.0, 5.0, 8.0):
        b = rng.dirichlet(alpha * np.ones(3), size=n)
        seeds.append(_barycentric_fold(b @ _VERTS))
    return seeds


def heilbronn_triangle11() -> np.ndarray:
    n = 11
    rng = np.random.default_rng(20240717)
    seeds = _make_seeds(rng)

    budget = 8.0
    deadline = time.time() + budget
    per_seed = budget / len(seeds)

    best = seeds[0]
    best_val = _min_area_and_triplet(best)[0]

    for seed in seeds:
        t0 = time.time()
        seed_deadline = min(deadline, t0 + per_seed)
        if time.time() > deadline or best_val >= 0.0365:
            break
        cur = seed.copy()
        cur_val, m = _min_area_and_triplet(cur)
        tri = _TRIPLETS[m]

        # --- soft phase: global annealed perturbations ---
        step = 0.06
        while time.time() < seed_deadline and step > 2e-4:
            improved = False
            i = int(rng.integers(n))
            cand = cur.copy()
            cand[i] = _barycentric_fold(cand[i] + rng.normal(0.0, step, 2))[0]
            v, m2 = _min_area_and_triplet(cand)
            if v > cur_val + 1e-13:
                cur, cur_val, tri = cand, v, _TRIPLETS[m2]
                improved = True
            if not improved:
                step *= 0.85

        # --- hard phase: targeted moves on the minimal triangle ---
        step = 0.02
        while time.time() < seed_deadline:
            improved = False
            i0, i1, i2 = tri
            cen = cur[[i0, i1, i2]].mean(axis=0)
            moves = []
            for vi in (i0, i1, i2):
                d = cur[vi] - cen
                nrm = np.hypot(*d)
                d = d / nrm if nrm > 1e-12 else np.array([0.0, 1.0])
                perp = np.array([-d[1], d[0]])
                for st in (step, 0.5 * step):
                    moves.append((vi, d * st))
                    moves.append((vi, d * st + perp * 0.4 * st))
                    moves.append((vi, d * st - perp * 0.4 * st))
                    moves.append((vi, -d * 0.3 * st))
            for _ in range(5):
                moves.append((int(rng.integers(n)), rng.normal(0.0, step, 2)))
            for vi, mv in moves:
                cand = cur.copy()
                cand[vi] = _barycentric_fold(cand[vi] + mv)[0]
                v, m2 = _min_area_and_triplet(cand)
                if v > cur_val + 1e-13:
                    cur, cur_val, tri = cand, v, _TRIPLETS[m2]
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-6:
                    break

        if cur_val > best_val:
            best, best_val = cur.copy(), cur_val

    pts = np.asarray(best, dtype=float)
    assert pts.shape == (n, 2)
    return pts


# EVOLVE-BLOCK-END
