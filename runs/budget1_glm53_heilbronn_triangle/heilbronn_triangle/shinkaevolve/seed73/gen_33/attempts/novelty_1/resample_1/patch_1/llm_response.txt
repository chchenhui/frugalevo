def _soft_search(pts, trip, step, iters, rng):
    """Annealed soft-min coordinate search: temp sharpens from 150 to 600
    over the iterations so the surrogate smoothly approaches hard-min."""
    pts = _project_inside(pts.copy())
    n = len(pts)
    temp0, temp1 = 150.0, 600.0
    NC = 8  # candidates per coordinate probe
    for it in range(iters):
        temp = temp0 + (temp1 - temp0) * it / max(1, iters - 1)
        best = _soft_objective(_all_areas(pts, trip), frac=0.20, temp=temp)

def _midk_refine(pts, trip, rounds, rng):
    """Transition phase: soft-min over a wider set (frac=0.25) with high
    sharpness, lifting all near-minimal triangles before hard polish."""
    pts = _project_inside(pts.copy())
    best = _soft_objective(_all_areas(pts, trip), frac=0.25, temp=600.0)
    step = 0.008
    for _ in range(rounds):
        improved = False
        for i in range(11):
            for d in range(2):
                for s in (+1, -1):
                    cand = pts.copy()
                    cand[i, d] += s * step
                    cand = _project_inside(cand)
                    v = _soft_objective(_all_areas(cand, trip), frac=0.25, temp=600.0)
                    if v > best + 1e-13:
                        pts, best = cand, v
                        improved = True
        if not improved:
            step *= 0.5
            if step < 1e-6:
                break
    return pts
