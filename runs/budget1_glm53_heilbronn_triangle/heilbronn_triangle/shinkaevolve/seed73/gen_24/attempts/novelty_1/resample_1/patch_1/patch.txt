 import numpy as np
 
 
 def _clamp_to_triangle(pts):
     """Project points back inside the unit equilateral triangle."""
     s3 = np.sqrt(3.0)
     x = np.clip(pts[:, 0], 0.0, 1.0)
     y = pts[:, 1]
     y = np.clip(y, 0.0, s3 * np.minimum(x, 1.0 - x))
     return np.stack([x, y], axis=1)
