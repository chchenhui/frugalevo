<DESCRIPTION>
Combines the best ideas from the context programs: Gen 16's targeted moves on the minimal-area triplet (its key edge), the current program's structured seed layouts, and the recommendation to expand to 6+ diverse seeds. Each start gets a coarse targeted phase (moves that expand the minimal triangle) followed by a fine coordinate-descent polish, all within a fixed time budget (~7.5s). Vectorized min-area over precomputed triplet indices keeps each evaluation fast, so many targeted iterations fit per start. Seeds include: vertices+medial layout, hexagonal-perturbation, concentric-ring layout, and three Dirichlet samples with fixed seeds.

# EVOLVE-BLOCK-START
import time
import numpy as np
from itertools import combinations

_TRI_AREA = np.sqrt(3.0) / 4.0
_TRIPLETS = np.array(list(combinations(range(11), 3)), dtype=int)
_N_TRIPLETS = len(_TRIPLETS)

_A = np.array([0.0, 0.0])
_B = np.array([1.0, 0.0])
_C = np.array([0.5, np.sqrt(3.0) / 2.0])
_VERTS = np.stack([_A, _B, _C])

_V0 = _B - _A
_V1 = _C - _A
_D00 = _V0 @ _V0
_D01 = _V0 @ _V1
_D11 = _V1 @ _V1
_DEN = _D00 * _D11 - _D01 * _D01


def _fold(pts):
    """Project points into the triangle via barycentric clamping."""
    v2 = pts - _A
    d20 = v2 @ _V0
    d21 = v2 @ _V1
    b1 = (_D11 * d20 - _D01 * d21) / _DEN
    b2 = (_D00 * d21 - _D01 * d20) / _DEN
    b0 = 1.0 - b1 - b2
    b = np.stack([b0, b1, b2], axis=1)
    b = np.maximum(b, 0.0)
    b /= b.sum(axis=1, keepdims=True)
    return b @ _VERTS


def _min_area_and_idx(pts):
    """Vectorized minimal normalized area + triplet index."""
    p = pts[_TRIPLETS]
    a = np.abs((p[:, 1, 0] - p[:, 0, 0]) * (p[:, 2, 1] - p[:, 0, 1]) -
               (p[:, 1, 1] - p[:, 0, 1]) * (p[:, 2, 0] - p[:, 0, 0]))
    m = a.argmin()
    return a[m] / (2.0 * _TRI_AREA), m


def _seed_set():
    h = np.sqrt(3.0) / 2.0
    rng = np.random.default_rng(20240711)
    seeds = []

    # 1. vertices + medial symmetric layout
    s1 = np.array([
        [0.0, 0.0], [1.0, 0.0], [0.5, h],
        [0.25, h / 2], [0.75, h / 2], [0.5, 0.0],
        [0.125, h / 4], [0.875, h / 4],
        [0.375, 3 * h / 4], [0.625, 3 * h / 4],
        [0.5, h / 4],
    ])
    seeds.append(_fold(s1))

    # 2. vertices + edge thirds + interior
    s2 = np.array([
        [0.0, 0.0], [1.0, 0.0], [0.5, h],
        [1/3, 0.0], [2/3, 0.0],
        [1/6, h / 3], [5/6, h / 3],
        [1/3, 2 * h / 3], [2/3, 2 * h / 3],
        [0.5, h / 3], [0.25, h / 6],
    ])
    seeds.append(_fold(s2))

    # 3. hexagonal-perturbation seed
    s3 = s1 + rng.normal(0.0, 0.025, s1.shape)
    seeds.append(_fold(s3))

    # 4. concentric rings: vertices, ring at 2/3, ring at 1/3
    ring = []
    for t in (1.0, 2.0 / 3.0, 1.0 / 3.0):
        for u in (0.0, 1.0 / 3.0, 2.0 / 3.0):
            b = np.array([1 - u, u, 0.0]) * t + (1 - t) / 3.0
            ring.append(b)
    for u in (1/6, 3/6, 5/6):
        b = np.array([0.5, u, 1 - 0.5 - u])
        ring.append(b)
    s4 = np.array(ring) @ _VERTS
    seeds.append(_fold(s4))

    # 5-7. Dirichlet samples with distinct fixed seeds
    for sd in (7, 42, 991):
        r2 = np.random.default_rng(sd)
        bary = r2.dirichlet(np.ones(3) * 3.0, size=11)
        bary = 0.5 + 0.6 * (bary - bary.mean(axis=0))
        bary = np.maximum(bary, 0.03)
        bary /= bary.sum(axis=1, keepdims=True)
        seeds.append(bary @ _VERTS)

    return seeds


def _optimize(cur, rng, deadline, phase_budget):
    """Targeted expansion of the minimal triplet + coordinate polish."""
    n = 11
    cur_val, m = _min_area_and_idx(cur)
    tri = _TRIPLETS[m]
    best = cur.copy()
    best_val = cur_val

    # ---- phase 1: targeted moves on the minimal triangle ----
    step = 0.03
    while time.time() < deadline and phase_budget > 0 and step > 1e-5:
        phase_budget -= 1
        i0, i1, i2 = tri
        cen = cur[[i0, i1, i2]].mean(axis=0)
        improved = False
        for vi in (i0, i1, i2):
            d = cur[vi] - cen
            nrm = np.hypot(*d)
            d = d / nrm if nrm > 1e-12 else np.array([0.0, 1.0])
            perp = np.array([-d[1], d[0]])
            moves = [d * step, d * step + 0.5 * step * perp,
                     d * step - 0.5 * step * perp, -0.4 * step * d,
                     step * perp, -step * perp]
            for mv in moves:
                cand = cur.copy()
                cand[vi] = _fold((cand[vi] + mv).reshape(1, 2))[0]
                v, m2 = _min_area_and_idx(cand)
                if v > cur_val + 1e-12:
                    cur, cur_val, tri = cand, v, _TRIPLETS[m2]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # random exploration move
            for _ in range(8):
                cand = cur.copy()
                j = int(rng.integers(n))
                cand[j] = _fold((cand[j] + rng.normal(0.0, step, 2)).reshape(1, 2))[0]
                v, m2 = _min_area_and_idx(cand)
                if v > cur_val + 1e-12:
                    cur, cur_val, tri = cand, v, _TRIPLETS[m2]
                    improved = True
                    break
        if not improved:
            step *= 0.6
        if cur_val > best_val:
            best, best_val = cur.copy(), cur_val

    # ---- phase 2: fine coordinate polish ----
    step = 2e-3
    cur, cur_val = best.copy(), best_val
    while time.time() < deadline and step > 1e-6:
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                break
            better = False
            for _ in range(4):
                cand = cur.copy()
                cand[i] = _fold((cand[i] + rng.normal(0.0, step, 2)).reshape(1, 2))[0]
                v, m2 = _min_area_and_idx(cand)
                if v > cur_val + 1e-14:
                    cur, cur_val = cand, v
                    improved_any = True
                    better = True
                    break
            _ = better
        if not improved_any:
            step *= 0.5
    if cur_val > best_val:
        best, best_val = cur.copy(), cur_val
    return best, best_val


def heilbronn_triangle11() -> np.ndarray:
    n = 11
    seeds = _seed_set()
    rng = np.random.default_rng(20240711)

    deadline = time.time() + 7.5
    best = seeds[0].copy()
    best_val = _min_area_and_idx(best)[0]

    per_start = max(len(seeds), 1)
    for si, seed in enumerate(seeds):
        if time.time() > deadline:
            break
        share = (deadline - time.time()) / max(1, per_start - si)
        sub_deadline = min(time.time() + share, deadline)
        out, val = _optimize(seed.copy(), rng, sub_deadline, 6000)
        if val > best_val:
            best, best_val = out, val
        if best_val >= 0.0365:
            break

    # safety fallback
    if best is None or not np.all(np.isfinite(best)):
        best = _fold(np.tile(np.array([0.4, 0.3]), (n, 1)))

    pts = np.asarray(best, dtype=float)
    assert pts.shape == (n, 2)
    return pts
