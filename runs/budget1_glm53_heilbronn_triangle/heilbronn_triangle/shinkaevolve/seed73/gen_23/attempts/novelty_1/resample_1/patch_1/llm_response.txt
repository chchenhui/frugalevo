<DESCRIPTION>
The recommendation notes that single-point moves plateau around min_area ≈ 0.02 (well below the known ~0.037 optimum) because individual perturbations cannot escape local optima where the minimal-triangle constraint involves tight clusters of points. I add a simulated-annealing refinement phase that (a) with ~25% probability perturbs 2–3 nearby points simultaneously with correlated Gaussian deltas scaled to the current step size, and (b) accepts uphill moves with a standard Metropolis criterion. This lets coordinated clusters shift together. To keep the evaluator's normalization convention untouched (it normalizes by the triangle area itself), we keep raw areas. The annealing runs after the hard polish, uses a wall-clock budget for robustness, and the best-ever configuration is always kept, so failures degrade gracefully to the previous pipeline output.</DESCRIPTION>

import time as _time

def _anneal_group(pts, rng, t_budget=2.0, step0=0.008):
    """Simulated annealing with correlated multi-point (group) moves.

    With ~25% probability, 2-3 nearby points are perturbed simultaneously
    with correlated Gaussian deltas; otherwise a single point moves. This
    escapes local optima where single-point moves cannot improve.
    """
    import math
    pts = _project(pts.copy())
    cur, cur_val = pts.copy(), _min_area(pts)[0]
    best, best_val = pts.copy(), cur_val
    step = step0
    T = step * 0.25          # temperature scale
    deadline = _time.time() + t_budget
    it = 0
    while _time.time() < deadline:
        it += 1
        cand = cur.copy()
        if rng.random() < 0.25:
            # group move: pick a point and its 1-2 nearest neighbours,
            # apply correlated deltas
            i = int(rng.integers(N))
            d2 = np.sum((cur - cur[i]) ** 2, axis=1)
            near = np.argsort(d2)[1:1 + int(rng.integers(1, 3))]
            group = np.concatenate(([i], near))
            shared = rng.normal(0.0, step * 0.6, size=2)
            for g in group:
                cand[g] += shared + rng.normal(0.0, step * 0.4, size=2)
        else:
            i = int(rng.integers(N))
            cand[i] += rng.normal(0.0, step, size=2)
        cand = _project(cand)
        v = _min_area(cand)[0]
        if v >= cur_val or rng.random() < math.exp((v - cur_val) / max(T, 1e-12)):
            cur, cur_val = cand, v
            if v > best_val:
                best, best_val = cand.copy(), v
        if it % 3000 == 0:
            step *= 0.85
            T *= 0.85
            if step < 1e-5:
                break
    # final hard polish from the annealing winner
    best = _hard_phase(best, 0.002, 40)
    return best

def _optimize():
    rng = np.random.default_rng(20240517)
    starts = [_lattice_seed(), _vertex_seed()]
    for _ in range(8):
        starts.append(_random_seed(rng))
    best_pts, best_val = None, -1.0
    for s in starts:
        c = _soft_phase(s, 0.02, 25)
        c = _hard_phase(c, 0.004, 70)
        v, _ = _min_area(c)
        if v > best_val:
            best_pts, best_val = c, v
    # group-move annealing refinements from the top starts
    top = sorted(starts, key=lambda s: -_min_area(_hard_phase(
        _soft_phase(s, 0.02, 10), 0.004, 30))[0])[:3]
    cands = [best_pts] + top
    for c0 in cands:
        try:
            c = _anneal_group(c0, rng, t_budget=1.2)
            v, _ = _min_area(c)
            if v > best_val:
