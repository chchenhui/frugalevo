All top variants plateau near min_area ≈ 0.03, suggesting the coordinate descent gets trapped in a local basin. Following the recommendation, I add a final structured escape phase: compute a per-vertex "binding score" (mean of the 20 smallest triangle areas containing each vertex), pick the binding vertex from the smallest triangle and swap it with a non-binding vertex, then re-polish with a short hard-min refinement. Keep the swap only if the hard minimum improves. This reuses the existing vectorized area machinery and stays deterministic.

def _vertex_binding_scores(pts, trip, k=20):
    """Mean of the k smallest triangle areas containing each vertex.
    Low score = vertex participates in many near-minimal (binding) triangles."""
    areas = _all_areas(pts, trip)
    n = len(pts)
    scores = np.zeros(n)
    for i in range(n):
        mask = (trip[0, :] == i) | (trip[1, :] == i) | (trip[2, :] == i)
        a = areas[mask]
        kk = min(k, len(a))
        scores[i] = np.mean(np.partition(a, kk - 1)[:kk])
    return scores


def _pair_exchange(pts, trip, rng, attempts=40):
    """Structured escape: swap a binding vertex (from the smallest triangle)
    with a non-binding vertex, then short hard polish. Keep only improvements."""
    pts = _project_inside(pts.copy())
    best_val, _ = _min_area(pts, trip)
    n = len(pts)
    for _ in range(attempts):
        _, kmin = _min_area(pts, trip)
        tri = trip[:, kmin]
        scores = _vertex_binding_scores(pts, trip)
        # binding vertex: smallest score among minimal-triangle vertices
        bi = min(tri, key=lambda v: scores[v])
        # non-binding candidates: top-scoring vertices not in tri
        cand = [i for i in range(n) if i not in tri]
        cand.sort(key=lambda v: -scores[v])
        ni = cand[rng.integers(0, min(5, len(cand)))]
        sw = pts.copy()
        sw[bi], sw[ni] = sw[ni].copy(), sw[bi].copy()
        # small displacement of the moved point to break symmetry
        sw[bi] = sw[bi] + rng.normal(0.0, 0.01, size=2)
        sw = _project_inside(sw)
        sw = _hard_refine(sw, trip, 0.01, 20, rng)
        val, _ = _min_area(sw, trip)
        if val > best_val + 1e-12:
            pts, best_val = sw, val
    return pts
