def _directed_polish(pts, trip, step, rounds, rng, k_tri=10):
    """Directed sweep over the k_tri smallest triangles: for each vertex of a
    binding triangle, probe moves along the outward normal of each adjacent
    edge (which raises the triangle's altitude), with step-halving on
    failure. 20% of budget keeps random kicks for exploration."""
    pts = _project_inside(pts.copy())
    best, _ = _min_area(pts, trip)
    for r in range(rounds):
        improved = False
        areas = _all_areas(pts, trip)
        order = np.argsort(areas)[:k_tri]
        for tk in order:
            i0, i1, i2 = trip[:, tk]
            verts = [i0, i1, i2]
            for vi in range(3):
                i = verts[vi]
                p = pts[verts[(vi + 1) % 3]] - pts[verts[(vi - 1) % 3]]
                plen = np.hypot(p[0], p[1])
                if plen < 1e-12:
                    continue
                # outward normal of the opposite edge (away from vertex)
                nrm = np.array([p[1], -p[0]]) / plen
                if np.dot(nrm, pts[i] - pts[verts[(vi + 1) % 3]]) < 0.0:
                    nrm = -nrm
                for s in (1.0, 0.5, 0.25, -0.5, -0.25):
                    cand = pts.copy()
                    cand[i] = cand[i] + s * step * nrm
                    cand = _project_inside(cand)
                    val, _ = _min_area(cand, trip)
                    if val > best + 1e-13:
                        pts, best, improved = cand, val, True
                        break
        # exploration budget: random kicks on binding points
        for t in range(12):
            tk = order[rng.integers(0, len(order))]
            i = trip[rng.integers(0, 3), tk]
            cand = pts.copy()
            cand[i] += rng.normal(0.0, step, size=2)
            cand = _project_inside(cand)
            val, _ = _min_area(cand, trip)
            if val > best + 1e-13:
                pts, best, improved = cand, val, True
        if not improved:
            step *= 0.5
            if step < 1e-7:
                break
    return pts
