<DESCRIPTION>
Add a targeted greedy hill-climb stage that directly manipulates the distance pairs that matter: the pair realizing the current minimum distance (push them apart along their separation direction) and the pair realizing the maximum distance (pull them toward each other), plus a coordinated triple move doing both simultaneously. Single-coordinate or smooth-gradient polishes often stall on plateaus because improving the ratio requires coordinated displacement of both endpoints. This stage uses exact ratio evaluation with an accept/reject shrinking step, is scale-invariant (re-normalizes diameter to 1), and is cheap (91 pairs). It is appended after Stage 5 so it can only improve the result.

    def renorm(P):
        P = P - P.mean(axis=0)
        return P / pair_dists(P).max()

    def hillclimb(P0, iters=400, step0=0.02):
        P = renorm(P0.copy())
        r_best = ratio(P)
        step = step0
        iu = np.array([(i, j) for i in range(N) for j in range(i + 1, N)])
        for it in range(iters):
            d = pair_dists(P)
            ia, ib = iu[d.argmin()]
            ja, jb = iu[d.argmax()]
            improved = False
            u_min = P[ia] - P[ib]
            n_min = np.linalg.norm(u_min)
            if n_min > 1e-12:
                u_min /= n_min
            u_max = P[ja] - P[jb]
            n_max = np.linalg.norm(u_max)
            if n_max > 1e-12:
                u_max /= n_max
            for mag in (step, step / 2, step / 4):
                # move 1: push min pair apart
                Q = P.copy()
                Q[ia] += mag * u_min
                Q[ib] -= mag * u_min
                Q = renorm(Q)
                r = ratio(Q)
                if r > r_best + 1e-14:
                    P, r_best, improved = Q, r, True
                    break
                # move 2: pull max pair inward
                Q = P.copy()
                Q[ja] -= mag * u_max
                Q[jb] += mag * u_max
                Q = renorm(Q)
                r = ratio(Q)
                if r > r_best + 1e-14:
                    P, r_best, improved = Q, r, True
                    break
                # move 3 (triple): do both simultaneously
                if (ia, ib) != (ja, jb):
                    Q = P.copy()
                    Q[ia] += mag * u_min
                    Q[ib] -= mag * u_min
                    Q[ja] -= mag * u_max
                    Q[jb] += mag * u_max
                    Q = renorm(Q)
                    r = ratio(Q)
                    if r > r_best + 1e-14:
                        P, r_best, improved = Q, r, True
                        break
