The best prior program (score 0.99, ratio 0.24) started from an icosahedron + 2 polar points and used an annealed soft objective followed by greedy coordinate descent. The current program's D6 double-ring family gets stuck at ratio 0.22 (score 0.92). I therefore replace the main construction with: (1) an icosahedron-plus-poles initialization (also compared against the D6 parametric optimum so we never regress), (2) the existing annealed soft-min/soft-max `polish`, and (3) a greedy coordinate descent on the exact ratio with adaptive per-point step sizes as recommended: binding points (those in the min-distance and max-distance pairs) get larger steps (0.05–0.1) while non-binding points get small steps (0.01), concentrating the 14×3×2 sweep budget on coordinates that actually affect the binding constraints. This preserves the exact-ratio convergence of the greedy stage while cutting wasted evaluations and improving the achievable ratio.

def icosa_plus_poles():
    """Icosahedron vertices (12) + 2 points near opposite z poles."""
    phi = (1 + np.sqrt(5)) / 2
    base = np.array([
        [-1, phi, 0], [1, phi, 0], [-1, -phi, 0], [1, -phi, 0],
        [0, -1, phi], [0, 1, phi], [0, -1, -phi], [0, 1, -phi],
        [phi, 0, -1], [phi, 0, 1], [-phi, 0, -1], [-phi, 0, 1],
    ], dtype=float)
    base /= np.linalg.norm(base, axis=1, keepdims=True)
    extra = np.array([[0.0, 0.0, 1.05], [0.1, -0.1, -1.05]])
    P = np.vstack([base, extra])
    P = P - P.mean(axis=0)
    return P / pair_dists(P).max()


def greedy_adaptive(P0, iters=600):
    """Greedy coordinate descent on the exact ratio with adaptive per-point steps.

    Points involved in the current min-distance or max-distance pair get
    larger steps (0.05-0.1); other points get small steps (0.01). This
    concentrates effort on the binding constraints.
    """
    P = P0 - P0.mean(axis=0)
    P = P / pair_dists(P).max()
    r_best = ratio(P)
    iu = np.array([(i, j) for i in range(N) for j in range(i + 1, N)])
    # per-point step sizes
    step = np.full(N, 0.01)
    for _ in range(iters):
        d = pair_dists(P)
        ia, ib = iu[d.argmin()]
        ja, jb = iu[d.argmax()]
        # adaptive per-point steps: binding points get big steps
        step[:] = 0.01
        for b in (ia, ib):
            step[b] = 0.1
        for b in (ja, jb):
            step[b] = max(step[b], 0.05)
        improved = False
        for i in range(N):
            for ax in range(D):
                for sgn in (1, -1):
                    Q = P.copy()
                    Q[i, ax] += sgn * step[i]
                    Q = Q - Q.mean(axis=0)
                    Q = Q / pair_dists(Q).max()
                    r = ratio(Q)
                    if r > r_best + 1e-9:
                        P, r_best, improved = Q, r, True
        if not improved:
            step *= 0.5
            if step.max() < 1e-6:
                break
    return P, r_best
