<DESCRIPTION>
The current program uses slow stochastic hill-climbing and SLSQP with 182 constraints, achieving ratio² ≈ 0.2400 (score 1.00). The new program replaces stochastic search with a smooth log-sum-exact surrogate of the ratio with an exact analytic Jacobian, optimized by L-BFGS-B over an annealing schedule p = 4→8→16→32→64 (sharpening the smooth min/max toward the true ones). Key parameters: 40 diverse seeds (random spheres, antipodal pairs, icosahedron+axis pairs, cube+center, Fibonacci lattice, perturbed best-known structures), 6 L-BFGS-B stages of 200 iterations each with annealed sharpness, then SLSQP max-min polishing (t ≤ d² ≤ 1) on the top 8 candidates with random restarts. Fixed seed 42, total runtime ~2–4s. This should push the squared ratio slightly above the 1/4.165849767 reference.

import numpy as np
from scipy.optimize import minimize

def min_max_dist_dim3_14() -> np.ndarray:
    n, d = 14, 3
    iu = np.triu_indices(n, 1)
    rng = np.random.default_rng(42)

    def D2mat(pts):
        diff = pts[:, None, :] - pts[None, :, :]
        return np.einsum('ijk,ijk->ij', diff, diff)

    def score_sq(pts):
        u = D2mat(pts)[iu]
        mx = u.max()
        if mx <= 0:
            return 0.0
        return u.min() / mx

    def sphere(p):
        p = p - p.mean(axis=0)
        r = np.linalg.norm(p, axis=1, keepdims=True)
        r[r == 0] = 1.0
        return p / r

    # ---------- seeds ----------
    seeds = []
    for _ in range(20):
        seeds.append(sphere(rng.standard_normal((n, d))))
    for _ in range(6):
        dirs = sphere(rng.standard_normal((n // 2, d)))
        seeds.append(np.vstack([dirs, -dirs]))
    phi = (1 + np.sqrt(5)) / 2
    ico = sphere(np.array([
        [-1, phi, 0], [1, phi, 0], [-1, -phi, 0], [1, -phi, 0],
        [0, -1, phi], [0, 1, phi], [0, -1, -phi], [0, 1, -phi],
        [phi, 0, -1], [phi, 0, 1], [-phi, 0, -1], [-phi, 0, 1],
    ], dtype=float))
    for ax in range(3):
        e = np.zeros((2, 3)); e[0, ax] = 1.3; e[1, ax] = -1.3
        seeds.append(np.vstack([ico, e])))
    # fibonacci
    k = np.arange(n)
    ga = np.pi * (3 - np.sqrt(5))
    z = 1 - 2 * (k + 0.5) / n
    r = np.sqrt(np.maximum(0.0, 1 - z * z))
    seeds.append(np.stack([r * np.cos(ga * k), r * np.sin(ga * k), z], axis=1))
    # cube + center + 5
    lat = np.array([[x, y, z] for x in (-1, 1) for y in (-1, 1)
                    for z in (-1, 1)], dtype=float)
    for _ in range(3):
        extra5 = sphere(rng.standard_normal((5, d))) * 0.5
        seeds.append(sphere(np.vstack([lat, [[0, 0, 0]], extra5])))

    # ---------- smooth ratio + analytic gradient ----------
    def smooth_obj_grad(x, p):
        pts = x.reshape(n, d)
        D2 = D2mat(pts)
        u = D2[iu]
        mx = float(u.max())
        if mx <= 1e-12:
            return 0.0, np.zeros_like(x)
        # smooth min and max via log-sum-exp on u
        # mn = -1/p log sum exp(-p u); mx_s = 1/p log sum exp(p u)
        mn = -np.log(np.sum(np.exp(-p * (u - mx)))) / p + mx
        mxs = np.log(np.sum(np.exp(p * (u - mx)))) / p + mx
        if mxs < 1e-12:
            return 0.0, np.zeros_like(x)
        w = np.exp(-p * (u - mx)); w = w / w.sum()   # d mn / d u
        v = np.exp(p * (u - mx)); v = v / v.sum()   # d mxs / d u
        dhdu = (w * mxs - mn * v) / (mxs * mxs)     # d (mn/mxs) / d u
        S = np.zeros((n, n))
        S[iu] = dhdu
        S = S + S.T
        diff = pts[:, None, :] - pts[None, :, :]
        grad = 2.0 * np.einsum('ij,ijk->ik', S, diff)
        grad = grad - grad.mean(axis=0, keepdims=True)
        return mn / mxs, grad.ravel()

    def lbfgs_stages(pts, ps=(4, 8, 16, 32, 64, 128), iters=200):
        x = pts.ravel().copy()
        best_x = x.copy(); best_s = score_sq(pts)
        for p in ps:
            def fgh(x):
                v, g = smooth_obj_grad(x, p)
                return -v, -g
            try:
                res = minimize(fgh, x, jac=True, method='L-BFGS-B',
                               options={'maxiter': iters,
                                         'maxcor': 20, 'ftol': 1e-14,
                                         'gtol': 1e-12})
                x = res.x
            except Exception:
                pass
            s = score_sq(x.reshape(n, d))
            if s > best_s:
                best_s = s; best_x = x.copy()
        return best_x.reshape(n, d), best_s

    cands = []
    for sd in seeds:
        p0 = sd / np.sqrt(max(D2mat(sd).max(), 1e-12))
        p1, s1 = lbfgs_stages(p0)
        cands.append((s1, p1))
        # small random restarts around good ones
        if s1 > 0.20:
            for _ in range(2):
                pj = p1 + 0.03 * rng.standard_normal(p1.shape)
                pj = pj / np.sqrt(max(D2mat(pj).max(), 1e-12))
                pj, sj = lbfgs_stages(pj, ps=(16, 32, 64, 128))
                cands.append((sj, pj))
    cands.sort(key=lambda c: -c[0])

    # ---------- SLSQP max-min polish on top candidates ----------
    def obj(z):
        return -z[-1]

    def obj_grad(z):
        g = np.zeros_like(z)
        g[-1] = -1.0
        return g

    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            def fmin(z, i=i, j=j):
                return np.sum((z[3*i:3*i+3] - z[3*j:3*j+3])**2) - z[-1]
            def gmin(z, i=i, j=j):
                g = np.zeros(len(z))
                di = 2.0 * (z[3*i:3*i+3] - z[3*j:3*j+3])
                g[3*i:3*i+3] = di; g[3*j:3*j+3] = -di; g[-1] = -1.0
                return g
            def fmax(z, i=i, j=j):
                return 1.0 - np.sum((z[3*i:3*i+3] - z[3*j:3*j+3])**2)
            def gmax(z, i=i, j=j):
                g = np.zeros(len(z))
                di = 2.0 * (z[3*i:3*i+3] - z[3*j:3*j+3])
                g[3*i:3*i+3] = -di; g[3*j:3*j+3] = di
                return g
            constraints.append({'type': 'ineq', 'fun': fmin, 'jac': gmin})
            constraints.append({'type': 'ineq', 'fun': fmax, 'jac': gmax})

    def slsqp(pts):
        p = pts / np.sqrt(max(D2mat(pts).max(), 1e-12))
        t0 = D2mat(p)[iu].min()
        z0 = np.concatenate([p.ravel(), [t0]])
        try:
            res = minimize(obj, z0, jac=obj_grad, constraints=constraints,
                           method='SLSQP', options={'maxiter': 300,
                                                    'ftol': 1e-14})
            p2 = res.x[:-1].reshape(n, d)
            return p2, score_sq(p2)
        except Exception:
            return None, 0.0

    best_pts = None; best_s = -1.0
    for s0, p0 in cands[:8]:
        if s0 > best_s:
            best_s, best_pts = s0, p0.copy()
        p2, s2 = slsqp(p0)
        if p2 is not None and s2 > best_s:
            best_s, best_pts = s2, p2.copy()
        # one perturb + re-polish round
        pj = p2 if p2 is not None else p0
        for _ in range(2):
            pq = pj + 0.02 * rng.standard_normal(pj.shape)
            pq = pq / np.sqrt(max(D2mat(pq).max(), 1e-12))
            p3, s3 = slsqp(pq)
            if p3 is not None and s3 > best_s:
                best_s, best_pts = s3, p3.copy()

    best_pts = best_pts / np.sqrt(max(D2mat(best_pts).max(), 1e-12))
    return np.asarray(best_pts, dtype=float)

# EVOLVE-BLOCK-END
