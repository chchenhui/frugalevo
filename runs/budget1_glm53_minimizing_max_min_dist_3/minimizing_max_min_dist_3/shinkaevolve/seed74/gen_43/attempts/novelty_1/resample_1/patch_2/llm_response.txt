def _ratio_sq(points):
    n = points.shape[0]
    iu = np.triu_indices(n, 1)
    diff = points[:, None, :] - points[None, :, :]
    d2 = (diff ** 2).sum(-1)
    u = d2[iu]
    mx = u.max()
    if mx <= 0:
        return 0.0
    return u.min() / mx
