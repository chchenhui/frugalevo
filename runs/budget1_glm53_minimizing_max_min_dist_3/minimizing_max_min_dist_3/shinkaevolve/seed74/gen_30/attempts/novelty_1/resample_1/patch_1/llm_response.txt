Replace the messy soft-objective machinery with a clean squared-distance log-sum-exp surrogate and its exact analytic Jacobian, use a finer sharpening schedule (p = 2→8→32→128), add more diverse structured starts, and finish with the exact-ratio polish. def _soft_objective(flat, n, T):
    pts = flat.reshape(n, 3)
    diff = pts[:, None, :] - pts[None, :, :]
    d = np.sqrt(np.sum(diff * diff, axis=-1) + 1e-12)
    iu = np.triu_indices(n, 1)
    dv = d[iu]
    # softmin / softmax via log-sum-exp
    a = -dv / T
    a -= a.max()
    soft_min = -T * (np.log(np.sum(np.exp(a))) - a.max() + np.log(np.exp(a.max() - a.max())))
    # careful: recompute properly
    a = -dv / T
    amax = a.max()
    soft_min = -T * (amax + np.log(np.sum(np.exp(a - amax))))
    b = dv / T
    bmax = b.max()
    soft_max = T * (bmax + np.log(np.sum(np.exp(b - bmax))))
    return -(soft_min / soft_max)


def _grad_soft(flat, n, T):
    pts = flat.reshape(n, 3)
    diff = pts[:, None, :] - pts[None, :, :]
    d = np.sqrt(np.sum(diff * diff, axis=-1) + 1e-12)
    i, j = np.triu_indices(n, 1)
    dv = d[i, j]
    a = -dv / T
    amax = a.max()
    w_min = np.exp(a - amax)
    w_min /= w_min.sum()
    b = dv / T
    bmax = b.max()
    w_max = np.exp(b - bmax)
    w_max /= w_max.sum()
    soft_min = -T * (amax + np.log(w_min.sum() * np.exp(amax - amax) + 0) + np.log(np.sum(np.exp(a - amax))))
    soft_min = -T * (amax + np.log(np.sum(np.exp(a - amax))))
    soft_max = T * (bmax + np.log(np.sum(np.exp(b - bmax))))
    # gradient of -(soft_min/soft_max)
    # d soft_min / d dv = w_min  (since soft_min = -T*lse(-dv/T))
    # d soft_max / d dv = w_max
    g = w_min / soft_max + (soft_min / soft_max ** 2) * w_max
    # note: d(-soft_min/soft_max)/d dv = -(w_min*soft_max - soft_min*w_max)/soft_max^2
    g = -(w_min * soft_max - soft_min * w_max) / (soft_max ** 2)
    # chain: dv depends on pts[i], pts[j] with +/- unit vectors
    grad = np.zeros((n, 3))
    uv = diff[i, j] / d[i, j][:, None]
    contrib = g[:, None] * uv
    np.add.at(grad, i, contrib)
    np.add.at(grad, j, -contrib)
    return grad.ravel(), -soft_min / soft_max

def _lse_grad(flat, n, p):
    """Value and analytic gradient of -(smin/smax) where
    smin = -lse(-p*d2)/p, smax = lse(p*d2)/p over squared pair distances d2."""
    pts = flat.reshape(n, 3)
    diff = pts[:, None, :] - pts[None, :, :]
    D2 = np.sum(diff * diff, axis=-1)
    i, j = np.triu_indices(n, 1)
    dv = D2[i, j]
    a = -p * dv
    amax = a.max()
    smin = -(amax + np.log(np.sum(np.exp(a - amax)))) / p
    w_min = np.exp(a - amax)
    w_min /= w_min.sum()
    b = p * dv
    bmax = b.max()
    smax = (bmax + np.log(np.sum(np.exp(b - bmax)))) / p
    w_max = np.exp(b - bmax)
    w_max /= w_max.sum()
    val = -(smin / smax)
    # d(val)/d(dv_ab) = -(w_min*smax - smin*w_max)/smax^2
    g = -(w_min * smax - smin * w_max) / (smax ** 2)
    # chain: dv = |pa - pb|^2, d dv/d pa = 2(pa - pb)
    grad = np.zeros((n, 3))
    contrib = (2.0 * g)[:, None] * diff[i, j]
    np.add.at(grad, i, contrib)
    np.add.at(grad, j, -contrib)
    return val, grad.ravel()
