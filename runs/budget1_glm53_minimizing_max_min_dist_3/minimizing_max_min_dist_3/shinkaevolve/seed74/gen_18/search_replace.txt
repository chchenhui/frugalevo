<DESCRIPTION>
Replace the slow numerical-gradient L-BFGS-B and ineffective Nelder-Mead stages with (a) a fast annealed analytic-gradient ascent on a soft-minimum of the pairwise distances (with points kept centered and scaled so dmax=1 after every step, which handles scale invariance exactly), and (b) the exact SLSQP polish (maximize t subject to t^2 <= ||pi-pj||^2 <= 1) from the 1.00-scoring program, plus jitter restarts of the polish. This is ~20x faster per optimization run than L-BFGS-B with numerical gradients, freeing budget for many more structured/random starts, while the SLSQP stage polishes to the best known configuration.

    from scipy.optimize import minimize

    rng = np.random.default_rng(42)
    I, J = np.triu_indices(n, k=1)

    def true_ratio(P):
        dd = np.linalg.norm(P[I] - P[J], axis=1)
        mx = dd.max()
        if mx <= 0:
            return 0.0
        return dd.min() / mx

    def normalize(P):
        P = P - P.mean(axis=0)
        mx = np.linalg.norm(P[I] - P[J], axis=1).max()
        if mx <= 1e-12:
            return P
        return P / mx

    # Fast annealed analytic-gradient ascent on a soft-minimum of the
    # pairwise distances. Points are re-centered and re-scaled to dmax=1
    # after every step, so the ratio equals the minimum distance.
    def grad_ascend(P0, ks=(3.0, 6.0, 12.0, 25.0, 50.0), iters=200,
                    lr=0.02):
        P = normalize(np.asarray(P0, dtype=float).copy())
        for k in ks:
            for _ in range(iters):
                diff = P[I] - P[J]
                dd = np.linalg.norm(diff, axis=1)
                dd = np.maximum(dd, 1e-9)
                w = np.exp(-k * (dd - dd.min()))
                W = w.sum()
                u = diff / dd[:, None]
                coef = (w / W)[:, None]
                G = np.zeros_like(P)
                np.add.at(G, I, coef * u)
                np.add.at(G, J, -coef * u)
                P = normalize(P + lr * G)
        return P
