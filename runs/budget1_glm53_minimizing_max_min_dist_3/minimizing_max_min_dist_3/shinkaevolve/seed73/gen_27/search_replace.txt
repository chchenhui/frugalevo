    def fibonacci_sphere(k):
        i = np.arange(k) + 0.5
        phi = np.pi * (1.0 + 5.0 ** 0.5) * i
        z = 1.0 - 2.0 * i / k
        r = np.sqrt(np.maximum(0.0, 1.0 - z * z))
        return np.stack([r * np.cos(phi), r * np.sin(phi), z], axis=1)

    # structured seed: icosahedron (12 verts) + 2 poles
