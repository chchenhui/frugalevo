def _slsqp_stage(pts, beta_unused=None):
    """Directly maximize min pairwise distance subject to max pairwise
    distance <= 1, via hard inequality constraints (exactly the true
    objective). Returns improved points (or the input if no gain)."""
    if not HAVE_SCIPY:
        return pts
    n = len(pts)
    idx_i, idx_j = np.triu_indices(n, 1)
    m = len(idx_i)
    pts = pts - pts.mean(axis=0)
    # normalize so max pair dist == 1
    d2 = np.sum((pts[:, None] - pts[None]) ** 2, axis=-1)
    dx = np.sqrt(d2[np.triu_indices(n, 1)].max())
    if dx <= 0:
        return pts
    pts = pts / dx
    d2 = np.sum((pts[:, None] - pts[None]) ** 2, axis=-1)
    t0 = d2[np.triu_indices(n, 1)].min()

    def unpack(x):
        return x[:3 * n].reshape(n, 3), x[3 * n]

    def obj(x):
        g = np.zeros(3 * n + 1)
        g[-1] = -1.0
        return -x[-1], g

    rows_i = 3 * idx_i
    rows_j = 3 * idx_j

    def cons_f(x):
        p, t = unpack(x)
        dd = np.sum((p[:, None] - p[None]) ** 2, axis=-1)[idx_i, idx_j]
        return np.concatenate([dd - t, 1.0 - dd])

    def cons_j(x):
        p, _ = unpack(x)
        diff = p[idx_i] - p[idx_j]
        J = np.zeros((2 * m, 3 * n + 1))
        for k in range(m):
            i3, j3 = rows_i[k], rows_j[k]
            J[k, i3:i3 + 3] = 2.0 * diff[k]
            J[k, j3:j3 + 3] = -2.0 * diff[k]
            J[k, -1] = -1.0
            J[m + k, i3:i3 + 3] = -2.0 * diff[k]
            J[m + k, j3:j3 + 3] = 2.0 * diff[k]
        return J

    cons = {"type": "ineq", "fun": cons_f, "jac": cons_j}
    x0 = np.concatenate([pts.ravel(), [t0]])
    try:
        res = minimize(obj, x0, jac=True, method="SLSQP",
                       constraints=[cons],
                       options={"maxiter": 300, "ftol": 1e-12})
        cand, _ = unpack(res.x)
        cand = cand - cand.mean(axis=0)
        return cand
    except Exception:
        return pts
