<DESCRIPTION>
Replace the random guess with a direct stochastic optimization of the target ratio dmin/dmax. Since the metric is exactly (dmin/dmax)^2, we optimize that ratio directly with a greedy random-perturbation hill-climb over many restarts (random starts plus structured starts from an icosahedron-based configuration), keeping the best configuration found. Pairwise distance computations for 14 points are cheap (91 pairs), so tens of thousands of proposals fit easily in the time budget. Hill-climbing on the true objective is far more reliable than penalized or surrogate objectives here.

    n = 14
    d = 3
    rng = np.random.default_rng(42)

    def ratio(pts):
        dm = pts[:, None, :] - pts[None, :, :]
        dists = np.sqrt((dm * dm).sum(-1))
        iu = np.triu_indices(n, k=1)
        pv = dists[iu]
        dmin = pv.min()
        dmax = pv.max()
        if dmax <= 0:
            return 0.0, pv
        return dmin / dmax, pv

    def normalize(pts):
        c = pts.mean(0)
        pts = pts - c
        iu = np.triu_indices(len(pts), k=1)
        dmax = np.sqrt(((pts[:, None] - pts[None]) ** 2).sum(-1))[iu].max()
        if dmax <= 0:
            return pts
        return pts / dmax

    # Structured start: icosahedron vertices + 2 antipodal axis points
    phi = (1 + np.sqrt(5)) / 2
    ico = np.array([
        [-1, phi, 0], [1, phi, 0], [-1, -phi, 0], [1, -phi, 0],
        [0, -1, phi], [0, 1, phi], [0, -1, -phi], [0, 1, -phi],
        [phi, 0, -1], [phi, 0, 1], [-phi, 0, -1], [-phi, 0, 1],
        [0, 0, 2.2], [0, 0, -2.2],
    ], dtype=float)
    ico = normalize(ico)

    starts = [ico]
    # Random starts on the unit sphere and in the ball
    for _ in range(30):
        v = rng.standard_normal((n, d))
        v /= np.linalg.norm(v, axis=1, keepdims=True)
        starts.append(v * rng.uniform(0.3, 1.0))
    for _ in range(20):
        starts.append(rng.standard_normal((n, d)))

    best_pts = None
    best_val = -1.0

    for start in starts:
        pts = normalize(np.array(start, dtype=float))
        cur, _ = ratio(pts)
        step = 0.05
        while step > 1e-6:
            improved = False
            for trial in range(400):
                i = rng.integers(n)
                cand = pts.copy()
                dirv = rng.standard_normal(d)
                dirv /= np.linalg.norm(dirv)
                cand[i] = cand[i] + step * dirv
                cand = normalize(cand)
                val, _ = ratio(cand)
                if val > cur + 1e-12:
                    pts = cand
                    cur = val
                    improved = True
                    if trial >= 399:
                        break
            if not improved:
                step *= 0.5
        if cur > best_val:
            best_val = cur
            best_pts = pts.copy()

    # Final fine refinement from the best configuration
    pts = best_pts.copy()
    cur, _ = ratio(pts)
    step = 0.01
    while step > 1e-8:
        improved = False
        for trial in range(600):
            i = rng.integers(n)
            cand = pts.copy()
            dirv = rng.standard_normal(d)
            dirv /= np.linalg.norm(dirv)
            cand[i] = cand[i] + step * dirv
            cand = normalize(cand)
            val, _ = ratio(cand)
            if val > cur + 1e-14:
                pts = cand
                cur = val
                improved = True
        if not improved:
            step *= 0.5
    if cur > best_val:
        best_pts = pts.copy()

    return np.asarray(best_pts, dtype=float)
