<DESCRIPTION>
Replace the random point placement with a numerical optimizer. Since the metric is (dmin/dmax)^2, we can normalize scale each iteration so dmax=1 and run gradient ascent on a soft-min of squared pairwise distances (log-sum-exp surrogate with annealing). We seed from several structured 14-point configurations (cube vertices + face centers, hexagonal prism + axis points, icosahedron + poles, random) and keep the best result over restarts, making it deterministic via fixed seeds.

    n = 14
    d = 3
    rng = np.random.RandomState(0)

    def ratio(pts):
        dm = pts - pts[:, None, :]
        dist = np.sqrt((dm * dm).sum(-1))
        iu = np.triu_indices(n, 1)
        dd = dist[iu]
        return dd.min() / dd.max(), dist

    def optimize(pts, iters=2500, t0=0.05, t1=0.002, lr=0.02):
        pts = pts.copy()
        for it in range(iters):
            # normalize so dmax = 1 (scale invariance)
            dm = pts - pts[:, None, :]
            dist = np.sqrt((dm * dm).sum(-1) + 1e-12)
            dmax = dist.max()
            pts = pts / dmax
            t = t0 * (t1 / t0) ** (it / (iters - 1))
            q = dist ** 2
            w = np.exp(-q / t)
            np.fill_diagonal(w, 0.0)
            w = w / (w.sum() + 1e-18)
            # gradient of soft-min objective wrt points
            grad = np.zeros_like(pts)
            for i in range(n):
                g = (2.0 * t) * w[i, :, None] * (pts[i] - pts)
                grad[i] = g.sum(0)
            # step + keep points spread (repulsion when very close)
            pts = pts + lr * grad
        return pts

    seeds = []

    # 1) cube vertices + face centers (14 pts)
    c = np.array([[x, y, z] for x in (-1, 1) for y in (-1, 1) for z in (-1, 1)], float)
    fc = []
    for ax in range(3):
        for s in (-1, 1):
            p = np.zeros(3)
            p[ax] = s * 1.3
            fc.append(p)
    seeds.append(np.vstack([c, np.array(fc)]))

    # 2) hexagonal prism (12) + 2 axis points
    ang = np.arange(6) * np.pi / 3.0
    ring = np.stack([np.cos(ang), np.sin(ang)], axis=1)
    hp = []
    for z in (-0.6, 0.6):
        hp.append(np.hstack([ring, np.full((6, 1), z)]))
    hp = np.vstack(hp)
    ax = np.array([[0, 0, 1.4], [0, 0, -1.4]])
    seeds.append(np.vstack([hp, ax]))

    # 3) icosahedron (12) + 2 poles
    phi = (1 + np.sqrt(5)) / 2
    ico = np.array([
        [-1, phi, 0], [1, phi, 0], [-1, -phi, 0], [1, -phi, 0],
        [0, -1, phi], [0, 1, phi], [0, -1, -phi], [0, 1, -phi],
        [phi, 0, -1], [phi, 0, 1], [-phi, 0, -1], [-phi, 0, 1]], float)
    seeds.append(np.vstack([ico, np.array([[0, 0, 2.2], [0, 0, -2.2]])]))

    # 4) cuboctahedron-like (12) + 2 random-ish extras
    cu = np.array([[x, y, z] for x in (-1, 0, 1) for y in (-1, 0, 1)
                   for z in (-1, 0, 1) if (x, y, z).count(0) >= 2 and any((x, y, z))], float)
    extra = np.array([[1.5, 1.5, 1.5], [-1.5, -1.5, -1.5]], float)
    seeds.append(np.vstack([cu, extra]))

    # 5) random seeds
    for _ in range(6):
        seeds.append(rng.randn(n, d))

    best_pts, best_r = None, -1.0
    for s in seeds:
        p = optimize(s)
        r, _ = ratio(p)
        # local polish with small t
        p2 = optimize(p, iters=800, t0=0.002, t1=0.0005, lr=0.005)
        r2, _ = ratio(p2)
        if max(r, r2) > best_r:
            best_r = max(r, r2)
            best_pts = p if r > r2 else p2

    return best_pts
