def _safe_output(pts, fallback):
    """Validate optimizer output: shape (N, D) and all finite."""

    if pts.shape != (N, D) or not np.all(np.isfinite(pts)):
        return np.asarray(fallback, dtype=float).copy()
    return pts

def _dual_ring():
    """14 points as two staggered rings (anti-prism topology)."""
    k = np.arange(7)
    th = 2 * np.pi * k / 7
    z = 0.55
    ring1 = np.stack([np.cos(th), np.sin(th), z * np.ones(7)], axis=1)
    th2 = th + np.pi / 7
    ring2 = np.stack([np.cos(th2), np.sin(th2), -z * np.ones(7)], axis=1)
    return _unit(np.vstack([ring1, ring2]))


def _two_shell():
    """7 points on an outer cap ring, 7 near center: a 'two-shell' basin."""
    k = np.arange(7)
    th = 2 * np.pi * k / 7
    outer = np.stack([np.cos(th), np.sin(th), 0.4 * np.ones(7)], axis=1)
    th2 = th + np.pi / 7
    inner = 0.45 * np.stack([np.cos(th2), np.sin(th2), -0.4 * np.ones(7)], axis=1)
    return np.vstack([outer, inner])
