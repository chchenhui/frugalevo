+def _project_simplex(values: np.ndarray, margin: float = 0.0) -> np.ndarray:
+    """Project affine point coordinates into u>=0, v>=0, u+v<=1."""
+    z = np.asarray(values, dtype=float).copy()
+    z = np.maximum(z, margin)
+
+    total = z[..., 0] + z[..., 1]
+    cap = 1.0 - margin
+    outside = total > cap
+
+    if np.any(outside):
+        z[outside] *= (cap / total[outside])[..., None]
+
+    return z

+def _gradient_field(layout: np.ndarray, signed: np.ndarray) -> np.ndarray:
+    """
+    Aggregate determinant-ascent directions from the active small triangles.
+
+    A determinant is linear in each one of its three vertices.  For every
+    restrictive triangle, its signed gradient is added to each participating
+    movable point.  The resulting vector field proposes coordinated repairs
+    rather than only random coordinate replacement.
+    """
+    absolute = np.abs(signed)
+    active_count = min(28, len(absolute))
+    active_indices = np.argpartition(absolute, active_count - 1)[:active_count]
+
+    points = np.vstack((_CORNERS_UV, layout))
+    direction = np.zeros((_FREE, 2), dtype=float)
+
+    limit = max(float(np.min(absolute)) + 0.009, 0.006)
+
+    for tri_index in active_indices:
+        i, j, k = _TRIPLES[tri_index]
+        a, b, c = points[i], points[j], points[k]
+
+        sign = 1.0 if signed[tri_index] >= 0.0 else -1.0
+        weight = 1.0 / max(absolute[tri_index], 0.004)
+        if absolute[tri_index] > limit:
+            weight *= 0.25
+
+        gradients = (
+            np.array((b[1] - c[1], c[0] - b[0])),
+            np.array((c[1] - a[1], a[0] - c[0])),
+            np.array((a[1] - b[1], b[0] - a[0])),
+        )
+
+        for point_index, gradient in zip((i, j, k), gradients):
+            if point_index >= 3:
+                direction[point_index - 3] += sign * weight * gradient
+

+    good = norm > 1.0e-14
+    direction[good] /= norm[good, None]
+    return direction
