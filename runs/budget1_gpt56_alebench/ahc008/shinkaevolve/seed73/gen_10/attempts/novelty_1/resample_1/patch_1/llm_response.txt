<DESCRIPTION>
This approach replaces perimeter-strip construction with a dense horizontal partition strategy.

Humans are sorted by initial row and assigned two nearby horizontal barrier rows each. Building full-width barriers creates many narrow independent regions, which is much more favorable for the scoring function: smaller regions have a substantially higher probability of containing zero pets, avoiding the exponential `2^-n` penalty.

Each human builds its first barrier while deliberately closing the final gap from the side of its second assigned barrier, ensuring it is not trapped away from its remaining work. For the final barrier, it evaluates both possible sides using a flood-fill estimate of reachable area and contained pets, then closes from the better side. This is fundamentally different from the previous strategy of enclosing one large strip per human.

</DESCRIPTION>

// EVOLVE-BLOCK-START
#include <bits/stdc++.h>
using namespace std;

static constexpr int S = 30;
static constexpr int TURNS = 300;

struct P {
    int x, y;
    bool operator==(const P& o) const { return x == o.x && y == o.y; }
};

const int DX[4] = {-1, 1, 0, 0};
const int DY[4] = {0, 0, -1, 1};
const char BM[4] = {'u', 'd', 'l', 'r'};
const char MV[4] = {'U', 'D', 'L', 'R'};

struct Pet {
    P p;
    int type;
};

struct Human {
    P p;
    vector<int> rows;
    int phase = 0;
};

int N, M;
vector<Pet> pets;
vector<Human> humans;
bool wall_[31][31];

bool inside(int x, int y) {
    return 1 <= x && x <= S && 1 <= y && y <= S;
}

bool occupiedByHuman(int x, int y, int except) {
    for (int i = 0; i < M; ++i) {
        if (i != except && humans[i].p.x == x && humans[i].p.y == y) return true;
    }
    return false;
}

bool legalBuild(int x, int y, int who) {
    if (!inside(x, y) || wall_[x][y]) return false;
    if (occupiedByHuman(x, y, who)) return false;
    for (const auto& pet : pets) {
        int d = abs(pet.p.x - x) + abs(pet.p.y - y);
        if (d <= 1) return false;
    }
    return true;
}

char directionMove(P a, P b) {
    for (int d = 0; d < 4; ++d) {
        if (a.x + DX[d] == b.x && a.y + DY[d] == b.y) return MV[d];
    }
    return '.';
}

char directionBuild(P a, P b) {
    for (int d = 0; d < 4; ++d) {
        if (a.x + DX[d] == b.x && a.y + DY[d] == b.y) return BM[d];
    }
    return '.';
}

char bfsMove(P start, P goal, const bool blockedNow[31][31]) {
    if (start == goal) return '.';

    static int dist[31][31];
    static P parent[31][31];

    for (int x = 1; x <= S; ++x) {
        for (int y = 1; y <= S; ++y) {
            dist[x][y] = -1;
            parent[x][y] = {-1, -1};
        }
    }

    queue<P> q;
    q.push(start);
    dist[start.x][start.y] = 0;

    while (!q.empty()) {
        P cur = q.front();
        q.pop();

        for (int d = 0; d < 4; ++d) {
            int nx = cur.x + DX[d], ny = cur.y + DY[d];
            if (!inside(nx, ny) || wall_[nx][ny] || blockedNow[nx][ny]) continue;
            if (dist[nx][ny] != -1) continue;

            dist[nx][ny] = dist[cur.x][cur.y] + 1;
            parent[nx][ny] = cur;
            if (nx == goal.x && ny == goal.y) {
                P z = goal;
                while (!(parent[z.x][z.y] == start)) z = parent[z.x][z.y];
                return directionMove(start, z);
            }
            q.push({nx, ny});
        }
    }
    return '.';
}

int bfsDistance(P start, P goal, const bool blockedNow[31][31]) {
    if (start == goal) return 0;
    static int dist[31][31];
    for (int x = 1; x <= S; ++x)
        for (int y = 1; y <= S; ++y)
            dist[x][y] = -1;

    queue<P> q;
    q.push(start);
    dist[start.x][start.y] = 0;

    while (!q.empty()) {
        P cur = q.front();
        q.pop();
        for (int d = 0; d < 4; ++d) {
            int nx = cur.x + DX[d], ny = cur.y + DY[d];
            if (!inside(nx, ny) || wall_[nx][ny] || blockedNow[nx][ny]) continue;
            if (dist[nx][ny] != -1) continue;
            dist[nx][ny] = dist[cur.x][cur.y] + 1;
            if (nx == goal.x && ny == goal.y) return dist[nx][ny];
            q.push({nx, ny});
        }
    }
    return 1e9;
}

bool rowDone(int r) {
    for (int y = 1; y <= S; ++y) {
        if (!wall_[r][y]) return false;
    }
    return true;
}

double componentValue(P start, int extraWallX, int extraWallY) {
    if (!inside(start.x, start.y) || wall_[start.x][start.y]) return -1.0;

    bool vis[31][31] = {};
    queue<P> q;
    q.push(start);
    vis[start.x][start.y] = true;
    int area = 0;

    while (!q.empty()) {
        P cur = q.front();
        q.pop();
        ++area;
        for (int d = 0; d < 4; ++d) {
            int nx = cur.x + DX[d], ny = cur.y + DY[d];
            if (!inside(nx, ny) || vis[nx][ny] || wall_[nx][ny]) continue;
            if (nx == extraWallX && ny == extraWallY) continue;
            vis[nx][ny] = true;
            q.push({nx, ny});
        }
    }

    int pc = 0;
    for (const auto& pet : pets) {
        if (vis[pet.p.x][pet.p.y]) ++pc;
    }
    return area / pow(2.0, pc);
}

void updatePhase(Human& h) {
    while (h.phase < (int)h.rows.size() && rowDone(h.rows[h.phase])) ++h.phase;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> N;
    pets.resize(N);
    for (int i = 0; i < N; ++i) {
        cin >> pets[i].p.x >> pets[i].p.y >> pets[i].type;
    }

    cin >> M;
    humans.resize(M);
    for (int i = 0; i < M; ++i) cin >> humans[i].p.x >> humans[i].p.y;

    vector<int> order(M);
    iota(order.begin(), order.end(), 0);
    sort(order.begin(), order.end(), [&](int a, int b) {
        return humans[a].p.x < humans[b].p.x;
    });

    vector<int> dividerRows;
    int K = 2 * M;
    for (int k = 1; k <= K; ++k) {
        int r = 1 + (k * 29) / (K + 1);
        r = max(2, min(29, r));
        if (dividerRows.empty() || dividerRows.back() != r) dividerRows.push_back(r);
    }

    // If integer rounding produced fewer distinct rows, fill unused rows.
    for (int r = 2; (int)dividerRows.size() < K && r <= 29; ++r) {
        if (find(dividerRows.begin(), dividerRows.end(), r) == dividerRows.end())
            dividerRows.push_back(r);
    }
    sort(dividerRows.begin(), dividerRows.end());

    for (int rank = 0; rank < M; ++rank) {
        humans[order[rank]].rows.push_back(dividerRows[2 * rank]);
        humans[order[rank]].rows.push_back(dividerRows[2 * rank + 1]);
    }

    for (int turn = 0; turn < TURNS; ++turn) {
        for (auto& h : humans) updatePhase(h);

        bool plannedWall[31][31] = {};
        string act(M, '.');

        for (int i = 0; i < M; ++i) {
            Human& h = humans[i];
            updatePhase(h);
            if (h.phase >= (int)h.rows.size()) continue;

            int r = h.rows[h.phase];

            vector<int> columns;
            for (int y = 1; y <= 29; ++y) {
                if (!wall_[r][y]) columns.push_back(y);
            }
            if (columns.empty() && !wall_[r][30]) columns.push_back(30);

            P bestWall = {-1, -1};
            P bestStand = {-1, -1};
            int bestScore = 1e9;

            for (int y : columns) {
                if (!legalBuild(r, y, i)) continue;

                for (int d = 0; d < 4; ++d) {
                    P stand = {r + DX[d], y + DY[d]};
                    if (!inside(stand.x, stand.y) || wall_[stand.x][stand.y]) continue;
                    if (plannedWall[stand.x][stand.y]) continue;

                    int dist = bfsDistance(h.p, stand, plannedWall);
                    if (dist >= 1000000000) continue;

                    int score = dist * 100;

                    // The very last cell closes the barrier. Choose the useful side.
                    if (y == 30 && columns.size() == 1) {
                        double val = componentValue(stand, r, y);
                        score -= (int)min(90000.0, val * 1000.0);

                        // For the first line, prefer the side containing the next line.
                        if (h.phase + 1 < (int)h.rows.size()) {
                            int nr = h.rows[h.phase + 1];
                            if ((nr > r && stand.x > r) || (nr < r && stand.x < r))
                                score -= 2000;
                        }
                    } else {
                        // Prefer working from close to the active divider.
                        score += abs(stand.x - r) * 5;
                    }

                    if (score < bestScore) {
                        bestScore = score;
                        bestWall = {r, y};
                        bestStand = stand;
                    }
                }
            }

            if (bestWall.x == -1) continue;

            if (h.p == bestStand) {
                act[i] = directionBuild(h.p, bestWall);
                if (act[i] != '.') plannedWall[bestWall.x][bestWall.y] = true;
            } else {
                act[i] = bfsMove(h.p, bestStand, plannedWall);
            }
        }

        cout << act << endl;

        for (int i = 0; i < M; ++i) {
            char a = act[i];
            for (int d = 0; d < 4; ++d) {
                if (a == BM[d]) {
                    int nx = humans[i].p.x + DX[d];
                    int ny = humans[i].p.y + DY[d];
                    if (inside(nx, ny)) wall_[nx][ny] = true;
                }
            }
        }

        for (int i = 0; i < M; ++i) {
            char a = act[i];
            for (int d = 0; d < 4; ++d) {
                if (a == MV[d]) {
                    int nx = humans[i].p.x + DX[d];
                    int ny = humans[i].p.y + DY[d];
                    if (inside(nx, ny) && !wall_[nx][ny]) humans[i].p = {nx, ny};
                }
            }
        }

        for (int i = 0; i < N; ++i) {
            string s;
            cin >> s;
            for (char c : s) {
                if (c == 'U') --pets[i].p.x;
                else if (c == 'D') ++pets[i].p.x;
                else if (c == 'L') --pets[i].p.y;
                else if (c == 'R') ++pets[i].p.y;
            }
        }
    }

    return 0;
}
