<DESCRIPTION>
Replace the single shuffled greedy pass with repeated randomized erosion passes that stop when deletions have become unproductive. A cell that was not removable early in a pass may become removable after nearby cells are deleted, so revisiting cells can improve the initial zero region substantially. Conversely, several consecutive zero-yield blocks indicate that greedy erosion is exhausted; stopping then preserves the remaining budget for simulated annealing, where non-monotone restructuring is more valuable.
</DESCRIPTION>

    // often unlock cells that failed when they were tested earlier.
    // Stop quickly once randomized blocks no longer produce any deletions,
    // retaining most of the time budget for non-monotone SA moves.
    const double greedy_limit = 500.0;
    const int greedy_block_size = 256;
    const int max_empty_greedy_blocks = 3;
    int empty_greedy_blocks = 0;
    bool stop_greedy = false;

    while (!stop_greedy && elapsed_ms() < greedy_limit) {
        shuffle(order.begin(), order.end(), rng);
        int block_attempts = 0;
        int block_successes = 0;

        for (int p : order) {
            if (elapsed_ms() >= greedy_limit) {
                stop_greedy = true;
                break;
            }

            int old_color = grid[p];
            if (old_color != 0 && attempt_change(p, old_color, 0)) {
                ++block_successes;
                int score = (int)cells[0].size();

                    memcpy(best_grid, grid, sizeof(grid));
