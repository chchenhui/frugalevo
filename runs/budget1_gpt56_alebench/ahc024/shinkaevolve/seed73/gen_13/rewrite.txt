// EVOLVE-BLOCK-START
#pragma GCC optimize("O3,unroll-loops")

#include <bits/stdc++.h>
using namespace std;

static constexpr int N = 50;
static constexpr int M = 100;

struct RNG {
    uint64_t x = 88172645463325252ULL;
    uint64_t next() {
        x ^= x << 7;
        x ^= x >> 9;
        return x;
    }
    int nextInt(int n) {
        return (int)(next() % (uint64_t)n);
    }
};

int a[N][N], original[N][N], bestg[N][N];
bool req[M + 1][M + 1];
int edgeCnt[M + 1][M + 1];
int colorCnt[M + 1];
int reqDeg[M + 1];

int seen[N][N];
int stamp = 1;
int qr[N * N], qc[N * N];

const int dr[4] = {-1, 1, 0, 0};
const int dc[4] = {0, 0, -1, 1};

RNG rng;
chrono::steady_clock::time_point startTime;

inline bool inside(int r, int c) {
    return (unsigned)r < N && (unsigned)c < N;
}

inline void normPair(int &x, int &y) {
    if (x > y) swap(x, y);
}

inline void addEdge(int x, int y, int delta) {
    if (x == y) return;
    normPair(x, y);
    edgeCnt[x][y] += delta;
}

inline bool requiredEdge(int x, int y) {
    if (x == y) return false;
    normPair(x, y);
    return req[x][y];
}

inline double elapsedMs() {
    return chrono::duration<double, milli>(chrono::steady_clock::now() - startTime).count();
}

void buildInitialData() {
    memset(req, 0, sizeof(req));

    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++) {
            int x = original[r][c];

            if (r == 0 || r == N - 1 || c == 0 || c == N - 1) {
                int u = 0, v = x;
                normPair(u, v);
                req[u][v] = true;
            }
            if (r + 1 < N && x != original[r + 1][c]) {
                int u = x, v = original[r + 1][c];
                normPair(u, v);
                req[u][v] = true;
            }
            if (c + 1 < N && x != original[r][c + 1]) {
                int u = x, v = original[r][c + 1];
                normPair(u, v);
                req[u][v] = true;
            }
        }
    }

    memset(reqDeg, 0, sizeof(reqDeg));
    for (int i = 0; i <= M; i++) {
        for (int j = i + 1; j <= M; j++) {
            if (req[i][j]) {
                reqDeg[i]++;
                reqDeg[j]++;
            }
        }
    }
}

void resetState() {
    memcpy(a, original, sizeof(a));
    memset(edgeCnt, 0, sizeof(edgeCnt));
    memset(colorCnt, 0, sizeof(colorCnt));

    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++) {
            colorCnt[a[r][c]]++;

            int x = a[r][c];
            if (r == 0) addEdge(0, x, 1);
            if (r == N - 1) addEdge(0, x, 1);
            if (c == 0) addEdge(0, x, 1);
            if (c == N - 1) addEdge(0, x, 1);

            if (r + 1 < N) addEdge(x, a[r + 1][c], 1);
            if (c + 1 < N) addEdge(x, a[r][c + 1], 1);
        }
    }
}

bool oldColorStaysConnected(int r, int c, int oldColor) {
    int nrList[4], ncList[4], kcnt = 0;

    for (int d = 0; d < 4; d++) {
        int nr = r + dr[d], nc = c + dc[d];
        if (inside(nr, nc) && a[nr][nc] == oldColor) {
            nrList[kcnt] = nr;
            ncList[kcnt] = nc;
            kcnt++;
        }
    }

    if (kcnt <= 1) return true;

    ++stamp;
    if (stamp == INT_MAX) {
        memset(seen, 0, sizeof(seen));
        stamp = 1;
    }

    int head = 0, tail = 0;
    qr[tail] = nrList[0];
    qc[tail++] = ncList[0];
    seen[nrList[0]][ncList[0]] = stamp;

    while (head < tail) {
        int cr = qr[head], cc = qc[head++];
        for (int d = 0; d < 4; d++) {
            int rr = cr + dr[d], ss = cc + dc[d];
            if (inside(rr, ss) && seen[rr][ss] != stamp && a[rr][ss] == oldColor) {
                seen[rr][ss] = stamp;
                qr[tail] = rr;
                qc[tail++] = ss;
            }
        }
    }

    for (int i = 1; i < kcnt; i++) {
        if (seen[nrList[i]][ncList[i]] != stamp) return false;
    }
    return true;
}

bool touchesExteriorZero(int r, int c) {
    for (int d = 0; d < 4; d++) {
        int nr = r + dr[d], nc = c + dc[d];
        if (!inside(nr, nc) || a[nr][nc] == 0) return true;
    }
    return false;
}

bool tryMove(int r, int c, int nw) {
    int old = a[r][c];
    if (old == nw || old == 0) return false;

    // We never remove zero cells. This keeps zero connected to the outside.
    if (nw == 0 && !touchesExteriorZero(r, c)) return false;

    // A color with required adjacencies must not disappear.
    if (colorCnt[old] == 1 && reqDeg[old] > 0) return false;

    int pu[8], pv[8], pd[8], pcnt = 0;

    auto record = [&](int x, int y, int delta) {
        if (x == y) return;
        normPair(x, y);
        for (int i = 0; i < pcnt; i++) {
            if (pu[i] == x && pv[i] == y) {
                pd[i] += delta;
                return;
            }
        }
        pu[pcnt] = x;
        pv[pcnt] = y;
        pd[pcnt] = delta;
        pcnt++;
    };

    for (int d = 0; d < 4; d++) {
        int nr = r + dr[d], nc = c + dc[d];
        int y = inside(nr, nc) ? a[nr][nc] : 0;
        record(old, y, -1);
        record(nw, y, +1);
    }

    for (int i = 0; i < pcnt; i++) {
        edgeCnt[pu[i]][pv[i]] += pd[i];
    }

    bool ok = true;
    for (int i = 0; i < pcnt; i++) {
        bool exists = edgeCnt[pu[i]][pv[i]] > 0;
        if (exists != req[pu[i]][pv[i]]) {
            ok = false;
            break;
        }
    }

    if (ok) {
        a[r][c] = nw;
        colorCnt[old]--;
        colorCnt[nw]++;

        if (colorCnt[old] == 0 && reqDeg[old] > 0) ok = false;
        if (ok && !oldColorStaysConnected(r, c, old)) ok = false;

        if (!ok) {
            a[r][c] = old;
            colorCnt[old]++;
            colorCnt[nw]--;
        }
    }

    if (!ok) {
        for (int i = 0; i < pcnt; i++) {
            edgeCnt[pu[i]][pv[i]] -= pd[i];
        }
        return false;
    }

    return true;
}

int zeroScore() {
    return colorCnt[0];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;
    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++) {
            cin >> original[r][c];
        }
    }

    startTime = chrono::steady_clock::now();
    buildInitialData();

    int bestScore = -1;
    vector<int> order(N * N);
    iota(order.begin(), order.end(), 0);

    while (elapsedMs() < 1920.0) {
        resetState();

        // Multiple randomized erosion passes. Repeated passes matter because
        // removing an obstructing compatible ward can unlock a previously
        // illegal frontier deletion.
        for (int pass = 0; pass < 18 && elapsedMs() < 1920.0; pass++) {
            for (int i = N * N - 1; i > 0; i--) {
                swap(order[i], order[rng.nextInt(i + 1)]);
            }

            int changed = 0;

            for (int ii = 0; ii < N * N; ii++) {
                if ((ii & 63) == 0 && elapsedMs() >= 1920.0) break;

                int r = order[ii] / N;
                int c = order[ii] % N;
                int old = a[r][c];
                if (old == 0) continue;

                // Main move: exterior-connected erosion.
                if (tryMove(r, c, 0)) {
                    changed++;
                    continue;
                }

                // Small amount of nonzero reshaping. It can move articulation
                // bottlenecks and make later erosion possible while retaining
                // all exact adjacency constraints.
                if ((rng.next() & 15ULL) == 0ULL) {
                    int cand[4], cnt = 0;
                    for (int d = 0; d < 4; d++) {
                        int nr = r + dr[d], nc = c + dc[d];
                        if (inside(nr, nc) && a[nr][nc] != 0 && a[nr][nc] != old) {
                            bool duplicate = false;
                            for (int z = 0; z < cnt; z++) {
                                if (cand[z] == a[nr][nc]) duplicate = true;
                            }
                            if (!duplicate) cand[cnt++] = a[nr][nc];
                        }
                    }
                    if (cnt) {
                        if (tryMove(r, c, cand[rng.nextInt(cnt)])) changed++;
                    }
                }
            }

            if (changed == 0 && pass >= 3) break;
        }

        int sc = zeroScore();
        if (sc > bestScore) {
            bestScore = sc;
            memcpy(bestg, a, sizeof(a));
        }
    }

    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++) {
            if (c) cout << ' ';
            cout << bestg[r][c];
        }
        cout << '\n';
    }
    return 0;
}
