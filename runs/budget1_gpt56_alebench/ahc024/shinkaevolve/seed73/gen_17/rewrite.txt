// EVOLVE-BLOCK-START
#pragma GCC optimize("O3,unroll-loops")

#include <bits/stdc++.h>
using namespace std;

static constexpr int N = 50;
static constexpr int M = 100;
static constexpr int SZ = N * N;

struct RNG {
    uint64_t x;
    RNG() {
        x = chrono::high_resolution_clock::now().time_since_epoch().count();
        x ^= (uint64_t)(uintptr_t)this;
        if (!x) x = 88172645463325252ULL;
    }
    uint64_t next() {
        x ^= x << 7;
        x ^= x >> 9;
        return x;
    }
    int operator()(int n) {
        return (int)(next() % n);
    }
};

RNG rng;
int g[SZ];
bool req[M + 1][M + 1];
int edgeCnt[M + 1][M + 1];

vector<int> cells[M + 1];
int posInColor[SZ];

int seen[SZ], stamp = 1;
int que[SZ];

const int dr[4] = {-1, 1, 0, 0};
const int dc[4] = {0, 0, -1, 1};

inline bool inside(int r, int c) {
    return (unsigned)r < N && (unsigned)c < N;
}
inline int id(int r, int c) {
    return r * N + c;
}
inline void norm(int &a, int &b) {
    if (a > b) swap(a, b);
}
inline void addCell(int v, int col) {
    posInColor[v] = (int)cells[col].size();
    cells[col].push_back(v);
}
inline void removeCell(int v, int col) {
    int p = posInColor[v];
    int z = cells[col].back();
    cells[col][p] = z;
    posInColor[z] = p;
    cells[col].pop_back();
}

bool connectedColor(int col) {
    if (cells[col].empty()) return false;

    ++stamp;
    if (stamp == INT_MAX) {
        memset(seen, 0, sizeof(seen));
        stamp = 1;
    }

    int head = 0, tail = 0;
    int start = cells[col][0];
    que[tail++] = start;
    seen[start] = stamp;

    while (head < tail) {
        int v = que[head++];
        int r = v / N, c = v % N;
        for (int k = 0; k < 4; ++k) {
            int nr = r + dr[k], nc = c + dc[k];
            if (!inside(nr, nc)) continue;
            int u = id(nr, nc);
            if (seen[u] != stamp && g[u] == col) {
                seen[u] = stamp;
                que[tail++] = u;
            }
        }
    }
    return tail == (int)cells[col].size();
}

/*
  Changes one cell from oldCol to newCol.
  Preconditions used by callers:
    - for nonzero target, target is a neighboring cell, hence target stays connected.
    - for zero target, it is boundary-adjacent or zero-adjacent, hence zero stays
      connected to the outside.
*/
bool tryMove(int v, int newCol) {
    int oldCol = g[v];
    if (oldCol == newCol) return false;

    int r = v / N, c = v % N;

    int ka[8], kb[8], delta[8], cnt = 0;

    auto addDelta = [&](int a, int b, int d) {
        if (a == b) return;
        norm(a, b);
        for (int i = 0; i < cnt; ++i) {
            if (ka[i] == a && kb[i] == b) {
                delta[i] += d;
                return;
            }
        }
        ka[cnt] = a;
        kb[cnt] = b;
        delta[cnt] = d;
        ++cnt;
    };

    for (int k = 0; k < 4; ++k) {
        int nr = r + dr[k], nc = c + dc[k];
        int z = inside(nr, nc) ? g[id(nr, nc)] : 0;
        addDelta(oldCol, z, -1);
        addDelta(newCol, z, +1);
    }

    for (int i = 0; i < cnt; ++i) {
        if (delta[i] == 0) continue;
        int after = edgeCnt[ka[i]][kb[i]] + delta[i];
        if (after < 0) return false;
        if ((after > 0) != req[ka[i]][kb[i]]) return false;
    }

    g[v] = newCol;
    removeCell(v, oldCol);
    addCell(v, newCol);
    for (int i = 0; i < cnt; ++i) edgeCnt[ka[i]][kb[i]] += delta[i];

    // Every nonzero ward must remain nonempty and connected.
    bool ok = !cells[oldCol].empty() && connectedColor(oldCol);

    if (!ok) {
        for (int i = 0; i < cnt; ++i) edgeCnt[ka[i]][kb[i]] -= delta[i];
        removeCell(v, newCol);
        addCell(v, oldCol);
        g[v] = oldCol;
        return false;
    }
    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    memset(req, 0, sizeof(req));
    memset(edgeCnt, 0, sizeof(edgeCnt));

    for (int r = 0; r < N; ++r) {
        for (int c = 0; c < N; ++c) {
            int v = id(r, c);
            cin >> g[v];
            addCell(v, g[v]);
        }
    }

    auto addReq = [&](int a, int b) {
        if (a == b) return;
        norm(a, b);
        req[a][b] = true;
    };
    auto addEdge = [&](int a, int b) {
        if (a == b) return;
        norm(a, b);
        ++edgeCnt[a][b];
    };

    for (int r = 0; r < N; ++r) {
        for (int c = 0; c < N; ++c) {
            int v = id(r, c);
            int a = g[v];

            if (r == 0) {
                addReq(0, a);
                addEdge(0, a);
            }
            if (r == N - 1) {
                addReq(0, a);
                addEdge(0, a);
            }
            if (c == 0) {
                addReq(0, a);
                addEdge(0, a);
            }
            if (c == N - 1) {
                addReq(0, a);
                addEdge(0, a);
            }

            if (r + 1 < N) {
                addReq(a, g[id(r + 1, c)]);
                addEdge(a, g[id(r + 1, c)]);
            }
            if (c + 1 < N) {
                addReq(a, g[id(r, c + 1)]);
                addEdge(a, g[id(r, c + 1)]);
            }
        }
    }

    const auto start = chrono::steady_clock::now();
    const double LIMIT_MS = 1930.0;

    auto elapsed = [&]() -> double {
        return chrono::duration<double, milli>(
            chrono::steady_clock::now() - start
        ).count();
    };

    int attempts = 0;
    while (elapsed() < LIMIT_MS) {
        ++attempts;

        /*
          82%: outside frontier peeling.
          If no zero exists yet, use a random boundary cell as the first legal
          candidate connected to outside.
        */
        if (rng(100) < 82) {
            int v = -1;

            if (cells[0].empty()) {
                int side = rng(4);
                if (side == 0) v = id(0, rng(N));
                else if (side == 1) v = id(N - 1, rng(N));
                else if (side == 2) v = id(rng(N), 0);
                else v = id(rng(N), N - 1);
            } else {
                int z = cells[0][rng((int)cells[0].size())];
                int r = z / N, c = z % N;
                int cand[4], cc = 0;
                for (int k = 0; k < 4; ++k) {
                    int nr = r + dr[k], nc = c + dc[k];
                    if (inside(nr, nc)) {
                        int u = id(nr, nc);
                        if (g[u] != 0) cand[cc++] = u;
                    }
                }
                if (cc) v = cand[rng(cc)];
            }

            if (v >= 0 && g[v] != 0) {
                tryMove(v, 0);
            }
        } else {
            /*
              Boundary transfer diversification.
              A random nonzero cell adopts an adjacent nonzero color. This
              preserves target connectivity by construction and may expose
              future removable frontier cells.
            */
            int v = rng(SZ);
            int old = g[v];
            if (old == 0) continue;

            int r = v / N, c = v % N;
            int cand[4], cc = 0;
            for (int k = 0; k < 4; ++k) {
                int nr = r + dr[k], nc = c + dc[k];
                if (!inside(nr, nc)) continue;
                int u = id(nr, nc);
                if (g[u] != 0 && g[u] != old) cand[cc++] = g[u];
            }
            if (cc) tryMove(v, cand[rng(cc)]);
        }

        // Occasionally force a broad random boundary sweep, useful if the
        // zero frontier has a narrow and repeatedly sampled boundary.
        if ((attempts & 2047) == 0) {
            for (int t = 0; t < 200 && elapsed() < LIMIT_MS; ++t) {
                int v = rng(SZ);
                if (g[v] == 0) continue;
                int r = v / N, c = v % N;
                bool touchesZero = (r == 0 || r == N - 1 || c == 0 || c == N - 1);
                for (int k = 0; k < 4 && !touchesZero; ++k) {
                    int nr = r + dr[k], nc = c + dc[k];
                    if (inside(nr, nc) && g[id(nr, nc)] == 0) touchesZero = true;
                }
                if (touchesZero) tryMove(v, 0);
            }
        }
    }

    for (int r = 0; r < N; ++r) {
        for (int c = 0; c < N; ++c) {
            if (c) cout << ' ';
            cout << g[id(r, c)];
        }
        cout << '\n';
    }
    return 0;
}
