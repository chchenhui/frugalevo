<DESCRIPTION>
Replace the single shuffled greedy deletion pass with repeated randomized erosion passes. A cell that cannot be removed early may become removable after neighboring wards have been simplified, so revisiting all cells substantially improves the monotonic construction before simulated annealing begins. The larger greedy budget still leaves over one second for SA-based rearrangement, while every accepted greedy move directly improves the score and preserves validity.
</DESCRIPTION>

    // Repeated erosion is much stronger than one sweep: deletions made during a
    // pass often turn previously protected cells into valid zero candidates.
    const double GREEDY_PASS_BUDGET_MS = 700.0;
    double greedy_pass_start_abs_time = time_elapsed_ms();

    std::vector<std::pair<int,int>> all_cells_shuffled;
    all_cells_shuffled.reserve(N_FIXED * N_FIXED);
    for(int r_idx=0; r_idx<N_FIXED; ++r_idx) for(int c_idx=0; c_idx<N_FIXED; ++c_idx) {
        all_cells_shuffled.push_back({r_idx,c_idx});
    }

    while (time_elapsed_ms() - greedy_pass_start_abs_time < GREEDY_PASS_BUDGET_MS) {
        std::shuffle(all_cells_shuffled.begin(), all_cells_shuffled.end(), rnd_gen);
        bool improved_this_pass = false;
        for (const auto& cell_coords : all_cells_shuffled) {
            if (time_elapsed_ms() - greedy_pass_start_abs_time >= GREEDY_PASS_BUDGET_MS) break;

        }
