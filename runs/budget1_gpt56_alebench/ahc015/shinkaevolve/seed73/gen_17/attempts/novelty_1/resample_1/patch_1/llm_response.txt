<DESCRIPTION>
This crossover keeps the compact fixed-array board representation and efficient in-place tilt/evaluation logic from the current implementation, while restoring the higher-performing inspiration program’s Monte-Carlo policy: its xorshift generator and conservative `{18, 7}` two-level random lookahead sampling. This avoids vector rebuilding and dynamic allocations during search, leaving more runtime margin while preserving the successful randomized expectimax behavior and strip-based clustering evaluation.
</DESCRIPTION>

# EVOLVE-BLOCK-START
#include <iostream>
#include <array>
#include <algorithm>
#include <cmath>
#include <cstdint>
#include <limits>
#include <chrono>

constexpr int N = 10;
constexpr int CELLS = 100;
constexpr int TURNS = 100;
constexpr int FLAVORS = 3;
constexpr int DIRECTIONS = 4;
constexpr int LOOKAHEAD = 2;

constexpr char DIRS[DIRECTIONS] = {'F', 'B', 'L', 'R'};
constexpr std::array<int, LOOKAHEAD> SAMPLES = {18, 7};

std::array<int, TURNS> flavor_seq;
std::array<int, FLAVORS + 1> flavor_total{};
std::array<int, FLAVORS + 1> strip_lo{};
std::array<int, FLAVORS + 1> strip_hi{};

struct XorShift {
    uint64_t x;

    XorShift()
        : x(static_cast<uint64_t>(
              std::chrono::steady_clock::now().time_since_epoch().count())) {}

    uint64_t next() {
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        return x;
    }

    int uniform(int lo, int hi) {
        if (lo >= hi) return lo;
        return lo + static_cast<int>(next() % static_cast<uint64_t>(hi - lo + 1));
    }
};

XorShift rng;

struct State {
    std::array<uint8_t, CELLS> board{};
    int occupied = 0;
    int turn = 0;

    int find_empty(int rank) const {
        int count = 0;
        for (int p = 0; p < CELLS; ++p) {
            if (board[p] == 0 && ++count == rank) return p;
        }
        return -1;
    }

    int only_empty() const {
        for (int p = 0; p < CELLS; ++p) {
            if (board[p] == 0) return p;
        }
        return -1;
    }

    void place(int pos, int flavor) {
        board[pos] = static_cast<uint8_t>(flavor);
        ++occupied;
    }

    void tilt(int dir) {
        if (dir == 0) {
            for (int c = 0; c < N; ++c) {
                int write = c;
                for (int r = 0; r < N; ++r) {
                    const int pos = r * N + c;
                    if (board[pos]) {
                        const uint8_t value = board[pos];
                        board[pos] = 0;
                        board[write] = value;
                        write += N;
                    }
                }
            }
        } else if (dir == 1) {
            for (int c = 0; c < N; ++c) {
                int write = (N - 1) * N + c;
                for (int r = N - 1; r >= 0; --r) {
                    const int pos = r * N + c;
                    if (board[pos]) {
                        const uint8_t value = board[pos];
                        board[pos] = 0;
                        board[write] = value;
                        write -= N;
                    }
                }
            }
        } else if (dir == 2) {
            for (int r = 0; r < N; ++r) {
                const int base = r * N;
                int write = base;
                for (int c = 0; c < N; ++c) {
                    const int pos = base + c;
                    if (board[pos]) {
                        const uint8_t value = board[pos];
                        board[pos] = 0;
                        board[write++] = value;
                    }
                }
            }
        } else {
            for (int r = 0; r < N; ++r) {
                const int base = r * N;
                int write = base + N - 1;
                for (int c = N - 1; c >= 0; --c) {
                    const int pos = base + c;
                    if (board[pos]) {
                        const uint8_t value = board[pos];
                        board[pos] = 0;
                        board[write--] = value;
                    }
                }
            }
        }
    }

    double evaluate() const {
        std::array<int, FLAVORS + 1> count{};
        std::array<int, FLAVORS + 1> sum_r{};
        std::array<int, FLAVORS + 1> sum_c{};

        int region_penalty = 0;
        int edge_units = 0;

        for (int p = 0; p < CELLS; ++p) {
            const int flavor = board[p];
            if (!flavor) continue;

            const int r = p / N;
            const int c = p % N;

            ++count[flavor];
            sum_r[flavor] += r;
            sum_c[flavor] += c;

            if (c < strip_lo[flavor]) {
                region_penalty += strip_lo[flavor] - c;
            } else if (c > strip_hi[flavor]) {
                region_penalty += c - strip_hi[flavor];
            } else {
                if (r == 0 || r == N - 1) ++edge_units;
                if ((c == 0 && strip_lo[flavor] == 0) ||
                    (c == N - 1 && strip_hi[flavor] == N - 1)) {
                    ++edge_units;
                }
            }
        }

        double com_penalty = 0.0;
        for (int p = 0; p < CELLS; ++p) {
            const int flavor = board[p];
            if (!flavor || count[flavor] <= 1) continue;

            const int r = p / N;
            const int c = p % N;
            const double center_r =
                static_cast<double>(sum_r[flavor]) / count[flavor];
            const double center_c =
                static_cast<double>(sum_c[flavor]) / count[flavor];

            com_penalty += std::abs(static_cast<double>(r) - center_r);
            com_penalty += std::abs(static_cast<double>(c) - center_c);
        }

        std::array<uint8_t, CELLS> visited{};
        std::array<int, CELLS> queue{};
        int component_square_sum = 0;

        for (int start = 0; start < CELLS; ++start) {
            const int flavor = board[start];
            if (!flavor || visited[start]) continue;

            int head = 0;
            int tail = 0;
            queue[tail++] = start;
            visited[start] = 1;

            while (head < tail) {
                const int p = queue[head++];
                const int r = p / N;
                const int c = p % N;

                if (r > 0) {
                    const int q = p - N;
                    if (!visited[q] && board[q] == flavor) {
                        visited[q] = 1;
                        queue[tail++] = q;
                    }
                }
                if (r + 1 < N) {
                    const int q = p + N;
                    if (!visited[q] && board[q] == flavor) {
                        visited[q] = 1;
                        queue[tail++] = q;
                    }
                }
                if (c > 0) {
                    const int q = p - 1;
                    if (!visited[q] && board[q] == flavor) {
                        visited[q] = 1;
                        queue[tail++] = q;
                    }
                }
                if (c + 1 < N) {
                    const int q = p + 1;
                    if (!visited[q] && board[q] == flavor) {
                        visited[q] = 1;
                        queue[tail++] = q;
                    }
                }
            }

            component_square_sum += tail * tail;
        }

        const double t = static_cast<double>(turn);
        const double connection_weight = 15.0 + 1.1 * t;
        const double com_weight = std::max(0.0, 170.0 - 1.7 * t);
        const double region_weight = std::max(2.0, 27.0 - 0.17 * t);
        const double edge_weight = 5.0 + 0.2 * t;

        return connection_weight * component_square_sum
             - com_weight * com_penalty
             - region_weight * region_penalty
             + edge_weight * 0.5 * edge_units;
    }
};

double search_value(const State& state, int depth) {
    if (depth == 0 || state.turn == TURNS || state.occupied == CELLS) {
        return state.evaluate();
    }

    const int empty_count = CELLS - state.occupied;
    const int next_flavor = flavor_seq[state.turn];

    if (empty_count == 1) {
        State next = state;
        next.place(next.only_empty(), next_flavor);
        next.turn = state.turn + 1;
        return next.evaluate();
    }

    const int sample_count =
        std::min(SAMPLES[LOOKAHEAD - depth], empty_count);

    double total = 0.0;

    for (int sample = 0; sample < sample_count; ++sample) {
        const int rank = (sample_count == empty_count)
            ? sample + 1
            : rng.uniform(1, empty_count);

        State placed = state;
        placed.place(placed.find_empty(rank), next_flavor);
        placed.turn = state.turn + 1;

        double best = -std::numeric_limits<double>::infinity();

        for (int dir = 0; dir < DIRECTIONS; ++dir) {
            State moved = placed;
            moved.tilt(dir);

            const double value = search_value(moved, depth - 1);
            if (value > best) best = value;
        }

        total += best;
    }

    return total / static_cast<double>(sample_count);
}

int choose_direction(const State& state) {
    double best_value = -std::numeric_limits<double>::infinity();
    int best_direction = 0;

    for (int dir = 0; dir < DIRECTIONS; ++dir) {
        State moved = state;
        moved.tilt(dir);

        const double value = search_value(moved, LOOKAHEAD);
        if (value > best_value) {
            best_value = value;
            best_direction = dir;
        }
    }

    return best_direction;
}

void initialize() {
    for (int t = 0; t < TURNS; ++t) {
        std::cin >> flavor_seq[t];
        ++flavor_total[flavor_seq[t]];
    }

    std::array<int, FLAVORS> order = {1, 2, 3};
    std::sort(order.begin(), order.end(), [](int a, int b) {
        if (flavor_total[a] != flavor_total[b]) {
            return flavor_total[a] > flavor_total[b];
        }
        return a < b;
    });

    std::array<int, FLAVORS + 1> width{};
    int assigned = 0;

    for (int flavor = 1; flavor <= FLAVORS; ++flavor) {
        width[flavor] = N * flavor_total[flavor] / TURNS;
        assigned += width[flavor];
    }

    for (int i = 0; assigned < N; ++i, ++assigned) {
        ++width[order[i % FLAVORS]];
    }

    int col = 0;
    for (int flavor : order) {
        strip_lo[flavor] = col;
        strip_hi[flavor] = col + width[flavor] - 1;
        col += width[flavor];
    }
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    initialize();

    State current;

    for (int t = 0; t < TURNS; ++t) {
        int placement_rank;
        std::cin >> placement_rank;

        current.turn = t + 1;
        current.place(current.find_empty(placement_rank), flavor_seq[t]);

        const int direction = (t == TURNS - 1)
            ? 0
            : choose_direction(current);

        std::cout << DIRS[direction] << std::endl;
        current.tilt(direction);
    }

    return 0;
}
