<DESCRIPTION>
This redesign separates board mechanics, heuristic evaluation, spatial sampling, and expectimax control into compact modules. It replaces phase-based rank sampling with distinct antithetic spatial samples: randomly selected empty cells are paired with cells nearest to their 180-degree mirrored positions. This improves coverage of both board regions while avoiding repeated chance outcomes. The heuristic returns to the stronger connectivity/center/strip shaping structure without the weaker mixed-edge penalty, while using a compact fixed-size board representation for faster simulation.
</DESCRIPTION>

// EVOLVE-BLOCK-START
#include <algorithm>
#include <array>
#include <chrono>
#include <cmath>
#include <cstdint>
#include <iostream>
#include <limits>

using namespace std;

namespace {

constexpr int N = 10;
constexpr int CELLS = 100;
constexpr int FLAVORS = 3;
constexpr int DIRS = 4;
constexpr array<char, DIRS> DIR_CHAR = {'F', 'B', 'L', 'R'};
constexpr array<int, 2> SAMPLE_COUNT = {20, 7};

struct RNG {
    uint64_t x;

    RNG() {
        x = static_cast<uint64_t>(
            chrono::steady_clock::now().time_since_epoch().count()
        ) | 1ULL;
    }

    uint64_t next() {
        x ^= x << 7;
        x ^= x >> 9;
        x ^= x << 8;
        return x;
    }

    int nextInt(int n) {
        return n <= 1 ? 0 : static_cast<int>(next() % static_cast<uint64_t>(n));
    }
};

RNG rng;

struct FlavorPlan {
    array<int, FLAVORS + 1> total{};
    array<int, FLAVORS + 1> left{};
    array<int, FLAVORS + 1> right{};

    void build(const array<int, CELLS>& sequence) {
        total.fill(0);
        left.fill(0);
        right.fill(-1);

        for (int flavor : sequence) ++total[flavor];

        array<int, FLAVORS> order = {1, 2, 3};
        sort(order.begin(), order.end(), [&](int a, int b) {
            if (total[a] != total[b]) return total[a] > total[b];
            return a < b;
        });

        array<int, FLAVORS + 1> width{};
        int allocated = 0;

        for (int flavor = 1; flavor <= FLAVORS; ++flavor) {
            width[flavor] = N * total[flavor] / CELLS;
            allocated += width[flavor];
        }

        for (int i = 0; allocated < N; ++i, ++allocated) {
            ++width[order[i % FLAVORS]];
        }

        int col = 0;
        for (int flavor : order) {
            left[flavor] = col;
            right[flavor] = col + width[flavor] - 1;
            col += width[flavor];
        }
    }
};

struct Board {
    array<uint8_t, CELLS> cells{};
    int placed = 0;

    int emptyCount() const {
        return CELLS - placed;
    }

    int nthEmpty(int rank) const {
        int seen = 0;
        for (int p = 0; p < CELLS; ++p) {
            if (cells[p] == 0 && ++seen == rank) return p;
        }
        return -1;
    }

    void put(int pos, int flavor) {
        cells[pos] = static_cast<uint8_t>(flavor);
        ++placed;
    }

    void tilt(int dir) {
        array<uint8_t, CELLS> next{};

        if (dir < 2) {
            const bool forward = (dir == 0);
            for (int c = 0; c < N; ++c) {
                int write = forward ? 0 : N - 1;
                const int delta = forward ? 1 : -1;

                for (int k = 0; k < N; ++k) {
                    const int r = forward ? k : N - 1 - k;
                    const uint8_t candy = cells[r * N + c];
                    if (candy != 0) {
                        next[write * N + c] = candy;
                        write += delta;
                    }
                }
            }
        } else {
            const bool leftward = (dir == 2);
            for (int r = 0; r < N; ++r) {
                int write = leftward ? 0 : N - 1;
                const int delta = leftward ? 1 : -1;

                for (int k = 0; k < N; ++k) {
                    const int c = leftward ? k : N - 1 - k;
                    const uint8_t candy = cells[r * N + c];
                    if (candy != 0) {
                        next[r * N + write] = candy;
                        write += delta;
                    }
                }
            }
        }

        cells = next;
    }
};

struct Heuristic {
    const FlavorPlan& plan;

    long long componentSquareSum(const Board& board) const {
        array<uint8_t, CELLS> visited{};
        array<int, CELLS> queue{};
        long long result = 0;

        for (int start = 0; start < CELLS; ++start) {
            const int flavor = board.cells[start];
            if (flavor == 0 || visited[start]) continue;

            int head = 0;
            int tail = 0;
            int size = 0;

            queue[tail++] = start;
            visited[start] = 1;

            while (head < tail) {
                const int p = queue[head++];
                ++size;

                const int r = p / N;
                const int c = p % N;

                if (r > 0) {
                    const int q = p - N;
                    if (!visited[q] && board.cells[q] == flavor) {
                        visited[q] = 1;
                        queue[tail++] = q;
                    }
                }
                if (r + 1 < N) {
                    const int q = p + N;
                    if (!visited[q] && board.cells[q] == flavor) {
                        visited[q] = 1;
                        queue[tail++] = q;
                    }
                }
                if (c > 0) {
                    const int q = p - 1;
                    if (!visited[q] && board.cells[q] == flavor) {
                        visited[q] = 1;
                        queue[tail++] = q;
                    }
                }
                if (c + 1 < N) {
                    const int q = p + 1;
                    if (!visited[q] && board.cells[q] == flavor) {
                        visited[q] = 1;
                        queue[tail++] = q;
                    }
                }
            }

            result += 1LL * size * size;
        }

        return result;
    }

    double evaluate(const Board& board) const {
        const long long connected = componentSquareSum(board);

        if (board.placed == CELLS) {
            return static_cast<double>(connected) * 10000.0;
        }

        array<int, FLAVORS + 1> count{};
        array<int, FLAVORS + 1> sumR{};
        array<int, FLAVORS + 1> sumC{};

        double stripPenalty = 0.0;
        double edgeBonus = 0.0;

        for (int p = 0; p < CELLS; ++p) {
            const int flavor = board.cells[p];
            if (flavor == 0) continue;

            const int r = p / N;
            const int c = p % N;

            ++count[flavor];
            sumR[flavor] += r;
            sumC[flavor] += c;

            if (c < plan.left[flavor]) {
                stripPenalty += plan.left[flavor] - c;
            } else if (c > plan.right[flavor]) {
                stripPenalty += c - plan.right[flavor];
            } else {
                if (r == 0 || r == N - 1) edgeBonus += 0.5;
                if ((c == 0 && plan.left[flavor] == 0) ||
                    (c == N - 1 && plan.right[flavor] == N - 1)) {
                    edgeBonus += 0.5;
                }
            }
        }

        double centerPenalty = 0.0;
        for (int p = 0; p < CELLS; ++p) {
            const int flavor = board.cells[p];
            if (flavor == 0 || count[flavor] <= 1) continue;

            const int r = p / N;
            const int c = p % N;

            const double centerR = static_cast<double>(sumR[flavor]) / count[flavor];
            const double centerC = static_cast<double>(sumC[flavor]) / count[flavor];

            centerPenalty += abs(r - centerR) + abs(c - centerC);
        }

        const double turn = static_cast<double>(board.placed);
        const double connectivityWeight = 15.0 + 1.1 * turn;
        const double centerWeight = max(0.0, 170.0 - 1.7 * turn);
        const double stripWeight = max(2.0, 27.0 - 0.17 * turn);
        const double edgeWeight = 5.0 + 0.2 * turn;

        return connectivityWeight * connected
             - centerWeight * centerPenalty
             - stripWeight * stripPenalty
             + edgeWeight * edgeBonus;
    }
};

struct AntitheticSampler {
    static int collect(const Board& board, int desired, array<int, CELLS>& out) {
        array<int, CELLS> empty{};
        int emptyCount = 0;

        for (int p = 0; p < CELLS; ++p) {
            if (board.cells[p] == 0) empty[emptyCount++] = p;
        }

        const int samples = min(desired, emptyCount);
        if (samples == emptyCount) {
            for (int i = 0; i < samples; ++i) out[i] = empty[i];
            return samples;
        }

        array<uint8_t, CELLS> selected{};
        int used = 0;

        auto addRandomUnused = [&]() {
            int remaining = emptyCount - used;
            int pick = rng.nextInt(remaining);

            for (int i = 0; i < emptyCount; ++i) {
                const int p = empty[i];
                if (selected[p]) continue;
                if (pick-- == 0) {
                    selected[p] = 1;
                    out[used++] = p;
                    return;
                }
            }
        };

        while (used + 1 < samples) {
            addRandomUnused();

            const int source = out[used - 1];
            const int targetR = N - 1 - source / N;
            const int targetC = N - 1 - source % N;

            int bestPos = -1;
            int bestDistance = numeric_limits<int>::max();

            for (int i = 0; i < emptyCount; ++i) {
                const int p = empty[i];
                if (selected[p]) continue;

                const int dist =
                    abs(p / N - targetR) + abs(p % N - targetC);

                if (dist < bestDistance ||
                    (dist == bestDistance && (rng.next() & 3ULL) == 0)) {
                    bestDistance = dist;
                    bestPos = p;
                }
            }

            selected[bestPos] = 1;
            out[used++] = bestPos;
        }

        if (used < samples) addRandomUnused();
        return used;
    }
};

struct Expectimax {
    const array<int, CELLS>& sequence;
    const Heuristic& heuristic;

    double search(const Board& state, int depth) const {
        if (depth == 0 || state.placed >= CELLS) {
            return heuristic.evaluate(state);
        }

        const int sampleTarget = SAMPLE_COUNT[2 - depth];
        array<int, CELLS> positions{};
        const int samples = AntitheticSampler::collect(state, sampleTarget, positions);

        const int flavor = sequence[state.placed];
        double expectedValue = 0.0;

        for (int s = 0; s < samples; ++s) {
            Board afterPut = state;
            afterPut.put(positions[s], flavor);

            double bestChild = -numeric_limits<double>::infinity();

            for (int dir = 0; dir < DIRS; ++dir) {
                Board child = afterPut;
                child.tilt(dir);
                bestChild = max(bestChild, search(child, depth - 1));
            }

            expectedValue += bestChild;
        }

        return expectedValue / samples;
    }

    int decide(const Board& afterPut) const {
        int bestDirection = 0;
        double bestValue = -numeric_limits<double>::infinity();

        for (int dir = 0; dir < DIRS; ++dir) {
            Board child = afterPut;
            child.tilt(dir);

            const double value = search(child, 2);
            if (value > bestValue) {
                bestValue = value;
                bestDirection = dir;
            }
        }

        return bestDirection;
    }
};

} // namespace

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    array<int, CELLS> sequence{};
    for (int i = 0; i < CELLS; ++i) cin >> sequence[i];

    FlavorPlan plan;
    plan.build(sequence);

    Heuristic heuristic{plan};
    Expectimax policy{sequence, heuristic};
    Board board;

    for (int turn = 0; turn < CELLS; ++turn) {
        int rank;
        cin >> rank;

        board.put(board.nthEmpty(rank), sequence[turn]);

        const int direction = policy.decide(board);
        cout << DIR_CHAR[direction] << '\n' << flush;

        board.tilt(direction);
    }

    return 0;
}
