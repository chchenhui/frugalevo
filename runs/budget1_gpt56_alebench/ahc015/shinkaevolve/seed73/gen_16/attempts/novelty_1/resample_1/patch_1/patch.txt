+                if (selected[p]) continue;

+            addRandomUnused();

+            const int source = out[used - 1];
+            const int targetR = N - 1 - source / N;
+            const int targetC = N - 1 - source % N;
+
+            int bestPos = -1;

+            for (int i = 0; i < emptyCount; ++i) {
