close_root_resampling
</NAME>

<DESCRIPTION>
Add a targeted second Monte-Carlo batch for ambiguous root decisions. The existing search already uses distinct chance ranks, which is good for coverage, but each root direction currently receives an independent sample set, so close root values can be selected due to sampling noise.

First evaluate all four directions with the established `{18,7}` budget. If the best and second-best directions are sufficiently close, independently re-evaluate only those two with a smaller `{8,3}` batch and combine the estimates weighted by their first-chance sample counts. This preserves full search quality for clear decisions while spending modest additional computation only where it can change the selected action. The worst-case added leaf work is small relative to the baseline and is avoided entirely on unambiguous turns.
</DESCRIPTION>

char decide_tilt_direction_logic(const GameState& current_gs_after_placement) {
    std::array<double, NUM_DIRECTIONS> values;
    int turn_T_for_lookahead_base = current_gs_after_placement.turn_num_1_indexed;

    for (int i = 0; i < NUM_DIRECTIONS; ++i) {
        GameState gs_after_tilt_T = current_gs_after_placement;
        gs_after_tilt_T.apply_tilt(i);
        values[i] = eval_lookahead(gs_after_tilt_T, turn_T_for_lookahead_base,
                                   MAX_LOOKAHEAD_DEPTH, NUM_SAMPLES_CONFIG);
    }

    int best_dir_idx = 0;
    int second_dir_idx = 1;
    if (values[second_dir_idx] > values[best_dir_idx]) {
        std::swap(best_dir_idx, second_dir_idx);
    }
    for (int i = 2; i < NUM_DIRECTIONS; ++i) {
        if (values[i] > values[best_dir_idx]) {
            second_dir_idx = best_dir_idx;
            best_dir_idx = i;
        } else if (values[i] > values[second_dir_idx]) {
            second_dir_idx = i;
        }
    }

    // A clear winner needs no extra work.  For close calls, use fresh,
    // independent unique ranks for just the leading candidates, then blend
    // the two estimates according to their first chance-layer sample counts.
    if (values[best_dir_idx] - values[second_dir_idx] < ROOT_CLOSE_MARGIN &&
        turn_T_for_lookahead_base < NUM_TURNS) {
        for (int dir_idx : {best_dir_idx, second_dir_idx}) {
            GameState gs_after_tilt_T = current_gs_after_placement;
            gs_after_tilt_T.apply_tilt(dir_idx);
            double extra = eval_lookahead(gs_after_tilt_T, turn_T_for_lookahead_base,
                                          MAX_LOOKAHEAD_DEPTH, ROOT_RESAMPLE_CONFIG);
            values[dir_idx] =
                (NUM_SAMPLES_CONFIG[0] * values[dir_idx] +
                 ROOT_RESAMPLE_CONFIG[0] * extra) /
                static_cast<double>(NUM_SAMPLES_CONFIG[0] + ROOT_RESAMPLE_CONFIG[0]);
        }

        best_dir_idx = 0;
        for (int i = 1; i < NUM_DIRECTIONS; ++i) {
            if (values[i] > values[best_dir_idx]) {
                best_dir_idx = i;
            }
        }
    }

    return DIR_CHARS[best_dir_idx];
}
