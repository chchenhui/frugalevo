<DESCRIPTION>
Replace the per-score dynamic DSU allocations with fixed-size arrays. The board has at most 100 cells, but the previous implementation allocated and initialized three vectors for every uncached candidate evaluation. Beam search evaluates a very large number of such candidates, so this allocation overhead materially reduces the number of useful states explored before the time limit.

Use the saved time to widen the beam for larger boards. The history-storage limit remains safe: at the maximum relevant depths these widths still fit below the existing 3,000,000-entry capacity. A wider beam is particularly valuable here because many locally plausible partial forests lead to dead ends or cycles later.
</DESCRIPTION>

struct DSU {
    std::array<int, N_MAX_CONST * N_MAX_CONST> parent;
    std::array<int, N_MAX_CONST * N_MAX_CONST> nodes_in_set;
    std::array<int, N_MAX_CONST * N_MAX_CONST> edges_in_set;
    int N_sq_total_cells;

    DSU(int current_N) : N_sq_total_cells(current_N * current_N) {
        for (int i = 0; i < N_sq_total_cells; ++i) {
            parent[i] = i;
            nodes_in_set[i] = 0;
            edges_in_set[i] = 0;
        }
    }

    int find(int i) {
        if (parent[i] == i) return i;
        return parent[i] = find(parent[i]);
    }

    void unite(int i_idx, int j_idx) {
        int root_i = find(i_idx);
        int root_j = find(j_idx);

        if (nodes_in_set[root_i] < nodes_in_set[root_j]) std::swap(root_i, root_j);

        parent[root_j] = root_i;
        nodes_in_set[root_i] += nodes_in_set[root_j];
        edges_in_set[root_i] += edges_in_set[root_j];
    }

    void add_edge(int u_idx, int v_idx) {
        int root_u = find(u_idx);
        int root_v = find(v_idx);
        if (root_u != root_v) {
            unite(root_u, root_v);
            edges_in_set[find(root_u)]++;
        } else {
            edges_in_set[root_u]++;
        }
    }
};
