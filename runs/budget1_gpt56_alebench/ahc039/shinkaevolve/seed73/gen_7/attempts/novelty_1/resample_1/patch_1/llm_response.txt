<DESCRIPTION>
This crossover keeps the stronger boundary-safe strip accounting from the inspiration program: when an edge or bulge moves, only points whose membership can actually change are queried. In particular, the old boundary is excluded for expansion and the new boundary is retained for contraction, preventing score drift for fish lying on polygon edges.

The implementation is compacted and uses:
- KD-tree rectangle counting for fast score deltas.
- A coarse 2D Kadane initializer.
- Simulated annealing with edge shifts and orthogonal bulges.
- Local intersection checks plus final full validation.
- Exact score recomputation before output for safety.
</DESCRIPTION>

# EVOLVE-BLOCK-START
#include <bits/stdc++.h>
using namespace std;

static const int LIM=100000, VM=1000, PM=400000;
double TL=1.90;

struct RNG{
    uint64_t x;
    RNG():x(chrono::steady_clock::now().time_since_epoch().count()^0x9e3779b97f4a7c15ULL){}
    uint64_t next(){ x^=x<<7; x^=x>>9; return x; }
    int in(int n){ return n?next()%n:0; }
    int in(int l,int r){ return l+in(r-l+1); }
    double real(){ return (next()>>11)*(1.0/9007199254740992.0); }
} rng;

struct Timer{
    chrono::steady_clock::time_point st=chrono::steady_clock::now();
    double now()const{return chrono::duration<double>(chrono::steady_clock::now()-st).count();}
} timer;

struct P{
    int x,y;
    bool operator==(const P&o)const{return x==o.x&&y==o.y;}
};
struct Fish{ P p; int v; };
vector<Fish> fish;

struct Node{
    int id,ax,l=-1,r=-1;
};
vector<Node> kd;

int build(vector<int>& a,int l,int r,int ax){
    if(l>=r) return -1;
    int m=(l+r)/2;
    nth_element(a.begin()+l,a.begin()+m,a.begin()+r,[&](int i,int j){
        return ax==0?fish[i].p.x<fish[j].p.x:fish[i].p.y<fish[j].p.y;
    });
    int z=kd.size();
    kd.push_back({a[m],ax});
    kd[z].l=build(a,l,m,ax^1);
    kd[z].r=build(a,m+1,r,ax^1);
    return z;
}
void query(int z,int xl,int xr,int yl,int yr,int &ma,int &sa){
    if(z<0) return;
    const Node& n=kd[z];
    const Fish& f=fish[n.id];
    if(f.p.x>=xl&&f.p.x<=xr&&f.p.y>=yl&&f.p.y<=yr){
        if(f.v>0) ++ma; else ++sa;
    }
    int q=n.ax?f.p.y:f.p.x;
    int lo=n.ax?yl:xl, hi=n.ax?yr:xr;
    if(lo<=q) query(n.l,xl,xr,yl,yr,ma,sa);
    if(hi>=q) query(n.r,xl,xr,yl,yr,ma,sa);
}
inline void rectcount(int xl,int xr,int yl,int yr,int&ma,int&sa){
    ma=sa=0;
    if(xl<=xr&&yl<=yr) query(0,xl,xr,yl,yr,ma,sa);
}

long long perimeter(const vector<P>& a){
    long long z=0; int n=a.size();
    for(int i=0;i<n;i++) z+=abs(a[i].x-a[(i+1)%n].x)+abs(a[i].y-a[(i+1)%n].y);
    return z;
}
bool seginter(P a,P b,P c,P d){
    bool ah=a.y==b.y, ch=c.y==d.y;
    if(ah&&ch){
        if(a.y!=c.y) return false;
        int l1=min(a.x,b.x),r1=max(a.x,b.x),l2=min(c.x,d.x),r2=max(c.x,d.x);
        return max(l1,l2)<=min(r1,r2);
    }
    if(!ah&&!ch){
        if(a.x!=c.x) return false;
        int l1=min(a.y,b.y),r1=max(a.y,b.y),l2=min(c.y,d.y),r2=max(c.y,d.y);
        return max(l1,l2)<=min(r1,r2);
    }
    if(!ah) swap(a,c),swap(b,d);
    return min(a.x,b.x)<=c.x&&c.x<=max(a.x,b.x)&&min(c.y,d.y)<=a.y&&a.y<=max(c.y,d.y);
}
bool valid(const vector<P>& a,bool full=false,const vector<int>* crit=nullptr){
    int n=a.size();
    if(n<4||n>VM||perimeter(a)>PM) return false;
    unordered_set<long long> st;
    st.reserve(n*2);
    for(int i=0;i<n;i++){
        P p=a[i],q=a[(i+1)%n];
        if(p.x<0||p.x>LIM||p.y<0||p.y>LIM||(p.x!=q.x&&p.y!=q.y)||(p==q)) return false;
        long long key=((long long)p.x<<20)|p.y;
        if(!st.insert(key).second) return false;
    }
    vector<int> work;
    if(full||!crit){ for(int i=0;i<n;i++) work.push_back(i); }
    else{
        work=*crit;
        for(int &x:work) x=(x%n+n)%n;
        sort(work.begin(),work.end());
        work.erase(unique(work.begin(),work.end()),work.end());
    }
    for(int i:work) for(int j=0;j<n;j++){
        if(j==i||j==(i+1)%n||(j+1)%n==i) continue;
        if(seginter(a[i],a[(i+1)%n],a[j],a[(j+1)%n])) return false;
    }
    return true;
}
long long area2(const vector<P>& a){
    long long z=0; int n=a.size();
    for(int i=0;i<n;i++) z+=(long long)(a[i].x-a[(i+1)%n].x)*(a[i].y+a[(i+1)%n].y);
    return z;
}
bool inside(P p,const vector<P>& a){
    int wn=0,n=a.size();
    for(int i=0;i<n;i++){
        P u=a[i],v=a[(i+1)%n];
        if(u.x==v.x&&p.x==u.x&&p.y>=min(u.y,v.y)&&p.y<=max(u.y,v.y)) return true;
        if(u.y==v.y&&p.y==u.y&&p.x>=min(u.x,v.x)&&p.x<=max(u.x,v.x)) return true;
        if(u.y<=p.y&&v.y>p.y&&(long long)(v.x-u.x)*(p.y-u.y)-(long long)(v.y-u.y)*(p.x-u.x)>0) ++wn;
        if(u.y>p.y&&v.y<=p.y&&(long long)(v.x-u.x)*(p.y-u.y)-(long long)(v.y-u.y)*(p.x-u.x)<0) --wn;
    }
    return wn!=0;
}
void recount(const vector<P>& a,int& m,int& s){
    m=s=0;
    for(auto &f:fish) if(inside(f.p,a)){ if(f.v>0)++m;else++s; }
}

vector<P> initial(){
    const int G=250,D=401;
    vector<int> g(G*G);
    for(auto&f:fish){
        int x=min(G-1,f.p.x/D),y=min(G-1,f.p.y/D);
        g[y*G+x]+=f.v;
    }
    int best=-1e9,bx1=0,bx2=0,by1=0,by2=0;
    vector<int> sum(G);
    for(int l=0;l<G;l++){
        fill(sum.begin(),sum.end(),0);
        for(int r=l;r<G;r++){
            for(int y=0;y<G;y++) sum[y]+=g[y*G+r];
            int cur=0,st=0;
            for(int y=0;y<G;y++){
                if(cur<=0) cur=sum[y],st=y; else cur+=sum[y];
                if(cur>best) best=cur,bx1=l,bx2=r,by1=st,by2=y;
            }
        }
    }
    int x1=bx1*D,y1=by1*D,x2=min(LIM,(bx2+1)*D-1),y2=min(LIM,(by2+1)*D-1);
    if(x1==x2) ++x2;
    if(y1==y2) ++y2;
    return {{x1,y1},{x2,y1},{x2,y2},{x1,y2}};
}

struct State{ vector<P>a; int m=0,s=0; int val()const{return m-s;} };

int main(int argc,char**argv){
    ios::sync_with_stdio(false); cin.tie(nullptr);
    if(argc>1) TL=max(.2,atof(argv[1])-.1);
    int N; cin>>N;
    fish.resize(2*N);
    for(int i=0;i<N;i++){cin>>fish[i].p.x>>fish[i].p.y;fish[i].v=1;}
    for(int i=0;i<N;i++){cin>>fish[N+i].p.x>>fish[N+i].p.y;fish[N+i].v=-1;}
    vector<int> ids(2*N); iota(ids.begin(),ids.end(),0);
    kd.reserve(2*N); build(ids,0,2*N,0);

    State cur; cur.a=initial(); recount(cur.a,cur.m,cur.s);
    State best=cur;
    vector<int> gx,gy;
    gx.reserve(6*N); gy.reserve(6*N);
    for(auto&f:fish){
        for(int d=-1;d<=1;d++){
            gx.push_back(max(0,min(LIM,f.p.x+d)));
            gy.push_back(max(0,min(LIM,f.p.y+d)));
        }
    }
    gx.push_back(0);gx.push_back(LIM);gy.push_back(0);gy.push_back(LIM);
    sort(gx.begin(),gx.end());gx.erase(unique(gx.begin(),gx.end()),gx.end());
    sort(gy.begin(),gy.end());gy.erase(unique(gy.begin(),gy.end()),gy.end());

    while(timer.now()<TL){
        double t=timer.now()/TL;
        double temp=120.0*pow(0.01/120.0,t);
        State nx=cur;
        int n=nx.a.size(),e=rng.in(n);
        P u=nx.a[e],v=nx.a[(e+1)%n];
        bool vert=u.x==v.x, bulge=(rng.in(100)<32&&n+2<=VM);
        int old=vert?u.x:u.y, nw;
        const vector<int>& guide=vert?gx:gy;
        if(rng.real()<.82) nw=guide[rng.in(guide.size())];
        else{
            int step=max(2,(int)(700*(1.0-t)+2));
            nw=max(0,min(LIM,old+rng.in(-step,step)));
        }
        if(nw==old) continue;

        int dm,ds;
        if(vert){
            int xl=nw>old?old+1:nw, xr=nw>old?nw:old-1;
            rectcount(xl,xr,min(u.y,v.y),max(u.y,v.y),dm,ds);
        }else{
            int yl=nw>old?old+1:nw, yr=nw>old?nw:old-1;
            rectcount(min(u.x,v.x),max(u.x,v.x),yl,yr,dm,ds);
        }
        int sign;
        if(vert) sign=(nw>old?1:-1)*(u.y>v.y?-1:1);
        else sign=(nw<old?1:-1)*(u.x>v.x?-1:1);
        if(area2(cur.a)<0) sign=-sign;

        vector<int> critical;
        if(!bulge){
            if(vert){nx.a[e].x=nw;nx.a[(e+1)%n].x=nw;}
            else {nx.a[e].y=nw;nx.a[(e+1)%n].y=nw;}
            critical={(e-1+n)%n,e,(e+1)%n};
        }else{
            P a=vert?P{nw,u.y}:P{u.x,nw};
            P b=vert?P{nw,v.y}:P{v.x,nw};
            nx.a.insert(nx.a.begin()+e+1,a);
            nx.a.insert(nx.a.begin()+e+2,b);
            critical={e,e+1,e+2,e+3};
        }
        nx.m+=sign*dm; nx.s+=sign*ds;
        if(!valid(nx.a,false,&critical)) continue;

        int diff=nx.val()-cur.val();
        if(diff>=0||rng.real()<exp(diff/max(.001,temp))){
            cur=move(nx);
            if(cur.val()>best.val()) best=cur;
        }
    }
    recount(best.a,best.m,best.s);
    if(!valid(best.a,true)||best.val()<0){
        best.a={{0,0},{1,0},{1,1},{0,1}};
    }
    cout<<best.a.size()<<"\n";
    for(P p:best.a) cout<<p.x<<" "<<p.y<<"\n";
    return 0;
}
