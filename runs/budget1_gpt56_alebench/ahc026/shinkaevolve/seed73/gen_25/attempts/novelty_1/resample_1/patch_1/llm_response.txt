<DESCRIPTION>
No specific runtime error was captured, so the validation failure must be addressed by guaranteeing operation legality rather than relying on the complex heuristic/partial-simulation logic. The original program has many state-copy, timeout, and fallback paths that can make correctness difficult to ensure.

The replacement uses a direct deterministic simulation: for each required box `1..n`, it finds that box, moves the entire block above it to a different stack, then carries the target box out. Every move is legal, preserves order, and every carry-out occurs only when the smallest remaining box is on top. The total operation count is at most `2*n`, well below 5000.
</DESCRIPTION>

# EVOLVE-BLOCK-START
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    int h = n / m;
    vector<vector<int>> stacks(m, vector<int>(h));
    for (int i = 0; i < m; ++i) {
        for (int j = 0; j < h; ++j) {
            cin >> stacks[i][j];
        }
    }

    vector<pair<int, int>> operations;
    operations.reserve(2 * n);

    for (int target = 1; target <= n; ++target) {
        int src = -1;
        int pos = -1;

        for (int i = 0; i < m && src == -1; ++i) {
            for (int j = 0; j < (int)stacks[i].size(); ++j) {
                if (stacks[i][j] == target) {
                    src = i;
                    pos = j;
                    break;
                }
            }
        }

        // Move every box above target as one legal block.
        if (pos + 1 < (int)stacks[src].size()) {
            int dst = (src + 1) % m;
            if (dst == src) dst = (src + 1) % m;

            int first_moved_box = stacks[src][pos + 1];

            vector<int> block(
                stacks[src].begin() + pos + 1,
                stacks[src].end()
            );
            stacks[src].erase(stacks[src].begin() + pos + 1, stacks[src].end());
            stacks[dst].insert(stacks[dst].end(), block.begin(), block.end());

            operations.push_back({first_moved_box, dst + 1});
        }

        // Target is now at the top and is the smallest remaining box.
        stacks[src].pop_back();
        operations.push_back({target, 0});
    }

    for (const auto& [v, destination] : operations) {
        cout << v << ' ' << destination << '\n';
    }

    return 0;
}
# EVOLVE-BLOCK-END
