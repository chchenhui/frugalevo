<DESCRIPTION>
The original program relied on C++20 `std::span`, which may fail under the contest's standard C++17 compilation environment. It also used a complex beam-search/SA simulation pipeline despite validation only requiring a legal sequence; this increases the risk of timeout or state inconsistencies without providing correctness guarantees.

The fix replaces the fragile heuristic pipeline with a direct, fully simulated greedy algorithm. For each required box in ascending order, it moves exactly the boxes above it to the currently shortest different stack, then carries the target box out. All stack contents and box positions are updated after every move, ensuring every emitted operation is legal. The solution uses at most `2*n` operations.
</DESCRIPTION>

```cpp
# EVOLVE-BLOCK-START
#include <iostream>
#include <vector>
#include <utility>
#include <algorithm>

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int n, m;
    if (!(std::cin >> n >> m)) return 0;

    const int h = n / m;
    std::vector<std::vector<int>> stacks(m);
    std::vector<std::pair<int, int>> pos(n + 1);

    for (int i = 0; i < m; ++i) {
        stacks[i].resize(h);
        for (int j = 0; j < h; ++j) {
            std::cin >> stacks[i][j];
            pos[stacks[i][j]] = {i, j};
        }
    }

    std::vector<std::pair<int, int>> operations;
    operations.reserve(2 * n);

    for (int target = 1; target <= n; ++target) {
        int src = pos[target].first;
        int target_height = pos[target].second;

        // Move all boxes above target to another stack.
        if (target_height + 1 < static_cast<int>(stacks[src].size())) {
            int dst = -1;
            for (int i = 0; i < m; ++i) {
                if (i == src) continue;
                if (dst == -1 || stacks[i].size() < stacks[dst].size()) {
                    dst = i;
                }
            }

            // Under the stated constraints m=10, so a distinct destination exists.
            if (dst == -1) return 0;

            int old_dst_size = static_cast<int>(stacks[dst].size());
            int first_moved = stacks[src][target_height + 1];

            operations.push_back({first_moved, dst + 1});

            for (int i = target_height + 1; i < static_cast<int>(stacks[src].size()); ++i) {
                int box = stacks[src][i];
                stacks[dst].push_back(box);
                pos[box] = {dst, old_dst_size + (i - (target_height + 1))};
            }
            stacks[src].resize(target_height + 1);
        }

        // Target is now the top box of its stack and can legally be carried out.
        src = pos[target].first;
        stacks[src].pop_back();
        operations.push_back({target, 0});
    }

    for (const auto& op : operations) {
        std::cout << op.first << ' ' << op.second << '\n';
    }

    return 0;
}
# EVOLVE-BLOCK-END
``` 
