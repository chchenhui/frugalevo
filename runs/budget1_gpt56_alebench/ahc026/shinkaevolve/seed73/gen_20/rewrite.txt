#include <iostream>
#include <vector>
#include <utility>
#include <algorithm>

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int n, m;
    if (!(std::cin >> n >> m)) return 0;

    const int height = n / m;

    // Each stack is stored from bottom to top.
    std::vector<std::vector<int> > stacks(m);
    std::vector<std::pair<int, int> > pos(n + 1);

    for (int i = 0; i < m; ++i) {
        stacks[i].reserve(n);
        for (int j = 0; j < height; ++j) {
            int box;
            std::cin >> box;
            stacks[i].push_back(box);
            pos[box] = std::make_pair(i, j);
        }
    }

    std::vector<std::pair<int, int> > answer;
    answer.reserve(2 * n);

    for (int target = 1; target <= n; ++target) {
        int src = pos[target].first;
        int target_height = pos[target].second;
        int above_count = static_cast<int>(stacks[src].size()) - 1 - target_height;

        if (above_count > 0) {
            // Select a different destination stack. Prefer a shortest stack,
            // which also tends to reduce the amount moved later.
            int dst = -1;
            for (int i = 0; i < m; ++i) {
                if (i == src) continue;
                if (dst == -1 || stacks[i].size() < stacks[dst].size()) {
                    dst = i;
                }
            }

            // The original constraints have m = 10, so dst always exists.
            // Move precisely the suffix directly above target.
            int first_moved_height = target_height + 1;
            int first_moved_box = stacks[src][first_moved_height];

            answer.push_back(std::make_pair(first_moved_box, dst + 1));

            int old_dst_size = static_cast<int>(stacks[dst].size());
            for (int i = first_moved_height; i < static_cast<int>(stacks[src].size()); ++i) {
                int box = stacks[src][i];
                stacks[dst].push_back(box);
                pos[box] = std::make_pair(dst, old_dst_size + (i - first_moved_height));
            }
            stacks[src].resize(first_moved_height);
        }

        // target is now necessarily on top of its stack.
        answer.push_back(std::make_pair(target, 0));
        stacks[src].pop_back();
    }

    for (const auto& op : answer) {
        std::cout << op.first << ' ' << op.second << '\n';
    }

    return 0;
}
