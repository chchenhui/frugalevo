<DESCRIPTION>
No runtime error was captured, so the validation failure must be prevented by ensuring every emitted operation is legal. The original program used a large heuristic/beam-search/SA implementation with fragile state transitions and C++20 `std::span`; a corrupted simulated state could cause an invalid move or carry-out operation.

The fix replaces that fragile planning path with a direct state simulation. For each required box in ascending order, it finds the box, moves every box above it as one legal block to a different currently shortest stack, then carries out the target box. This always produces at most `2*n` operations and keeps the stack state synchronized with output operations.
</DESCRIPTION>

<CODE>
```cpp
// EVOLVE-BLOCK-START
#include <iostream>
#include <vector>
#include <utility>
#include <algorithm>

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int n, m;
    if (!(std::cin >> n >> m)) return 0;

    const int h = n / m;
    std::vector<std::vector<int>> stacks(m);
    for (int i = 0; i < m; ++i) {
        stacks[i].resize(h);
        for (int j = 0; j < h; ++j) {
            std::cin >> stacks[i][j];
        }
    }

    std::vector<std::pair<int, int>> operations;
    operations.reserve(2 * n);

    for (int target = 1; target <= n; ++target) {
        int src = -1;
        int pos = -1;

        for (int i = 0; i < m && src == -1; ++i) {
            for (int j = 0; j < (int)stacks[i].size(); ++j) {
                if (stacks[i][j] == target) {
                    src = i;
                    pos = j;
                    break;
                }
            }
        }

        // Valid inputs always contain every not-yet-carried-out target.
        if (src == -1) return 0;

        const int above = (int)stacks[src].size() - pos - 1;

        if (above > 0) {
            int dest = -1;
            for (int i = 0; i < m; ++i) {
                if (i == src) continue;
                if (dest == -1 || stacks[i].size() < stacks[dest].size()) {
                    dest = i;
                }
            }

            // The official constraints have m = 10, so a distinct stack exists.
            if (dest == -1) return 0;

            int first_moved_box = stacks[src][pos + 1];
            operations.push_back({first_moved_box, dest + 1});

            stacks[dest].insert(
                stacks[dest].end(),
                stacks[src].begin() + pos + 1,
                stacks[src].end()
            );
            stacks[src].erase(stacks[src].begin() + pos + 1, stacks[src].end());
        }

        // target is now guaranteed to be the top box of its stack.
        operations.push_back({target, 0});
        stacks[src].pop_back();
    }

    for (const auto& op : operations) {
        std::cout << op.first << ' ' << op.second << '\n';
    }

    return 0;
}
// EVOLVE-BLOCK-END
``` 
