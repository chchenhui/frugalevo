<DESCRIPTION>
No concrete validator message was captured, so the failure must be addressed by removing fragile heuristic/state-copy logic rather than relying on a specific runtime trace. The original program performs complex beam search and simulated annealing despite correctness only requiring a legal sequence; this creates unnecessary timeout and state-consistency risk.

The replacement uses a direct simulation. For each required box in ascending order, it finds that box, moves every box above it to the shortest different stack, then carries out the target box. Every move is legal, all carry-outs occur only when the required smallest box is on top, and the operation count is at most 400 for the given constraints.
</DESCRIPTION>

// EVOLVE-BLOCK-START
#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int n, m;
    if (!(std::cin >> n >> m)) return 0;

    const int h = n / m;
    std::vector<std::vector<int>> stacks(m, std::vector<int>(h));

    for (int i = 0; i < m; ++i) {
        for (int j = 0; j < h; ++j) {
            std::cin >> stacks[i][j];
        }
    }

    for (int target = 1; target <= n; ++target) {
        int src = -1;
        int pos = -1;

        for (int i = 0; i < m && src == -1; ++i) {
            for (int j = 0; j < (int)stacks[i].size(); ++j) {
                if (stacks[i][j] == target) {
                    src = i;
                    pos = j;
                    break;
                }
            }
        }

        // Under valid input every remaining target is present.
        if (src == -1) return 0;

        if (pos + 1 < (int)stacks[src].size()) {
            int dest = -1;
            for (int i = 0; i < m; ++i) {
                if (i == src) continue;
                if (dest == -1 || stacks[i].size() < stacks[dest].size()) {
                    dest = i;
                }
            }

            // The official constraints guarantee m = 10, hence dest exists.
            if (dest == -1) return 0;

            int first_moved_box = stacks[src][pos + 1];
            std::cout << first_moved_box << ' ' << (dest + 1) << '\n';

            stacks[dest].insert(
                stacks[dest].end(),
                stacks[src].begin() + pos + 1,
                stacks[src].end()
            );
            stacks[src].erase(stacks[src].begin() + pos + 1, stacks[src].end());
        }

        // target is now the topmost box in its stack.
        std::cout << target << " 0\n";
        stacks[src].pop_back();
    }

    return 0;
}
// EVOLVE-BLOCK-END
