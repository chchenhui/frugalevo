<DESCRIPTION>
No specific runtime error was captured, so the failure must be treated as a validation failure. The original program used a time-based beam search and simulated annealing loop very close to the 2-second limit. This is unnecessary for validity and can make behavior timing-dependent; its complex partial-state simulation also increases the risk of producing an invalid sequence.

The fix replaces the heuristic search with a deterministic direct simulation. For each required box in ascending order, it finds that box, moves every box above it as one legal block to a different stack, then carries the target box out. Since `m=10`, another stack always exists. This produces at most 400 operations for `n=200`, well below the 5000-operation limit, and every emitted operation is legal.
</DESCRIPTION>

# EVOLVE-BLOCK-START
#include <iostream>
#include <vector>

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int n, m;
    if (!(std::cin >> n >> m)) return 0;

    const int height = n / m;
    std::vector<std::vector<int>> stacks(m, std::vector<int>(height));

    for (int i = 0; i < m; ++i) {
        for (int j = 0; j < height; ++j) {
            std::cin >> stacks[i][j];
        }
    }

    /*
     * Process boxes in the required order.
     *
     * Before carrying out box v, move the entire contiguous block above it
     * to a different stack.  This is exactly operation 1.  Then v is on
     * top of its stack and can legally be carried out by operation 2.
     *
     * At most one move and one carry-out are used per box, hence at most
     * 2*n = 400 operations under the official constraints.
     */
    for (int v = 1; v <= n; ++v) {
        int src = -1;
        int pos = -1;

        for (int s = 0; s < m && src == -1; ++s) {
            for (int p = 0; p < static_cast<int>(stacks[s].size()); ++p) {
                if (stacks[s][p] == v) {
                    src = s;
                    pos = p;
                    break;
                }
            }
        }

        // Input is guaranteed to be a permutation, so box v must be found.
        if (src == -1) return 0;

        if (pos + 1 < static_cast<int>(stacks[src].size())) {
            int dest = (src + 1) % m;

            // m is 10 in the official constraints, so dest != src.
            int first_moved_box = stacks[src][pos + 1];

            std::cout << first_moved_box << ' ' << (dest + 1) << '\n';

            stacks[dest].insert(
                stacks[dest].end(),
                stacks[src].begin() + pos + 1,
                stacks[src].end()
            );
            stacks[src].erase(stacks[src].begin() + pos + 1, stacks[src].end());
        }

        // Box v is now necessarily the top box of src.
        std::cout << v << " 0\n";
        stacks[src].pop_back();
    }

    return 0;
}
# EVOLVE-BLOCK-END
