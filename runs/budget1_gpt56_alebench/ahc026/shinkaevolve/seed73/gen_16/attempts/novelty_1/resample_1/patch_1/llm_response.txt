<DESCRIPTION>
The original program uses a complex beam-search and simulated-annealing pipeline, including C++20 `std::span`, despite heuristic solutions only needing to output a legal sequence. With no captured runtime error, the likely validation failure is caused by unsupported C++20 compilation settings, timeout risk from repeated full simulations, or an edge case in the optimization simulation.

The fix replaces the fragile optimization pipeline with a direct legal simulation. For each required box in ascending order, all boxes above it are moved as one block to a different stack, then the target box is carried out. This always satisfies the operation rules and stays well below the 5000-operation limit (at most 400 move operations plus 200 carry-out operations for the given constraints).
</DESCRIPTION>

# EVOLVE-BLOCK-START
#include <iostream>
#include <vector>
#include <utility>
#include <limits>

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int n, m;
    if (!(std::cin >> n >> m)) return 0;

    const int height = n / m;
    std::vector<std::vector<int>> stacks(m);
    std::vector<int> position_stack(n + 1);
    std::vector<int> position_height(n + 1);

    for (int i = 0; i < m; ++i) {
        stacks[i].resize(height);
        for (int j = 0; j < height; ++j) {
            std::cin >> stacks[i][j];
            position_stack[stacks[i][j]] = i;
            position_height[stacks[i][j]] = j;
        }
    }

    std::vector<std::pair<int, int>> operations;
    operations.reserve(2 * n);

    for (int target = 1; target <= n; ++target) {
        int src = position_stack[target];
        int target_height = position_height[target];
        int above_count = (int)stacks[src].size() - target_height - 1;

        if (above_count > 0) {
            int dest = -1;
            int best_size = std::numeric_limits<int>::max();

            // Select a different stack. Prefer the shortest one to keep
            // future blocks reasonably small.
            for (int i = 0; i < m; ++i) {
                if (i == src) continue;
                if ((int)stacks[i].size() < best_size) {
                    best_size = (int)stacks[i].size();
                    dest = i;
                }
            }

            // Under the stated constraints m = 10, so a destination always exists.
            // Move exactly the boxes above target, preserving their order.
            int first_moved_height = target_height + 1;
            for (int j = first_moved_height; j < (int)stacks[src].size(); ++j) {
                int box = stacks[src][j];
                position_stack[box] = dest;
                position_height[box] = (int)stacks[dest].size();
                stacks[dest].push_back(box);
            }

            stacks[src].resize(first_moved_height);

            // Operation 1 specifies the bottom box of the moved block.
            operations.push_back({stacks[dest][position_height[stacks[dest][best_size]]], dest + 1});
        }

        // The target is now the top box of its stack and can be carried out.
        stacks[src].pop_back();
        operations.push_back({target, 0});
    }

    for (const auto& op : operations) {
        std::cout << op.first << ' ' << op.second << '\n';
    }

    return 0;
}
# EVOLVE-BLOCK-END
