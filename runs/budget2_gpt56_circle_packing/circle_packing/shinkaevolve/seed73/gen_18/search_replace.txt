<DESCRIPTION>
Supply the exact Jacobian of every containment and pair-separation inequality to SLSQP. The current optimizer otherwise numerically differentiates 78 variables for each of many multistart solves, despite all constraint derivatives being elementary. This substantially reduces optimization cost and makes the contact-graph refinement more reliable, allowing the existing diverse asymmetric hexagonal seeds to converge more accurately within the evaluator time budget. Pair constraints retain Euclidean distance rather than squared distance, so their scale remains well conditioned near the active tangencies that define the packing.
</DESCRIPTION>

    def inequality_jacobian(z):
        """Exact derivatives of walls and circle-contact inequalities."""
        points = z[:2 * n].reshape(n, 2)
        rows = 4 * n + len(ii)
        jac = np.zeros((rows, 3 * n))

        indices = np.arange(n)
        xcols = 2 * indices
        ycols = xcols + 1
        rcols = 2 * n + indices

        # Constraint ordering matches inequalities exactly.
        jac[indices, xcols] = 1.0
        jac[indices, rcols] = -1.0
        jac[n + indices, ycols] = 1.0
        jac[n + indices, rcols] = -1.0
        jac[2 * n + indices, xcols] = -1.0
        jac[2 * n + indices, rcols] = -1.0
        jac[3 * n + indices, ycols] = -1.0
        jac[3 * n + indices, rcols] = -1.0

        dx = points[ii, 0] - points[jj, 0]
        dy = points[ii, 1] - points[jj, 1]
        lengths = np.sqrt(dx * dx + dy * dy)
        # Distinct lattice centers ensure positive lengths; the guard also
        # makes a perturbed degenerate start harmless.
        lengths = np.maximum(lengths, 1.0e-14)
        q = 4 * n + np.arange(len(ii))
        ux, uy = dx / lengths, dy / lengths
        jac[q, 2 * ii] = ux
        jac[q, 2 * ii + 1] = uy
        jac[q, 2 * jj] = -ux
        jac[q, 2 * jj + 1] = -uy
        jac[q, 2 * n + ii] = -1.0
        jac[q, 2 * n + jj] = -1.0
        return jac
