<DESCRIPTION>
Replace the single symmetric lattice optimization with a small deterministic multistart search over several 26-circle row distributions and perturbed staggered seeds. Finite square packings are strongly affected by which row contains the extra circle and by boundary contacts; the best solution is generally not reached from one perfectly symmetric seed. Every candidate is explicitly checked against all border and pairwise constraints before selection, retaining the existing final numerical safety scaling. This is the successful strategy indicated by the stronger prior constructor while remaining far below the timeout.
</DESCRIPTION>

    # These small initial circles make every lattice and perturbation feasible.
    initial_radii = np.full(n, 0.025)
    initial = np.concatenate((seed_centers.ravel(), initial_radii))

    try:
        from scipy.optimize import minimize

        rng = np.random.default_rng(26031991)
        starts = []
        for layout in row_layouts:
            base = make_seed(layout)
            starts.append(base)
            for _ in range(2):
                starts.append(base + rng.uniform(-0.012, 0.012, base.shape))

        candidate = initial
        best_value = float(np.sum(initial_radii))
        for points in starts:
            trial = np.concatenate((points.ravel(), initial_radii))
            result = minimize(
                lambda z: -np.sum(z[2 * n:]),
                trial,
                method="SLSQP",
                bounds=[(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n,
                constraints={"type": "ineq", "fun": inequalities},
                options={"maxiter": 2200, "ftol": 1e-11, "disp": False},
            )
            proposal = result.x
            # Reject unfinished SLSQP iterates rather than allowing a
            # subsequent global scale-down to hide constraint violations.
            if (np.all(np.isfinite(proposal))
                    and np.min(inequalities(proposal)) >= -1e-7):
                value = float(np.sum(proposal[2 * n:]))

                    candidate = proposal
                    best_value = value

        candidate = initial
