<DESCRIPTION>
This constructor replaces purely radius-uniform starts with a two-level approach. Each layered, asymmetric topology first receives radii from a fixed-center linear program: wall limits and every pairwise separation become linear radius constraints once centers are fixed. This produces unequal boundary-aware radii before nonlinear optimization begins.

The nonlinear stage then performs simultaneous center/radius contact optimization from a portfolio of twelve asymmetric hexagonal-row seeds, including the recommended five-row layouts and their vertical reversals. Unlike the previous equal-radius continuation, the LP initialization deliberately introduces useful radius asymmetry early, allowing SLSQP to explore contact graphs with small sacrificial boundary circles and larger interior circles. Candidates are conservatively normalized before comparison and again before return, ensuring strict validity.

# EVOLVE-BLOCK-START
"""Boundary-aware LP initialized circle packing constructor for n=26."""
import numpy as np


def construct_packing():
    """
    Construct 26 unequal non-overlapping circles in the unit square.

    A fixed-center linear program first distributes radii optimally for each
    layered topology.  Those feasible unequal-radius states are subsequently
    refined by a nonlinear contact optimization over both centers and radii.
    """
    from scipy.optimize import minimize, linprog

    n = 26
    ii, jj = np.triu_indices(n, 1)
    pair_count = len(ii)
    indices = np.arange(n)

    # These deliberately asymmetric five-row count patterns all total 26.
    # Their different locations for the two dense six-circle layers induce
    # different wall/diagonal contact graphs.
    layouts = (
        (4, 6, 6, 5, 5),
        (5, 5, 6, 6, 4),
        (6, 4, 6, 5, 5),
        (5, 6, 5, 6, 4),
        (5, 5, 6, 5, 5),
        (5, 6, 5, 5, 5),
    )

    def layered_seed(counts, reverse=False):
        """Make a slightly staggered triangular-lattice style seed."""
        points = []
        rows = len(counts)
        order = range(rows - 1, -1, -1) if reverse else range(rows)

        for out_row, source_row in enumerate(order):
            count = counts[source_row]
            y = (out_row + 0.5) / rows

            # Alternating offsets make diagonal contacts available immediately.
            # The mild source-row term prevents artificial reflection symmetry.
            phase = (0.18 if out_row % 2 else -0.18)
            phase += 0.035 * (source_row - 2)
            xs = (np.arange(count) + 0.5 + phase) / count
            xs = np.clip(xs, 0.028, 0.972)
            points.extend((x, y) for x in xs)

        return np.asarray(points, dtype=float)

    def contact_constraints(z):
        c = z[:2 * n].reshape(n, 2)
        r = z[2 * n:]

        dx = c[ii, 0] - c[jj, 0]
        dy = c[ii, 1] - c[jj, 1]
        d = np.sqrt(dx * dx + dy * dy)

        return np.concatenate((
            c[:, 0] - r,
            c[:, 1] - r,
            1.0 - c[:, 0] - r,
            1.0 - c[:, 1] - r,
            d - r[ii] - r[jj],
        ))

    def contact_jacobian(z):
        c = z[:2 * n].reshape(n, 2)
        jac = np.zeros((4 * n + pair_count, 3 * n), dtype=float)

        xcols = 2 * indices
        ycols = xcols + 1
        rcols = 2 * n + indices

        jac[indices, xcols] = 1.0
        jac[indices, rcols] = -1.0
        jac[n + indices, ycols] = 1.0
        jac[n + indices, rcols] = -1.0
        jac[2 * n + indices, xcols] = -1.0
        jac[2 * n + indices, rcols] = -1.0
        jac[3 * n + indices, ycols] = -1.0
        jac[3 * n + indices, rcols] = -1.0

        dx = c[ii, 0] - c[jj, 0]
        dy = c[ii, 1] - c[jj, 1]
        d = np.sqrt(dx * dx + dy * dy)
        d = np.maximum(d, 1.0e-14)

        rows = 4 * n + np.arange(pair_count)
        ux = dx / d
        uy = dy / d
        jac[rows, 2 * ii] = ux
        jac[rows, 2 * ii + 1] = uy
        jac[rows, 2 * jj] = -ux
        jac[rows, 2 * jj + 1] = -uy
        jac[rows, 2 * n + ii] = -1.0
        jac[rows, 2 * n + jj] = -1.0
        return jac

    def lp_radii(centers):
        """
        Maximize total radius with centers fixed.

        This is a genuine linear radius allocation problem:
        r_i <= wall_i and r_i + r_j <= distance(i,j).
        """
        wall = np.minimum.reduce((
            centers[:, 0], centers[:, 1],
            1.0 - centers[:, 0], 1.0 - centers[:, 1],
        ))

        distances = np.hypot(
            centers[ii, 0] - centers[jj, 0],
            centers[ii, 1] - centers[jj, 1],
        )

        matrix = np.zeros((pair_count, n), dtype=float)
        matrix[np.arange(pair_count), ii] = 1.0
        matrix[np.arange(pair_count), jj] = 1.0

        result = linprog(
            -np.ones(n),
            A_ub=matrix,
            b_ub=distances,
            bounds=[(0.004, float(w)) for w in wall],
            method="highs",
        )
        if result.success and np.all(np.isfinite(result.x)):
            return result.x

        # Conservative fallback remains feasible for every layered seed.
        return np.full(n, 0.022)

    def normalized(centers, radii):
        """Globally shrink radii only as needed for strict final feasibility."""
        centers = np.clip(np.asarray(centers, dtype=float), 0.0, 1.0)
        radii = np.maximum(np.asarray(radii, dtype=float), 0.0)

        wall = np.minimum.reduce((
            centers[:, 0], centers[:, 1],
            1.0 - centers[:, 0], 1.0 - centers[:, 1],
        ))
        scale = np.min(wall / np.maximum(radii, 1.0e-15))

        distances = np.hypot(
            centers[ii, 0] - centers[jj, 0],
            centers[ii, 1] - centers[jj, 1],
        )
        pair_scale = np.min(
            distances / np.maximum(radii[ii] + radii[jj], 1.0e-15)
        )
        scale = max(0.0, min(1.0, scale, pair_scale))
        return centers, radii * scale * (1.0 - 2.0e-10)

    objective_gradient = np.zeros(3 * n)
    objective_gradient[2 * n:] = -1.0

    objective = lambda z: -np.sum(z[2 * n:])
    bounds = [(0.0, 1.0)] * (2 * n) + [(1.0e-7, 0.5)] * n
    constraint = {
        "type": "ineq",
        "fun": contact_constraints,
        "jac": contact_jacobian,
    }

    # Six topologies and their vertical reversals form a compact twelve-start
    # portfolio without spending time on redundant random jitter.
    starts = []
    for layout in layouts:
        for reverse in (False, True):
            centers = layered_seed(layout, reverse)
            radii = lp_radii(centers)
            # Deflation gives the nonlinear solver maneuvering room while
            # preserving the useful unequal LP radius proportions.
            starts.append(np.concatenate((centers.ravel(), 0.90 * radii)))

    fallback_centers = layered_seed(layouts[0])
    fallback_radii = np.full(n, 0.02)
    best_centers, best_radii = normalized(fallback_centers, fallback_radii)
    best_value = float(np.sum(best_radii))

    for start in starts:
        result = minimize(
            objective,
            start,
            jac=lambda z: objective_gradient,
            method="SLSQP",
            bounds=bounds,
            constraints=constraint,
            options={"maxiter": 1150, "ftol": 6.0e-12, "disp": False},
        )

        if result.x is None or not np.all(np.isfinite(result.x)):
            continue

        candidate_centers, candidate_radii = normalized(
            result.x[:2 * n].reshape(n, 2),
            result.x[2 * n:],
        )
        value = float(np.sum(candidate_radii))
        if value > best_value:
            best_centers = candidate_centers
            best_radii = candidate_radii
            best_value = value

    # A final high-accuracy solve is reserved for the strongest basin found.
    polish_start = np.concatenate((best_centers.ravel(), best_radii * 0.999))
    polished = minimize(
        objective,
        polish_start,
        jac=lambda z: objective_gradient,
        method="SLSQP",
        bounds=bounds,
        constraints=constraint,
        options={"maxiter": 2800, "ftol": 8.0e-14, "disp": False},
    )

    if polished.x is not None and np.all(np.isfinite(polished.x)):
        candidate_centers, candidate_radii = normalized(
            polished.x[:2 * n].reshape(n, 2),
            polished.x[2 * n:],
        )
        value = float(np.sum(candidate_radii))
        if value > best_value:
            best_centers, best_radii, best_value = (
                candidate_centers, candidate_radii, value
            )

    return best_centers, best_radii, float(best_value)


def compute_max_radii(centers):
    """
    Compute conservative radii for fixed centers such that circles remain
    inside the unit square and do not overlap.
    """
    n = centers.shape[0]
    radii = np.minimum.reduce((
        centers[:, 0], centers[:, 1],
        1.0 - centers[:, 0], 1.0 - centers[:, 1],
    )).copy()

    for i in range(n):
        for j in range(i + 1, n):
            distance = np.linalg.norm(centers[i] - centers[j])
            if radii[i] + radii[j] > distance:
                factor = distance / (radii[i] + radii[j])
                radii[i] *= factor
                radii[j] *= factor

    return radii


# EVOLVE-BLOCK-END


def run_packing():
    """Run the circle packing constructor for n=26."""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """Visualize the unit-square circle packing."""
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    for i, (center, radius) in enumerate(zip(centers, radii)):
        ax.add_patch(Circle(center, radius, alpha=0.5))
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
