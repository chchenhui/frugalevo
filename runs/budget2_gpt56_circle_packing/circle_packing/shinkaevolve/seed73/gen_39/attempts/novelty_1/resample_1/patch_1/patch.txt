 def compute_max_radii(centers):
     """
     Compute conservative radii for supplied centers while preserving the
     original public helper interface.
     """
     centers = np.asarray(centers, dtype=float)
     n = centers.shape[0]
     radii = np.minimum.reduce((
         centers[:, 0],
         centers[:, 1],
         1.0 - centers[:, 0],
         1.0 - centers[:, 1],
     ))
 
     for i in range(n):
         for j in range(i + 1, n):
             dist = np.linalg.norm(centers[i] - centers[j])
             if radii[i] + radii[j] > dist:
                 scale = dist / (radii[i] + radii[j])
                 radii[i] *= scale
                 radii[j] *= scale
 
     return radii
