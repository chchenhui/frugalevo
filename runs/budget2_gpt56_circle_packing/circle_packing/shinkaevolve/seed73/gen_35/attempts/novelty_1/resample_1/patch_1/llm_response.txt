"""Gradient-assisted multistart circle packing constructor for 26 circles."""
import numpy as np


N_CIRCLES = 26


class PackingModel:
    """Vectorized squared-distance nonlinear model with exact derivatives."""

    def __init__(self, n):
        self.n = n
        self.dimension = 3 * n
        self.i, self.j = np.triu_indices(n, 1)
        self.pair_count = len(self.i)
        self.indices = np.arange(n)
        self.pair_rows = 4 * n + np.arange(self.pair_count)

    def objective(self, z):
        return -float(np.sum(z[2 * self.n:]))

    def objective_jacobian(self, z):
        grad = np.zeros(self.dimension, dtype=float)
        grad[2 * self.n:] = -1.0
        return grad

    def constraints(self, z):
        points = z[:2 * self.n].reshape(self.n, 2)
        radii = z[2 * self.n:]
        x, y = points[:, 0], points[:, 1]

        dx = x[self.i] - x[self.j]
        dy = y[self.i] - y[self.j]
        pair_gap = dx * dx + dy * dy - (radii[self.i] + radii[self.j]) ** 2

        return np.concatenate((
            x - radii,
            y - radii,
            1.0 - x - radii,
            1.0 - y - radii,
            pair_gap,
        ))

    def constraint_jacobian(self, z):
        n = self.n
        points = z[:2 * n].reshape(n, 2)
        radii = z[2 * n:]
        ind = self.indices

        jac = np.zeros((4 * n + self.pair_count, self.dimension), dtype=float)

        jac[ind, 2 * ind] = 1.0
        jac[ind, 2 * n + ind] = -1.0

        jac[n + ind, 2 * ind + 1] = 1.0
        jac[n + ind, 2 * n + ind] = -1.0

        jac[2 * n + ind, 2 * ind] = -1.0
        jac[2 * n + ind, 2 * n + ind] = -1.0

        jac[3 * n + ind, 2 * ind + 1] = -1.0
        jac[3 * n + ind, 2 * n + ind] = -1.0

        dx = points[self.i, 0] - points[self.j, 0]
        dy = points[self.i, 1] - points[self.j, 1]
        radius_sum = radii[self.i] + radii[self.j]
        rows = self.pair_rows

        jac[rows, 2 * self.i] = 2.0 * dx
        jac[rows, 2 * self.i + 1] = 2.0 * dy
        jac[rows, 2 * self.j] = -2.0 * dx
        jac[rows, 2 * self.j + 1] = -2.0 * dy
        jac[rows, 2 * n + self.i] = -2.0 * radius_sum
        jac[rows, 2 * n + self.j] = -2.0 * radius_sum

        return jac


def _row_layout(row_sizes, phase):
    """Construct a finite staggered triangular-lattice seed."""
    rows = len(row_sizes)
    centers = []

    for row, count in enumerate(row_sizes):
        y = (row + 0.5) / rows
        shift = phase if row % 2 else -phase
        for col in range(count):
            centers.append(((col + 0.5 + shift) / count, y))

    return np.asarray(centers, dtype=float)


def _make_seed(row_sizes, seed, phase):
    """Break lattice symmetry while retaining ample initial clearance."""
    rng = np.random.default_rng(seed)
    centers = _row_layout(row_sizes, phase)

    jitter = rng.normal(0.0, 0.016, centers.shape)
    jitter[:, 1] *= 0.72
    centers = np.clip(centers + jitter, 0.042, 0.958)

    wall_x = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    wall_y = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    corner_bias = np.clip((0.24 - wall_x) / 0.24, 0.0, 1.0)
    corner_bias *= np.clip((0.24 - wall_y) / 0.24, 0.0, 1.0)

    radii = 0.0118 + 0.0020 * corner_bias + rng.uniform(-0.0008, 0.0008, N_CIRCLES)
    return np.concatenate((centers.ravel(), np.clip(radii, 0.008, 0.015)))


def _repair(z, model):
    """Uniformly shrink radii enough to make every returned contact strict."""
    if z is None or np.shape(z) != (model.dimension,) or not np.all(np.isfinite(z)):
        return None

    n = model.n
    centers = np.clip(np.asarray(z[:2 * n], dtype=float).reshape(n, 2), 0.0, 1.0)
    radii = np.maximum(np.asarray(z[2 * n:], dtype=float), 0.0)

    border = np.minimum.reduce((
        centers[:, 0], centers[:, 1],
        1.0 - centers[:, 0], 1.0 - centers[:, 1],
    ))
    scale = float(np.min(border / np.maximum(radii, 1e-15)))

    dx = centers[model.i, 0] - centers[model.j, 0]
    dy = centers[model.i, 1] - centers[model.j, 1]
    distances = np.sqrt(dx * dx + dy * dy)
    pair_scale = distances / np.maximum(radii[model.i] + radii[model.j], 1e-15)

    scale = min(1.0, scale, float(np.min(pair_scale)))
    radii *= max(0.0, scale) * (1.0 - 3e-10)
    return np.concatenate((centers.ravel(), radii))


def _solve(model, initial, maxiter):
    """Run one exact-gradient SLSQP solve and return a safe candidate."""
    try:
        from scipy.optimize import minimize

        result = minimize(
            model.objective,
            initial,
            method="SLSQP",
            jac=model.objective_jacobian,
            bounds=[(0.0, 1.0)] * (2 * model.n) + [(1e-7, 0.5)] * model.n,
            constraints={
                "type": "ineq",
                "fun": model.constraints,
                "jac": model.constraint_jacobian,
            },
            options={"maxiter": maxiter, "ftol": 1.5e-11, "disp": False},
        )
        raw = result.x if np.all(np.isfinite(result.x)) else initial
    except Exception:
        raw = initial

    return _repair(raw, model)


def _score(z):
    return float(np.sum(z[2 * N_CIRCLES:]))


def construct_packing():
    """
    Construct 26 non-overlapping circles in the unit square.

    Returns:
        centers: array of shape (26, 2)
        radii: array of shape (26,)
        sum_of_radii: float
    """
    model = PackingModel(N_CIRCLES)

    # Five-row patterns preserve a dense hexagonal interior.  Six-row patterns
    # are included because their extra boundary freedom can create better
    # unequal-circle defects near corners.
    layouts = (
        (5, 6, 5, 5, 5),
        (5, 5, 6, 5, 5),
        (5, 5, 5, 6, 5),
        (6, 5, 5, 5, 5),
        (5, 6, 5, 6, 4),
        (4, 6, 5, 6, 5),
        (4, 5, 6, 6, 5),
        (5, 6, 6, 5, 4),
    )

    candidates = []
    phases = (-0.105, 0.040, 0.160)

    for layout_index, layout in enumerate(layouts):
        for variant, phase0 in enumerate(phases):
            phase = phase0 + 0.020 * (layout_index % 3)
            initial = _make_seed(
                layout,
                seed=26419 + 97 * layout_index + variant,
                phase=phase,
            )
            candidate = _solve(model, initial, maxiter=900)
            if candidate is not None:
                candidates.append(candidate)

    candidates.sort(key=_score, reverse=True)

    # Polish only the strongest basins with nearly contact-feasible radii.
    for rank, candidate in enumerate(candidates[:5]):
        continuation = candidate.copy()
        continuation[2 * N_CIRCLES:] *= 0.999995
        polished = _solve(model, continuation, maxiter=1900 + 150 * rank)
        if polished is not None:
            candidates.append(polished)

    if not candidates:
        fallback = _make_seed((5, 6, 5, 5, 5), seed=1, phase=0.0)
        candidates = [_repair(fallback, model)]

    best = max(candidates, key=_score)
    centers = best[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best[2 * N_CIRCLES:]
    return centers, radii, float(np.sum(radii))


def compute_max_radii(centers):
    """
    Compute conservative feasible radii for arbitrary fixed centers.

    Args:
        centers: np.array of shape (n, 2)

    Returns:
        np.array of shape (n)
    """
    centers = np.asarray(centers, dtype=float)
    n = len(centers)

    radii = np.minimum.reduce((
        centers[:, 0],
        centers[:, 1],
        1.0 - centers[:, 0],
        1.0 - centers[:, 1],
    ))
    radii = np.maximum(radii, 0.0)

    for _ in range(2):
        for i in range(n):
            for j in range(i + 1, n):
                distance = float(np.linalg.norm(centers[i] - centers[j]))
                total = radii[i] + radii[j]
                if total > distance and total > 0.0:
                    factor = distance / total
                    radii[i] *= factor
                    radii[j] *= factor

    return radii

def run_packing():
    """Run the circle packing constructor for n=26"""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """
    Visualize the circle packing

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle
    """

    fig, ax = plt.subplots(figsize=(8, 8))

    # Draw unit square
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    # Draw circles
    for i, (center, radius) in enumerate(zip(centers, radii)):
        circle = Circle(center, radius, alpha=0.5)
        ax.add_patch(circle)
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()
