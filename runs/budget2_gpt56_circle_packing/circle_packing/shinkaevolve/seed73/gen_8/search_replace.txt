<DESCRIPTION>
Replace the single highly symmetric initialization with a deterministic multi-start search over several finite hexagonal row layouts and small symmetry-breaking perturbations. The objective has many local optima: a single lattice seed tends to retain row artifacts and converges near 2.50, while staggered alternatives let SLSQP discover beneficial boundary-radius variation and asymmetric local rearrangements.

All starts are safely interior with small feasible radii. Only numerically feasible optimizer outputs are retained, and the existing final conservative scaling remains responsible for strict evaluator validity. The number of starts is modest and deterministic, so this remains comfortably within the timeout while substantially improving the chance of reaching the known approximately 2.635-quality basin.
</DESCRIPTION>

    # Optimize centers and radii simultaneously.  In contrast to successively
    # shrinking radii, these are the actual geometric packing constraints.
    ii, jj = np.triu_indices(n, 1)

    def inequalities(z):
        points = z[:2 * n].reshape(n, 2)
        rs = z[2 * n:]
        dx = points[ii, 0] - points[jj, 0]
        dy = points[ii, 1] - points[jj, 1]
        separations = np.sqrt(dx * dx + dy * dy) - rs[ii] - rs[jj]

    # Finite square packings have several distinct local optima.  In
    # particular, which rows receive the extra circle materially changes the
    # eventual boundary contacts.  Use several staggered hexagonal seeds and
    # small deterministic perturbations to avoid preserving artificial mirror
    # symmetries during the nonlinear optimization.
    row_layouts = (
        (5, 6, 5, 5, 5),
        (5, 6, 5, 6, 4),
        (6, 5, 5, 5, 5),
        (5, 5, 6, 5, 5),
    )

    def make_seed(row_sizes):
        points = []
        rows = len(row_sizes)
        for row, count in enumerate(row_sizes):
            y = (row + 0.5) / rows
            for col in range(count):
                points.append(((col + 0.5) / count, y))
        return np.asarray(points, dtype=float)

    initial = np.concatenate((seed_centers.ravel(), initial_radii))

        rng = np.random.default_rng(26031991)
        starts = []
        for layout in row_layouts:
            base = make_seed(layout)
            starts.append(base)
            # Two inequivalent small perturbations per layout are enough to
            # release lattice symmetry without making the initial point tight.
            for _ in range(2):
                starts.append(base + rng.uniform(-0.012, 0.012, base.shape))

        candidate = initial
        best_value = float(np.sum(initial_radii))
        for points in starts:
            trial = np.concatenate((points.ravel(), initial_radii))
            result = minimize(

                trial,
                method="SLSQP",
                bounds=[(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n,
                constraints={"type": "ineq", "fun": inequalities},
                options={"maxiter": 2200, "ftol": 1e-11, "disp": False},
            )
            proposal = result.x
            # Do not select a high-objective iterate that stopped before
            # satisfying its contacts; the final scaling is only a rounding
            # safeguard, not a substitute for feasible optimization.
            if (np.all(np.isfinite(proposal))
                    and np.min(inequalities(proposal)) >= -1e-7):
                value = float(np.sum(proposal[2 * n:]))
                if value > best_value:
                    candidate = proposal
                    best_value = value

        candidate = initial
