<DESCRIPTION>
This version uses a topology-diverse multistart strategy rather than refining only two nearly symmetric lattice states.  It generates several staggered layered seeds, jointly improves each geometry with smooth squared-distance constraints, then solves an exact linear program for the radii at the resulting centers.  The LP removes radius-allocation bias from nonlinear optimization and lets asymmetric boundary circles receive larger radii when beneficial.  The best valid LP-scored layout is returned with a strict safety margin.

# EVOLVE-BLOCK-START
"""Multistart LP-scored circle packing constructor for 26 circles."""
import numpy as np


def construct_packing():
    """
    Construct 26 non-overlapping circles in the unit square.

    Several different staggered-layer contact topologies are refined.  For
    every refined center configuration, the best possible radii are computed
    by a linear program:
        maximize sum(r_i)
        r_i + r_j <= distance(center_i, center_j)
        r_i <= distance to each wall.
    """
    from scipy.optimize import linprog, minimize

    n = 26
    ii, jj = np.triu_indices(n, 1)
    npairs = len(ii)

    pair_matrix = np.zeros((npairs, n), dtype=float)
    pair_matrix[np.arange(npairs), ii] = 1.0
    pair_matrix[np.arange(npairs), jj] = 1.0

    def layered_seed(row_sizes, phase_shift=0.0, jitter=0.0):
        """
        Make a broad triangular-like seed.  Row populations intentionally vary
        so that the nonlinear solver can discover different boundary contact
        graphs instead of being locked into a square lattice.
        """
        rows = len(row_sizes)
        points = []
        index = 0

        for row, count in enumerate(row_sizes):
            y = (row + 0.5) / rows
            phase = 0.50 if row % 2 == 0 else 0.36

            for col in range(count):
                x = (col + phase) / count

                # Deterministic asymmetric perturbation breaks reflections and
                # permits unequal boundary radii.
                dx = jitter * np.sin(1.731 * index + phase_shift)
                dy = jitter * np.cos(1.213 * index + 0.71 * phase_shift)

                points.append([
                    np.clip(x + dx, 0.025, 0.975),
                    np.clip(y + dy, 0.025, 0.975),
                ])
                index += 1

        return np.asarray(points, dtype=float)

    def improve_centers(seed):
        """
        Improve centers and provisional radii with smooth squared contact
        constraints.  Squared distances avoid repeated square-root Jacobian
        work and are well-conditioned away from coincident centers.
        """
        provisional_radius = 0.021
        z0 = np.r_[seed.ravel(), np.full(n, provisional_radius)]

        def constraints(z):
            c = z[:2 * n].reshape(n, 2)
            r = z[2 * n:]

            x = c[:, 0]
            y = c[:, 1]
            walls = np.r_[
                x - r,
                y - r,
                1.0 - x - r,
                1.0 - y - r,
            ]

            delta = c[ii] - c[jj]
            total = r[ii] + r[jj]
            contacts = 10.0 * (
                np.einsum("ij,ij->i", delta, delta) - total * total
            )

            return np.r_[r, walls, contacts]

        def jacobian(z):
            c = z[:2 * n].reshape(n, 2)
            r = z[2 * n:]

            rows = n + 4 * n + npairs
            jac = np.zeros((rows, 3 * n), dtype=float)
            disk = np.arange(n)

            # Positivity.
            jac[disk, 2 * n + disk] = 1.0

            offset = n

            # x-r, y-r, 1-x-r, 1-y-r.
            jac[offset + disk, 2 * disk] = 1.0
            jac[offset + disk, 2 * n + disk] = -1.0

            jac[offset + n + disk, 2 * disk + 1] = 1.0
            jac[offset + n + disk, 2 * n + disk] = -1.0

            jac[offset + 2 * n + disk, 2 * disk] = -1.0
            jac[offset + 2 * n + disk, 2 * n + disk] = -1.0

            jac[offset + 3 * n + disk, 2 * disk + 1] = -1.0
            jac[offset + 3 * n + disk, 2 * n + disk] = -1.0

            # Squared pair-contact constraints.
            delta = c[ii] - c[jj]
            total = r[ii] + r[jj]
            contact_rows = offset + 4 * n + np.arange(npairs)

            jac[contact_rows, 2 * ii] = 20.0 * delta[:, 0]
            jac[contact_rows, 2 * ii + 1] = 20.0 * delta[:, 1]
            jac[contact_rows, 2 * jj] = -20.0 * delta[:, 0]
            jac[contact_rows, 2 * jj + 1] = -20.0 * delta[:, 1]
            jac[contact_rows, 2 * n + ii] = -20.0 * total
            jac[contact_rows, 2 * n + jj] = -20.0 * total

            return jac

        objective_gradient = np.r_[np.zeros(2 * n), -np.ones(n)]

        result = minimize(
            fun=lambda z: -np.sum(z[2 * n:]),
            x0=z0,
            jac=lambda z: objective_gradient,
            method="SLSQP",
            bounds=[(0.0, 1.0)] * (2 * n) + [(0.0, 0.35)] * n,
            constraints={
                "type": "ineq",
                "fun": constraints,
                "jac": jacobian,
            },
            options={
                "maxiter": 1000,
                "ftol": 1.0e-11,
                "disp": False,
            },
        )

        if result.x is None or not np.all(np.isfinite(result.x)):
            return seed

        return np.clip(result.x[:2 * n].reshape(n, 2), 0.0, 1.0)

    def optimal_radii(centers):
        """Solve the exact fixed-center maximum-total-radius LP."""
        wall = np.minimum.reduce((
            centers[:, 0],
            centers[:, 1],
            1.0 - centers[:, 0],
            1.0 - centers[:, 1],
        ))
        distances = np.linalg.norm(centers[ii] - centers[jj], axis=1)

        safety = 1.0 - 2.0e-9
        result = linprog(
            c=-np.ones(n),
            A_ub=pair_matrix,
            b_ub=safety * distances,
            bounds=[
                (0.0, max(0.0, safety * float(limit)))
                for limit in wall
            ],
            method="highs",
        )

        if result.success and result.x is not None:
            return result.x

        return compute_max_radii(centers)

    # Each template has exactly 26 points.  The first three are five-row
    # layouts; the latter two have six rows and expose different wall effects.
    templates = (
        (5, 6, 5, 6, 4),
        (5, 5, 6, 5, 5),
        (5, 6, 5, 5, 5),
        (4, 5, 4, 5, 4, 4),
        (4, 5, 5, 4, 4, 4),
    )

    best_centers = None
    best_radii = None
    best_value = -np.inf

    # Two deterministic variants per topology are inexpensive and commonly
    # converge to different local contact graphs.
    for template_index, template in enumerate(templates):
        for variant in range(2):
            jitter = 0.0025 + 0.0017 * variant
            phase = 0.83 * template_index + 1.37 * variant
            seed = layered_seed(template, phase, jitter)

            centers = improve_centers(seed)
            radii = optimal_radii(centers)
            value = float(np.sum(radii))

            if value > best_value:
                best_value = value
                best_centers = centers
                best_radii = radii

    # Final common scale protects against strict geometric checkers while
    # preserving all pairwise and boundary inequalities.
    wall = np.minimum.reduce((
        best_centers[:, 0],
        best_centers[:, 1],
        1.0 - best_centers[:, 0],
        1.0 - best_centers[:, 1],
    ))
    distances = np.linalg.norm(best_centers[ii] - best_centers[jj], axis=1)
    positive = best_radii > 0.0

    scale = 1.0
    if np.any(positive):
        scale = min(scale, float(np.min(wall[positive] / best_radii[positive])))

    totals = best_radii[ii] + best_radii[jj]
    active_pairs = totals > 0.0
    if np.any(active_pairs):
        scale = min(scale, float(np.min(distances[active_pairs] / totals[active_pairs])))

    best_radii = best_radii * min(1.0, scale) * (1.0 - 1.0e-10)

    return best_centers, best_radii, float(np.sum(best_radii))


def compute_max_radii(centers):
    """
    Conservative fallback radius assignment for arbitrary centers.
    """
    n = len(centers)
    radii = np.minimum.reduce((
        centers[:, 0],
        centers[:, 1],
        1.0 - centers[:, 0],
        1.0 - centers[:, 1],
    )).astype(float)

    for i in range(n):
        for j in range(i + 1, n):
            distance = float(np.linalg.norm(centers[i] - centers[j]))
            total = radii[i] + radii[j]
            if total > distance and total > 0.0:
                factor = (1.0 - 1.0e-10) * distance / total
                radii[i] *= factor
                radii[j] *= factor

    return np.maximum(radii, 0.0)


# EVOLVE-BLOCK-END


def run_packing():
    """Run the circle packing constructor for n=26."""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """Visualize a circle packing in the unit square."""
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    for i, (center, radius) in enumerate(zip(centers, radii)):
        ax.add_patch(Circle(center, radius, alpha=0.5))
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
