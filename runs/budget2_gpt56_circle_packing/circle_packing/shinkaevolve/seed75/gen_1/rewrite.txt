"""Adaptive nonlinear constructor for 26 circles in a unit square."""
import numpy as np


def _safe_radii(centers, radii):
    """Project an approximate optimizer result onto a strictly valid packing."""
    centers = np.asarray(centers, dtype=float).copy()
    radii = np.maximum(np.asarray(radii, dtype=float).copy(), 0.0)

    # Keep centers in the square.  The subsequent border cap handles radii.
    centers = np.clip(centers, 0.0, 1.0)

    border = np.minimum.reduce(
        [centers[:, 0], centers[:, 1], 1.0 - centers[:, 0], 1.0 - centers[:, 1]]
    )
    radii = np.minimum(radii, np.maximum(border, 0.0))

    # A common scaling factor preserves all already valid border constraints and
    # makes every pair constraint valid, including small optimizer roundoff.
    n = len(radii)
    scale = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(
                centers[i, 0] - centers[j, 0],
                centers[i, 1] - centers[j, 1],
            )
            s = radii[i] + radii[j]
            if s > 0.0:
                scale = min(scale, d / s)

    # Tiny safety margin is deliberately far below evaluator precision.
    radii *= max(0.0, min(1.0, scale * (1.0 - 2.0e-10)))
    return centers, radii


def _row_seed(row_sizes, phase=0.0):
    """
    Produce a feasible low-radius staggered seed.

    Unequal row sizes are important: they let the optimizer discover large
    boundary circles instead of forcing all 26 circles into an equal grid.
    """
    centers = []
    rows = len(row_sizes)
    for row, count in enumerate(row_sizes):
        y = 0.105 + 0.79 * row / max(1, rows - 1)
        # Alternate horizontal phase, resembling a triangular lattice without
        # imposing exact symmetry on the final optimization.
        inset = 0.075 + 0.008 * ((row + int(phase * 10)) % 2)
        xs = np.linspace(inset, 1.0 - inset, count)
        for col, x in enumerate(xs):
            # Small deterministic offsets remove artificial multiple contacts.
            dx = 0.009 * np.sin(1.73 * (row + 1) * (col + 1) + phase)
            dy = 0.007 * np.cos(1.21 * (row + 2) * (col + 1) + phase)
            centers.append([np.clip(x + dx, 0.055, 0.945),
                            np.clip(y + dy, 0.055, 0.945)])
    centers = np.asarray(centers, dtype=float)
    radii = np.full(len(centers), 0.041, dtype=float)
    return centers, radii


def _fallback():
    """A dependency-free valid staggered packing."""
    centers, radii = _row_seed([5, 6, 5, 5, 5], 0.0)
    # Uniformly enlarge only as far as the actual seed permits.
    border = np.minimum.reduce(
        [centers[:, 0], centers[:, 1], 1 - centers[:, 0], 1 - centers[:, 1]]
    )
    limit = float(np.min(border))
    for i in range(26):
        for j in range(i):
            limit = min(limit, 0.5 * np.linalg.norm(centers[i] - centers[j]))
    radii[:] = 0.999999 * limit
    return centers, radii


def construct_packing():
    """
    Construct a high-sum, valid packing of 26 circles.

    Returns:
        (centers, radii, sum_of_radii)
    """
    # A selection of layered triangular-lattice seeds.  They have the same
    # circle count but different boundary defects, which is where unequal
    # radii offer most of their advantage.
    templates = [
        ([5, 6, 5, 5, 5], 0.00),
        ([5, 5, 6, 5, 5], 0.37),
        ([5, 6, 5, 6, 4], 0.71),
        ([6, 5, 5, 5, 5], 1.13),
        ([4, 6, 6, 5, 5], 1.67),
    ]

    best_centers, best_radii = _fallback()
    best_value = float(np.sum(best_radii))

    try:
        from scipy.optimize import minimize

        n = 26
        ii, jj = np.triu_indices(n, 1)
        pair_count = len(ii)

        def objective(z):
            return -float(np.sum(z[2 * n:]))

        def objective_jac(z):
            out = np.zeros(3 * n)
            out[2 * n:] = -1.0
            return out

        def inequalities(z):
            x = z[:n]
            y = z[n:2 * n]
            r = z[2 * n:]

            # Four containment inequalities per circle.
            walls = np.concatenate((x - r, 1.0 - x - r,
                                    y - r, 1.0 - y - r))

            # Squared-distance form is smooth away from coincident centers.
            dx = x[ii] - x[jj]
            dy = y[ii] - y[jj]
            pairs = dx * dx + dy * dy - (r[ii] + r[jj]) ** 2
            return np.concatenate((walls, pairs))

        def inequalities_jac(z):
            x = z[:n]
            y = z[n:2 * n]
            r = z[2 * n:]
            jac = np.zeros((4 * n + pair_count, 3 * n))

            k = np.arange(n)
            jac[k, k] = 1.0
            jac[k, 2 * n + k] = -1.0

            jac[n + k, k] = -1.0
            jac[n + k, 2 * n + k] = -1.0

            jac[2 * n + k, n + k] = 1.0
            jac[2 * n + k, 2 * n + k] = -1.0

            jac[3 * n + k, n + k] = -1.0
            jac[3 * n + k, 2 * n + k] = -1.0

            row = 4 * n + np.arange(pair_count)
            dx = x[ii] - x[jj]
            dy = y[ii] - y[jj]
            sr = r[ii] + r[jj]

            jac[row, ii] = 2.0 * dx
            jac[row, jj] = -2.0 * dx
            jac[row, n + ii] = 2.0 * dy
            jac[row, n + jj] = -2.0 * dy
            jac[row, 2 * n + ii] = -2.0 * sr
            jac[row, 2 * n + jj] = -2.0 * sr
            return jac

        constraints = {
            "type": "ineq",
            "fun": inequalities,
            "jac": inequalities_jac,
        }
        bounds = [(0.0, 1.0)] * (2 * n) + [(1.0e-7, 0.5)] * n

        for template, phase in templates:
            centers, radii = _row_seed(template, phase)
            z0 = np.concatenate((centers[:, 0], centers[:, 1], radii))

            result = minimize(
                objective,
                z0,
                jac=objective_jac,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={
                    "maxiter": 1100,
                    "ftol": 2.0e-11,
                    "disp": False,
                },
            )

            candidate = result.x if result.x is not None else z0
            candidate_centers = np.column_stack((candidate[:n], candidate[n:2 * n]))
            candidate_radii = candidate[2 * n:]
            candidate_centers, candidate_radii = _safe_radii(
                candidate_centers, candidate_radii
            )
            value = float(np.sum(candidate_radii))

            if value > best_value:
                best_centers = candidate_centers
                best_radii = candidate_radii
                best_value = value

    except Exception:
        # The explicit staggered fallback remains valid on installations where
        # SciPy is unavailable.
        pass

    return best_centers, best_radii, float(np.sum(best_radii))


def compute_max_radii(centers):
    """
    Retained public helper: conservative radii based on the supplied centers.
    """
    n = centers.shape[0]
    radii = np.minimum.reduce(
        [centers[:, 0], centers[:, 1], 1.0 - centers[:, 0], 1.0 - centers[:, 1]]
    ).astype(float)

    # Repeated global projection is conservative and avoids the order dependence
    # of sequential pair scaling.
    for _ in range(3):
        factor = 1.0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                s = radii[i] + radii[j]
                if s > d and s > 0:
                    factor = min(factor, d / s)
        radii *= factor
    return radii


# EVOLVE-BLOCK-END


def run_packing():
    """Run the circle packing constructor for n=26."""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    for i, (center, radius) in enumerate(zip(centers, radii)):
        ax.add_patch(Circle(center, radius, alpha=0.5))
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
