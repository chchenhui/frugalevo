<DESCRIPTION>
Replace the clipped concentric-ring layout with a six-layer staggered (hexagonal) seed and optimize both centers and unequal radii directly. The staggered seed captures the efficient local density of the triangular lattice while allowing the optimizer to exploit square-boundary effects and the missing-circle asymmetry.

The nonlinear program maximizes the total radius subject to all boundary and pairwise non-overlap inequalities. Several deterministic, lightly perturbed starts reduce the chance of retaining an overly symmetric local optimum. A final global radius certification scale makes the returned solution robust to optimizer tolerances, guaranteeing that every circle remains inside the unit square and all pairwise distances satisfy non-overlap constraints. If SciPy is unavailable, the valid hexagonal seed is still returned.
</DESCRIPTION>

    n = 26

    # A truncated triangular lattice is a substantially better seed than
    # concentric rings in a square.  Five alternating rows would contain 27
    # circles; the short final row supplies the useful asymmetry for n = 26.
    row_lengths = (5, 4, 5, 4, 5, 3)
    seed_centers = []
    vertical = np.sqrt(3.0) * 0.09
    for row, count in enumerate(row_lengths):
        y = 0.09 + row * vertical
        if count == 5:
            xs = np.linspace(0.10, 0.90, count)
        elif count == 4:
            xs = np.linspace(0.20, 0.80, count)
        else:
            xs = np.linspace(0.30, 0.70, count)
        seed_centers.extend((x, y) for x in xs)
    seed_centers = np.asarray(seed_centers, dtype=float)

    def certify(centers, radii):
        """Uniformly shrink radii by the exact worst constraint ratio."""
        centers = np.clip(np.asarray(centers, dtype=float), 0.0, 1.0)
        radii = np.maximum(np.asarray(radii, dtype=float), 1.0e-12)
        ratios = [
            np.min(centers[:, 0] / radii),
            np.min((1.0 - centers[:, 0]) / radii),
            np.min(centers[:, 1] / radii),
            np.min((1.0 - centers[:, 1]) / radii),
        ]
        for i in range(n):
            for j in range(i):
                ratios.append(
                    np.linalg.norm(centers[i] - centers[j]) / (radii[i] + radii[j])
                )
        radii *= min(1.0, min(ratios)) * (1.0 - 1.0e-10)
        return centers, radii

    # Keep a good fully feasible answer even on installations without SciPy.
    best_centers, best_radii = certify(seed_centers, np.full(n, 0.09))
    best_value = float(np.sum(best_radii))

    try:
        from scipy.optimize import minimize

        pairs_i, pairs_j = np.triu_indices(n, 1)

        def constraints(z):
            c = z[:2 * n].reshape(n, 2)
            r = z[2 * n:]
            border = np.concatenate((
                c[:, 0] - r, 1.0 - c[:, 0] - r,
                c[:, 1] - r, 1.0 - c[:, 1] - r,
            ))
            delta = c[pairs_i] - c[pairs_j]
            separation = np.einsum("ij,ij->i", delta, delta) - (r[pairs_i] + r[pairs_j]) ** 2
            return np.concatenate((border, separation))

        # The regular start and three reproducible symmetry-breaking starts
        # explore distinct boundary assignments without an expensive global
        # search.
        rng = np.random.default_rng(260319)
        starts = [(seed_centers, np.full(n, 0.09))]
        for _ in range(3):
            starts.append((
                seed_centers + rng.normal(0.0, 0.009, size=(n, 2)),
                np.full(n, 0.070),
            ))

        bounds = [(0.0, 1.0)] * (2 * n) + [(1.0e-7, 0.5)] * n
        for initial_centers, initial_radii in starts:
            z0 = np.concatenate((initial_centers.ravel(), initial_radii))
            result = minimize(
                lambda z: -np.sum(z[2 * n:]),
                z0,
                method="SLSQP",
                bounds=bounds,
                constraints={"type": "ineq", "fun": constraints},
                options={"maxiter": 1400, "ftol": 1.0e-11, "disp": False},
            )
            candidate_centers, candidate_radii = certify(
                result.x[:2 * n].reshape(n, 2), result.x[2 * n:]
            )
            value = float(np.sum(candidate_radii))
            if value > best_value:
                best_centers, best_radii, best_value = (
                    candidate_centers, candidate_radii, value
                )
    except ImportError:
        pass

    return best_centers, best_radii, best_value
