<DESCRIPTION>
The previous constructor used 24 broadly similar SLSQP runs with 700 iterations and retained only the single best LP-certified result. This version uses a two-stage strategy:

1. **Exploration stage:** 20 deterministic layered and triangular-lattice seeds, with asymmetric perturbations, are optimized for 600 iterations.
2. **LP certification and structural selection:** each candidate is reallocated using an exact fixed-center linear program. Candidates are ranked by LP sum, raw nonlinear objective, contact-count diagnostics, and distinct near-active contact signatures.
3. **Polishing stage:** the best structurally different candidates are restarted from their LP-optimal radii and given 2400 additional SLSQP iterations. This focuses expensive optimization on promising contact graphs instead of spending all iterations on near-duplicate row arrangements.

The main changed parameters are: more diverse row templates, 20 initial starts instead of 24 nearly related starts, 600 initial iterations, up to 8 structurally distinct refinement candidates, 2400 polishing iterations, tighter LP-start scaling, and a smaller final safety margin.

# EVOLVE-BLOCK-START
"""Layered LP-polished constructor for 26 circles in a unit square."""
import numpy as np


def construct_packing():
    """
    Construct a valid unequal-radius packing of 26 circles in the unit square.

    The optimization is deliberately split into a broad contact-graph search
    followed by long restarts from LP-certified radius allocations.
    """
    n = 26
    ii, jj = np.triu_indices(n, 1)
    pair_count = len(ii)

    def row_seed(row_sizes, phase=0.0, jitter=0.0):
        """Build a layered triangular-lattice-like seed with mild asymmetry."""
        points = []
        rows = len(row_sizes)
        for row, count in enumerate(row_sizes):
            y = 0.085 + 0.830 * row / max(1, rows - 1)

            # Alternate insets emulate triangular layers.  Different row
            # counts place the unavoidable boundary defects in different rows.
            base_inset = {
                3: 0.275,
                4: 0.175,
                5: 0.105,
                6: 0.070,
                7: 0.052,
            }.get(count, 0.10)

            stagger = 0.010 * ((row + int(3.0 * phase)) % 2)
            xs = np.linspace(base_inset + stagger,
                             1.0 - base_inset + stagger, count)
            for col, x in enumerate(xs):
                dx = jitter * np.sin(1.713 * (row + 1) * (col + 1) + phase)
                dy = jitter * np.cos(1.271 * (row + 2) * (col + 1) + phase)
                points.append([
                    np.clip(x + dx, 0.025, 0.975),
                    np.clip(y + dy, 0.025, 0.975),
                ])
        return np.asarray(points, dtype=float)

    # A conservative dependency-free fallback.  The rows are separated enough
    # for radius .04 even after the small deterministic perturbation.
    fallback_centers = row_seed((5, 6, 5, 6, 4), 0.0, 0.0)
    fallback_radii = np.full(n, 0.040, dtype=float)

    try:
        from scipy.optimize import linprog, minimize
    except Exception:
        return fallback_centers, fallback_radii, float(np.sum(fallback_radii))

    pair_matrix = np.zeros((pair_count, n), dtype=float)
    q = np.arange(pair_count)
    pair_matrix[q, ii] = 1.0
    pair_matrix[q, jj] = 1.0

    def optimal_radii(centers):
        """Solve the exact radius LP at fixed center locations."""
        centers = np.asarray(centers, dtype=float)
        dx = centers[ii, 0] - centers[jj, 0]
        dy = centers[ii, 1] - centers[jj, 1]
        distances = np.hypot(dx, dy)
        walls = np.minimum.reduce((
            centers[:, 0], centers[:, 1],
            1.0 - centers[:, 0], 1.0 - centers[:, 1],
        ))
        walls = np.maximum(walls, 0.0)

        result = linprog(
            -np.ones(n),
            A_ub=pair_matrix,
            b_ub=distances,
            bounds=[(0.0, float(w)) for w in walls],
            method="highs",
        )
        if result.success and result.x is not None:
            return np.maximum(result.x, 0.0)
        return np.zeros(n, dtype=float)

    def inequalities(z):
        c = z[:2 * n].reshape(n, 2)
        r = z[2 * n:]

        walls = np.column_stack((
            c[:, 0] - r,
            c[:, 1] - r,
            1.0 - c[:, 0] - r,
            1.0 - c[:, 1] - r,
        )).ravel()

        dx = c[ii, 0] - c[jj, 0]
        dy = c[ii, 1] - c[jj, 1]
        pair_gaps = dx * dx + dy * dy - (r[ii] + r[jj]) ** 2
        return np.concatenate((walls, pair_gaps))

    def inequalities_jac(z):
        c = z[:2 * n].reshape(n, 2)
        r = z[2 * n:]
        jac = np.zeros((4 * n + pair_count, 3 * n), dtype=float)

        k = np.arange(n)
        # Wall ordering agrees with the flattened row-wise wall array.
        wall_rows = 4 * k
        jac[wall_rows, 2 * k] = 1.0
        jac[wall_rows, 2 * n + k] = -1.0

        jac[wall_rows + 1, 2 * k + 1] = 1.0
        jac[wall_rows + 1, 2 * n + k] = -1.0

        jac[wall_rows + 2, 2 * k] = -1.0
        jac[wall_rows + 2, 2 * n + k] = -1.0

        jac[wall_rows + 3, 2 * k + 1] = -1.0
        jac[wall_rows + 3, 2 * n + k] = -1.0

        rows = 4 * n + np.arange(pair_count)
        dx = c[ii, 0] - c[jj, 0]
        dy = c[ii, 1] - c[jj, 1]
        sr = r[ii] + r[jj]

        jac[rows, 2 * ii] = 2.0 * dx
        jac[rows, 2 * ii + 1] = 2.0 * dy
        jac[rows, 2 * jj] = -2.0 * dx
        jac[rows, 2 * jj + 1] = -2.0 * dy
        jac[rows, 2 * n + ii] = -2.0 * sr
        jac[rows, 2 * n + jj] = -2.0 * sr
        return jac

    def objective(z):
        return -float(np.sum(z[2 * n:]))

    def objective_jac(z):
        ans = np.zeros(3 * n, dtype=float)
        ans[2 * n:] = -1.0
        return ans

    constraints = {
        "type": "ineq",
        "fun": inequalities,
        "jac": inequalities_jac,
    }
    bounds = (
        [(1.0e-8, 1.0 - 1.0e-8)] * (2 * n)
        + [(1.0e-9, 0.5)] * n
    )

    def contact_signature(centers, radii):
        """
        A compact active-set signature used to avoid polishing many copies of
        the same local contact graph.
        """
        dx = centers[ii, 0] - centers[jj, 0]
        dy = centers[ii, 1] - centers[jj, 1]
        slacks = np.hypot(dx, dy) - radii[ii] - radii[jj]
        active_pairs = tuple(np.flatnonzero(slacks < 1.8e-4)[:90])

        wall = np.minimum.reduce((
            centers[:, 0] - radii,
            centers[:, 1] - radii,
            1.0 - centers[:, 0] - radii,
            1.0 - centers[:, 1] - radii,
        ))
        active_walls = tuple(np.flatnonzero(wall < 1.8e-4))
        return active_pairs, active_walls

    # All templates contain exactly 26 circles.  Their unequal layers move
    # boundary defects and make distinct local optima substantially more likely.
    templates = [
        ((5, 6, 5, 6, 4), 0.00),
        ((4, 6, 5, 6, 5), 0.31),
        ((5, 5, 6, 5, 5), 0.62),
        ((5, 6, 5, 5, 5), 0.93),
        ((5, 5, 5, 6, 5), 1.24),
        ((6, 5, 5, 5, 5), 1.55),
        ((5, 4, 6, 5, 6), 1.86),
        ((6, 5, 4, 6, 5), 2.17),
        ((5, 6, 5, 4, 6), 2.48),
        ((6, 5, 6, 5, 4), 2.79),
        ((4, 6, 6, 5, 5), 3.10),
        ((5, 4, 6, 6, 5), 3.41),
        ((5, 5, 4, 6, 6), 3.72),
        ((6, 6, 5, 5, 4), 4.03),
        ((5, 4, 5, 6, 6), 4.34),
        ((6, 4, 5, 5, 6), 4.65),
    ]

    starts = []
    for shape, phase in templates:
        starts.append(row_seed(shape, phase, 0.009))

    # Two extra perturbed starts around the historically strong 5/6/5/6/4
    # layout provide non-reflection-symmetric alternatives.
    rng = np.random.default_rng(26032617)
    base = row_seed((5, 6, 5, 6, 4), 0.18, 0.0)
    for amount in (0.011, 0.017, 0.023, 0.029):
        c = base + rng.normal(0.0, amount, size=(n, 2))
        starts.append(np.clip(c, 0.020, 0.980))

    candidates = []
    for start in starts:
        r0 = optimal_radii(start)
        z0 = np.concatenate((start.ravel(), 0.992 * r0))

        result = minimize(
            objective,
            z0,
            jac=objective_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 600, "ftol": 4.0e-12, "disp": False},
        )

        z = result.x if result.x is not None and np.all(np.isfinite(result.x)) else z0
        centers = np.clip(z[:2 * n].reshape(n, 2), 0.0, 1.0)
        radii = optimal_radii(centers)
        lp_value = float(np.sum(radii))
        raw_value = float(np.sum(np.maximum(z[2 * n:], 0.0)))

        dx = centers[ii, 0] - centers[jj, 0]
        dy = centers[ii, 1] - centers[jj, 1]
        min_pair_slack = float(np.min(np.hypot(dx, dy) - radii[ii] - radii[jj]))
        signature = contact_signature(centers, radii)

        candidates.append({
            "centers": centers,
            "radii": radii,
            "lp_value": lp_value,
            "raw_value": raw_value,
            "min_slack": min_pair_slack,
            "signature": signature,
        })

    # Preserve top candidates from distinct active sets, then fill remaining
    # slots by score.  LP score is primary; raw nonlinear score breaks ties.
    candidates.sort(
        key=lambda item: (
            item["lp_value"],
            item["raw_value"],
            item["min_slack"],
        ),
        reverse=True,
    )

    selected = []
    signatures = set()
    for item in candidates:
        if item["signature"] not in signatures:
            selected.append(item)
            signatures.add(item["signature"])
        if len(selected) >= 8:
            break

    if len(selected) < 8:
        for item in candidates:
            if item not in selected:
                selected.append(item)
            if len(selected) >= 8:
                break

    best_centers = fallback_centers.copy()
    best_radii = fallback_radii.copy()
    best_value = float(np.sum(best_radii))

    # Long restarts are much more effective than increasing every exploratory
    # run: each begins from an LP-optimal allocation for its discovered graph.
    for item in selected:
        z0 = np.concatenate((
            item["centers"].ravel(),
            0.9992 * item["radii"],
        ))
        result = minimize(
            objective,
            z0,
            jac=objective_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 2400, "ftol": 8.0e-13, "disp": False},
        )

        z = result.x if result.x is not None and np.all(np.isfinite(result.x)) else z0
        centers = np.clip(z[:2 * n].reshape(n, 2), 0.0, 1.0)
        radii = optimal_radii(centers)
        value = float(np.sum(radii))

        if value > best_value:
            best_centers = centers
            best_radii = radii
            best_value = value

    # Final common scale eliminates LP/SLSQP and floating-point roundoff while
    # preserving every non-overlap and containment inequality.
    scale = 1.0
    for i in range(n):
        if best_radii[i] > 0.0:
            scale = min(
                scale,
                best_centers[i, 0] / best_radii[i],
                best_centers[i, 1] / best_radii[i],
                (1.0 - best_centers[i, 0]) / best_radii[i],
                (1.0 - best_centers[i, 1]) / best_radii[i],
            )

    dx = best_centers[ii, 0] - best_centers[jj, 0]
    dy = best_centers[ii, 1] - best_centers[jj, 1]
    distance = np.hypot(dx, dy)
    total = best_radii[ii] + best_radii[jj]
    mask = total > 0.0
    if np.any(mask):
        scale = min(scale, float(np.min(distance[mask] / total[mask])))

    best_radii *= max(0.0, min(1.0, scale * (1.0 - 2.0e-10)))
    return best_centers, best_radii, float(np.sum(best_radii))


def compute_max_radii(centers):
    """
    Compute conservative valid radii for supplied center positions.

    This retained helper keeps the public interface of the original program.
    """
    centers = np.asarray(centers, dtype=float)
    n = len(centers)
    radii = np.minimum.reduce((
        centers[:, 0], centers[:, 1],
        1.0 - centers[:, 0], 1.0 - centers[:, 1],
    ))
    radii = np.maximum(radii, 0.0)

    # Common scaling is order-independent and always preserves validity.
    factor = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = float(np.linalg.norm(centers[i] - centers[j]))
            s = radii[i] + radii[j]
            if s > 0.0:
                factor = min(factor, d / s)
    return radii * max(0.0, min(1.0, factor * (1.0 - 1.0e-10)))


# EVOLVE-BLOCK-END


def run_packing():
    """Run the circle packing constructor for n=26."""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """Visualize a packing in the unit square."""
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    for i, (center, radius) in enumerate(zip(centers, radii)):
        ax.add_patch(Circle(center, radius, alpha=0.5))
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
