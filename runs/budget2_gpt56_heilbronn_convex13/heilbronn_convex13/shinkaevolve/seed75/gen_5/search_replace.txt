deterministic_triangle_annealing</NAME>

<DESCRIPTION>
Replace the random square sample with a deterministic multi-start simulated-annealing construction inside a fixed equilateral triangular hull. Fixing three hull vertices gives a stable convex region, while all remaining points are represented by barycentric samples and are therefore always inside that hull.

The optimization evaluates all 286 triangle areas simultaneously. Its annealing objective is a smooth soft-minimum of normalized triangle areas, which gives useful progress even when the identity of the currently smallest triangle changes. Several deterministic restarts, decreasing perturbation radii, and a final exact-minimum selection substantially improve on an unstructured random configuration while retaining reproducibility and avoiding external dependencies.
</DESCRIPTION>

    # Use a fixed equilateral hull.  Its area is sqrt(3)/4, so all
    # determinant areas are divided by this value during optimization.
    n = 13
    hull = np.array(
        [[0.0, 0.0], [1.0, 0.0], [0.5, np.sqrt(3.0) * 0.5]],
        dtype=float,
    )
    hull_area = np.sqrt(3.0) * 0.25
    triples = np.asarray(
        [(i, j, k) for i in range(n - 2) for j in range(i + 1, n - 1)
         for k in range(j + 1, n)],
        dtype=np.intp,
    )
    rng = np.random.default_rng(13031957)

    def areas(p: np.ndarray) -> np.ndarray:
        a = p[triples[:, 0]]
        b = p[triples[:, 1]]
        c = p[triples[:, 2]]
        return np.abs(
            (b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1])
            - (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0])
        ) / (2.0 * hull_area)

    def random_in_hull(count: int) -> np.ndarray:
        # Folding the unit square produces uniform barycentric samples.
        uv = rng.random((count, 2))
        fold = uv.sum(axis=1) > 1.0
        uv[fold] = 1.0 - uv[fold]
        return hull[0] + uv[:, :1] * (hull[1] - hull[0]) + uv[:, 1:] * (
            hull[2] - hull[0]
        )

    best_points = None
    best_minimum = -np.inf

    # Multiple fixed-seed starts are inexpensive for only 286 constraints,
    # and are much less susceptible to a poor local arrangement.
    for restart in range(6):
        points = np.vstack((hull, random_in_hull(n - 3)))
        current = areas(points)

        for step in range(7000):
            fraction = step / 6999.0
            # Early large moves arrange the global pattern; late moves tune
            # active near-degenerate triples.
            move_scale = 0.085 * (1.0 - fraction) + 0.0025
            temperature = 0.018 * (1.0 - fraction) + 0.0007
            index = 3 + rng.integers(n - 3)
            proposal = points.copy()
            candidate = proposal[index] + rng.normal(scale=move_scale, size=2)

            # Barycentric containment test in the fixed equilateral hull.
            v = candidate - hull[0]
            bary2 = 2.0 * v[1] / np.sqrt(3.0)
            bary1 = v[0] - 0.5 * bary2
            if bary1 < 0.0 or bary2 < 0.0 or bary1 + bary2 > 1.0:
                continue

            proposal[index] = candidate
            candidate_areas = areas(proposal)

            # Smooth minimum retains information from several nearly active
            # constraints rather than chasing one triangle at a time.
            def soft_min(x: np.ndarray) -> float:
                z = -x / temperature
                zmax = np.max(z)
                return -(zmax + np.log(np.exp(z - zmax).sum())) * temperature

            gain = soft_min(candidate_areas) - soft_min(current)
            if gain >= 0.0 or rng.random() < np.exp(gain / temperature):
                points = proposal
                current = candidate_areas

        exact_minimum = float(current.min())
        if exact_minimum > best_minimum:
            best_minimum = exact_minimum
            best_points = points.copy()

    return best_points
