<DESCRIPTION>
Replace repeated selection of the single currently worst triangle with probabilistic sampling from a low-area active set. Maximin optima are governed by several nearly tight triangle constraints; always moving only the unique worst triangle tends to oscillate between a few constraints. Sampling low-area triangles with inverse-area weights retains strong pressure on bottlenecks while allowing the search to resolve competing active constraints. The discovery stage uses a narrow active set and the longer refinement stage uses a broader one.
</DESCRIPTION>

    def evaluate(p: np.ndarray) -> tuple[float, int, np.ndarray]:
        """Return the primary score, worst triangle, and normalized areas."""
        a = p[triangles[:, 0]]
        b = p[triangles[:, 1]]
        c = p[triangles[:, 2]]
        double_areas = np.abs(
            (b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1])
            - (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0])
        )
        worst = int(np.argmin(double_areas))
        area = hull_area(p)
        if area <= 1.0e-14:
            return -np.inf, worst, double_areas
        normalized = 0.5 * double_areas / area
        return float(normalized[worst]), worst, normalized

    def anneal(points: np.ndarray, score: float, worst: int,
               normalized: np.ndarray, iterations: int, initial_step: float,
               initial_temperature: float,
               active_count: int) -> tuple[np.ndarray, float, int, np.ndarray]:
        """
        Metropolis refinement with sampled near-active constraints.

        A maximin configuration generally has several equal or nearly equal
        bottleneck triangles.  Sampling among them avoids repeatedly undoing
        the previous move made for a single argmin constraint.
        """
        for iteration in range(iterations):
            fraction = iteration / max(1, iterations - 1)
            step = initial_step * (1.0 - fraction) ** 1.7 + 0.0020
            temperature = (initial_temperature * (1.0 - fraction) ** 2
                           + 0.000012)

            candidate = points.copy()
            if rng.random() < 0.82:
                active = np.argsort(normalized)[:active_count]
                weights = 1.0 / (normalized[active] + 1.0e-12)
                selected = int(active[rng.choice(len(active), p=weights / weights.sum())])
                moved = int(triangles[selected, rng.integers(0, 3)])
            else:
                moved = int(rng.integers(n))
            candidate[moved] += rng.normal(0.0, step, 2)

            candidate_score, candidate_worst, candidate_normalized = evaluate(candidate)
            delta = candidate_score - score
            if (delta >= 0.0
                    or rng.random() < np.exp(max(-700.0, delta / temperature))):
                points = candidate
                score = candidate_score
                worst = candidate_worst
                normalized = candidate_normalized
        return points, score, worst, normalized
