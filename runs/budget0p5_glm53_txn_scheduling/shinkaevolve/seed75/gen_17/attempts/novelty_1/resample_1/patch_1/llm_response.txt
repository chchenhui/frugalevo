<DESCRIPTION>
Crossover combines the current program's strengths (full greedy construction with cost caching, 2-opt swap local search) with the inspiration program's stochastic multi-restart idea. The new version: (1) builds greedy solutions from each starting transaction, (2) improves each with cached 2-opt local search, (3) then applies iterated perturbation — randomly shuffling a small window of the best sequence and re-running local search — to escape local minima like the sampled greedy did, but more effectively since it starts from a good solution. Cost caching avoids recomputing makespans for repeated sequences.

import random

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Get optimal schedule using greedy construction + cached 2-opt local
    search + perturbation-based iterated local search.

    Returns:
        Tuple of (lowest makespan, corresponding schedule)
    """
    _cost_cache = {}

    def eval_cost(seq):
        key = tuple(seq)
        c = _cost_cache.get(key)
        if c is None:
            c = workload.get_opt_seq_cost(seq)
            _cost_cache[key] = c
        return c

    def greedy_from(start_txn):
        # Greedy construction: at each step, pick the transaction whose
        # addition to the current prefix yields the lowest makespan.
        txn_seq = [start_txn]
        remaining_txns = [x for x in range(0, workload.num_txns)]
        remaining_txns.remove(start_txn)

        while remaining_txns:
            min_cost = float("inf")
            min_txn = -1
            for t in remaining_txns:
                test_seq = txn_seq + [t]
                cost = eval_cost(test_seq)
                if cost < min_cost:
                    min_cost = cost
                    min_txn = t
            txn_seq.append(min_txn)
            remaining_txns.remove(min_txn)

        assert len(set(txn_seq)) == workload.num_txns
        return eval_cost(txn_seq), txn_seq

    def local_search(seq, cost, max_iters=30):
        # 2-opt style swap local search: try swapping pairs of transactions
        # (with emphasis on nearby positions) and accept the best swap that
        # lowers makespan. Repeat until no improvement.
        n = len(seq)
        seq = seq[:]
        for _ in range(max_iters):
            best_pair = None
            best_pair_cost = cost
            for i in range(n - 1):
                for span in (1, 2, 3, 5, 8, 13):
                    j = i + span
                    if j >= n:
                        break
                    trial = seq[:]
                    trial[i], trial[j] = trial[j], trial[i]
                    c = eval_cost(trial)
                    if c < best_pair_cost:
                        best_pair_cost = c
                        best_pair = (i, j)
            if best_pair is None:
                break
            i, j = best_pair
            seq[i], seq[j] = seq[j], seq[i]
            cost = best_pair_cost
        return cost, seq

    def perturb(seq):
        # Randomly shuffle a small window of the sequence to escape
        # local minima, then local search re-optimizes.
        seq = seq[:]
        n = len(seq)
        window = random.randint(2, max(2, min(6, n)))
        start = random.randint(0, n - window)
        chunk = seq[start:start + window]
        random.shuffle(chunk)
        seq[start:start + window] = chunk
        return seq

    best_cost = float("inf")
    best_seq = None

    # Phase 1: greedy construction from each starting transaction,
    # each improved with local search.
    for start_txn in range(workload.num_txns):
        cost, seq = greedy_from(start_txn)
        cost, seq = local_search(seq, cost)
        if cost < best_cost:
            best_cost = cost
            best_seq = seq

    # Phase 2: iterated local search — perturb the best solution and
    # re-optimize, keeping improvements.
    num_perturbs = max(20, num_seqs * 10)
    for _ in range(num_perturbs):
        cand = perturb(best_seq)
        cost = eval_cost(cand)
        cost, cand = local_search(cand, cost)
        if cost < best_cost:
            best_cost = cost
            best_seq = cand

    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]


if __name__ == "__main__":
    makespan, schedule = get_random_costs()
    print(f"Makespan: {makespan}")
