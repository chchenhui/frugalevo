import random
import time

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Get optimal schedule using greedy construction + simulated annealing
    with insertion moves and basin-hopping restarts.
    """
    n = workload.num_txns
    cost_fn = workload.get_opt_seq_cost

    def greedy_construct(sample_rate):
        start_txn = random.randint(0, n - 1)
        seq = [start_txn]
        remaining = [x for x in range(n) if x != start_txn]
        while remaining:
            if random.random() > sample_rate:
                # explore: random pick
                idx = random.randint(0, len(remaining) - 1)
                seq.append(remaining.pop(idx))
                continue
            num_samples = min(10, len(remaining))
            min_cost = float('inf')
            min_txn = None
            pool = random.sample(remaining, num_samples)
            for t in pool:
                c = cost_fn(seq + [t])
                if c < min_cost:
                    min_cost = c
                    min_txn = t
            seq.append(min_txn)
            remaining.remove(min_txn)
        return cost_fn(seq), seq

    def insertion_move(seq):
        cand = seq[:]
        i = random.randint(0, len(cand) - 1)
        t = cand.pop(i)
        j = random.randint(0, len(cand))
        cand.insert(j, t)
        return cand

    def swap_move(seq):
        cand = seq[:]
        i = random.randint(0, len(cand) - 1)
        j = random.randint(0, len(cand) - 1)
        cand[i], cand[j] = cand[j], cand[i]
        return cand

    def anneal(seq, cost, temp, deadline, max_iter=2000):
        cur_seq, cur_cost = seq, cost
        best_seq, best_cost = seq[:], cost
        it = 0
        while it < max_iter and time.time() < deadline:
            it += 1
            cand = insertion_move(cur_seq) if random.random() < 0.7 else swap_move(cur_seq)
            c = cost_fn(cand)
            delta = c - cur_cost
            if delta < 0 or random.random() < pow(2.718281828, -delta / max(temp, 1e-9)):
                cur_seq, cur_cost = cand, c
                if c < best_cost:
                    best_cost = c
                    best_seq = cand[:]
            temp *= 0.995
            if temp < 0.5:
                temp = 0.5
        return best_cost, best_seq

    def descend(seq, cost, deadline, max_rounds=8):
        # first-improvement insertion descent
        best_seq, best_cost = seq[:], cost
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            improved = False
            for i in range(len(best_seq)):
                t = best_seq[i]
                rest = best_seq[:i] + best_seq[i+1:]
                best_j_cost = None
                best_j_seq = None
                for j in range(len(rest) + 1):
                    cand = rest[:j] + [t] + rest[j:]
                    c = cost_fn(cand)
                    if best_j_cost is None or c < best_j_cost:
                        best_j_cost = c
                        best_j_seq = cand
                if best_j_cost < best_cost:
                    best_cost = best_j_cost
                    best_seq = best_j_seq
                    improved = True
            if not improved:
                break
        return best_cost, best_seq

    deadline = time.time() + 100

    best_cost = float('inf')
    best_seq = None
    restart = 0
    while time.time() < deadline:
        restart += 1
        # decay exploration rate: early restarts greedy, later more random
        rate = max(0.7, 0.95 - 0.03 * restart)
        cost, seq = greedy_construct(rate)
        # quick descent
        cost, seq = descend(seq, cost, deadline, max_rounds=3)
        # anneal from there
        cost, seq = anneal(seq, cost, 5.0, deadline, max_iter=600)
        # final descent polish
        cost, seq = descend(seq, cost, deadline, max_rounds=3)
        if cost < best_cost:
            best_cost = cost
            best_seq = seq
        # basin hopping: occasionally perturb best and continue
        if restart % 3 == 0 and best_seq is not None:
            pert = best_seq[:]
            for _ in range(3):
                pert = insertion_move(pert)
            c = cost_fn(pert)
            c, s = descend(pert, c, deadline, max_rounds=2)
            if c < best_cost:
                best_cost = c
                best_seq = s
        if restart >= 40:
            break

    if best_seq is None:
        best_seq = list(range(n))
        best_cost = cost_fn(best_seq)
    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]
