import random
import time

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Get optimal schedule using multi-restart greedy construction with
    true incremental cost evaluation, followed by pairwise-swap local search.

    Returns:
        Tuple of (lowest makespan, corresponding schedule)
    """
    n = workload.num_txns
    time_budget = 8.0  # seconds per workload
    start_time = time.time()

    def greedy_from(start_txn):
        txn_seq = [start_txn]
        remaining = [x for x in range(n) if x != start_txn]
        while remaining:
            best_cost = None
            best_txns = []
            for t in remaining:
                test_seq = txn_seq + [t]
                cost = workload.get_opt_seq_cost(test_seq)
                if best_cost is None or cost < best_cost:
                    best_cost = cost
                    best_txns = [t]
                elif cost == best_cost:
                    best_txns.append(t)
            pick = random.choice(best_txns)
            txn_seq.append(pick)
            remaining.remove(pick)
            if time.time() - start_time > time_budget:
                # finish greedily-quickly with random order to stay valid
                txn_seq.extend(remaining)
                remaining = []
        return txn_seq

    def local_search(seq, deadline):
        cur_cost = workload.get_opt_seq_cost(seq)
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # try all pairwise swaps
            best_delta = 0
            best_move = None
            for i in range(n - 1):
                for j in range(i + 1, n):
                    cand = seq.copy()
                    cand[i], cand[j] = cand[j], cand[i]
                    c = workload.get_opt_seq_cost(cand)
                    if c < cur_cost + best_delta:
                        best_delta = c - cur_cost
                        best_move = (i, j, c)
                    if time.time() >= deadline:
                        break
                if time.time() >= deadline:
                    break
            if best_move:
                i, j, c = best_move
                seq[i], seq[j] = seq[j], seq[i]
                cur_cost = c
                improved = True
        return seq, cur_cost

    best_seq = None
    best_cost = None
    deadline = start_time + time_budget

    # multi-restart greedy with different starting transactions
    restart_idx = 0
    while time.time() < deadline:
        start_txn = restart_idx % n
        seq = greedy_from(start_txn)
        cost = workload.get_opt_seq_cost(seq)
        if best_cost is None or cost < best_cost:
            best_cost = cost
            best_seq = seq
        restart_idx += 1
        if restart_idx >= n:
            break

    # local search polish
    if best_seq is not None:
        remaining_time = deadline - time.time()
        if remaining_time > 0.1:
            best_seq, best_cost = local_search(best_seq, deadline)

    if best_seq is None:
        best_seq = list(range(n))
        best_cost = workload.get_opt_seq_cost(best_seq)

    assert len(set(best_seq)) == n
    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100

    makespan1, schedule1 = get_best_schedule(workload, 10)

    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]
