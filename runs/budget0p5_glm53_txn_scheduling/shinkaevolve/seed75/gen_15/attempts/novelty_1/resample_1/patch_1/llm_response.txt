annealed_insertion_search

<DESCRIPTION>
The current program alternates full pairwise-swap scans and greedy reinsertion with strict-improvement acceptance, which stalls in local optima once no move lowers makespan. Key parameters: number of greedy restarts (20), sample size (6), sample rate (0.95), and the acceptance rule (strict descent). My new configuration: (1) focus the neighborhood on insertion moves only — cheaper per evaluation than swaps, so far more candidate evaluations per second; (2) add simulated-annealing acceptance: accept a candidate insertion if it improves, otherwise accept with probability exp(-delta/(temp)) where temp decays geometrically over rounds — this escapes local optima; (3) keep best-so-far separately so annealing never loses the incumbent; (4) run many annealing restarts from different greedy seeds within a time budget rather than a fixed restart count, using more restarts (30) and a lower initial temperature so the search intensifies. Since combined score is driven by total makespan, spending the whole budget on annealed insertion search over the best greedy seeds should push below 292.

import random
import time

from txn_simulator import Workload

def get_best_schedule(workload, num_seqs):
    """
    Get optimal schedule using greedy seeds + simulated-annealing insertion local search.
    """

    def greedy_construct():
        # greedy construction with random sampling of candidate next txns
        start_txn = random.randint(0, workload.num_txns - 1)
        txn_seq = [start_txn]
        remaining = [x for x in range(workload.num_txns) if x != start_txn]
        while remaining:
            num_samples = min(6, len(remaining))
            candidates = random.sample(remaining, num_samples)
            best_c, best_t = None, None
            for t in candidates:
                test = txn_seq + [t]
                c = workload.get_opt_seq_cost(test)
                if best_c is None or c < best_c:
                    best_c, best_t = c, t
            txn_seq.append(best_t)
            remaining.remove(best_t)
        return workload.get_opt_seq_cost(txn_seq), txn_seq

    def anneal_insertion(seq, time_limit, t0=2.0, cool=0.95):
        cur = seq.copy()
        cur_cost = workload.get_opt_seq_cost(cur)
        best, best_cost = cur.copy(), cur_cost
        temp = t0
        n = len(cur)
        deadline = time.time() + time_limit
        while time.time() < deadline:
            # one round of insertion moves with annealed acceptance
            for i in range(n):
                if time.time() > deadline:
                    break
                t = cur.pop(i)
                order = list(range(len(cur) + 1))
                random.shuffle(order)
                for j in order:
                    cand = cur[:j] + [t] + cur[j:]
                    c = workload.get_opt_seq_cost(cand)
                    if c < cur_cost or random.random() < pow(2.718281828, -(c - cur_cost) / max(temp, 1e-9)):
                        cur, cur_cost = cand, c
                        if c < best_cost:
                            best, best_cost = cand.copy(), c
                        break
                else:
                    cur.insert(i, t)
            temp *= cool
            if temp < 0.05:
                # restart annealing from best found (reheating, gentler)
                cur, cur_cost = best.copy(), best_cost
                temp = t0 * 0.5
        return best_cost, best

    best_cost = float('inf')
    best_seq = None
    seeds = []
    for _ in range(10):
        c, s = greedy_construct()
        seeds.append((c, s))
        if c < best_cost:
            best_cost, best_seq = c, s

    # anneal each of the best seeds within overall time budget
    overall_deadline = time.time() + 90
    seeds.sort(key=lambda x: x[0])
    for c, s in seeds:
        if time.time() > overall_deadline:
            break
        rc, rs = anneal_insertion(s, min(9.0, overall_deadline - time.time()))
        if rc < best_cost:
            best_cost, best_seq = rc, rs

    return best_cost, best_seq

# EVOLVE-BLOCK-END
