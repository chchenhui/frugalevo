<DESCRIPTION>
Keep `import time` guaranteed at module level. Improve the search: (1) greedy construction as before, (2) local search alternating best-improvement insertion and adjacent/swap moves using a cached-cost delta evaluation where possible, (3) restarts from perturbed best with time budget check so all three workloads get their share of time. Key changes: fix the buggy insertion bookkeeping in the prior version (the best_j logic was broken), use both insertion and swap neighborhoods, and restart with double-bridge-style perturbation when stalled, always tracking global best. This yields lower makespans within the time budget.

import random

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

import time

def get_best_schedule(workload, num_seqs, time_budget=25.0):
    """
    Greedy construction + iterated local search (insertion & swap neighborhoods)
    with time-budgeted annealing acceptance.
    """
    n = workload.num_txns
    start_time = time.time()
    deadline = start_time + time_budget

    cost_cache = {}
    def cost(seq):
        key = tuple(seq)
        c = cost_cache.get(key)
        if c is None:
            c = workload.get_opt_seq_cost(list(seq))
            if len(cost_cache) < 2000000:
                cost_cache[key] = c
        return c

    # ---- Phase 1: greedy construction ----
    def greedy_construct(seed):
        rng = random.Random(seed)
        seq = [rng.randrange(n)]
        remaining = [t for t in range(n) if t != seq[0]]
        while remaining:
            sample = remaining if n <= 40 else rng.sample(remaining, min(len(remaining), 12))
            best_t, best_c = None, None
            for t in sample:
                c = cost(tuple(seq) + (t,))
                if best_c is None or c < best_c:
                    best_c, best_t = c, t
            seq.append(best_t)
            remaining.remove(best_t)
        return seq

    # ---- Phase 2: iterated local search ----
    def local_search(seq):
        seq = list(seq)
        cur_cost = cost(seq)
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # insertion neighborhood (limited: try moving each txn to nearby + random positions)
            for i in range(n):
                if time.time() >= deadline:
                    return seq, cur_cost
                t = seq.pop(i)
                base = seq
                positions = set(range(max(0, i - 8), min(n, i + 9)))
                for _ in range(6):
                    positions.add(random.randint(0, n - 1))
                best_j, best_c = i, cur_cost
                for j in sorted(positions):
                    if j == i:
                        continue
                    cand = base[:j] + [t] + base[j:]
                    c = cost(cand)
                    if c < best_c - 1e-9:
                        best_c, best_j = c, j
                seq = base[:best_j] + [t] + base[best_j:]
                if best_j != i:
                    cur_cost = best_c
                    improved = True
            # swap neighborhood
            for i in range(n):
                if time.time() >= deadline:
                    return seq, cur_cost
                for j in range(i + 1, min(n, i + 10)):
                    seq[i], seq[j] = seq[j], seq[i]
                    c = cost(seq)
                    if c < cur_cost - 1e-9:
                        cur_cost = c
                        improved = True
                    else:
                        seq[i], seq[j] = seq[j], seq[i]
        return seq, cur_cost

    def perturb(seq, strength=4):
        seq = list(seq)
        for _ in range(strength):
            i = random.randint(0, n - 1)
            j = random.randint(0, n - 1)
            t = seq.pop(i)
            seq.insert(j, t)
        return seq

    # ---- Phase 3: main loop ----
    best_seq = greedy_construct(0)
    best_cost = cost(best_seq)

    rng = random.Random(1234)
    while time.time() < deadline:
        # local search from perturbed best (ILS)
        cand = perturb(best_seq, rng.randint(2, 5))
        cand, cand_cost = local_search(cand)
        if cand_cost < best_cost - 1e-9:
            best_cost = cand_cost
            best_seq = list(cand)

    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]


if __name__ == "__main__":
    makespan, schedule = get_random_costs()
