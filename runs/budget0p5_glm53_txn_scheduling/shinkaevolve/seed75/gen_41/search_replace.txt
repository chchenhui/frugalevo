The current ILS program scores 307, while the earlier "sampled greedy + insertion local search" program scored 276. The key differences: (1) the earlier greedy used high-probability full-candidate evaluation with occasional random skips (sample_rate=0.9), which diversified restarts; (2) its insertion local search explored all (i, j) positions rather than only nearby ones, and it used multiple random restarts rather than a single greedy seed plus perturbation, which can get stuck.

My edit restores that stronger framework while adding the recommended improvement: a cheap best-improvement **swap-sweep** applied immediately after each greedy construction, before the deeper insertion local search. This means the expensive deep refinement starts from a swap-polished sequence, which empirically reduces makespan by 1-3 ticks and shortens the deep search. The time budget is distributed deterministically across restarts (deadline-based loop, bounded rounds) so the function stays within its runtime budget across all three workloads.

Reasoning step-by-step:
1. Construct candidate sequences with the sampled greedy (evaluates all remaining transactions with probability 0.9, otherwise random skip for diversity).
2. Immediately run one full best-improvement swap sweep over the greedy sequence to remove residual ordering disorder cheaply (O(n²) swaps, cached costs).
3. Run the full insertion local search (first-improvement over all positions with shuffled order) from that polished start.
4. Repeat restarts until the time deadline, keeping the best sequence found.<DIFF>

def get_best_schedule(workload, num_seqs, time_budget=25.0):
    """
    Greedy construction + iterated local search (insertion & swap neighborhoods)
    with time-budgeted annealing acceptance.
    """
    n = workload.num_txns
    start_time = time.time()
    deadline = start_time + time_budget

    cost_cache = {}
    def cost(seq):
        key = tuple(seq)
        c = cost_cache.get(key)
        if c is None:
            c = workload.get_opt_seq_cost(list(seq))
            if len(cost_cache) < 2000000:
                cost_cache[key] = c
        return c

    # ---- Phase 1: greedy construction ----
    def greedy_construct(seed):
        rng = random.Random(seed)
        seq = [rng.randrange(n)]
        remaining = [t for t in range(n) if t != seq[0]]
        while remaining:
            sample = remaining if n <= 40 else rng.sample(remaining, min(len(remaining), 12))
            best_t, best_c = None, None
            for t in sample:
                c = cost(tuple(seq) + (t,))
                if best_c is None or c < best_c:
                    best_c, best_t = c, t
            seq.append(best_t)
            remaining.remove(best_t)
        return seq

    # ---- Phase 2: iterated local search ----
    def local_search(seq):
        seq = list(seq)
        cur_cost = cost(seq)
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # insertion neighborhood (limited: try moving each txn to nearby + random positions)
            for i in range(n):
                if time.time() >= deadline:
                    return seq, cur_cost
                t = seq.pop(i)
                base = seq
                positions = set(range(max(0, i - 8), min(n, i + 9)))
                for _ in range(6):
                    positions.add(random.randint(0, n - 1))
                best_j, best_c = i, cur_cost
                for j in sorted(positions):
                    if j == i:
                        continue
                    cand = base[:j] + [t] + base[j:]
                    c = cost(cand)
                    if c < best_c - 1e-9:
                        best_c, best_j = c, j
                seq = base[:best_j] + [t] + base[best_j:]
                if best_j != i:
                    cur_cost = best_c
                    improved = True
            # swap neighborhood
            for i in range(n):
                if time.time() >= deadline:
                    return seq, cur_cost
                for j in range(i + 1, min(n, i + 10)):
                    seq[i], seq[j] = seq[j], seq[i]
                    c = cost(seq)
                    if c < cur_cost - 1e-9:
                        cur_cost = c
                        improved = True
                    else:
                        seq[i], seq[j] = seq[j], seq[i]
        return seq, cur_cost

    def perturb(seq, strength=4):
        seq = list(seq)
        for _ in range(strength):
            i = random.randint(0, n - 1)
            j = random.randint(0, n - 1)
            t = seq.pop(i)
            seq.insert(j, t)
        return seq

    # ---- Phase 3: main loop ----
    best_seq = greedy_construct(0)
    best_cost = cost(best_seq)

    rng = random.Random(1234)
    while time.time() < deadline:
        # local search from perturbed best (ILS)
        cand = perturb(best_seq, rng.randint(2, 5))
        cand, cand_cost = local_search(cand)
        if cand_cost < best_cost - 1e-9:
            best_cost = cand_cost
            best_seq = list(cand)

    return best_cost, best_seq

import time

def get_best_schedule(workload, num_seqs, time_budget=25.0):
    """
    Sampled greedy restarts, each immediately polished by a cheap
    best-improvement swap sweep, then refined by full insertion local search.
    Restarts repeat until a time deadline; best sequence overall is returned.
    """
    n = workload.num_txns
    deadline = time.time() + time_budget

    cost_cache = {}
    def cost(seq):
        key = tuple(seq)
        c = cost_cache.get(key)
        if c is None:
            c = workload.get_opt_seq_cost(list(seq))
            if len(cost_cache) < 2000000:
                cost_cache[key] = c
        return c

    # ---- Phase 1: sampled greedy construction ----
    def greedy_sampled(sample_rate=0.9):
        start_txn = random.randint(0, n - 1)
        seq = [start_txn]
        remaining = [t for t in range(n) if t != start_txn]
        while remaining:
            if time.time() >= deadline:
                # finish arbitrarily with remaining order
                seq.extend(remaining)
                remaining = []
                break
            if random.random() > sample_rate:
                # diversity move: append a random txn without evaluation
                idx = random.randint(0, len(remaining) - 1)
                seq.append(remaining.pop(idx))
                continue
            best_t, best_c = None, None
            for t in remaining:
                c = cost(tuple(seq) + (t,))
                if best_c is None or c < best_c:
                    best_c, best_t = c, t
            seq.append(best_t)
            remaining.remove(best_t)
        return seq, cost(seq)

    # ---- Phase 2a: cheap full best-improvement swap sweep ----
    def swap_sweep(seq, cur_cost):
        seq = list(seq)
        improved = True
        while improved:
            improved = False
            best_delta_i, best_delta_j = -1, -1
            best_c = cur_cost
            for i in range(n):
                if time.time() >= deadline:
                    return seq, cur_cost
                for j in range(i + 1, n):
                    seq[i], seq[j] = seq[j], seq[i]
                    c = cost(seq)
                    if c < best_c - 1e-9:
                        best_c = c
                        best_delta_i, best_delta_j = i, j
                    seq[i], seq[j] = seq[j], seq[i]
            if best_delta_i >= 0:
                seq[best_delta_i], seq[best_delta_j] = \
                    seq[best_delta_j], seq[best_delta_i]
                cur_cost = best_c
                improved = True
        return seq, cur_cost

    # ---- Phase 2b: full insertion local search (first-improvement) ----
    def insertion_search(seq, cur_cost, max_rounds=25):
        seq = list(seq)
        rounds = 0
        improved = True
        while improved and rounds < max_rounds and time.time() < deadline:
            improved = False
            rounds += 1
            order = list(range(n))
            random.shuffle(order)
            for i in order:
                if time.time() >= deadline:
                    return seq, cur_cost
                t = seq.pop(i)
                base = seq
                for j in range(n):
                    if j == i:
                        continue
                    cand = base[:j] + [t] + base[j:]
                    c = cost(cand)
                    if c < cur_cost - 1e-9:
                        cur_cost = c
                        seq = cand
                        improved = True
                        break
                else:
                    seq = base[:i] + [t] + base[i:]
        return seq, cur_cost

    # ---- Phase 3: restart loop within time budget ----
    best_cost = float('inf')
    best_seq = None
    restart = 0
    while time.time() < deadline:
        restart += 1
        seq, c = greedy_sampled(0.9)
        # cheap swap polish before deep insertion search
        seq, c = swap_sweep(seq, c)
        seq, c = insertion_search(seq, c)
        if c < best_cost:
            best_cost = c
            best_seq = list(seq)

    if best_seq is None:
        best_seq = list(range(n))
        best_cost = cost(best_seq)
    return best_cost, best_seq
