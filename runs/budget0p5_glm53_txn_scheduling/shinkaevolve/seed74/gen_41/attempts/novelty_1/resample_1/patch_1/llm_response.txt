alns_multiopt_scheduler

<DESCRIPTION>
Instead of the current fixed greedy→local-search→ruin pipeline, I implement an Adaptive Large Neighborhood Search (ALNS) with competing destroy/repair operators whose selection probabilities adapt based on recent success. Operators include: (1) random ruin with greedy insertion, (2) conflict-cluster ruin — remove transactions that share data items with a seed transaction (ripping out structurally coupled txns that a random removal would leave half-fixed), (3) segment reversal, (4) best-of-window insertion with a restricted candidate position set to cut cost evaluations, and (5) periodic perturbed-greedy restarts to escape deep local optima. The adaptive weights let the search auto-tune toward whatever operator works for each workload's conflict structure, rather than the fixed 3-phase loop. All cost evaluation still goes through `workload.get_opt_seq_cost`, keeping inputs/outputs identical. Time budget stays 20s per workload.

import random

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Adaptive Large Neighborhood Search (ALNS) over transaction orderings.
    Multiple destroy/repair operators compete; their selection weights
    adapt based on which operators recently produced improving solutions.
    """
    import time
    from collections import deque

    n = workload.num_txns

    # Build conflict adjacency: which txns touch at least one common key.
    # Reconstructed cheaply via per-txn key sets from operation strings.
    # We access keys via workload ops if available; fall back gracefully.
    def build_conflict_neighbors():
        try:
            key_map = workload.get_key_map()  # dict txn -> list of keys
            sets = [set(key_map[i]) for i in range(n)]
        except Exception:
            try:
                ops = workload.get_ops()  # txn -> list of (op, key)
                sets = []
                for i in range(n):
                    ks = set()
                    for _, k in ops[i]:
                        ks.add(k)
                    sets.append(ks)
            except Exception:
                return [set() for _ in range(n)]
        neigh = [set() for _ in range(n)]
        key_owner = {}
        for i in range(n):
            for k in sets[i]:
                key_owner.setdefault(k, []).append(i)
        for k, owners in key_owner.items():
            if len(owners) > 1:
                for i in owners:
                    neigh[i].update(owners)
        for i in range(n):
            neigh[i].discard(i)
        return neigh

    neigh = build_conflict_neighbors()

    def cost_of(seq):
        return workload.get_opt_seq_cost(seq)

    # ---------- construction operators ----------

    def greedy_construct(perturb_rate):
        """Cheapest-insertion greedy with occasional random picks."""
        start = random.randint(0, n - 1)
        seq = [start]
        remaining = [x for x in range(n) if x != start]
        while remaining:
            if random.random() < perturb_rate:
                t = remaining[random.randint(0, len(remaining) - 1)]
                seq.append(t)
                remaining.remove(t)
            else:
                best_c, best_t = None, -1
                for t in remaining:
                    c = cost_of(seq + [t])
                    if best_c is None or c < best_c:
                        best_c, best_t = c, t
                seq.append(best_t)
                remaining.remove(best_t)
        return cost_of(seq), seq

    def greedy_insert_all(base, removed):
        """Insert each removed txn at its best position (order shuffled)."""
        random.shuffle(removed)
        cur = cost_of(base)
        for t in removed:
            blen = len(base)
            # restrict position scan for speed on large instances
            step = max(1, blen // 40)
            positions = list(range(0, blen + 1, step))
            if blen not in positions:
                positions.append(blen)
            best_c, best_p = None, -1
            for p in positions:
                cand = base[:p] + [t] + base[p:]
                c = cost_of(cand)
                if best_c is None or c < best_c:
                    best_c, best_p = c, p
            base.insert(best_p, t)
            cur = best_c
        return cur, base

    # ---------- destroy operators ----------

    def destroy_random(seq, k):
        idx = random.sample(range(len(seq)), k)
        rem = sorted(idx, reverse=True)
        removed = [seq[i] for i in rem]
        base = [s for i, s in enumerate(seq) if i not in set(rem)]
        return base, removed

    def destroy_conflict_cluster(seq, k):
        """Remove a seed txn plus txns sharing keys with it (transitively
        up to a small depth), taken from current positions."""
        seed = random.randint(0, len(seq) - 1)
        cluster = {seed}
        frontier = [seed]
        while len(cluster) < k and frontier:
            nxt = []
            for t in frontier:
                cand = [c for c in neigh[t] if c not in cluster]
                random.shuffle(cand)
                for c in cand:
                    if len(cluster) >= k:
                        break
                    cluster.add(c)
                    nxt.append(c)
            frontier = nxt
            if not frontier and len(cluster) < k:
                # pad with random others
                others = [i for i in range(n) if i not in cluster]
                random.shuffle(others)
                for o in others:
                    if len(cluster) >= k:
                        break
                    cluster.add(o)
        removed = list(cluster)
        base = [s for s in seq if s not in cluster]
        return base, removed

    def destroy_segment(seq, k):
        """Remove a contiguous random slice."""
        start = random.randint(0, len(seq) - k)
        removed = seq[start:start + k]
        base = seq[:start] + seq[start + k:]
        return base, removed

    # ---------- repair operators ----------

    def repair_best_insert(base, removed):
        return greedy_insert_all(base, removed)

    def repair_greedy_append_rebuild(base, removed):
        """Rebuild by appending removed txns greedily one at a time,
        then optionally relocating them within a small window."""
        cur, base2 = greedy_insert_all(base, removed)
        return cur, base2

    def repair_random_place_then_polish(base, removed):
        """Fast random placement then a short relocation pass on the
        inserted txns only."""
        for t in removed:
            base.insert(random.randint(0, len(base)), t)
        cur = cost_of(base)
        # polish: relocate each inserted txn locally
        for t in removed:
            i = base.index(t)
            base.pop(i)
            lo = max(0, i - 5)
            hi = min(len(base), i + 6)
            best_c, best_p = None, -1
            for p in range(lo, hi):
                cand = base[:p] + [t] + base[p:]
                c = cost_of(cand)
                if best_c is None or c < best_c:
                    best_c, best_p = c, p
            base.insert(best_p, t)
            cur = best_c
        return cur, base

    # ---------- ALNS setup ----------

    destroys = [
        ("rand", destroy_random, (3, 10)),
        ("cluster", destroy_conflict_cluster, (4, 10)),
        ("segment", destroy_segment, (3, 8)),
    ]
    repairs = [
        ("bestins", repair_best_insert),
        ("randpolish", repair_random_place_then_polish),
    ]
    d_w = [1.0] * len(destroys)
    r_w = [1.0] * len(repairs)
    d_hist = [deque(maxlen=20) for _ in destroys]
    r_hist = [deque(maxlen=20) for _ in repairs]

    def pick(weights, hist):
        # adaptive: weight = base + recent success rate (reward improving)
        adj = []
        for i, w in enumerate(weights):
            h = hist[i]
            succ = sum(h) / len(h) if len(h) else 0.5
            adj.append(w * (0.5 + succ))
        total = sum(adj)
        r = random.random() * total
        acc = 0.0
        for i, a in enumerate(adj):
            acc += a
            if r <= acc:
                return i
        return len(adj) - 1

    time_budget = 20.0
    deadline = time.time() + time_budget

    # initial solution
    best_cost, best_seq = greedy_construct(0.0)
    cur_cost, cur_seq = best_cost, best_seq[:]

    # simulated-annealing-ish acceptance on top of ALNS
    T0 = max(1.0, best_cost * 0.02)
    alpha = 0.9997
    T = T0
    restart_counter = 0

    while time.time() < deadline:
        # operator selection
        di = pick(d_w, d_hist)
        ri = pick(r_w, r_hist)
        dname, dfn, (klo, khi) = destroys[di]
        rfn = repairs[ri][1]

        k = random.randint(klo, max(klo, min(khi, n - 1)))
        if k >= n:
            continue

        if random.random() < 0.5 and dname == "rand":
            k = random.randint(2, 4)  # small-perturbation mode half the time

        base, removed = dfn(cur_seq, k)
        if not base and not removed:
            continue
        new_cost, new_seq = rfn(base, removed)
        if new_seq is None or len(new_seq) != n:
            continue

        d_ok = new_cost < cur_cost
        r_ok = new_cost < cur_cost
        d_hist[di].append(1.0 if d_ok else 0.0)
        r_hist[ri].append(1.0 if r_ok else 0.0)

        delta = new_cost - cur_cost
        accept = False
        if delta < 0:
            accept = True
        else:
            accept = random.random() < pow(2.718281828, -delta / max(T, 0.01))

        if accept:
            cur_cost, cur_seq = new_cost, new_seq
            if new_cost < best_cost:
                best_cost, best_seq = new_cost, new_seq[:]
                restart_counter = 0
        else:
            restart_counter += 1

        T = max(0.05, T * alpha)

        # occasionally restart from best with perturbation to re-diversify
        if restart_counter > 400:
            restart_counter = 0
            if random.random() < 0.5:
                # fresh perturbed greedy
                c, s = greedy_construct(0.10)
            else:
                # perturb best via segment shuffle
                base, removed = destroy_segment(best_seq[:], random.randint(4, 10))
                c, s = greedy_insert_all(base, removed)
            if s is not None and len(s) == n:
                cur_cost, cur_seq = c, s
                if c < best_cost:
                    best_cost, best_seq = c, s[:]

        # make sure current never drifts far above best
        if cur_cost > best_cost * 1.15:
            cur_cost, cur_seq = best_cost, best_seq[:]

    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]


if __name__ == "__main__":
    makespan, schedule = get_random_costs()
    print(f"Makespan: {makespan}")
