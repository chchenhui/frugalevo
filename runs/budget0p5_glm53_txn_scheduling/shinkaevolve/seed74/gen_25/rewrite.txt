import random
import time

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Get optimal schedule using greedy cost sampling, time-budgeted restarts,
    shuffled-best perturbation, and swap/Or-opt local search.
    """

    def get_greedy_cost_sampled(num_samples, sample_rate):
        start_txn = random.randint(0, workload.num_txns - 1)
        txn_seq = [start_txn]
        remaining_txns = [x for x in range(0, workload.num_txns)]
        remaining_txns.remove(start_txn)
        for i in range(0, workload.num_txns - 1):
            min_cost = 100000
            min_txn = -1
            holdout_txns = []
            done = False

            sample = random.random()
            if sample > sample_rate:
                idx = random.randint(0, len(remaining_txns) - 1)
                t = remaining_txns[idx]
                txn_seq.append(t)
                remaining_txns.pop(idx)
                continue

            for j in range(0, num_samples):
                idx = 0
                if len(remaining_txns) > 1:
                    idx = random.randint(0, len(remaining_txns) - 1)
                else:
                    done = True
                t = remaining_txns[idx]
                holdout_txns.append(remaining_txns.pop(idx))
                test_seq = txn_seq.copy()
                test_seq.append(t)
                cost = workload.get_opt_seq_cost(test_seq)
                if cost < min_cost:
                    min_cost = cost
                    min_txn = t
                if done:
                    break
            assert min_txn != -1
            txn_seq.append(min_txn)
            holdout_txns.remove(min_txn)
            remaining_txns.extend(holdout_txns)
        assert len(set(txn_seq)) == workload.num_txns
        overall_cost = workload.get_opt_seq_cost(txn_seq)
        return overall_cost, txn_seq

    def shuffled_best(base_seq):
        # Perturb best: remove k random txns, greedy-reinsert each at cheapest spot.
        seq = base_seq[:]
        k = random.randint(4, 6)
        k = min(k, len(seq) - 2)
        removed = []
        for _ in range(k):
            if len(seq) <= 1:
                break
            idx = random.randint(0, len(seq) - 1)
            removed.append(seq.pop(idx))
        for t in removed:
            best_c = None
            best_j = 0
            for j in range(len(seq) + 1):
                cand = seq[:j] + [t] + seq[j:]
                c = workload.get_opt_seq_cost(cand)
                if best_c is None or c < best_c:
                    best_c = c
                    best_j = j
            seq.insert(best_j, t)
        return workload.get_opt_seq_cost(seq), seq

    def swap_pass(best_cost, best_sched, deadline):
        improved = False
        i = 1
        while i < len(best_sched) - 1:
            if time.time() > deadline:
                break
            cand = best_sched[:]
            cand[i], cand[i + 1] = cand[i + 1], cand[i]
            c = workload.get_opt_seq_cost(cand)
            if c < best_cost:
                best_cost = c
                best_sched = cand
                improved = True
            else:
                i += 1
        return best_cost, best_sched, improved

    def or_opt_pass(best_cost, best_sched, deadline):
        improved = False
        while time.time() < deadline:
            n = len(best_sched)
            found = False
            for seg_len in (1, 2, 3):
                if found:
                    break
                for i in range(n - seg_len + 1):
                    if found or time.time() > deadline:
                        break
                    seg = best_sched[i:i + seg_len]
                    rest = best_sched[:i] + best_sched[i + seg_len:]
                    for j in range(len(rest) + 1):
                        if j == i:
                            continue
                        if time.time() > deadline:
                            break
                        cand = rest[:j] + seg + rest[j:]
                        c = workload.get_opt_seq_cost(cand)
                        if c < best_cost:
                            best_cost = c
                            best_sched = cand
                            found = True
                            improved = True
                            break
            if not found:
                break
        return best_cost, best_sched, improved

    def local_search(best_cost, best_sched, deadline):
        rounds = 0
        while rounds < 100 and time.time() < deadline:
            rounds += 1
            best_cost, best_sched, imp1 = swap_pass(best_cost, best_sched, deadline)
            best_cost, best_sched, imp2 = or_opt_pass(best_cost, best_sched, deadline)
            if not imp1 and not imp2:
                break
        return best_cost, best_sched

    deadline = time.time() + 8.0
    best_cost = None
    best_sched = None
    restarts = 0
    while time.time() < deadline:
        restarts += 1
        if restarts % 3 == 0 and best_sched is not None:
            cost, sched = shuffled_best(best_sched)
        else:
            cost, sched = get_greedy_cost_sampled(8, 0.9)
        # quick local polish of each candidate
        cost, sched = local_search(cost, sched, min(deadline, time.time() + 0.5))
        if best_cost is None or cost < best_cost:
            best_cost, best_sched = cost, sched
        if time.time() > deadline:
            break

    # final deep local search on incumbent with remaining time
    best_cost, best_sched = local_search(best_cost, best_sched, deadline + 0.5)
    return best_cost, best_sched

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]
