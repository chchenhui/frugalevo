<DESCRIPTION>
Adjacent swaps can only fix local ordering issues; many suboptimal schedules have a small block of 1–3 transactions that belongs elsewhere. Adding an Or-opt pass (remove a segment of length 1–3 and try reinserting it at every other position), alternating with adjacent-swap hill climbing, explores a much richer neighborhood while staying within the time budget. Each candidate is evaluated exactly via `get_opt_seq_cost`, so improvements always reflect the true makespan. This should reduce makespan further and raise the combined score.
</DESCRIPTION>

    # local search: alternate adjacent swaps and Or-opt (segment relocation)
    def local_search(seq, cost, deadline):
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # adjacent swap pass
            for i in range(len(seq) - 1):
                if time.time() > deadline:
                    break
                cand = seq.copy()
                cand[i], cand[i + 1] = cand[i + 1], cand[i]
                c = workload.get_opt_seq_cost(cand)
                if c < cost:
                    cost, seq = c, cand
                    improved = True
                    break
            if improved or time.time() > deadline:
                continue
            # Or-opt pass: remove segment of length 1-3, reinsert elsewhere
            n = len(seq)
            found = False
            for seg_len in (1, 2, 3):
                if found or time.time() > deadline:
                    break
                for i in range(n - seg_len + 1):
                    if found or time.time() > deadline:
                        break
                    seg = seq[i:i + seg_len]
                    rest = seq[:i] + seq[i + seg_len:]
                    for j in range(len(rest) + 1):
                        if j == i:
                            continue
                        cand = rest[:j] + seg + rest[j:]
                        c = workload.get_opt_seq_cost(cand)
                        if c < cost:
                            cost, seq = c, cand
                            found = True
                            improved = True
                            break
        return cost, seq

    best_cost, best_seq = local_search(best_seq, best_cost, deadline)

    return best_cost, best_seq
