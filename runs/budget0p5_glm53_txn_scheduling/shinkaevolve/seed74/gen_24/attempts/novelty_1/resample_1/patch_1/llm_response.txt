ruin_recreate_hybrid_search</NAME>
<DESCRIPTION>
Restructure the solver into a pipeline of modular stages: (1) multi-restart randomized greedy construction, (2) ruin-and-recreate perturbation (remove k random transactions, greedily reinsert at best position), (3) adjacent-swap descent, all running under a unified time budget. This architecture escapes local optima that pure greedy+swap cannot, per the recommendation, and should lower makespan further.</DESCRIPTION>

import random
import time

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Pipeline: greedy restarts -> ruin-and-recreate -> adjacent-swap descent.
    Returns (lowest makespan, corresponding schedule).
    """

    def greedy_construct(sample_rate=0.9, num_samples=10):
        start_txn = random.randint(0, workload.num_txns - 1)
        txn_seq = [start_txn]
        remaining = [x for x in range(workload.num_txns) if x != start_txn]
        while remaining:
            if random.random() > sample_rate or len(remaining) == 1:
                t = remaining.pop(random.randrange(len(remaining)))
                txn_seq.append(t)
                continue
            holdout = []
            min_cost = None
            min_txn = None
            for _ in range(min(num_samples, len(remaining))):
                idx = random.randrange(len(remaining))
                t = remaining.pop(idx)
                holdout.append(t)
                cost = workload.get_opt_seq_cost(txn_seq + [t])
                if min_cost is None or cost < min_cost:
                    min_cost, min_txn = cost, t
                if not remaining:
                    break
            txn_seq.append(min_txn)
            holdout.remove(min_txn)
            remaining.extend(holdout)
        return workload.get_opt_seq_cost(txn_seq), txn_seq

    def ruin_and_recreate(seq, cost, k):
        seq = seq[:]
        removed = random.sample(seq, min(k, len(seq)))
        kept = [t for t in seq if t not in removed]
        # greedily reinsert each removed txn at best position
        for t in removed:
            best_c = None
            best_pos = 0
            for pos in range(len(kept) + 1):
                cand = kept[:pos] + [t] + kept[pos:]
                c = workload.get_opt_seq_cost(cand)
                if best_c is None or c < best_c:
                    best_c, best_pos = c, pos
            kept.insert(best_pos, t)
        c = workload.get_opt_seq_cost(kept)
        return (c, kept) if c < cost else (cost, seq)

    def swap_descent(seq, cost, deadline):
        improved = True
        n = len(seq)
        while improved and time.time() < deadline:
            improved = False
            for i in range(n - 1):
                if time.time() > deadline:
                    break
                seq[i], seq[i + 1] = seq[i + 1], seq[i]
                c = workload.get_opt_seq_cost(seq)
                if c < cost:
                    cost = c
                    improved = True
                else:
                    seq[i], seq[i + 1] = seq[i + 1], seq[i]
        return cost, seq

    deadline = time.time() + 25.0

    # Stage 1: greedy restarts, keep top-3
    pool = []
    while time.time() < deadline - 15.0:
        cost, seq = greedy_construct()
        pool.append((cost, seq))
        pool.sort(key=lambda x: x[0])
        pool = pool[:3]
    if not pool:
        cost, seq = greedy_construct()
        pool = [(cost, seq)]

    best_cost, best_seq = pool[0]

    # Stage 2: ruin-and-recreate perturbations on pool
    while time.time() < deadline - 5.0:
        for ci, (c, s) in enumerate(pool):
            if time.time() > deadline - 5.0:
                break
            nc, ns = ruin_and_recreate(s, c, random.randint(3, 8))
            pool[ci] = (nc, ns)
        pool.sort(key=lambda x: x[0])
        if pool[0][0] < best_cost:
            best_cost, best_seq = pool[0]

    # Stage 3: adjacent swap descent
    best_cost, best_seq = swap_descent(list(best_seq), best_cost, deadline)

    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]


if __name__ == "__main__":
    makespan, schedule = get_random_costs()
    print(f"Makespan: {makespan}")
