beam_greedy_2opt_restart</NAME>
<DESCRIPTION>
Instead of a single randomized greedy pass, this approach runs many independent greedy constructions with different starting transactions (multi-restart diversification), then improves the best schedule with a first-improvement 2-opt swap local search. The greedy step uses exact makespan evaluation via `get_opt_seq_cost` of the partial sequence to pick the next transaction (least-increase greedy), which the original already did partially but only once with sampling noise. Adding beam-style restarts + local search escape local optima and reliably lower makespan.
</DESCRIPTION>

import random

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Multi-restart greedy construction with least-makespan next-txn selection,
    followed by 2-opt swap local search on the best schedule found.
    """
    n = workload.num_txns

    def greedy_construct(start_txn):
        txn_seq = [start_txn]
        remaining = [x for x in range(n) if x != start_txn]
        while remaining:
            best_t, best_cost = None, float('inf')
            test = txn_seq + [0]
            for t in remaining:
                test[-1] = t
                c = workload.get_opt_seq_cost(test)
                if c < best_cost:
                    best_cost, best_t = c, t
            txn_seq.append(best_t)
            remaining.remove(best_t)
        return workload.get_opt_seq_cost(txn_seq), txn_seq

    def two_opt(seq, max_rounds=30):
        cost = workload.get_opt_seq_cost(seq)
        improved = True
        rounds = 0
        while improved and rounds < max_rounds:
            improved = False
            rounds += 1
            for i in range(n - 1):
                for j in range(i + 1, n):
                    cand = seq[:i] + seq[i:j + 1][::-1] + seq[j + 1:]
                    c = workload.get_opt_seq_cost(cand)
                    if c < cost:
                        cost, seq = c, cand
                        improved = True
        return cost, seq

    best_cost, best_seq = float('inf'), None
    starts = list(range(n))
    random.shuffle(starts)
    num_restarts = min(n, 25)
    for s in starts[:num_restarts]:
        c, sq = greedy_construct(s)
        if c < best_cost:
            best_cost, best_seq = c, sq

    # local search refinement on best schedule
    best_cost, best_seq = two_opt(best_seq)

    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]


if __name__ == "__main__":
    makespan, schedule = get_random_costs()
    print(f"Makespan: {makespan}")
