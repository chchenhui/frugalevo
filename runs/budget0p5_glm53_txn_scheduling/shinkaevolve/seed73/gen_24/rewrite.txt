import random
import time

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Get optimal schedule using a three-phase pipeline:
      1. Construct a candidate pool via multistart full-lookahead greedy
         (heuristic + random seeds).
      2. Rank candidates by makespan.
      3. Refine the top candidates with insertion + swap local search
         under a shared time budget.

    Returns:
        Tuple of (lowest makespan, corresponding schedule)
    """

    # ---------- Phase helpers ----------

    def heuristic_seed_txns(k=2):
        """Return the indices of the k transactions with the largest
        single-transaction makespan (a cheap one-step lookahead heuristic)."""
        costs = []
        for t in range(workload.num_txns):
            costs.append((workload.get_opt_seq_cost([t]), t))
        costs.sort(reverse=True)
        return [t for _, t in costs[:k]]

    def greedy_from(start_txn, sample_rate=0.9, num_samples=10):
        """Randomized full-lookahead greedy construction starting at
        start_txn. With probability sample_rate, evaluate num_samples random
        candidate next-transactions by actual partial-schedule makespan and
        pick the best; otherwise append a purely random pick for diversity."""
        txn_seq = [start_txn]
        remaining = [x for x in range(workload.num_txns) if x != start_txn]
        while remaining:
            if random.random() > sample_rate or len(remaining) == 1:
                idx = random.randrange(len(remaining))
                txn_seq.append(remaining.pop(idx))
                continue
            best_cost = None
            best_txn = None
            for _ in range(min(num_samples, len(remaining))):
                idx = random.randrange(len(remaining))
                t = remaining[idx]
                cost = workload.get_opt_seq_cost(txn_seq + [t])
                if best_cost is None or cost < best_cost:
                    best_cost = cost
                    best_txn = t
            txn_seq.append(best_txn)
            remaining.remove(best_txn)
        return workload.get_opt_seq_cost(txn_seq), txn_seq

    def local_search(seq, cost, deadline):
        """First-improvement local search: insertion moves, then swap moves,
        repeated until no improvement or deadline hit."""
        n = len(seq)
        improved = True
        while improved:
            improved = False
            # insertion moves
            for i in range(n):
                txn = seq[i]
                rest = seq[:i] + seq[i+1:]
                for j in range(n):
                    if time.time() >= deadline:
                        return seq, cost
                    if j == i:
                        continue
                    cand = rest[:j] + [txn] + rest[j:]
                    c = workload.get_opt_seq_cost(cand)
                    if c < cost:
                        seq, cost = cand, c
                        improved = True
                        break
                if improved:
                    break
            if improved:
                continue
            # swap moves
            for i in range(n - 1):
                for j in range(i + 1, n):
                    if time.time() >= deadline:
                        return seq, cost
                    cand = seq[:]
                    cand[i], cand[j] = cand[j], cand[i]
                    c = workload.get_opt_seq_cost(cand)
                    if c < cost:
                        seq, cost = cand, c
                        improved = True
                        break
                if improved:
                    break
        return seq, cost

    # ---------- Main pipeline ----------

    start_time = time.time()
    total_budget = 25.0
    best_cost = None
    best_seq = None

    # Phase 1: construction — heuristic seeds plus random restarts
    seeds = set(heuristic_seed_txns(2))
    while len(seeds) < 3:
        seeds.add(random.randrange(workload.num_txns))

    candidates = []
    for seed in seeds:
        cost, seq = greedy_from(seed)
        candidates.append((cost, seq))
        if best_cost is None or cost < best_cost:
            best_cost, best_seq = cost, seq

    construct_deadline = start_time + total_budget * 0.5
    while time.time() < construct_deadline:
        start_txn = random.randrange(workload.num_txns)
        cost, seq = greedy_from(start_txn)
        candidates.append((cost, seq))
        if cost < best_cost:
            best_cost, best_seq = cost, seq

    # Phase 2: rank candidates
    candidates.sort(key=lambda x: x[0])

    # Phase 3: refine top candidates with local search
    for c_cost, c_seq in candidates[:3]:
        remaining_time = total_budget - (time.time() - start_time)
        if remaining_time <= 0.2:
            break
        deadline = start_time + total_budget
        seq, cost = local_search(c_seq[:], c_cost, deadline)
        if cost < best_cost:
            best_cost, best_seq = cost, seq

    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]
