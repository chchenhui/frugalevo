<DESCRIPTION>
Crossover combining the current best (full-lookahead greedy with multistart restarts and a 25s budget) with the inspiration's insertion-move local search, enhanced with swap moves as well. Reasoning: greedy multistart provides diverse good starting schedules; local search (insertion relocations and adjacent/full swaps) then refines each schedule, escaping greedy's local optima. Time is budgeted per restart (remaining time split across remaining restarts) so every restart gets refinement while still covering many starting points. First-improvement moves keep the search fast; the best schedule found across all restart+refinement passes is returned, which should reduce makespan below 299.
</DESCRIPTION>

import random

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Get optimal schedule using full-lookahead greedy with multistart restarts,
    followed by insertion + swap local search refinement under a time budget.

    Returns:
        Tuple of (lowest makespan, corresponding schedule)
    """
    import time

    def greedy_from(start_txn):
        txn_seq = [start_txn]
        remaining = [x for x in range(workload.num_txns) if x != start_txn]
        while remaining:
            best_cost = None
            best_txn = None
            for t in remaining:
                test_seq = txn_seq + [t]
                cost = workload.get_opt_seq_cost(test_seq)
                if best_cost is None or cost < best_cost:
                    best_cost = cost
                    best_txn = t
            txn_seq.append(best_txn)
            remaining.remove(best_txn)
        return workload.get_opt_seq_cost(txn_seq), txn_seq

    def local_search(seq, cost, deadline):
        """First-improvement local search with insertion and swap moves."""
        n = len(seq)
        improved = True
        while improved and time.time() < deadline:
            improved = False
            # insertion moves
            for i in range(n):
                txn = seq[i]
                rest = seq[:i] + seq[i+1:]
                for j in range(n):
                    if time.time() >= deadline:
                        return seq, cost
                    if j == i:
                        continue
                    cand = rest[:j] + [txn] + rest[j:]
                    c = workload.get_opt_seq_cost(cand)
                    if c < cost:
                        seq, cost = cand, c
                        improved = True
                        break
                if improved:
                    break
            if improved:
                continue
            # swap moves
            for i in range(n - 1):
                for j in range(i + 1, n):
                    if time.time() >= deadline:
                        return seq, cost
                    cand = seq[:]
                    cand[i], cand[j] = cand[j], cand[i]
                    c = workload.get_opt_seq_cost(cand)
                    if c < cost:
                        seq, cost = cand, c
                        improved = True
                        break
                if improved:
                    break
        return seq, cost

    start_time = time.time()
    total_budget = 25
    best_cost = None
    best_seq = None
    order = list(range(workload.num_txns))
    random.shuffle(order)
    for idx, start_txn in enumerate(order):
        elapsed = time.time() - start_time
        remaining = total_budget - elapsed
        if best_seq is not None and remaining <= 0.5:
            break
        # split remaining time across remaining restarts
        per_run = max(remaining / max(1, (len(order) - idx)), 0.5)
        deadline = time.time() + per_run
        cost, seq = greedy_from(start_txn)
        if time.time() < deadline:
            seq, cost = local_search(seq, cost, deadline)
        if best_cost is None or cost < best_cost:
            best_cost = cost
            best_seq = seq

    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]


if __name__ == "__main__":
    makespan, schedule = get_random_costs()
