<DESCRIPTION>
The best prior programs reached makespan 261 by combining: (1) cheap, diverse full-lookahead greedy constructions (randomized tie-breaking, sampled candidate sets), (2) a diversity-preserving depth-adaptive beam search (top width-2 by cost plus 2 children with differing last transactions, avoiding prefix collapse), (3) simulated annealing with greedy insertion intensification on the top candidates, and (4) an LNS ruin-and-recreate phase on the incumbent. I combine all four phases with a 28s budget split roughly 15% greedy, 50% beam, 75% SA, and the remainder LNS. The greedy phase provides diverse, cheap seeds whose diversity drives beam and SA success; beam search with diversity-preserving pruning avoids converging on identical prefixes; SA with insertion sweeps escapes local optima; LNS block reinsertion provides large-neighborhood improvements. All moves are evaluated by actual makespan via get_opt_seq_cost, per the task guidance that heuristics like length/write-count don't correlate with makespan.

import random

from txn_simulator import Workload
from workloads import WORKLOAD_1, WORKLOAD_2, WORKLOAD_3

# EVOLVE-BLOCK-START

def get_best_schedule(workload, num_seqs):
    """
    Hybrid scheduler:
    1) Full-lookahead greedy constructions with randomized tie-breaking
       and sampled candidates for cheap diversity.
    2) Depth-adaptive, diversity-preserving beam search (keeps top
       width-2 by cost plus 2 children with differing appended txn).
    3) SA refinement with greedy insertion intensification.
    4) LNS ruin-and-recreate on the incumbent.
    All evaluated by actual makespan via get_opt_seq_cost.
    """
    import time
    import math

    n = workload.num_txns
    if n == 0:
        return 0, []

    start_time = time.time()
    time_budget = 28.0

    best_cost = float('inf')
    best_seq = None
    candidates = []

    def greedy_from(start_txn, sample_cap):
        txn_seq = [start_txn]
        remaining = [x for x in range(n) if x != start_txn]
        while remaining:
            cands = remaining
            if sample_cap and len(remaining) > sample_cap:
                cands = random.sample(remaining, sample_cap)
            bc = None
            best_txns = []
            for t in cands:
                cost = workload.get_opt_seq_cost(txn_seq + [t])
                if bc is None or cost < bc:
                    bc, best_txns = cost, [t]
                elif cost == bc:
                    best_txns.append(t)
            t = random.choice(best_txns)
            txn_seq.append(t)
            remaining.remove(t)
        return workload.get_opt_seq_cost(txn_seq), txn_seq

    # ---------- Phase 1a: greedy constructions ----------
    seen_starts = set()
    for _ in range(6):
        if time.time() - start_time >= time_budget * 0.12:
            break
        s = random.randrange(n)
        if s in seen_starts:
            continue
        seen_starts.add(s)
        cost, seq = greedy_from(s, sample_cap=None)
        candidates.append((cost, seq))
        if cost < best_cost:
            best_cost, best_seq = cost, seq

    # sampled greedy restarts
    while time.time() - start_time < time_budget * 0.18:
        s = random.randint(0, n - 1)
        cost, seq = greedy_from(s, sample_cap=15)
        candidates.append((cost, seq))
        if cost < best_cost:
            best_cost, best_seq = cost, seq

    # ---------- Phase 1b: diversity-preserving beam search ----------
    num_restarts = max(1, min(6, 150 // max(1, n)))
    max_beam_width = max(4, min(8, 600 // max(1, n)))

    for restart in range(num_restarts):
        if time.time() - start_time >= time_budget * 0.45:
            break
        seeds = {0, random.randrange(n)}
        if n > 1:
            seeds.add(random.randrange(n))
        beam = []
        for s in seeds:
            seq = [s]
            rem = [x for x in range(n) if x != s]
            beam.append((workload.get_opt_seq_cost(seq), seq, rem))
        beam.sort(key=lambda x: x[0])
        beam = beam[:max_beam_width]

        while beam and len(beam[0][1]) < n:
            depth = len(beam[0][1])
            frac = depth / max(1, n)
            if frac < 0.2:
                width = max_beam_width
            elif frac < 0.6:
                width = max(3, max_beam_width - 2)
            else:
                width = max(2, max_beam_width - 3)
            cands = []
            for c, seq, rem in beam:
                if not rem:
                    continue
                for t in rem:
                    new_seq = seq + [t]
                    new_rem = [x for x in rem if x != t]
                    cands.append((workload.get_opt_seq_cost(new_seq), new_seq, new_rem))
            if not cands:
                break
            cands.sort(key=lambda x: x[0])
            # diversity-preserving pruning: top (width-2) by cost
            # plus 2 children whose appended txn differs from the best
            kept = cands[:width - 2]
            used_last = {seq[-1] for _, seq, _ in kept}
            for c, seq, rem in cands[width - 2:]:
                if seq[-1] not in used_last:
                    kept.append((c, seq, rem))
                    used_last.add(seq[-1])
                if len(kept) >= width:
                    break
            beam = kept

        for c, seq, rem in beam:
            candidates.append((c, seq))
            if c < best_cost:
                best_cost, best_seq = c, seq

    if best_seq is None:
        best_seq = list(range(n))
        best_cost = workload.get_opt_seq_cost(best_seq)
        candidates.append((best_cost, best_seq[:]))

    # ---------- Phase 2: SA refinement with insertion intensification ----------
    candidates.sort(key=lambda x: x[0])
    seen = set()
    top = []
    for c, s in candidates:
        key = tuple(s)
        if key not in seen:
            seen.add(key)
            top.append((c, s))
        if len(top) >= 4:
            break

    rng = random.Random(1234)

    for c_cost, c_seq in top:
        if time.time() - start_time >= time_budget * 0.75:
            break
        seq = c_seq[:]
        cur_cost = c_cost
        local_best_cost, local_best_seq = cur_cost, seq[:]
        T0 = max(1.0, cur_cost * 0.05)
        cand_start = time.time()
        rem_budget = max(0.001, time_budget * 0.75 - (cand_start - start_time))
        while True:
            elapsed = time.time() - cand_start
            if elapsed >= rem_budget or time.time() - start_time >= time_budget * 0.75:
                break
            frac = elapsed / max(rem_budget, 1e-6)
            T = T0 * (1.0 - frac) + 1e-6
            if rng.random() < 0.5:
                i = rng.randrange(n)
                j = rng.randrange(n)
                if i == j:
                    continue
                cand = seq[:i] + seq[i+1:]
                cand.insert(j, seq[i])
            else:
                i = rng.randrange(n)
                j = rng.randrange(n)
                if i == j:
                    continue
                cand = seq[:]
                cand[i], cand[j] = cand[j], cand[i]
            c = workload.get_opt_seq_cost(cand)
            delta = c - cur_cost
            if delta < 0 or rng.random() < math.exp(-delta / T):
                seq, cur_cost = cand, c
                if cur_cost < local_best_cost:
                    local_best_cost, local_best_seq = cur_cost, seq[:]
                    # intensify: greedy insertion sweep around new best
                    for i2 in range(n):
                        best_c, best_cand = local_best_cost, None
                        for j2 in range(n):
                            if i2 == j2:
                                continue
                            trial = local_best_seq[:i2] + local_best_seq[i2+1:]
                            trial.insert(j2, local_best_seq[i2])
                            tc = workload.get_opt_seq_cost(trial)
                            if tc < best_c:
                                best_c, best_cand = tc, trial
                        if best_cand is not None:
                            local_best_cost = best_c
                            local_best_seq = best_cand
                    seq, cur_cost = local_best_seq[:], local_best_cost
        if local_best_cost < best_cost:
            best_cost, best_seq = local_best_cost, local_best_seq
        candidates.append((local_best_cost, local_best_seq))

    # ---------- Phase 3: LNS ruin-and-recreate on incumbent ----------
    candidates.sort(key=lambda x: x[0])
    inc_cost, inc_seq = candidates[0][0], candidates[0][1][:]
    if best_cost < inc_cost:
        inc_cost, inc_seq = best_cost, best_seq[:]

    def insertion_sweep(seq, cost, deadline):
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(len(seq)):
                if time.time() >= deadline:
                    return seq, cost
                txn = seq[i]
                rest = seq[:i] + seq[i+1:]
                bc, bpos = cost, None
                for j in range(len(seq)):
                    cand = rest[:j] + [txn] + rest[j:]
                    tc = workload.get_opt_seq_cost(cand)
                    if tc < bc:
                        bc, bpos = tc, j
                if bpos is not None:
                    seq = rest[:bpos] + [txn] + rest[bpos:]
                    cost = bc
                    improved = True
        return seq, cost

    while time.time() - start_time < time_budget:
        # ruin: remove a contiguous block of k transactions
        k = rng.randint(2, max(2, min(6, n // 4)))
        i = rng.randrange(0, n - k + 1)
        block = inc_seq[i:i+k]
        rest = inc_seq[:i] + inc_seq[i+k:]
        # recreate: greedily reinsert each removed txn at best position
        new_seq = rest
        for t in block:
            bc, bpos = None, None
            for j in range(len(new_seq) + 1):
                cand = new_seq[:j] + [t] + new_seq[j:]
                tc = workload.get_opt_seq_cost(cand)
                if bc is None or tc < bc:
                    bc, bpos = tc, j
            new_seq = new_seq[:bpos] + [t] + new_seq[bpos:]
        c = workload.get_opt_seq_cost(new_seq)
        if c < inc_cost:
            inc_cost, inc_seq = c, new_seq
            # intensify with a short insertion sweep
            inc_seq, inc_cost = insertion_sweep(inc_seq[:], inc_cost,
                                                time.time() + 1.0)
        elif c == inc_cost and rng.random() < 0.3:
            # sideways move for diversification
            inc_seq = new_seq

    if inc_cost < best_cost:
        best_cost, best_seq = inc_cost, inc_seq

    assert best_seq is not None and len(set(best_seq)) == n
    return best_cost, best_seq

# EVOLVE-BLOCK-END

def get_random_costs():
    workload_size = 100
    workload = Workload(WORKLOAD_1)

    makespan1, schedule1 = get_best_schedule(workload, 10)
    cost1 = workload.get_opt_seq_cost(schedule1)

    workload2 = Workload(WORKLOAD_2)
    makespan2, schedule2 = get_best_schedule(workload2, 10)
    cost2 = workload2.get_opt_seq_cost(schedule2)

    workload3 = Workload(WORKLOAD_3)
    makespan3, schedule3 = get_best_schedule(workload3, 10)
    cost3 = workload3.get_opt_seq_cost(schedule3)
    print(cost1, cost2, cost3)
    return cost1 + cost2 + cost3, [schedule1, schedule2, schedule3]


if __name__ == "__main__":
    makespan, schedule = get_random_costs()
