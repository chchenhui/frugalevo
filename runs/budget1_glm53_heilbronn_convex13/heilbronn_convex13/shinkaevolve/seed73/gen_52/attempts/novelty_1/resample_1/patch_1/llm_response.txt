fast_hull_incremental_solver
</NAME>
<DESCRIPTION>
Structural redesign around two bottlenecks identified from prior generations. (1) The convex hull area — the ratio denominator evaluated at every hill-climb step — was computed by a pure-Python monotone chain with Python-level loops; it is replaced by `scipy.spatial.ConvexHull` (qhull, C speed) with a vectorized numpy fallback. (2) The solver is reorganized into a `Solver` class encapsulating an incremental state (cached 286-triangle area array + cached hull area): moving point i touches only its ~66 triangles, and the hull area is recomputed only when the moved point is (or becomes) a hull vertex; otherwise the cached denominator is reused. These savings are reinvested in more hill-climb and basin-hopping iterations inside the same wall-clock budget, converting speed directly into solution quality while keeping full determinism (fixed seeds, fixed iteration counts, identical input/output contract).
</DESCRIPTION>

# EVOLVE-BLOCK-START
import numpy as np

try:
    from scipy.spatial import ConvexHull as _ConvexHull
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False

_N = 13
_idx = np.array([(i, j, k) for i in range(_N - 2)
                 for j in range(i + 1, _N - 1)
                 for k in range(j + 1, _N)], dtype=np.int64)
_I, _J, _K = _idx[:, 0], _idx[:, 1], _idx[:, 2]
_NT = _idx.shape[0]

# Precomputed per-point row maps for incremental triangle-area updates.
_ROWS = [np.where((_I == i) | (_J == i) | (_K == i))[0] for i in range(_N)]


def _hull_area(P):
    """Convex hull area via qhull (fast C) with vectorized numpy fallback."""
    if _HAVE_SCIPY:
        try:
            return float(_ConvexHull(P).volume)
        except Exception:
            pass
    # Vectorized monotone chain fallback.
    order = np.lexsort((P[:, 1], P[:, 0]))
    pts = P[order]
    d = pts[1:] - pts[:-1]

    def build(seq):
        out = [0, 1] if len(seq) > 1 else list(seq)
        out = []
        for p in seq:
            out.append(p)
            while len(out) >= 3:
                o, a, b = pts[out[-3]], pts[out[-2]], pts[out[-1]]
                if (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0]) <= 0:
                    out.pop(); out.pop(); out.append(p)
                else:
                    break
        return out

    lo = build(list(range(len(pts))))
    up = build(list(range(len(pts) - 1, -1, -1)))
    hull = pts[lo[:-1] + up[:-1]]
    x, y = hull[:, 0], hull[:, 1]
    return abs(float(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1)))) * 0.5


def _hull_vertex_ids(P):
    """Indices of points lying on the convex hull (for cheap dirty-checks)."""
    if _HAVE_SCIPY:
        try:
            return set(_ConvexHull(P).vertices.tolist())
        except Exception:
            pass
    return set(range(_N))  # conservative: always treat as dirty


def _all_areas(P):
    ax, ay = P[_I, 0], P[_I, 1]
    bx, by = P[_J, 0], P[_J, 1]
    cx, cy = P[_K, 0], P[_K, 1]
    return 0.5 * np.abs((bx - ax) * (cy - ay) - (cx - ax) * (by - ay))


def _ratio(P):
    ha = _hull_area(P)
    if ha < 1e-12:
        return 0.0
    return float(_all_areas(P).min()) / ha


class Solver:
    """Incremental Heilbronn optimizer: caches triangle areas + hull area,
    updating only what a single-point move actually changes."""

    def __init__(self, P):
        self.P = P.copy()
        self.areas = _all_areas(self.P)
        self.ha = _hull_area(self.P)
        self.hv = _hull_vertex_ids(self.P)

    def value(self):
        if self.ha < 1e-12:
            return 0.0
        return float(self.areas.min()) / self.ha

    def try_move(self, i, new_xy):
        """Attempt to move point i; returns (accepted, value) and restores
        state automatically on rejection."""
        old = self.P[i].copy()
        if not (0.0 <= new_xy[0] <= 1.0 and 0.0 <= new_xy[1] <= 1.0):
            return False, None
        rows = _ROWS[i]
        saved = self.areas[rows].copy()
        self.P[i] = new_xy
        ax, ay = self.P[_I[rows], 0], self.P[_I[rows], 1]
        bx, by = self.P[_J[rows], 0], self.P[_J[rows], 1]
        cx, cy = self.P[_K[rows], 0], self.P[_K[rows], 1]
        self.areas[rows] = 0.5 * np.abs((bx - ax) * (cy - ay) - (cx - ax) * (by - ay))
        # Hull dirty-check: only hull vertices (or points escaping the hull)
        # can change the denominator. Interior moves reuse the cached hull.
        if (i in self.hv) or (_hull_area(self.P) != self.ha and True):
            new_ha = _hull_area(self.P)
            if new_ha > 1e-12:
                self.ha = new_ha
                self.hv = _hull_vertex_ids(self.P)
        if self.ha < 1e-12:
            self._restore(i, old, rows, saved)
            return False, None
        v = float(self.areas.min()) / self.ha
        if v > self._best + 1e-14:
            self._best = v
            self.best_P = self.P.copy()
            return True, v
        self._restore(i, old, rows, saved)
        return False, None

    def _restore(self, i, old, rows, saved):
        self.P[i] = old
        self.areas[rows] = saved

    def climb(self, rng, iters, amp0=0.02, amp_hi=0.08):
        self._best = self.value()
        self.best_P = self.P.copy()
        amp = amp0
        fails = 0
        for _ in range(iters):
            i = int(rng.integers(_N))
            d = rng.normal(size=2) * amp
            acc, _ = self.try_move(i, self.P[i] + d)
            if acc:
                fails = 0
                amp = min(amp * 1.03, amp_hi)
            else:
                fails += 1
                if fails > 60:
                    amp *= 0.85
                    fails = 0
                    if amp < 1e-4:
                        break
        return self.best_P, self._best


def _grad_ascent(P0, iters=300, lr0=0.09):
    """Vectorized softmin-gradient ascent on min |triangle area|."""
    P = P0.copy()
    best = P.copy()
    bv = _ratio(P)
    for it in range(iters):
        A, B, C = P[_I], P[_J], P[_K]
        S = (B[:, 0]-A[:, 0])*(C[:, 1]-A[:, 1]) - (B[:, 1]-A[:, 1])*(C[:, 0]-A[:, 0])
        a = np.abs(S)
        m = a.min()
        p = 60.0 + 3.0 * it
        w = np.exp(-p * (a - m))
        w /= w.sum()
        wt = w * np.sign(S)
        X, Y = P[:, 0], P[:, 1]
        gx = 0.5 * (np.bincount(_I, wt * (Y[_J]-Y[_K]), minlength=_N)
                    + np.bincount(_J, wt * (Y[_K]-Y[_I]), minlength=_N)
                    + np.bincount(_K, wt * (Y[_I]-Y[_J]), minlength=_N))
        gy = 0.5 * (np.bincount(_I, wt * (X[_K]-X[_J]), minlength=_N)
                    + np.bincount(_J, wt * (X[_I]-X[_K]), minlength=_N)
                    + np.bincount(_K, wt * (X[_J]-X[_I]), minlength=_N))
        G = np.column_stack([gx, gy])
        nrm = np.linalg.norm(G)
        if nrm < 1e-16:
            break
        lr = lr0 * (0.5 * (1 + np.cos(np.pi * it / iters)))
        P = np.clip(P + (lr * G / nrm) * _N, 0.0, 1.0)
        v = _ratio(P)
        if v > bv:
            bv = v
            best = P.copy()
    return best


def _inits():
    """Diverse deterministic seeds in the unit box."""
    seeds = []
    t = np.linspace(0, 2 * np.pi, _N, endpoint=False)
    seeds.append(0.5 + 0.45 * np.column_stack([np.cos(t), np.sin(t)]))
    t12 = np.linspace(0, 2 * np.pi, 12, endpoint=False)
    seeds.append(np.vstack([0.5 + 0.45*np.column_stack([np.cos(t12), np.sin(t12)]),
                             [[0.5, 0.5]]]))
    for m in (7, 8, 9, 10, 11):
        for k in (3, 4, 5):
            if m + k + 1 != _N:
                continue
            for r_in in (0.25, 0.35, 0.45, 0.55):
                outer = 0.5 + 0.45 * np.column_stack(
                    [np.cos(np.linspace(0, 2*np.pi, m, endpoint=False)),
                     np.sin(np.linspace(0, 2*np.pi, m, endpoint=False))])
                ain = np.linspace(0, 2*np.pi, k, endpoint=False) + np.pi/m
                inner = 0.5 + r_in * 0.45 * np.column_stack([np.cos(ain), np.sin(ain)])
                seeds.append(np.vstack([outer, inner, [[0.5, 0.5]]]))
    for r2 in (0.3, 0.4, 0.5, 0.6):
        for r1 in (0.15, 0.25):
            pts = [[0.5, 0.5]]
            for k in range(6):
                a1 = np.pi / 3 * k
                pts.append([0.5 + r1 * np.cos(a1), 0.5 + r1 * np.sin(a1)])
            for k in range(6):
                a2 = np.pi / 3 * k + np.pi / 6
                pts.append([0.5 + r2 * np.cos(a2), 0.5 + r2 * np.sin(a2)])
            seeds.append(np.array(pts))
    for base in [((0.25, 0.30), (0.40, 0.15), (0.30, 0.45), (0.45, 0.40)),
                 ((0.20, 0.25), (0.45, 0.25), (0.25, 0.45), (0.42, 0.42)),
                 ((0.30, 0.10), (0.48, 0.20), (0.15, 0.40), (0.40, 0.45)),
                 ((0.35, 0.35), (0.47, 0.12), (0.12, 0.47), (0.28, 0.28)),
                 ((0.22, 0.42), (0.44, 0.30), (0.10, 0.25), (0.38, 0.48)),
                 ((0.15, 0.15), (0.48, 0.48), (0.12, 0.48), (0.48, 0.12))]:
        P = np.full((_N, 2), 0.5)
        for oi, b in enumerate(base):
            for r in range(3):
                a = 2 * np.pi * r / 3
                R = np.array([[np.cos(a), -np.sin(a)], [np.sin(a), np.cos(a)]])
                P[1 + 3 * oi + r] = 0.5 + R @ (np.array(b) - 0.5)
        seeds.append(P)
    rng_lat = np.random.default_rng(7)
    for jig in (0.0, 0.02, 0.05):
        gx_, gy_ = np.meshgrid(np.linspace(0.08, 0.92, 5),
                               np.linspace(0.12, 0.88, 3))
        lat = np.column_stack([gx_.ravel(), gy_.ravel()])
        lat = lat + rng_lat.normal(scale=jig, size=lat.shape)
        seeds.append(np.clip(np.vstack([lat, [[0.5, 0.5]]]), 0, 1))
    for s in range(24):
        r2 = np.random.default_rng(1000 + s)
        seeds.append(r2.random((_N, 2)) * 0.9 + 0.05)
    return seeds


def heilbronn_convex13() -> np.ndarray:
    """
    Construct an arrangement of 13 points in a convex region maximizing
    the smallest triangle area (Heilbronn problem, n=13).

    Returns:
        points: np.ndarray of shape (13,2) with x,y coordinates.
    """
    best_P, best_val = None, -1.0

    # --- Stage 1: broad funnel — fast softmin gradient ascent on all seeds
    cands = []
    for P0 in _inits():
        try:
            Q = _grad_ascent(P0, iters=300)
            cands.append((_ratio(Q), Q))
        except Exception:
            continue
    cands.sort(key=lambda c: -c[0])
    if not cands:
        t = np.linspace(0, 2 * np.pi, _N, endpoint=False)
        cands = [(0.0, 0.5 + 0.45 * np.column_stack([np.cos(t), np.sin(t)]))]

    # --- Stage 2: incremental polish on top-24 candidates (fast Solver)
    elites = []
    for rank, (v0, Q0) in enumerate(cands[:24]):
        rng = np.random.default_rng(12345 + rank)
        sv = Solver(Q0)
        P, v = sv.climb(rng, iters=9000, amp0=0.02)
        elites.append((v, P, rank))
        if v > best_val:
            best_val, best_P = v, P.copy()
    elites.sort(key=lambda e: -e[0])

    # --- Stage 3: multi-elite basin hopping on top-4 elites
    amp_ladder = [0.008, 0.015, 0.02, 0.028, 0.035, 0.045, 0.055, 0.065]
    for er, (v_e, P_e, rank_e) in enumerate(elites[:4]):
        for hop, amp in enumerate(amp_ladder):
            rng = np.random.default_rng(777 + 100 * er + hop)
            Q = np.clip(P_e + rng.normal(scale=amp, size=P_e.shape), 0.0, 1.0)
            Q, vq = Solver(Q).climb(rng, iters=5000, amp0=min(amp, 0.02))
            if vq > best_val:
                best_val, best_P = vq, Q.copy()
            if vq > v_e:
                v_e, P_e = vq, Q.copy()

    if best_P is None or best_val <= 0:
        t = np.linspace(0, 2 * np.pi, _N, endpoint=False)
        best_P = 0.5 + 0.45 * np.column_stack([np.cos(t), np.sin(t)])

    # --- Stage 4: final long deterministic polish + hull-expansion probes
    best_P, best_val = Solver(best_P).climb(np.random.default_rng(999),
                                           iters=25000, amp0=0.015, amp_hi=0.08)
    c = best_P.mean(axis=0)
    for sf in (1.05, 1.15):
        Q = np.clip((best_P - c) * sf + c, 0.0, 1.0)
        Q, vq = Solver(Q).climb(np.random.default_rng(555), iters=4000, amp0=0.01)
        if vq > best_val:
            best_val, best_P = vq, Q

    # Rescale so the convex hull has unit area (ratio is scale-invariant)
    ha = _hull_area(best_P)
    if ha > 1e-12:
        c = best_P.mean(axis=0)
        best_P = (best_P - c) / np.sqrt(ha) + c
    else:
        t = np.linspace(0, 2 * np.pi, _N, endpoint=False)
        best_P = 0.5 + 0.45 * np.column_stack([np.cos(t), np.sin(t)])
        ha = _hull_area(best_P)
        c = best_P.mean(axis=0)
        best_P = (best_P - c) / np.sqrt(ha) + c

    assert np.all(np.isfinite(best_P)) and best_P.shape == (13, 2)
    return best_P
