    # 3 outer vertices (hull), 6 mid-ring, 3 inner-ring, 1 center.
    # Coordinates are defined on a rough unit scale and then rescaled so the
    # convex hull area is exactly 1.
    import numpy as _np
    import math as _math

    _outer_r = 1.0
    _mid_r1 = 0.62
    _mid_r2 = 0.50
    _inner_r = 0.28

    _raw = []
    # Outer triangle (convex hull vertices) at 0, 120, 240 degrees
    for _k in range(3):
        _a = 2.0 * _math.pi * _k / 3.0
        _raw.append((_outer_r * _math.cos(_a), _outer_r * _math.sin(_a)))
    # Mid ring: 6 points alternating between two radii, offset by 60 deg
    for _k in range(6):
        _a = _math.pi / 3.0 * _k + _math.pi / 6.0
        _r = _mid_r1 if _k % 2 == 0 else _mid_r2
        _raw.append((_r * _math.cos(_a), _r * _math.sin(_a)))
    # Inner ring: 3 points at 90-degree offsets (breaking exact alignment
    # with outer vertices to avoid near-collinear triples)
    for _k in range(3):
        _a = 2.0 * _math.pi * _k / 3.0 + _math.pi / 2.0
        _raw.append((_inner_r * _math.cos(_a), _inner_r * _math.sin(_a)))
    # Center point
    _raw.append((0.0, 0.0))

    _P = _np.array(_raw, dtype=float)

    # Normalize so the convex hull (outer triangle) has area exactly 1.
    # Outer triangle side length for circumradius 1 is sqrt(3), area = 3*sqrt(3)/4

    _scale = 1.0 / _math.sqrt(_tri_area)
