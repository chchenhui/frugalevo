<DESCRIPTION>
Combines the soft-min subgradient ascent from the current program with the 16-direction pattern-search refinement stage from the higher-scoring crossover parent (0.49 vs 0.21), and adds a multi-start loop over several deterministic initializations (varying ring radii/jitters) with the best result kept. The pattern-search stage directly optimizes the true min-triangle objective and is the key contributor to the better score; the multi-start helps escape the poor local optimum that limited the parent. All randomness is seeded for determinism.

# EVOLVE-BLOCK-START
import numpy as np


def heilbronn_convex13() -> np.ndarray:
    """
    Construct an arrangement of n points on or inside a convex region in order to maximize the area of the
    smallest triangle formed by these points. Here n = 13.

    Returns:
        points: np.ndarray of shape (13,2) with the x,y coordinates of the points.
    """
    n = 13

    # Unit-area regular hexagon centered at origin as the convex region.
    R = np.sqrt(2.0 / (3.0 * np.sqrt(3.0)))
    hex_dir = np.array([np.cos(np.pi / 3 * k + np.pi / 6) for k in range(6)])
    hex_y = np.array([np.sin(np.pi / 3 * k + np.pi / 6) for k in range(6)])

    def clip(pts):
        out = pts.copy()
        for k in range(6):
            nx, ny = hex_dir[k], hex_y[k]
            d = out[:, 0] * nx + out[:, 1] * ny - R
            viol = d > 1e-12
            if np.any(viol):
                out[viol, 0] -= d[viol] * nx
                out[viol, 1] -= d[viol] * ny
        return out

    idx = np.array([(i, j, k) for i in range(n) for j in range(i + 1, n) for k in range(j + 1, n)])
    I0, I1, I2 = idx[:, 0], idx[:, 1], idx[:, 2]

    def min_tri(p):
        a = p[I0]
        b = p[I1]
        c = p[I2]
        return 0.5 * np.abs((b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) -
                            (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0]))

    def min_area_val(p):
        return min_tri(p).min()

    ang = np.arange(16) * np.pi / 16.0
    dirs = np.stack([np.cos(ang), np.sin(ang)], axis=1)

    def gradient_stage(pts):
        step = 0.02 * R
        best_pts = pts.copy()
        best_val = min_tri(pts).min()
        for it in range(400):
            areas = min_tri(pts)
            m = areas.min()
            if m > best_val:
                best_val = m
                best_pts = pts.copy()
            w = np.exp(-(areas - m) / max(1e-6, 0.02 * m + 1e-9))
            w /= w.sum()
            grads = np.zeros_like(pts)
            a = pts[I0]
            b = pts[I1]
            c = pts[I2]
            s = np.sign((b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) -
                        (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0]))
            ga = 0.5 * s[:, None] * np.stack([c[:, 1] - b[:, 1], b[:, 0] - c[:, 0]], axis=1)
            gb = 0.5 * s[:, None] * np.stack([a[:, 1] - c[:, 1], c[:, 0] - a[:, 0]], axis=1)
            gc = 0.5 * s[:, None] * np.stack([b[:, 1] - a[:, 1], a[:, 0] - b[:, 0]], axis=1)
            np.add.at(grads, I0, w[:, None] * ga)
            np.add.at(grads, I1, w[:, None] * gb)
            np.add.at(grads, I2, w[:, None] * gc)
            gn = np.linalg.norm(grads, axis=1, keepdims=True)
            gn[gn < 1e-12] = 1.0
            grads /= gn
            pts = clip(pts + step * grads)
            step *= 0.99
        return clip(best_pts)

    def pattern_stage(pts):
        cur_val = min_area_val(pts)
        step2 = 0.01 * R
        while step2 > 1e-5 * R:
            improved = False
            for i in range(n):
                cand = np.repeat(pts[i][None, :], 16, axis=0) + step2 * dirs
                cand = clip(cand)
                vals = np.empty(16)
                base = pts.copy()
                for d in range(16):
                    base[i] = cand[d]
                    vals[d] = min_tri(base).min()
                d = int(np.argmax(vals))
                if vals[d] > cur_val + 1e-15:
                    pts[i] = cand[d]
                    cur_val = vals[d]
                    improved = True
            if not improved:
                step2 *= 0.5
        return clip(pts), cur_val

    rng = np.random.default_rng(seed=42)
    best_global_pts = None
    best_global_val = -np.inf

    # Multi-start over a few deterministic initializations.
    starts = [(0.55, 0.95, 0.02), (0.50, 0.90, 0.03), (0.60, 1.00, 0.01)]
    for si, (r1f, r2f, jit) in enumerate(starts):
        rng_s = np.random.default_rng(seed=1000 + si)
        pts = np.zeros((n, 2))
        pts[0] = 0.0
        for i in range(6):
            a1 = np.pi / 3 * i
            a2 = np.pi / 3 * i + np.pi / 6
            pts[1 + i] = r1f * R * np.array([np.cos(a1), np.sin(a1)])
            pts[7 + i] = r2f * R * np.array([np.cos(a2), np.sin(a2)])
        pts += jit * R * rng_s.standard_normal((n, 2))
        pts = clip(pts)

        pts = gradient_stage(pts)
        pts, val = pattern_stage(pts)

        if val > best_global_val:
            best_global_val = val
            best_global_pts = pts.copy()

    pts = best_global_pts
    assert np.all(np.isfinite(pts))
    assert pts.shape == (13, 2)
    return pts


# EVOLVE-BLOCK-END
