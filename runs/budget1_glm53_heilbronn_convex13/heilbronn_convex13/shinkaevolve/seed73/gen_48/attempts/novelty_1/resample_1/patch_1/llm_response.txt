<DESCRIPTION>
The plateau at min_area_normalized ≈ 0.03 comes from two issues: (1) the softmin-free pattern search accepts moves on min-area alone, ignoring that score = min_area / hull_area, and (2) the hull denominator is never actively minimized. Key changes: (a) every accepted move in both the base refinement and the full polish is now gated on the exact true score (min_area / hull_area), with scipy's ConvexHull for fast hull-area evaluation; (b) a new hull-reduction move set — hull vertices get extra inward radial and centroid-pulling candidate directions (multiplicative contraction per hull point), letting the optimizer shrink the hull while holding min area, which directly raises the ratio; (c) periodic global contraction attempts (scale points toward centroid, repeat while the true score improves); (d) reduced seed count and a tighter polish min_step (5e-4) plus a bounded basin-hopping stage (2 elites × 3 amplitudes) to keep eval_time well under the previous 30s. All moves deterministic; seed 42 retained.

import numpy as np
from itertools import combinations

try:
    from scipy.spatial import ConvexHull as _ConvexHull
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False

n = 13
_IDX = np.array(list(combinations(range(n), 3)))
_I0, _I1, _I2 = _IDX[:, 0], _IDX[:, 1], _IDX[:, 2]


def _tri_areas(P: np.ndarray) -> np.ndarray:
    a = P[_I0]
    b = P[_I1]
    c = P[_I2]
    return 0.5 * np.abs((b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) -
                        (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0]))


def _hull_area(P: np.ndarray) -> float:
    if _HAVE_SCIPY and len(P) >= 3:
        try:
            return float(_ConvexHull(P).volume)
        except Exception:
            pass
    pts = P[np.lexsort((P[:, 1], P[:, 0]))]

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower, upper = [], []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    for p in pts[::-1]:
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 3:
        return 1e-12
    h = np.array(hull)
    x, y = h[:, 0], h[:, 1]
    return float(0.5 * abs(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1))))


def _true_score(P: np.ndarray) -> float:
    ha = _hull_area(P)
    if ha <= 1e-12:
        return 0.0
    return float(_tri_areas(P).min()) / ha


_thetas = np.arange(16) * (np.pi / 8.0)
_DIRS = np.stack([np.cos(_thetas), np.sin(_thetas)], axis=1)

_th = 2.0 * np.pi / 3.0
_c, _s = np.cos(_th), np.sin(_th)
_ROT = np.array([[_c, -_s], [_s, _c]])
_ROT2 = _ROT @ _ROT


def _expand(B: np.ndarray) -> np.ndarray:
    return np.vstack([B, B @ _ROT.T, B @ _ROT2.T, [[0.0, 0.0]]])


def _contract_global(P: np.ndarray, factor: float = 0.99) -> np.ndarray:
    """Pull ALL points toward the centroid by `factor`; repeat while the
    true score (min_area / hull_area) improves. Hull area shrinks
    quadratically while min area shrinks quadratically too, but when the
    extreme (hull-defining) points are the ones limiting hull size this
    move can strictly improve the ratio."""
    cen = P.mean(axis=0)
    best = _true_score(P)
    Q = cen + factor * (P - cen)
    sc = _true_score(Q)
    if sc <= best + 1e-15:
        return P
    for _ in range(200):
        Q2 = cen + factor * (Q - cen)
        sc2 = _true_score(Q2)
        if sc2 > sc + 1e-15:
            Q, sc = Q2, sc2
        else:
            break
    return Q


def _hull_vertex_pull(P: np.ndarray, step: float) -> np.ndarray:
    """One pass: try pulling each hull vertex toward the centroid by
    `step` (additively); accept each pull if the TRUE score improves."""
    try:
        if _HAVE_SCIPY:
            hv = _ConvexHull(P).vertices
        else:
            hv = range(len(P))
    except Exception:
        return P
    cen = P.mean(axis=0)
    for v in np.asarray(hv):
        d = cen - P[v]
        norm = np.linalg.norm(d)
        if norm < 1e-12:
            continue
        d = d / norm
        for sgn in (1.0, -1.0):
            Q = P.copy()
            Q[v] = P[v] + sgn * step * d
            if _true_score(Q) > _true_score(P) + 1e-15:
                P = Q
                break
    return P


def _refine_base(B: np.ndarray, step: float = 0.2,
                 min_step: float = 1e-3) -> np.ndarray:
    """Pattern search on the 8 base coords, gated on the TRUE score."""
    P = _expand(B)
    best = _true_score(P)
    while step > min_step:
        improved = False
        for i in range(len(B)):
            for j in range(2):
                for sgn in (1.0, -1.0):
                    cand = B.copy()
                    cand[i, j] += sgn * step
                    val = _true_score(_expand(cand))
                    if val > best + 1e-14:
                        B = cand
                        best = val
                        improved = True
        if not improved:
            step *= 0.5
    return B


def _polish_full(P: np.ndarray, step: float = 0.02,
                 min_step: float = 5e-4) -> np.ndarray:
    """Symmetry-breaking pattern search on all 26 coordinates, with every
    candidate evaluated on the exact min_area / hull_area score."""
    best = _true_score(P)
    P = _contract_global(P)
    best = _true_score(P)
    rounds = 0
    while step > min_step:
        improved = False
        for i in range(n):
            for d in range(16):
                trial = P.copy()
                trial[i] = P[i] + step * _DIRS[d]
                val = _true_score(trial)
                if val > best + 1e-14:
                    P = trial
                    best = val
                    improved = True
        rounds += 1
        if rounds % 2 == 0:
            Q = _contract_global(P)
            sc = _true_score(Q)
            if sc > best + 1e-14:
                P, best = Q, sc
                improved = True
        if not improved:
            step *= 0.5
    return P


def heilbronn_convex13() -> np.ndarray:
    rng = np.random.default_rng(seed=42)

    seeds = []
    ang = np.pi / 2.0
    for r4 in ([1.0, 0.72, 0.45, 0.2], [1.0, 0.8, 0.55, 0.3],
               [1.0, 0.85, 0.6, 0.35], [1.0, 0.75, 0.5, 0.25],
               [1.0, 0.78, 0.5, 0.15], [0.95, 0.7, 0.42, 0.1]):
        B = [[r * np.cos(ang - k * np.pi / 9.0),
              r * np.sin(ang - k * np.pi / 9.0)] for k, r in enumerate(r4)]
        seeds.append(np.array(B))
    for spread in (0.95, 0.75, 0.5, 0.3):
        seeds.append(np.array([[spread, 0.0],
                               [spread * 0.5, spread * 0.85],
                               [0.0, spread],
                               [spread * 0.4, spread * 0.3]]))
    for _ in range(12):
        r = rng.uniform(0.1, 1.0, 4)
        a = rng.uniform(0, 2 * np.pi, 4)
        seeds.append(np.column_stack([r * np.cos(a), r * np.sin(a)]))

    best_P, best_sc = None, -np.inf
    elites = []
    for B0 in seeds:
        try:
            B = _refine_base(B0.copy())
            P = _expand(B)
            P = _polish_full(P)
            P = _hull_vertex_pull(P, step=0.01)
            P = _polish_full(P, step=0.005, min_step=5e-4)
            sc = _true_score(P)
            if sc > best_sc:
                best_sc, best_P = sc, P.copy()
            elites.append((sc, P))
        except Exception:
            continue

    # deterministic basin hopping around top-2 elites
    elites.sort(key=lambda t: -t[0])
    kick_dirs = np.stack(
        [np.cos(np.arange(n) * 2.399963),
         np.sin(np.arange(n) * 2.399963)], axis=1)
    kick_dirs /= np.linalg.norm(kick_dirs, axis=1, keepdims=True)
    for sc0, elite in elites[:2]:
        for amp in (0.01, 0.03, 0.06):
            for sgn in (1.0, -1.0):
                try:
                    Q = elite + sgn * amp * kick_dirs
                    Q = _contract_global(Q)
                    Q = _polish_full(Q, step=0.01, min_step=5e-4)
                    sc = _true_score(Q)
                    if sc > best_sc + 1e-15:
                        best_sc, best_P = sc, Q.copy()
                except Exception:
                    continue

    if best_P is None:
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        best_P = np.column_stack([np.cos(t), np.sin(t)])

    for i in range(n):
        for j in range(i + 1, n):
            if np.linalg.norm(best_P[i] - best_P[j]) < 1e-9:
                best_P[j, 0] += 1e-6 * (j + 1)

    assert np.all(np.isfinite(best_P))
    assert best_P.shape == (n, 2)
    return best_P
