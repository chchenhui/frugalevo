<DESCRIPTION>
Key changes from the current approach: (1) Add hull-expansion moves inside the hill climb — periodically try scaling all points by 1.01x and accept only if the true min/hull score improves, letting the optimizer shrink hull area when possible. (2) Use a sharper guidance blend (0.8·min + 0.2·mean of 3 smallest) for hill-climb guidance while final selection still uses the exact min/hull ratio. (3) Expand the seed set (more random restarts plus 6-fold lattice-like seeds), use 8-directional (diagonal) moves instead of axis-only for faster/better local search, and an adaptive step schedule with faster geometric decay (0.6x) to reach fine precision quickly. Together these widen global coverage and exploit hull-shrinking, targeting a higher true normalized minimum triangle area at modest runtime.

import numpy as np
from itertools import combinations


def _tri_areas(P: np.ndarray) -> np.ndarray:
    idx = np.array(list(combinations(range(len(P)), 3)))
    A = P[idx[:, 0]]
    B = P[idx[:, 1]]
    C = P[idx[:, 2]]
    return 0.5 * np.abs(
        (B[:, 0] - A[:, 0]) * (C[:, 1] - A[:, 1])
        - (B[:, 1] - A[:, 1]) * (C[:, 0] - A[:, 0])
    )


def _hull_area(P: np.ndarray) -> float:
    pts = P[np.lexsort((P[:, 1], P[:, 0]))]
    if len(pts) < 3:
        return 1e-12

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 3:
        return 1e-12
    s = 0.0
    for i in range(len(hull)):
        x1, y1 = hull[i]
        x2, y2 = hull[(i + 1) % len(hull)]
        s += x1 * y2 - x2 * y1
    return abs(s) / 2.0


def heilbronn_convex13() -> np.ndarray:
    n = 13
    rng = np.random.default_rng(seed=7)

    th = 2.0 * np.pi / 3.0
    c, s = np.cos(th), np.sin(th)
    R = np.array([[c, -s], [s, c]])       # rotate by +120 deg
    R2 = R @ R

    # 8 move directions (axis + diagonal) for richer local search
    thetas = np.arange(8) * (np.pi / 8.0)
    DIRS = np.stack([np.cos(thetas), np.sin(thetas)], axis=1)

    def expand(B: np.ndarray) -> np.ndarray:
        # 4 base points -> 12 points by 3-fold rotation, plus center at origin
        return np.vstack([B, B @ R.T, B @ R2.T, [[0.0, 0.0]]])

    def score(P: np.ndarray) -> float:
        a = _tri_areas(P)
        h = _hull_area(P)
        if h <= 1e-12:
            return 0.0
        # sharper blend: mostly min, small tie-break from 3 smallest
        k = np.partition(a, 2)[:3]
        return (0.8 * a.min() + 0.2 * k.mean()) / h

    def true_score(P: np.ndarray) -> float:
        a = _tri_areas(P)
        h = _hull_area(P)
        return a.min() / h if h > 1e-12 else 0.0

    def hill_climb(P: np.ndarray, steps_max: int = 20,
                   allow_expand: bool = True) -> np.ndarray:
        step = 0.05
        best = score(P)
        ts_best = true_score(P)
        for _ in range(steps_max):
            improved = False
            # per-point moves along 8 directions
            for i in range(len(P)):
                for d in DIRS:
                    for sd in (step, 0.5 * step):
                        Q = P.copy()
                        Q[i] = Q[i] + sd * d
                        sc = score(Q)
                        if sc > best + 1e-13:
                            best = sc
                            P = Q
                            improved = True
            # hull-expansion move: scale all points, accept if TRUE score improves
            if allow_expand:
                for lam in (1.01, 0.99, 1.03, 0.97):
                    Q = P * lam
                    ts = true_score(Q)
                    if ts > ts_best + 1e-13:
                        ts_best = ts
                        P = Q
                        best = score(Q)
                        improved = True
            if not improved:
                step *= 0.6
                if step < 2e-4:
                    break
        return P

    # ---- Seeds for the 4 base points (3-fold rotation generates full config)
    seeds = []
    ang = np.pi / 2.0
    for r4 in ([1.0, 0.72, 0.45, 0.2], [1.0, 0.8, 0.55, 0.3],
               [1.0, 0.85, 0.6, 0.35], [1.0, 0.75, 0.5, 0.25],
               [1.0, 0.9, 0.7, 0.5], [0.95, 0.7, 0.45, 0.15]):
        B = []
        for k, r in enumerate(r4):
            a = ang - k * np.pi / 9.0
            B.append([r * np.cos(a), r * np.sin(a)])
        seeds.append(np.array(B))
    # spread along one edge direction
    for spread in (0.9, 0.6, 0.3):
        B = np.array([[spread, 0.0],
                      [spread * 0.5, spread * 0.85],
                      [0.0, spread],
                      [spread * 0.4, spread * 0.3]])
        seeds.append(B)
    # 6-fold-lattice-like seed (hexagonal packing flavor)
    B6 = np.array([[1.0, 0.0], [0.5, 0.5 * np.sqrt(3)],
                   [0.0, np.sqrt(3) / 3], [0.6, 0.35]])
    seeds.append(B6)
    # random seeds (more restarts for wider coverage)
    for _ in range(40):
        r = rng.uniform(0.1, 1.0, 4)
        a = rng.uniform(0, 2 * np.pi, 4)
        seeds.append(np.column_stack([r * np.cos(a), r * np.sin(a)]))

    best_P, best_ts = None, -1.0
    for B0 in seeds:
        try:
            P = expand(B0)
            P = hill_climb(P, steps_max=20)          # symmetric search w/ hull moves
            P = hill_climb(P, steps_max=8,
                           allow_expand=False)       # symmetry-breaking polish
            ts = true_score(P)
            if ts > best_ts:
                best_ts = ts
                best_P = P.copy()
        except Exception:
            continue

    if best_P is None:
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        best_P = np.column_stack([np.cos(t), np.sin(t)])

    # numerical safety: no duplicates
    for i in range(n):
        for j in range(i + 1, n):
            if np.linalg.norm(best_P[i] - best_P[j]) < 1e-9:
                best_P[j, 0] += 1e-6 * (j + 1)
    return best_P
