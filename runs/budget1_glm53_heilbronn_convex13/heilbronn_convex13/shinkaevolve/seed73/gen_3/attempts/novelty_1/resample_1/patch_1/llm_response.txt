<DESCRIPTION>
Replace the random point generation with a deterministic multi-restart projected gradient ascent optimizer. The objective is a smooth soft-min (LogSumExp) approximation of the smallest triangle area over all 286 triangles, with a sharpening (annealing) parameter p that increases over iterations so the soft-min converges to the true minimum. Gradients are computed by vectorized central finite differences over all 26 coordinates in a single batched area evaluation (52 perturbed configurations at once), which is very cheap. Points are projected back onto the unit square each step (the problem is scale-invariant, and fixing the containing region makes maximizing raw min-area equivalent to maximizing the normalized ratio up to hull shape). Multiple restarts (random plus structured seeds like the regular 13-gon and 12-gon+center, which exploit the known 3-fold/rotational symmetry heuristics) are evaluated with the true metric (smallest triangle area / convex hull area), and the best configuration is returned. A fixed seed guarantees reproducibility. This should raise the normalized minimum triangle area from ~0.00 to the ~0.02-0.03 range, pushing combined_score near or above 1.0.

import numpy as np
from itertools import combinations


def _hull_area(P: np.ndarray) -> float:
    """Area of the convex hull of the points (monotone chain)."""
    pts = sorted(set(map(tuple, np.round(P, 12))))
    if len(pts) < 3:
        return 1e-12

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 3:
        return 1e-12
    s = 0.0
    for i in range(len(hull)):
        x1, y1 = hull[i]
        x2, y2 = hull[(i + 1) % len(hull)]
        s += x1 * y2 - x2 * y1
    return abs(s) / 2.0


def heilbronn_convex13() -> np.ndarray:
    """
    Construct an arrangement of n points on or inside a convex region in order to maximize the area of the
    smallest triangle formed by these points. Here n = 13.

    Uses deterministic multi-restart projected gradient ascent on a smooth
    soft-min (LogSumExp) approximation of the smallest triangle area, with
    an annealed sharpness parameter, then selects the restart with the best
    true normalized score (min triangle area / convex hull area).

    Returns:
        points: np.ndarray of shape (13,2) with the x,y coordinates of the points.
    """
    n = 13
    rng = np.random.default_rng(seed=42)

    idx = np.array(list(combinations(range(n), 3)))
    I, J, K = idx[:, 0], idx[:, 1], idx[:, 2]

    def areas_batch(Ps: np.ndarray) -> np.ndarray:
        """Areas of all C(13,3) triangles for a batch of point sets."""
        A = Ps[:, I]
        B = Ps[:, J]
        C = Ps[:, K]
        return 0.5 * np.abs(
            (B[..., 0] - A[..., 0]) * (C[..., 1] - A[..., 1])
            - (B[..., 1] - A[..., 1]) * (C[..., 0] - A[..., 0])
        )

    def optimize(P0: np.ndarray, iters: int = 220) -> np.ndarray:
        P = P0.copy()
        h = 1e-5
        lr = 0.05
        for it in range(iters):
            a0 = areas_batch(P[None])[0]
            m = a0.min()
            p = 25.0 + 1.5 * it  # anneal softmin sharpness
            w = np.exp(-p * (a0 - m))
            w /= w.sum()
            # Vectorized central finite differences over all 26 coordinates.
            Ps = np.tile(P, (2 * n * 2, 1, 1))
            c = 0
            for i in range(n):
                for j in range(2):
                    Ps[2 * c, i, j] += h
                    Ps[2 * c + 1, i, j] -= h
                    c += 1
            ab = areas_batch(Ps)
            da = (ab[0::2] - ab[1::2]) / (2 * h)
            grad = da @ w
            G = grad.reshape(n, 2)
            nrm = np.linalg.norm(G)
            if nrm > 1e-14:
                P = P + (lr * G / nrm) * n
            # Project back into the unit square (scale-invariant problem).
            P = np.clip(P, -1.0, 1.0)
            lr *= 0.995
        return P

    # Structured seeds exploiting known symmetry heuristics + random restarts.
    seeds = []
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)
    seeds.append(np.column_stack([np.cos(t), np.sin(t)]))          # regular 13-gon
    t12 = np.linspace(0, 2 * np.pi, 12, endpoint=False)
    ring = np.column_stack([np.cos(t12), np.sin(t12)])
    seeds.append(np.vstack([ring, [[0.0, 0.0]]]))                  # 12-ring + center
    hexpts = []
    for r, cnt in ((1.0, 6), (0.55, 6)):
        for k in range(cnt):
            a = 2 * np.pi * k / cnt + (0.0 if r == 1.0 else np.pi / 6)
            hexpts.append([r * np.cos(a), r * np.sin(a)])
    seeds.append(np.array(hexpts) * 0.9 + [0.0, 0.0])              # hexagonal-ish

    best_P, best_score = None, -1.0
    for r in range(10):
        seeds.append(rng.uniform(-0.9, 0.9, (n, 2)))

    for P0 in seeds:
        try:
            P = optimize(P0)
            a = areas_batch(P[None])[0]
            ha = _hull_area(P)
            if ha <= 0:
                continue
            score = a.min() / ha
            if score > best_score:
                best_score = score
                best_P = P.copy()
        except Exception:
            continue

    if best_P is None:
        # Fallback: regular 13-gon (always feasible, deterministic).
        best_P = np.column_stack([np.cos(t), np.sin(t)])

    # Ensure no duplicate points (numerical safety).
    for i in range(n):
        for j in range(i + 1, n):
            if np.linalg.norm(best_P[i] - best_P[j]) < 1e-9:
                best_P[j, 0] += 1e-6 * (j + 1)
    return best_P
