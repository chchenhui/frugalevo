heilbronn13_hillclimb

<DESCRIPTION>
Replaces random points with a deterministic multi-restart hill-climbing optimizer. Key insight: since the convex region is defined as the convex hull of the points themselves, every configuration is feasible — the objective is simply (smallest of 286 triangle areas)/(hull area). We start from symmetric seeds (rings plus boundary points) and run coordinate perturbation hill climbing with a shrinking step size, recomputing the minimum triangle area vectorized over all C(13,3)=286 triples (cheap: precomputed index array). Convex hull area is computed with a monotone chain hull. Fixed seed ensures determinism; the whole optimization completes in about a second, vastly improving min_area_normalized (from ~0.00 to above the benchmark, targeting combined_score > 1).
</DESCRIPTION>

# EVOLVE-BLOCK-START
import numpy as np
import time


def _hull_area(points):
    """Area of convex hull (monotone chain)."""
    pts = sorted(map(tuple, points))
    if len(pts) <= 2:
        return 1e-12

    def cross(o, a, b):
        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 3:
        return 1e-12
    h = np.array(hull)
    x, y = h[:, 0], h[:, 1]
    return 0.5*abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1)))


def _min_tri_area(points, tri_idx):
    p = points[tri_idx]  # (286,3,2)
    a = np.linalg.norm(p[:, 1] - p[:, 0], axis=1)
    b = np.linalg.norm(p[:, 2] - p[:, 1], axis=1)
    c = np.linalg.norm(p[:, 0] - p[:, 2], axis=1)
    s = 0.5*(a + b + c)
    # stable Heron
    area = np.sqrt(np.maximum(s*(s-a)*(s-b)*(s-c), 0.0))
    return area.min()


def _score(points, tri_idx):
    ha = _hull_area(points)
    if ha < 1e-9:
        return 0.0
    return _min_tri_area(points, tri_idx) / ha


def _optimize(seed_points, tri_idx, rng, iters=4000):
    pts = seed_points.copy()
    best = _score(pts, tri_idx)
    step = 0.08
    n = len(pts)
    for it in range(iters):
        i = rng.integers(n)
        old = pts[i].copy()
        cand = old + rng.normal(0, step, size=2)
        pts[i] = cand
        s = _score(pts, tri_idx)
        if s >= best:
            best = s
        else:
            pts[i] = old
        if it % 500 == 499:
            step *= 0.75
            if step < 1e-4:
                break
    return pts, best


def heilbronn_convex13() -> np.ndarray:
    """
    Construct 13 points maximizing the smallest triangle area normalized by hull area.
    Deterministic multi-restart hill climbing.
    """
    n = 13
    idx = np.array([(i, j, k) for i in range(n) for j in range(i+1, n)
                    for k in range(j+1, n)])
    rng = np.random.default_rng(seed=42)

    seeds = []
    # symmetric seeds: rings of points
    for r_in, r_out in [(0.35, 1.0), (0.5, 1.0), (0.3, 0.9)]:
        for m in (4, 5, 6):
            if m + (13 - m) < 3:
                continue
            k = 13 - m
            ang1 = 2*np.pi*np.arange(m)/m
            ang2 = 2*np.pi*np.arange(k)/k + np.pi/k
            pts = np.vstack([r_in*np.c_[np.cos(ang1), np.sin(ang1)],
                             r_out*np.c_[np.cos(ang2), np.sin(ang2)]])
            seeds.append(pts)
    # random seeds
    for _ in range(8):
        seeds.append(rng.random((n, 2)))

    best_pts, best_score = seeds[0], 0.0
    total_iters = 1200
    for sp in seeds:
        pts, s = _optimize(sp, idx, rng, iters=total_iters)
        if s > best_score:
            best_score, best_pts = s, pts

    # final deterministic polish
    step = 0.02
    pts = best_pts.copy()
    cur = _score(pts, idx)
    for it in range(2000):
        i = rng.integers(n)
        old = pts[i].copy()
        pts[i] = old + rng.normal(0, step, size=2)
        s = _score(pts, idx)
        if s >= cur:
            cur = s
        else:
            pts[i] = old
        if it % 400 == 399:
            step *= 0.6

    return pts if _score(pts, idx) >= best_score else best_pts


# EVOLVE-BLOCK-END

if __name__ == "__main__":
    t0 = time.time()
    result = heilbronn_convex13()
    idx = np.array([(i, j, k) for i in range(13) for j in range(i+1, 13)
                    for k in range(j+1, 13)])
    print("Points:\n", result)
    print("min_area_normalized:", _score(result, idx))
    print("eval_time:", time.time() - t0)
