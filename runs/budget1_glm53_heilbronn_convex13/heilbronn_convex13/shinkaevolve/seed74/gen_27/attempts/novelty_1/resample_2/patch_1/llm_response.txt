softmin_gradient_continuation

<DESCRIPTION>
Instead of stochastic Metropolis annealing, this uses analytic gradient ascent on a smooth soft-minimum surrogate: weights w_t = softmax(-alpha * area_t) over all 286 triangles, with exact analytic gradients of the signed triangle areas (grad wrt vertex v = 0.5*w*sign*R(p_u - p_w)). A continuation schedule increases alpha over time so the surrogate tightens from "raise the average small triangles" to "raise the true minimum", sharpening the basin of attraction without random-walk noise. A per-restart adaptive step controller (grow 1.05 on improvement, shrink 0.7 on regression) replaces fixed cooling ladders. Periodic rescaling to unit hull area makes the optimized quantity equal to the normalized score directly. Multi-start over structured inits, followed by a short targeted greedy polish on the K smallest triangles to squeeze the exact minimum. This should outperform annealing because every step follows the exact gradient direction of the binding constraint set rather than relying on random proposals, converging in fewer, better-directed iterations.

import numpy as np
from itertools import combinations

N = 13
TRIPLES = np.array(list(combinations(range(N), 3)), dtype=int)
NT = len(TRIPLES)


def _areas(points: np.ndarray) -> np.ndarray:
    p = points[TRIPLES[:, 0]]
    q = points[TRIPLES[:, 1]]
    r = points[TRIPLES[:, 2]]
    return 0.5 * np.abs(
        (q[:, 0] - p[:, 0]) * (r[:, 1] - p[:, 1])
        - (q[:, 1] - p[:, 1]) * (r[:, 0] - p[:, 0])
    )


def _signed(points: np.ndarray) -> np.ndarray:
    p = points[TRIPLES[:, 0]]
    q = points[TRIPLES[:, 1]]
    r = points[TRIPLES[:, 2]]
    return 0.5 * (
        (q[:, 0] - p[:, 0]) * (r[:, 1] - p[:, 1])
        - (q[:, 1] - p[:, 1]) * (r[:, 0] - p[:, 0])
    )


def _hull_area(points: np.ndarray) -> float:
    pts = points[np.lexsort((points[:, 1], points[:, 0]))]
    if len(pts) <= 2:
        return 0.0

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = np.array(lower[:-1] + upper[:-1])
    x, y = hull[:, 0], hull[:, 1]
    return 0.5 * abs(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1)))


def _rescale_unit_hull(pts):
    ha = _hull_area(pts)
    if ha > 1e-12:
        s = 1.0 / np.sqrt(ha)
        c = pts.mean(axis=0)
        pts = (pts - c) * s + c
    return pts


def _softmin_value_grad(pts, alpha):
    """Smooth soft-min of triangle areas and its analytic gradient.

    F(pts) = -(1/alpha) * logsumexp(-alpha * |A_t|)
    grad wrt vertex v of triple t: 0.5 * w_t * s_t * R(p_u - p_w)
    where w_t = softmax weight, s_t = sign of signed area,
    R(x,y) = (y,-x), u=(v+1)%3, w=(v+2)%3.
    """
    sa = _signed(pts)
    a = np.abs(sa)
    sgn = np.where(sa >= 0.0, 1.0, -1.0)
    z = -alpha * a
    z = z - z.max()
    e = np.exp(z)
    Z = e.sum()
    w = e / Z
    val = -(np.log(Z) + (-z.max() * 0.0)) / alpha  # not used precisely; use below
    # softmin value (for monitoring): -(logsumexp)/alpha
    val = -(z.max() + np.log(Z)) / alpha

    grad = np.zeros((N, 2))
    P = pts[TRIPLES]  # (286, 3, 2)
    for pos in range(3):
        u = (pos + 1) % 3
        v2 = (pos + 2) % 3
        d = P[:, u] - P[:, v2]          # p_u - p_w
        R = np.empty_like(d)
        R[:, 0] = d[:, 1]
        R[:, 1] = -d[:, 0]
        contrib = (0.5 * w * sgn)[:, None] * R
        np.add.at(grad, TRIPLES[:, pos], contrib)
    return val, grad


def _gradient_ascent(init, rng, iters, alpha0, alpha1, step0, jitter=0.0):
    """Continuation-based softmin gradient ascent with adaptive step."""
    pts = _rescale_unit_hull(init.copy())
    step = step0
    best = pts.copy()
    best_min = float(_areas(pts).min())
    # logspace alpha schedule (continuation)
    alphas = np.exp(np.linspace(np.log(alpha0), np.log(alpha1), iters))
    prev_val = -np.inf
    for it in range(iters):
        alpha = alphas[it]
        val, grad = _softmin_value_grad(pts, alpha)
        gn = np.linalg.norm(grad) + 1e-12
        if val > prev_val:
            step *= 1.05
        else:
            step *= 0.7
        step = min(max(step, 1e-5), 0.05)
        prev_val = val
        cand = pts + (step / gn) * grad
        if jitter > 0.0:
            cand = cand + rng.normal(0.0, jitter * step, cand.shape)
        cand = _rescale_unit_hull(cand)
        _, g2 = _softmin_value_grad(cand, alpha)
        v2 = _softmin_value_grad(cand, alpha)[0]
        if v2 >= val:
            pts = cand
            m = float(_areas(pts).min())
            if m > best_min:
                best_min = m
                best = pts.copy()
        # occasional refresh of reference value (alpha changes anyway)
        if it % 400 == 399:
            prev_val = _softmin_value_grad(pts, alpha)[0]
    return best, best_min


def _polish(pts, rng, iters, K, step0=0.01, step_min=0.0004):
    """Targeted greedy polish: push a vertex of one of the K smallest
    triangles away from the opposite edge; accept strict improvements."""
    pts = pts.copy()
    a = _areas(pts)
    cur_min = float(a.min())
    best = pts.copy()
    best_min = cur_min
    step = step0
    for it in range(iters):
        order = np.argsort(a)
        t = TRIPLES[order[rng.integers(min(K, 6))]]
        i = int(t[rng.integers(3)])
        old = pts[i].copy()
        others = [int(x) for x in t if x != i]
        j, k2 = others[0], others[1]
        d = pts[k2] - pts[j]
        nrm = np.hypot(d[0], d[1]) + 1e-12
        nd = np.array([-d[1], d[0]]) / nrm
        side = d[0] * (old[1] - pts[j, 1]) - d[1] * (old[0] - pts[j, 0])
        if side < 0:
            nd = -nd
        new = old + step * 0.7 * nd + rng.normal(0, step * 0.2, 2)
        dists = np.hypot(*(pts - new).T)
        dists[i] = 10.0
        if dists.min() < 1e-6:
            continue
        pts[i] = new
        pts = _rescale_unit_hull(pts)
        a2 = _areas(pts)
        m2 = float(a2.min())
        if m2 > best_min:
            best_min = m2
            best = pts.copy()
            a = a2
        else:
            pts[i] = old
            pts = _rescale_unit_hull(pts)
        if it % 300 == 299:
            step = max(step * 0.9, step_min)
    return best, best_min


def _init_ring(alt: bool) -> np.ndarray:
    ang = np.arange(12) * (2.0 * np.pi / 12.0)
    rad = np.where(np.arange(12) % 2 == 0, 1.0, 0.72) if alt else np.ones(12)
    pts = np.zeros((N, 2))
    pts[:12, 0] = rad * np.cos(ang)
    pts[:12, 1] = rad * np.sin(ang)
    pts[12] = (0.0, 0.0)
    return pts


def _init_triangle() -> np.ndarray:
    pts = np.zeros((N, 2))
    a0 = np.pi / 2.0
    for i in range(3):
        a = a0 + i * 2.0 * np.pi / 3.0
        pts[i] = (1.05 * np.cos(a), 1.05 * np.sin(a))
    for i in range(6):
        a = a0 + np.pi / 6.0 + i * np.pi / 3.0
        pts[3 + i] = (0.62 * np.cos(a), 0.62 * np.sin(a))
    for i in range(3):
        a = a0 + i * 2.0 * np.pi / 3.0
        pts[9 + i] = (0.30 * np.cos(a), 0.30 * np.sin(a))
    pts[12] = (0.0, 0.0)
    return pts


def _init_grid(rng) -> np.ndarray:
    g = []
    for i in range(4):
        for j in range(4):
            g.append((i / 3.0, j / 3.0))
    g = np.array(g, dtype=float)
    keep = [0, 1, 2, 3, 4, 5, 6, 8, 9, 10, 12, 13, 14]
    pts = g[keep] * 2.0 - 1.0
    pts += rng.normal(0, 0.06, pts.shape)
    return pts


def _init_pentagon(rng) -> np.ndarray:
    pts = np.zeros((N, 2))
    for i in range(5):
        a = -np.pi / 2.0 + i * 2 * np.pi / 5.0
        pts[i] = (np.cos(a), np.sin(a))
    for i in range(5):
        a = -np.pi / 2.0 + 0.3 + i * 2 * np.pi / 5.0
        pts[5 + i] = (0.55 * np.cos(a), 0.55 * np.sin(a))
    pts[10] = (0.25, 0.15)
    pts[11] = (-0.22, -0.18)
    pts[12] = (0.0, 0.0)
    return pts + rng.normal(0, 0.02, pts.shape)


def heilbronn_convex13() -> np.ndarray:
    """
    Construct an arrangement of 13 points maximizing the smallest triangle
    area normalized by convex hull area.

    Approach: analytic gradient ascent on a soft-minimum surrogate of all
    C(13,3) triangle areas with a continuation schedule on the sharpness
    parameter alpha, adaptive step control, periodic unit-hull rescaling,
    structured multi-starts, and a final targeted greedy polish.
    """
    rng = np.random.default_rng(seed=42)
    inits = [_init_ring(False), _init_ring(True), _init_triangle(),
             _init_grid(rng), _init_pentagon(rng)]

    global_best = None
    global_min = -1.0

    for init in inits:
        # continuation: alpha from ~40 to ~4000 (soft -> sharp min)
        pts, m = _gradient_ascent(init, rng, iters=2500,
                                  alpha0=40.0, alpha1=4000.0,
                                  step0=0.01, jitter=0.25)
        pts, m = _polish(pts, rng, iters=1200, K=6)
        # second, sharper pass from the improved point
        pts2, m2 = _gradient_ascent(pts, rng, iters=1200,
                                    alpha0=2000.0, alpha1=20000.0,
                                    step0=0.004, jitter=0.1)
        if m2 > m:
            pts, m = pts2, m2
        pts, m = _polish(pts, rng, iters=1500, K=4)
        if m > global_min:
            global_min = m
            global_best = pts

    # one extra sharp refinement cycle from the global best
    pts, m = _gradient_ascent(global_best, rng, iters=1000,
                              alpha0=5000.0, alpha1=50000.0,
                              step0=0.002, jitter=0.05)
    pts, m = _polish(pts, rng, iters=1500, K=3)
    if m > global_min:
        global_min = m
        global_best = pts

    best_pts = _rescale_unit_hull(global_best)
    return best_pts
