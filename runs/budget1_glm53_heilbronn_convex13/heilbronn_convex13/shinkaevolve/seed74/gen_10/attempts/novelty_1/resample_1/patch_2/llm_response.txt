def solve():
    """Solve the 13-point Heilbronn triangle problem with deterministic
    multi-start simulated annealing. Returns a 13x2 numpy array of points."""
    import numpy as np
    from itertools import combinations

    rng = np.random.default_rng(12345)

    n = 13
    # Precompute all triangle index triples once
    tri_idx = np.array(list(combinations(range(n), 3)))

    def min_tri_area(P):
        # Vectorized minimum triangle area over all triples
        A = P[tri_idx[:, 0]]
        B = P[tri_idx[:, 1]]
        C = P[tri_idx[:, 2]]
        cross = (B[:, 0] - A[:, 0]) * (C[:, 1] - A[:, 1]) - \
                (B[:, 1] - A[:, 1]) * (C[:, 0] - A[:, 0])
        return 0.5 * np.abs(cross).min()

    def hull_area(P):
        # Convex hull area via monotone chain
        pts = sorted(map(tuple, P), key=lambda p: (p[0], p[1]))
        if len(pts) <= 2:
            return 1e-12
        def cross(o, a, b):
            return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
        lower = []
        for p in pts:
            while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 1e-15:
                lower.pop()
            lower.append(p)
        upper = []
        for p in reversed(pts):
            while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 1e-15:
                upper.pop()
            upper.append(p)
        hull = lower[:-1] + upper[:-1]
        if len(hull) < 3:
            return 1e-12
        s = 0.0
        for i in range(len(hull)):
            x1, y1 = hull[i]
            x2, y2 = hull[(i + 1) % len(hull)]
            s += x1 * y2 - x2 * y1
        return max(abs(s) / 2.0, 1e-12)

    def objective(P):
        return min_tri_area(P) / hull_area(P)

    def anneal(P, iters=6000, t0=0.05):
        P = P.copy()
        cur = objective(P)
        best_P, best = P.copy(), cur
        for it in range(iters):
            T = t0 * (0.9995 ** it)
            i = rng.integers(n)
            step = rng.normal(0, max(T, 1e-4), 2)
            Q = P.copy()
            Q[i] += step
            val = objective(Q)
            if val >= cur or rng.random() < np.exp((val - cur) / max(T, 1e-12)):
                P, cur = Q, val
                if val > best:
                    best_P, best = Q.copy(), val
        return best_P, best

    best_global, best_val = None, -np.inf

    for trial in range(30):
        mode = trial % 3
        if mode == 0:
            # Perturbed circle
            th = np.sort(rng.uniform(0, 2 * np.pi, n))
            r = 0.5 + rng.normal(0, 0.08, n)
            P = np.stack([r * np.cos(th), r * np.sin(th)], axis=1)
        elif mode == 1:
            # Random convex polygon via sorted angles from center
            P = rng.uniform(0, 1, (n, 2))
        else:
            # Uniform interior points
            P = rng.uniform(0.05, 0.95, (n, 2))
        P2, v = anneal(P, iters=4000, t0=0.04)
        if v > best_val:
            best_global, best_val = P2.copy(), v

    # Final local polish with small steps
    P, v = anneal(best_global, iters=3000, t0=0.005)
    if v > best_val:
        best_global, best_val = P.copy(), v

    # Ensure finite output
    best_global = np.nan_to_num(best_global, nan=0.0, posinf=1.0, neginf=0.0)
    return best_global
