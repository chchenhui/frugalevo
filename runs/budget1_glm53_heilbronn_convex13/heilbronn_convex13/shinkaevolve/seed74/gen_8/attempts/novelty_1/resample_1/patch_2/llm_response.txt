def solve():
    import math
    import numpy as np

    rng = np.random.default_rng(20240513)

    # ---- Unit-area equilateral triangle as the convex region ----
    side = math.sqrt(4.0 / math.sqrt(3.0))
    V = np.array([[0.0, 0.0],
                  [side, 0.0],
                  [side * 0.5, side * math.sqrt(3.0) * 0.5]])
    TRI_AREA = math.sqrt(3.0) / 4.0 * side * side  # == 1.0

    # ---- All C(13,3) = 286 index triples, vectorized ----
    triples = [(i, j, k) for i in range(13) for j in range(i + 1, 13)
               for k in range(j + 1, 13)]
    I = np.array([t[0] for t in triples])
    J = np.array([t[1] for t in triples])
    K = np.array([t[2] for t in triples])

    def min_tri(P):
        a, b, c = P[I], P[J], P[K]
        cross = ((b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) -
                 (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0]))
        return 0.5 * float(np.min(np.abs(cross)))

    def hull_area(P):
        pts = P[np.lexsort((P[:, 1], P[:, 0]))]
        pts = np.unique(pts, axis=0)
        if len(pts) < 3:
            return 1e-9
        def half(seq):
            h = []
            for p in seq:
                while len(h) >= 2:
                    ax, ay = h[-1]; bx, by = h[-2]
                    if (ax - bx) * (p[1] - by) - (ay - by) * (p[0] - bx) <= 1e-12:
                        h.pop()
                    else:
                        break
                h.append(p)
            return h
        lower = half([tuple(p) for p in pts])
        upper = half([tuple(p) for p in pts[::-1]])
        hull = np.array(lower[:-1] + upper[:-1])
        x, y = hull[:, 0], hull[:, 1]
        return 0.5 * abs(float(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1))))

    def bary(P):
        x1, y1 = V[0]; x2, y2 = V[1]; x3, y3 = V[2]
        x = P[..., 0]; y = P[..., 1]
        w1 = ((y2 - y3) * (x - x3) + (x3 - x2) * (y - y3)) / (2.0 * TRI_AREA)
        w2 = ((y3 - y1) * (x - x3) + (x1 - x3) * (y - y3)) / (2.0 * TRI_AREA)
        w3 = 1.0 - w1 - w2
        return np.stack([w1, w2, w3], axis=-1)

    def clip(P):
        W = np.clip(bary(P), 0.0, None)
        W = W / np.maximum(W.sum(axis=-1, keepdims=True), 1e-12)
        return W @ V
