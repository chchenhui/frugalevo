import numpy as np


# ---------------------------------------------------------------------------
# Stage 1: Geometric evaluation layer
# ---------------------------------------------------------------------------
class TriangleEvaluator:
    """Encapsulates all geometry: triangle areas, hull area, normalization."""

    def __init__(self, n=13):
        self.n = n
        i, j, k = np.meshgrid(np.arange(n), np.arange(n), np.arange(n), indexing="ij")
        mask = (i < j) & (j < k)
        self.tri_idx = np.stack([i[mask], j[mask], k[mask]], axis=1)  # (286,3)

    def triangle_areas(self, pts):
        p = pts[self.tri_idx]                     # (286,3,2)
        a, b, c = p[:, 0], p[:, 1], p[:, 2]
        cross = (b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) - (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0])
        return 0.5 * np.abs(cross)

    @staticmethod
    def hull_area(pts):
        # Convex hull via Andrew's monotone chain (deterministic, small n)
        P = sorted(map(tuple, np.round(pts, 12)))
        P = list(dict.fromkeys(P))
        if len(P) < 3:
            return 1e-12

        def cross(o, a, b):
            return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])

        lower, upper = [], []
        for p in P:
            while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
                lower.pop()
            lower.append(p)
        for p in reversed(P):
            while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
                upper.pop()
            upper.append(p)
        hull = lower[:-1] + upper[:-1]
        if len(hull) < 3:
            return 1e-12
        s = 0.0
        m = len(hull)
        for i in range(m):
            x1, y1 = hull[i]
            x2, y2 = hull[(i + 1) % m]
            s += x1 * y2 - x2 * y1
        return abs(s) / 2.0

    def score(self, pts):
        return self.triangle_areas(pts).min() / self.hull_area(pts)


# ---------------------------------------------------------------------------
# Stage 2: Smooth objective (differentiable surrogate for gradient ascent)
# ---------------------------------------------------------------------------
class SoftminObjective:
    """log-sum-exp softmin of triangle areas; gradients w.r.t. point positions."""

    def __init__(self, eval: TriangleEvaluator):
        self.eval = eval
        self.idx = eval.tri_idx

    def value_and_grad(self, pts, beta):
        p = pts[self.idx]                          # (T,3,2)
        a, b, c = p[:, 0], p[:, 1], p[:, 2]
        cr = (b[:, 0]-a[:, 0])*(c[:, 1]-a[:, 1]) - (b[:, 1]-a[:, 1])*(c[:, 0]-a[:, 0])
        signed = 0.5 * cr
        w = np.exp(beta * (signed - signed.max()))
        w /= w.sum()
        val = signed.max() + np.log(w.sum() * 0) if False else (signed * w).sum()
        # gradient of softmin wrt each coordinate
        grad = np.zeros_like(pts)
        # d(signed)/d(pts)
        for t in range(self.idx.shape[0]):
            ia, ib, ic = self.idx[t]
            ga = 0.5 * np.array([-(c[t,1]-b[t,1]), (c[t,0]-b[t,0])])
            gb = 0.5 * np.array([(c[t,1]-a[t,1]), -(c[t,0]-a[t,0])])
            gc = 0.5 * np.array([(b[t,1]-a[t,1]), -(b[t,0]-a[t,0])])
            wt = w[t]
            grad[ia] += wt * ga
            grad[ib] += wt * gb
            grad[ic] += wt * gc
        return val, grad


# ---------------------------------------------------------------------------
# Stage 3: Candidate generators (deterministic initialization layer)
# ---------------------------------------------------------------------------
class CandidateFactory:
    def __init__(self, n=13, seed=42):
        self.n = n
        self.seed = seed

    def triangular_lattice(self):
        pts = []
        rows = 4
        for r in range(rows):
            for c in range(rows - r):
                pts.append((c + 0.5 * r, r * 0.866))
        pts = np.array(pts[:13], dtype=float)
        return (pts - pts.min(0)) / (pts.ptp(0) + 1e-12)

    def ring_center(self):
        n_out, n_in = 9, 4
        ang_o = 2 * np.pi * np.arange(n_out) / n_out + 0.2
        ang_i = 2 * np.pi * np.arange(n_in) / n_in + np.pi / n_in
        outer = 0.5 + 0.48 * np.stack([np.cos(ang_o), np.sin(ang_o)], 1)
        inner = 0.5 + 0.22 * np.stack([np.cos(ang_i), np.sin(ang_i)], 1)
        return np.vstack([outer, inner])

    def perturbed_symmetric(self, variant=0):
        rng = np.random.default_rng(self.seed + variant)
        base = self.ring_center()
        return np.clip(base + 0.03 * rng.standard_normal(base.shape), 0.01, 0.99)

    def seeded_random(self, variant=0):
        rng = np.random.default_rng(self.seed + 100 + variant)
        pts = rng.random((self.n, 2))
        # rejection: avoid near-collinear triples via simple repulsion
        for _ in range(200):
            d = pts[:, None, :] - pts[None, :, :]
            dist = np.linalg.norm(d, axis=2) + np.eye(self.n)
            rep = (d / dist[:, :, None] ** 3).sum(axis=1) * 0.01
            pts = np.clip(pts + rep, 0.01, 0.99)
        return pts

    def all_candidates(self):
        return [
            self.triangular_lattice(),
            self.ring_center(),
            self.perturbed_symmetric(0),
            self.perturbed_symmetric(1),
            self.seeded_random(0),
            self.seeded_random(1),
        ]


# ---------------------------------------------------------------------------
# Stage 4: Refinement (projected gradient ascent with annealed softmin)
# ---------------------------------------------------------------------------
class RefinementStage:
    def __init__(self, eval: TriangleEvaluator, iters=140):
        self.eval = eval
        self.obj = SoftminObjective(eval)
        self.iters = iters

    def run(self, pts):
        pts = pts.copy()
        best, best_score = pts.copy(), self.eval.score(pts)
        step = 0.02
        for it in range(self.iters):
            beta = 20.0 + 480.0 * it / self.iters   # anneal softmin -> hard min
            _, g = self.obj.value_and_grad(pts, beta)
            gnorm = np.linalg.norm(g) + 1e-12
            pts = np.clip(pts + step * g / gnorm * self.n ** 0.5, 0.0, 1.0)
            s = self.eval.score(pts)
            if s > best_score:
                best_score, best = s, pts.copy()
            step *= 0.995
        return best, best_score


# ---------------------------------------------------------------------------
# Stage 5: Local polish (coordinate-descent micro-perturbations)
# ---------------------------------------------------------------------------
class PolishStage:
    def __init__(self, eval: TriangleEvaluator, rounds=60, seed=42):
        self.eval = eval
        self.rounds = rounds
        self.rng = np.random.default_rng(seed)

    def run(self, pts):
        best = pts.copy()
        bs = self.eval.score(best)
        eps = 0.008
        for _ in range(self.rounds):
            improved = False
            for i in range(len(best)):
                for dx, dy in ((eps, 0), (-eps, 0), (0, eps), (0, -eps),
                               (eps, eps), (-eps, -eps), (eps, -eps), (-eps, eps)):
                    trial = best.copy()
                    trial[i] = np.clip(trial[i] + [dx, dy], 0.0, 1.0)
                    s = self.eval.score(trial)
                    if s > bs + 1e-12:
                        bs, best, improved = s, trial, True
                        break
                if improved:
                    break
            if not improved:
                eps *= 0.5
                if eps < 1e-5:
                    break
        return best, bs


# ---------------------------------------------------------------------------
# Pipeline orchestrator
# ---------------------------------------------------------------------------
class HeilbronnPipeline:
    def __init__(self, n=13, seed=42):
        self.eval = TriangleEvaluator(n)
        self.factory = CandidateFactory(n, seed)
        self.refiner = RefinementStage(self.eval)
        self.polisher = PolishStage(self.eval, seed=seed)

    def solve(self):
        best_pts, best_score = None, -np.inf
        for cand in self.factory.all_candidates():
            p, s = self.refiner.run(cand)
            p, s = self.polisher.run(p)
            if s > best_score:
                best_pts, best_score = p, s
        # Deterministic post-condition: rescale so hull area = 1 (unit area region)
        ha = self.eval.hull_area(best_pts)
        if ha > 1e-9:
            best_pts = best_pts / np.sqrt(ha)
            best_pts = best_pts - best_pts.min(axis=0)
        return best_pts


def heilbronn_convex13() -> np.ndarray:
    """
    Construct an arrangement of 13 points on or inside a convex region in order
    to maximize the area of the smallest triangle formed by these points.

    Pipeline: deterministic candidates -> softmin gradient refinement ->
    coordinate polish -> best-of selection -> unit-area normalization.

    Returns:
        points: np.ndarray of shape (13,2) with the x,y coordinates of the points.
    """
    try:
        pipeline = HeilbronnPipeline(n=13, seed=42)
        pts = pipeline.solve()
        if pts is None or pts.shape != (13, 2) or not np.all(np.isfinite(pts)):
            raise RuntimeError("pipeline produced invalid output")
        return pts
    except Exception:
        # Fallback: deterministic lattice (always valid)
        pts = CandidateFactory(13, 42).triangular_lattice()
        return pts
