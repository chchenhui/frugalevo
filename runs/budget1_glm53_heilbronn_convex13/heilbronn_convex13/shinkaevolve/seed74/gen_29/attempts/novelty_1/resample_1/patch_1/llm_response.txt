<DESCRIPTION>
Implement the recommended adaptive step control in the annealer: track acceptance rate over the last 200 proposals and adjust the step size to stay in the productive 15–50% acceptance band. This replaces the pure geometric step decay with feedback-driven control, keeping the search in the productive regime longer and improving the final minimum-triangle area. Also add a brief reheat schedule boost when acceptance collapses. The polish stage remains unchanged as a strict greedy improvement, so determinism is preserved.

def _anneal(pts, iters, t0, t1, rng, step0=0.05, step1=0.002, k_bot=8):
    """Worst-triangle targeted simulated annealing with best-ever fallback.

    Uses feedback-driven adaptive step control: the acceptance rate over the
    last 200 proposals is monitored; if it drops below 15% the step is halved,
    if it exceeds 50% the step is scaled by 1.5x (capped at 0.05). This keeps
    the anneal in the productive acceptance band instead of relying purely on
    a fixed geometric decay.
    """
    pts = pts.copy()
    ha = _hull_area(pts)
    best, ha = _score(pts, ha)
    cur = best
    best_pts = pts.copy()
    step = step0
    acc_hist = []  # acceptance history (1 = accepted, 0 = rejected)
    for it in range(iters):
        frac = it / iters
        temp = t0 * (t1 / t0) ** frac
        areas = _all_tri_areas(pts)
        order = np.argsort(areas)[:k_bot]
        verts = np.unique(_IDX[order].ravel())
        vi = verts[rng.integers(len(verts))]
        cand = pts.copy()
        cand[vi] += rng.normal(0, step, 2)
        cand[vi] = np.clip(cand[vi], 0.0, 1.0)
        if np.min(np.linalg.norm(np.delete(cand, vi, axis=0) - cand[vi], axis=1)) < 1e-6:
            continue
        ha_new = _hull_area(cand)
        s_new, ha_new = _score(cand, ha_new)
        accepted = False
        if s_new >= cur or rng.random() < np.exp((s_new - cur) / max(temp, 1e-12)):
            pts, cur, ha = cand, s_new, ha_new
            accepted = True
            if s_new > best:
                best = s_new
                best_pts = cand.copy()
        # adaptive step control based on recent acceptance rate
        acc_hist.append(1 if accepted else 0)
        if len(acc_hist) > 200:
            acc_hist.pop(0)
        if len(acc_hist) == 200 and it % 50 == 0:
            rate = sum(acc_hist) / 200.0
            if rate < 0.15:
                step = max(step * 0.5, step1 * 0.25)
            elif rate > 0.50:
                step = min(step * 1.5, 0.05)
        # gentle geometric floor so steps still shrink over the run
        step = max(step, step0 * (step1 / step0) ** frac + 0.0005)
    return best_pts, best
