5. Error handling: if anything in the optimization fails, fall back to the deterministic symmetric seed configuration, which is already a valid (if suboptimal) answer.

def solve():
    import numpy as np
    import itertools

    rng = np.random.default_rng(42)

    # ---- Deterministic hybrid seed: 3-fold symmetric interior ring + boundary ----
    pts = []
    # boundary ring (slightly inset to keep points strictly usable)
    b = 0.02
    corners = [(b, b), (1 - b, b), (1 - b, 1 - b), (b, 1 - b)]
    edge_pts = []
    for (x1, y1), (x2, y2) in zip(corners, corners[1:] + corners[:1]):
        for t in (0.25, 0.5, 0.75):
            edge_pts.append((x1 + t * (x2 - x1), y1 + t * (y2 - y1)))
    pts.extend(corners)
    pts.extend(edge_pts)          # 4 + 12 = 16 candidates -> pick deterministic subset
    # interior 3-fold symmetric ring around center
    cx, cy = 0.5, 0.5
    for k in range(3):
        a = 2 * np.pi * k / 3 + np.pi / 6
        for r in (0.18, 0.33):
            pts.append((cx + r * np.cos(a), cy + r * np.sin(a)))
    # deterministic down-selection to exactly 13 points, well-spread
    idx = [0, 1, 2, 3, 5, 6, 8, 9, 11, 13, 14, 15, 16]
    P = np.array(pts, dtype=float)[idx]

    # ensure exactly 13 points
    assert P.shape == (13, 2), "must have exactly 13 points"

    tri_idx = np.array(list(itertools.combinations(range(13), 3)))

    def min_tri_area(P):
        # vectorized cross-product triangle areas
        A = P[tri_idx[:, 0]]
        B = P[tri_idx[:, 1]]
        C = P[tri_idx[:, 2]]
        cross = (B[:, 0] - A[:, 0]) * (C[:, 1] - A[:, 1]) - (B[:, 1] - A[:, 1]) * (C[:, 0] - A[:, 0])
        return 0.5 * np.abs(cross).min()

    # ---- Greedy multi-scale local refinement (deterministic) ----
    try:
        best = min_tri_area(P)
        step = 0.08
        n_iters = 0
        max_iters = 20000
        while step > 1e-5 and n_iters < max_iters:
            n_iters += 1
            improved = False
            order = rng.permutation(13)
            for i in order:
                for trial in range(24):
                    ang = 2 * np.pi * (trial / 24.0) + rng.uniform(-0.13, 0.13)
                    d = step * rng.uniform(0.3, 1.0)
                    Q = P.copy()
                    Q[i, 0] = np.clip(P[i, 0] + d * np.cos(ang), 0.0, 1.0)
                    Q[i, 1] = np.clip(P[i, 1] + d * np.sin(ang), 0.0, 1.0)
                    a = min_tri_area(Q)
                    if a > best + 1e-14:
                        P = Q
                        best = a
                        improved = True
            if not improved:
                step *= 0.6
    except Exception:
        # fallback: keep the deterministic seed configuration
        pass

    # ---- Normalize so convex hull area == 1 ----
    try:
        from scipy.spatial import ConvexHull
        hull = ConvexHull(P)
        hull_area = hull.volume  # 2D: volume is the polygon area
    except Exception:
        # shoelace fallback on hull points
        from scipy.spatial import ConvexHull as _CH
        try:
            h = _CH(P)
            H = P[h.vertices]
        except Exception:
            H = P
        hull_area = 0.5 * abs(
            np.dot(H[:, 0], np.roll(H[:, 1], -1)) - np.dot(H[:, 1], np.roll(H[:, 0], -1))
        )
    if hull_area <= 1e-12:
        hull_area = 1.0
    P = P / np.sqrt(hull_area)

    # final sanity checks
    assert P.shape == (13, 2)
    assert np.all(np.isfinite(P))
    # no duplicates
    D = np.linalg.norm(P[:, None, :] - P[None, :, :], axis=-1)
    D[D == 0] = np.inf
    assert D.min() > 1e-9, "duplicate points detected"

    return [tuple(p) for p in P]
