def create_points():
    """
    Construct 13 points maximizing the minimum triangle area (Heilbronn problem).
    Deterministic: fixed seeds, fixed initial configurations, exact arithmetic paths.
    Returns a list of 13 [x, y] pairs whose convex hull has unit area.
    """
    n = 13
    if _HAS_NUMPY:
        pts = _solve_with_numpy(n)
    else:
        pts = _solve_pure_python(n)
    _validate(pts, n)
    return pts


# ------------------------- geometry helpers -------------------------

def _hull_area(pts):
    """Area of the convex hull of pts (pure Python, works with lists)."""
    P = sorted(set((p[0], p[1]) for p in pts))
    if len(P) < 3:
        return 0.0

    def cross(o, a, b):
        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])

    lower = []
    for p in P:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(P):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 3:
        return 0.0
    a = 0.0
    m = len(hull)
    for i in range(m):
        x1, y1 = hull[i]
        x2, y2 = hull[(i+1) % m]
        a += x1*y2 - x2*y1
    return abs(a) / 2.0


def _min_triangle_area(pts):
    """Exact minimum area over all C(n,3) triangles."""
    n = len(pts)
    best = float("inf")
    for i in range(n):
        ax, ay = pts[i]
        for j in range(i+1, n):
            bx, by = pts[j]
            for k in range(j+1, n):
                cx, cy = pts[k]
                a = abs((bx-ax)*(cy-ay) - (by-ay)*(cx-ax)) / 2.0
                if a < best:
                    best = a
    return best


def _normalize_unit_hull(pts):
    """Scale configuration so its convex hull has area exactly 1."""
    area = _hull_area(pts)
    if area <= 0:
        return None
    s = 1.0 / math.sqrt(area)
    return [[p[0]*s, p[1]*s] for p in pts]


def _validate(pts, n):
    assert len(pts) == n, "point count mismatch"
    for p in pts:
        assert all(isinstance(c, float) and math.isfinite(c) for c in p)
    seen = set()
    for p in pts:
        key = (round(p[0], 9), round(p[1], 9))
        assert key not in seen, "duplicate points"
        seen.add(key)

def _refine(pts, iters):
    """Deterministic hill climbing with shrinking perturbation radius."""
    pts = [[float(p[0]), float(p[1])] for p in pts]
    best = _min_triangle_area(_normalize_unit_hull(pts) or pts)
    radius = 0.05
    step = 0.6180339887498949  # golden-ratio deterministic sequence
    counter = 0
    for it in range(iters):
        improved = False
        for idx in range(len(pts)):
            for (dx, dy) in ((radius, 0), (-radius, 0), (0, radius), (0, -radius),
                              (radius, radius), (-radius, -radius),
                              (radius, -radius), (-radius, radius)):
                counter += 1
                trial = [list(p) for p in pts]
                # deterministic pseudo-random jitter via golden ratio
                jx = ((step * counter * 0.6180339887) % 1.0) - 0.5
                jy = ((step * counter * 0.3819660113) % 1.0) - 0.5
                trial[idx][0] += dx + 0.01 * jx
                trial[idx][1] += dy + 0.01 * jy
                norm = _normalize_unit_hull(trial)
                if norm is None:
                    continue
                val = _min_triangle_area(norm)
                if val > best + 1e-12:
                    best = val
                    pts = trial
                    improved = True
        if not improved:
            radius *= 0.6
            if radius < 1e-7:
                break
    return pts
