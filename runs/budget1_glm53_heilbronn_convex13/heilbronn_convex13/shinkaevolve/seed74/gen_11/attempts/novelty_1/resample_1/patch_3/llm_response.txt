        def _hull_area(pts):
            pts = np.unique(np.round(np.asarray(pts), 12), axis=0)
            if len(pts) < 3:
                return 1e-12
            P = sorted(map(tuple, pts))
            def cross(o, a, b):
                return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
            lower = []
            for p in P:
                while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
                    lower.pop()
                lower.append(p)
            upper = []
            for p in reversed(P):
                while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
                    upper.pop()
                upper.append(p)

            if len(hull) < 3:
                return 1e-12

            s = 0.0
            for i in range(len(h)):
                x1, y1 = h[i]
                x2, y2 = h[(i+1) % len(h)]
                s += x1*y2 - x2*y1
            return max(abs(s) / 2.0, 1e-12)

        def _min_tri_area(pts):
            P = np.asarray(pts)
            m = len(P)
            best = np.inf
            for i in range(m - 2):
                xa, ya = P[i]
                for j in range(i + 1, m - 1):
                    xb, yb = P[j]
                    d1x = xb - xa
                    d1y = yb - ya
                    for k in range(j + 1, m):
                        xc, yc = P[k]
                        a = abs(d1x * (yc - ya) - d1y * (xc - xa))
                        if a < best:
                            best = a
            return best / 2.0
