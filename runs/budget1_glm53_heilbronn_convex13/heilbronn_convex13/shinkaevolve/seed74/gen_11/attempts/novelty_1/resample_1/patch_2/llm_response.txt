import numpy as np

def _hull_area(pts):
    # Shoelace on convex hull (Andrew's monotone chain)
    pts = np.unique(np.round(pts, 12), axis=0)
    if len(pts) < 3:
        return 1e-12
    P = sorted(map(tuple, pts))
    def cross(o, a, b):
        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
    lower = []
    for p in P:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(P):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 3:
        return 1e-12
    h = np.array(hull)
    s = 0.0
    for i in range(len(h)):
        x1, y1 = h[i]
        x2, y2 = h[(i+1) % len(h)]
        s += x1*y2 - x2*y1
    return max(abs(s) / 2.0, 1e-12)

def _min_tri_area(pts):
    n = len(pts)
    best = np.inf
    P = np.asarray(pts)
    for i in range(n - 2):
        xa, ya = P[i]
        for j in range(i + 1, n - 1):
            xb, yb = P[j]
            dx1 = xb - xa
            dy1 = yb - ya
            for k in range(j + 1, n):
                xc, yc = P[k]
                a = abs(dx1 * (yc - ya) - dy1 * (xc - xa))
                if a < best:
                    best = a
    return best / 2.0

def _score(pts):
    return _min_tri_area(pts) / _hull_area(pts)

def _starts(n):
    rng = np.random.RandomState(12345)
    cands = []
    # Hexagonal-ish grid, perturbed
    side = 4
    grid = [(i / (side - 1.0), j / (side - 1.0))
            for j in range(side) for i in range(side)]
    grid = grid[:n]
    g = np.array(grid) + rng.uniform(-0.02, 0.02, (n, 2))
    cands.append(np.clip(g, 0.0, 1.0))
    # Boundary-heavy: ring near perimeter + interior random
    t = np.linspace(0, 1, n, endpoint=False)
    ring = np.stack([0.5 + 0.48*np.cos(2*np.pi*t),
                     0.5 + 0.48*np.sin(2*np.pi*t)], axis=1)
    ring += rng.uniform(-0.02, 0.02, (n, 2))
    cands.append(np.clip(ring, 0.0, 1.0))
    # Random uniform
    cands.append(rng.uniform(0.05, 0.95, (n, 2)))
    return cands

def construct(n=13):
    try:
        rng = np.random.RandomState(20240517)
        best_pts, best_val = None, -1.0
        for start in _starts(n):
            pts = start.copy()
            val = _score(pts)
            step = 0.05
            T0 = 0.010
            iters = 3000
            for it in range(iters):
                T = T0 * (1.0 - it / float(iters)) + 1e-6
                i = rng.randint(n)
                old = pts[i].copy()
                cand = np.clip(old + rng.uniform(-step, step, 2), 0.0, 1.0)
                pts[i] = cand
                new_val = _score(pts)
                if new_val >= val or rng.rand() < np.exp((new_val - val) / max(T, 1e-12)):
                    val = new_val
                else:
                    pts[i] = old
                if it % 500 == 499:
                    step *= 0.85
            # Greedy polish
            step = 0.01
            for _ in range(1500):
                improved = False
                for i in range(n):
                    for dxy in ((step, 0), (-step, 0), (0, step), (0, -step),
                                (step, step), (-step, -step), (step, -step), (-step, step)):
                        old = pts[i].copy()
                        cand = np.clip(old + np.array(dxy), 0.0, 1.0)
                        pts[i] = cand
                        nv = _score(pts)
                        if nv > val + 1e-12:
                            val = nv
                            improved = True
                        else:
                            pts[i] = old
                if not improved:
                    step *= 0.5
                    if step < 1e-6:
                        break
            if val > best_val:
                best_val = val
                best_pts = pts.copy()
        if best_pts is None:
            raise RuntimeError("optimization failed")
        return [tuple(p) for p in best_pts]
    except Exception:
        # Deterministic fallback: near-regular ring configuration
        t = np.linspace(0, 2*np.pi, n, endpoint=False)
        pts = [(0.5 + 0.45*np.cos(a + 0.1), 0.5 + 0.45*np.sin(a + 0.1)) for a in t]
        return pts
