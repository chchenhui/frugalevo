<DESCRIPTION>
Implement the recommended "best-of-cycles" pool strategy: track the top-3 distinct configurations found across all seeds (diversity measured by mean pairwise point distance > 0.1), then run one short reheat from each pool member plus one from the global best, followed by greedy polish on each. This escapes the plateau at min_area_normalized ~0.02 by exploring multiple good basins instead of restarting only from single configs. Also add a few extra deterministic structured seeds (boundary-heavy and 13-gon) to broaden the initial basin search, and slightly increase anneal iterations within time budget.

def _pool_add(pool, pts, val, max_pool=3, min_dist=0.1):
    """Add (pts, val) to pool keeping top distinct configurations.
    Distinctness: mean pairwise point distance > min_dist."""
    for _, (qp, qv) in enumerate(pool):
        if qv == val:
            return
        # match configurations by sorted coordinates to skip near-duplicates
        d = np.mean(np.abs(np.sort(pts, axis=0) - np.sort(qp, axis=0)))
        if d < min_dist:
            if val > qv:
                pool.remove((qp, qv))
            else:
                return
    pool.append((pts, val))
    pool.sort(key=lambda t: -t[1])
    while len(pool) > max_pool:
        pool.pop()


def heilbronn_convex13() -> np.ndarray:
    """
    Construct an arrangement of 13 points maximizing the smallest triangle
    area normalized by convex hull area, via worst-triangle targeted
    simulated annealing with a best-of-cycles pool of distinct configurations
    (reheats from each pool member + global best) and a final greedy polish.
    """
    pool = []  # list of (points, score), top distinct basins across seeds
    seeds = _seeds()
    for si, cand in enumerate(seeds):
        rng = np.random.default_rng(100 + si)
        p, v = _anneal(cand, iters=2200, t0=0.004, t1=1e-5, rng=rng)
        # short reheat cycle from current result
        for r in range(2):
            rng_r = np.random.default_rng(300 + 10 * si + r)
            p2, v2 = _anneal(p, iters=500, t0=0.002, t1=1e-6,
                             rng=rng_r, step0=0.05, step1=0.002)
            if v2 > v:
                p, v = p2, v2
        _pool_add(pool, p, v)

    # reheat from every distinct pool basin plus global best
    restarts = [p for p, v in pool]
    if pool:
        restarts.append(pool[0][0])
    best_pts, best_val = None, -1.0
    for ri, p in enumerate(restarts):
        rng_r = np.random.default_rng(700 + ri)
        p2, v2 = _anneal(p, iters=900, t0=0.003, t1=1e-6,
                         rng=rng_r, step0=0.04, step1=0.001)
        p2, v2 = _polish(p2)
        if v2 > best_val:
            best_val, best_pts = v2, p2
    if best_pts is None:
        best_pts = np.random.default_rng(seed=42).random((13, 2))
    return best_pts
