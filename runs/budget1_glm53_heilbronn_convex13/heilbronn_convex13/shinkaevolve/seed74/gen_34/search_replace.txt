            # Vectorized minimum triangle area over all C(n,3) triples.
            m = pts.shape[0]
            i, j, k = np.triu_indices(m, 1)
            # all pairs (i<j), then combine with each k>j
            best = np.inf
            P = pts
            for b in range(2, m):
                # triples (a, b', c) with c = b fixed as apex
                ax = P[:b, 0] - P[b, 0]
                ay = P[:b, 1] - P[b, 1]
