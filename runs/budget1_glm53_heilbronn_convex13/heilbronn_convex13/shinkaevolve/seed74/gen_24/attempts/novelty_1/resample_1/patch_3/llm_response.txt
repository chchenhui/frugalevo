        # Deterministic multi-start simulated annealing for Heilbronn n=13
        def min_triangle_area(pts):
            # smallest area among all triples, via base*height/2 in vectorized form
            best = np.inf
            for i in range(n):
                for j in range(i + 1, n):
                    d = pts[j] - pts[i]
                    L = np.hypot(d[0], d[1])
                    if L < 1e-12:
                        return 0.0
                    # distance of all other points to line (i,j)
                    rel = pts - pts[i]
                    dist = np.abs(rel[:, 0] * d[1] - rel[:, 1] * d[0]) / L
                    mask = np.ones(n, dtype=bool)
                    mask[i] = mask[j] = False
                    if not mask.any():
                        continue
                    h = dist[mask].min()
                    best = min(best, 0.5 * L * h)
            return best

        def project(pts):
            return np.clip(pts, 0.0, 1.0)
