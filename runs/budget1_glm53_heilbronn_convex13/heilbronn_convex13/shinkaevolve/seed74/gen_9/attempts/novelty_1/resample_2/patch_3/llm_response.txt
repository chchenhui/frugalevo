        P = sorted(set(map(tuple, np.round(pts, 12))))
        if len(P) < 3:
            return 1e-12
        P = np.array(P)

        def cross(o, a, b):
            return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

        lower = []
        for p in P:
            while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
                lower.pop()
            lower.append(p)
        upper = []
        for p in reversed(P):
            while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
                upper.pop()
            upper.append(p)

            x2, y2 = h[(i + 1) % len(h)]
            s += x1 * y2 - x2 * y1
        return abs(s) * 0.5

                best = float("inf")
                for (i, j, k) in TRIPLES:
                    d = ((P[j, 0] - P[i, 0]) * (P[k, 1] - P[i, 1])
                         - (P[k, 0] - P[i, 0]) * (P[j, 1] - P[i, 1]))
                    a = abs(d) * 0.5

                            - (P[k, 0] - P[i, 0]) * (P[j, 1] - P[i, 1]))
                sgn = np.where(d >= 0.0, 1.0, -1.0)
                a = np.abs(d) * 0.5
                z = -a / lam
                zmax = z.max()
                ez = np.exp(z - zmax)
                S = -lam * (zmax + np.log(ez.sum()))
                w = ez / ez.sum()
                grad = np.zeros_like(P)

                    grad[i, 1] += wt * (xk - xj)

                    grad[j, 1] += wt * (xi - xk)
                    grad[k, 0] += wt * (yi - yj)
                    grad[k, 1] += wt * (xj - xi)
