def solve():
    import numpy as np
    import itertools

    N = 13
    RNG_SEED = 12345
    rng = np.random.default_rng(RNG_SEED)

    # All triangle index triples, vectorized area computation
    TRI = np.array(list(itertools.combinations(range(N), 3)), dtype=np.int64)

    def min_tri_area(P):
        a = P[TRI[:, 0]]
        b = P[TRI[:, 1]]
        c = P[TRI[:, 2]]
        areas = 0.5 * np.abs(
            (b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1])
            - (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0])
        )
        return areas.min(), areas

    def clip(P):
        P[:, 0] = np.clip(P[:, 0], 0.0, 1.0)
        P[:, 1] = np.clip(P[:, 1], 0.0, 1.0)
        return P

    def refine(P, steps=(0.05, 0.02, 0.008, 0.003, 0.001)):
        best_val, _ = min_tri_area(P)
        for s in steps:
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    dirs = np.array(
                        [[s, 0], [-s, 0], [0, s], [0, -s],
                         [s, s], [s, -s], [-s, s], [-s, -s],
                         [s * 0.5, s * 0.85], [-s * 0.85, s * 0.5]]
                    )
                    for d in dirs:
                        Q = P.copy()
                        Q[i] += d
                        clip(Q)
                        v, _ = min_tri_area(Q)
                        if v > best_val + 1e-15:
                            P = Q
                            best_val = v
                            improved = True
        return P, best_val

    # ---- Deterministic seeds ----
    seeds = []

    # 1) Triangular-ish lattice (4x4 minus 3 corners kept as 13 points)
    g = np.array([[x / 3.0, y / 3.0] for y in range(4) for x in range(4)])
    keep = [i for i in range(16)
            if (i % 4, i // 4) not in [(0, 0), (3, 3), (1, 3)]]
    seeds.append(clip(g[keep] + rng.normal(0, 0.02, (13, 2))))

    # 2) Boundary ring + interior (8 on boundary, 5 inside)
    ang = np.linspace(0, 2 * np.pi, 8, endpoint=False) + 0.15
    ring = np.stack([0.5 + 0.49 * np.cos(ang), 0.5 + 0.49 * np.sin(ang)], 1)
    inner = np.array([[0.5, 0.5], [0.3, 0.35], [0.7, 0.35],
                      [0.35, 0.7], [0.65, 0.68]])
    seeds.append(clip(np.vstack([ring, inner]) + rng.normal(0, 0.01, (13, 2))))

    # 3) 3-fold symmetric layout around center
    pts3 = []
    for r in (0.18, 0.34, 0.48):
        for k in range(3):
            a = 2 * np.pi * k / 3 + 0.4
            pts3.append([0.5 + r * np.cos(a), 0.5 + r * np.sin(a)])
    pts3.append([0.5, 0.5])
    pts3.append([0.5, 0.5 + 0.49])
    pts3.append([0.06, 0.06])
    seeds.append(clip(np.array(pts3) + rng.normal(0, 0.01, (13, 2))))

    # 4-10) Fixed-seed random restarts
    for _ in range(7):
        seeds.append(clip(rng.random((13, 2)) * 0.98 + 0.01))

    best_P = None
    best_val = -1.0
    for P0 in seeds:
        P, v = refine(np.array(P0, dtype=float))
        if v > best_val:
            best_val, best_P = v, P.copy()

    # Final fine polish with jittered restarts around the best
    for _ in range(6):
        P, v = refine(clip(best_P + rng.normal(0, 0.01, (13, 2))))
        if v > best_val:
            best_val, best_P = v, P.copy()

    # Validate: exactly N unique finite points
    assert best_P.shape == (N, 2)
    assert np.all(np.isfinite(best_P))
    assert min_tri_area(best_P)[0] == best_val

    return [tuple(p) for p in best_P]
