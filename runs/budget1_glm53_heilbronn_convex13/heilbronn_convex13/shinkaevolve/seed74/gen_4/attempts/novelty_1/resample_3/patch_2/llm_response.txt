def _convex_hull_area(P):
    """Area of convex hull of points P (n x 2) via monotone chain."""
    pts = sorted(map(tuple, P))
    if len(pts) <= 2:
        return 0.0

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    area = 0.0
    m = len(hull)
    for i in range(m):
        x1, y1 = hull[i]
        x2, y2 = hull[(i + 1) % m]
        area += x1 * y2 - x2 * y1
    return abs(area) / 2.0


def solve(n=13):
    import numpy as np
    from itertools import combinations

    # Fixed seed for full determinism
    rng = np.random.default_rng(42)

    # --- Initialization: uniform points on a circle (avoids collinearity) ---
    theta = 2.0 * np.pi * np.arange(n) / n
    pts = np.column_stack([np.cos(theta), np.sin(theta)])

    # Indices of all C(n,3) triangles
    tri = np.array(list(combinations(range(n), 3)), dtype=int)
    i0, i1, i2 = tri[:, 0], tri[:, 1], tri[:, 2]

    def min_tri_area(P):
        a = P[i0]
        b = P[i1]
        c = P[i2]
        return np.min(np.abs(np.cross(b - a, c - a))) / 2.0

    def score(P):
        hull_a = _convex_hull_area(P)
        if hull_a <= 1e-12:
            return 0.0
        return min_tri_area(P) / hull_a

    best = pts.copy()
    best_val = score(best)

    # --- Deterministic hill-climb refinement with decaying step size ---
    step = 0.05
    for it in range(3000):
        cand = best + rng.normal(0.0, step, size=(n, 2))
        v = score(cand)
        if v > best_val:
            best = cand
            best_val = v
        if (it + 1) % 300 == 0:
            step *= 0.75
        if step < 1e-5:
            break

    # Fallback safety: if refinement somehow degraded, revert to circle
    if not np.all(np.isfinite(best)) or best_val <= score(pts):
        best = pts.copy()

    # --- Normalize so the convex hull has exactly unit area ---
    hull_a = _convex_hull_area(best)
    if hull_a <= 1e-12:
        # degenerate fallback: regular polygon has area pi/... guaranteed > 0
        best = pts.copy()
        hull_a = _convex_hull_area(best)
    best = best / np.sqrt(hull_a)

    # Deduplicate / sanity checks (keep exactly n unique finite points)
    assert best.shape == (n, 2)
    assert np.all(np.isfinite(best))
    return best
