        # Strategy: put all 13 points on a circle (cyclic 13-gon, so all
        # points lie on the boundary of the convex hull). Optimize the
        # angular positions to maximize the smallest triangle area divided
        # by the hull area, then rescale so the hull has area exactly 1.
        TWO_PI = 2.0 * np.pi
        m = n  # number of points on the circle

        # Deterministic seeded RNG
        rng = np.random.default_rng(seed=42)
        angles = np.sort(rng.random(m) * TWO_PI)

        # Enforce minimum angular spacing to avoid coincident points
        for _ in range(300):
            changed = False
            for i in range(m):
                gp = (angles[i] - angles[(i - 1) % m]) % TWO_PI
                gn = (angles[(i + 1) % m] - angles[i]) % TWO_PI
                if gp < 0.05 or gn < 0.05:
                    angles[i] = (angles[i] + 0.06) % TWO_PI
                    changed = True
            angles = np.sort(angles)
            if not changed:
                break

        def hull_area(a):
            # Cyclic polygon area for points on the unit circle
            gaps = np.diff(np.concatenate([a, [a[0] + TWO_PI]]))
            return 0.5 * float(np.abs(np.sum(np.sin(gaps))))

        def min_tri_area(a):
            pts = np.column_stack((np.cos(a), np.sin(a)))
            best = np.inf
            for i in range(m):
                xi, yi = pts[i]
                for j in range(i + 1, m):
                    xj, yj = pts[j]
                    dxj, dyj = xj - xi, yj - yi
                    for k in range(j + 1, m):
                        xk, yk = pts[k]
                        area2 = abs(dxj * (yk - yi) - (xk - xi) * dyj)
                        if area2 < best:

            return best / 2.0
