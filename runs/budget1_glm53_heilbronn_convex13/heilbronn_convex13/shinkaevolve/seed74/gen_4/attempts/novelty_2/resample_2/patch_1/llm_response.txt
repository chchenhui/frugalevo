import numpy as np


def _hull_area(points):
    p = points[np.argsort(points[:, 0])]
    # monotone chain
    pts = sorted(map(tuple, points), key=lambda q: (q[0], q[1]))
    if len(pts) <= 2:
        return 1e-12
    def cross(o, a, b):
        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
    lower = []
    for q in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], q) <= 0:
            lower.pop()
        lower.append(q)
    upper = []
    for q in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], q) <= 0:
            upper.pop()
        upper.append(q)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 3:
        return 1e-12
    h = np.array(hull)
    s = 0.0
    for i in range(len(h)):
        s += np.cross(h[i], h[(i+1) % len(h)])
    return abs(s) / 2.0


def _min_triangle_info(points):
    n = len(points)
    best = np.inf
    idx = (0, 1, 2)
    for i in range(n - 2):
        for j in range(i + 1, n - 1):
            a = points[j] - points[i]
            for k in range(j + 1, n):
                b = points[k] - points[i]
                ar = abs(a[0]*b[1] - a[1]*b[0]) / 2.0
                if ar < best:
                    best = ar
                    idx = (i, j, k)
    return best, idx


def heilbronn_convex13() -> np.ndarray:
    """
    Construct an arrangement of 13 points on or inside a convex region to
    maximize the smallest triangle area. Deterministic symmetric init +
    projected local search, normalized to unit hull area.
    """
    n = 13
    # Symmetric initialization: 12 points on a slightly irregular circle + center
    rng = np.random.default_rng(seed=7)
    angles = np.linspace(0, 2*np.pi, 12, endpoint=False)
    radii = np.ones(12) * 0.5 + rng.uniform(-0.03, 0.03, 12)
    pts = np.zeros((n, 2))
    pts[:12, 0] = radii * np.cos(angles)
    pts[:12, 1] = radii * np.sin(angles)
    # center point slightly offset
    pts[12] = [0.02, -0.01]

    # Local search: move vertices of the minimal triangle outward from its centroid
    step = 0.05
    best_area, idx = _min_triangle_info(pts)
    while step > 1e-6:
        improved = False
        i, j, k = idx
        for trial in range(6):
            cand = pts.copy()
            c = (pts[i] + pts[j] + pts[k]) / 3.0
            for t in (i, j, k):
                d = cand[t] - c
                nd = np.linalg.norm(d)
                if nd > 1e-12:
                    cand[t] = cand[t] + step * d / nd
            # also try moving each minimal-triangle vertex away from the other two
            a2, idx2 = _min_triangle_info(cand)
            if a2 > best_area + 1e-12:
                pts = cand
                best_area, idx = a2, idx2
                improved = True
                break
        if not improved:
            step *= 0.6

    # Also try random small perturbations to escape local traps (deterministic seed)
    rng = np.random.default_rng(seed=12345)
    for it in range(2000):
        cand = pts + rng.normal(0, step * 3 + 1e-4, pts.shape)
        a2, idx2 = _min_triangle_info(cand)
        if a2 > best_area + 1e-12:
            pts = cand
            best_area, idx = a2, idx2
            idx = idx2
        elif step > 1e-6:
            step *= 0.99

    # Normalize so convex hull area = 1 (scale-invariant objective)
    hull_area = _hull_area(pts)
    if hull_area > 1e-12:
        pts = pts / np.sqrt(hull_area)

    # Validate
    assert pts.shape == (13, 2)
    assert np.all(np.isfinite(pts))
    assert len(np.unique(np.round(pts, 9), axis=0)) == 13
    return pts
