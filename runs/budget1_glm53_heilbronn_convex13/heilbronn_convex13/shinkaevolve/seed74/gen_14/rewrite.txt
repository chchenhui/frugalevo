import numpy as np
from itertools import combinations


def _min_triangle_area(points: np.ndarray) -> float:
    """Exact minimum area over all triangles formed by triples of points."""
    idx = np.array(list(combinations(range(len(points)), 3)))
    p, q, r = points[idx[:, 0]], points[idx[:, 1]], points[idx[:, 2]]
    areas = 0.5 * np.abs(
        (q[:, 0] - p[:, 0]) * (r[:, 1] - p[:, 1])
        - (q[:, 1] - p[:, 1]) * (r[:, 0] - p[:, 0])
    )
    return float(areas.min())


def _hull_area(points: np.ndarray) -> float:
    """Area of the convex hull of the points (monotone chain + shoelace)."""
    pts = points[np.lexsort((points[:, 1], points[:, 0]))]
    if len(pts) <= 2:
        return 0.0

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = np.array(lower[:-1] + upper[:-1])
    x, y = hull[:, 0], hull[:, 1]
    return 0.5 * abs(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1)))


def heilbronn_convex13() -> np.ndarray:
    """
    Construct an arrangement of n points on or inside a convex region in order to maximize the area of the
    smallest triangle formed by these points. Here n = 13.

    Returns:
        points: np.ndarray of shape (13,2) with the x,y coordinates of the points.
    """
    n = 13
    rng = np.random.default_rng(seed=42)

    # Initialization: quasi-uniform points in unit square (perturbed grid)
    gx, gy = 4, 4  # 16 grid slots, drop 3
    grid_pts = []
    for i in range(gx):
        for j in range(gy):
            grid_pts.append((i, j))
    grid_pts = np.array(grid_pts, dtype=float)
    # drop three slots to leave 13, spread out drops
    keep = [0, 1, 2, 3, 4, 5, 6, 8, 9, 10, 12, 13, 14]
    pts = grid_pts[keep].copy()
    pts = pts / 3.0  # scale to [0,1]
    pts += rng.normal(0, 0.03, pts.shape)
    pts = np.clip(pts, 0.0, 1.0)

    best_pts = pts.copy()
    best_score = _min_triangle_area(best_pts)
    cur_pts = pts.copy()
    cur_score = best_score

    triples = np.array(list(combinations(range(n), 3)))
    tri = np.stack([cur_pts[triples[:, 0]],
                    cur_pts[triples[:, 1]],
                    cur_pts[triples[:, 2]]])

    def anneal(points, score, iterations, step0, temp0, temp_decay, reheat_step=None):
        nonlocal best_pts, best_score
        step = step0
        temp = temp0
        for it in range(iterations):
            i = rng.integers(n)
            old = points[i].copy()
            new = old + rng.normal(0, step, 2)
            if np.any(new < 0) or np.any(new > 1):
                continue
            points[i] = new
            new_score = _min_triangle_area(points)
            if new_score >= score or rng.random() < np.exp((new_score - score) / temp):
                score = new_score
                if new_score > best_score:
                    best_score = new_score
                    best_pts = points.copy()
            else:
                points[i] = old
            temp *= temp_decay
            if it % 200 == 199:
                step = max(step * 0.85, 0.002)
        return points, score, step

    # Main anneal: 2500 iterations
    cur_pts, cur_score, _ = anneal(cur_pts, cur_score, 2500, 0.05, 0.008, 0.999)

    # Adaptive restarts: 3 reheat cycles from the best-ever solution
    for cycle in range(3):
        cur_pts = best_pts.copy()
        cur_score = best_score
        cur_pts, cur_score, _ = anneal(cur_pts, cur_score, 500, 0.05,
                                       0.006 * (0.5 ** cycle), 0.995)

    # Rescale so hull area = 1 (unit-area convex region), preserving shape
    hull = _hull_area(best_pts)
    if hull > 1e-12:
        s = 1.0 / np.sqrt(hull)
        best_pts = (best_pts - best_pts.mean(axis=0)) * s + best_pts.mean(axis=0)

    return best_pts
