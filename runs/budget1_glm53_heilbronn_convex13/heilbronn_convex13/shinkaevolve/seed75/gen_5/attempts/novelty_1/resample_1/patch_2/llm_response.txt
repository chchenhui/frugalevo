    import numpy as np

    rng = np.random.default_rng(12345)

    N = 13

    def _hull_area(pts):
        # Shoelace area of convex hull (use full point set; hull computed via angles)
        try:
            from scipy.spatial import ConvexHull
            return ConvexHull(pts).volume
        except Exception:
            # fallback: monotone chain hull then shoelace
            P = sorted(map(tuple, pts))
            def cross(o, a, b):
                return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
            lower = []
            for p in P:
                while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
                    lower.pop()
                lower.append(p)
            upper = []
            for p in reversed(P):
                while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
                    upper.pop()
                upper.append(p)
            hull = lower[:-1] + upper[:-1]
            s = 0.0
            for i in range(len(hull)):
                x1, y1 = hull[i]; x2, y2 = hull[(i+1) % len(hull)]
                s += x1*y2 - x2*y1
            return abs(s) / 2.0

    def _min_tri_area(pts):
        # vectorized min triangle area
        n = len(pts)
        idx = np.array([(i, j, k) for i in range(n) for j in range(i+1, n) for k in range(j+1, n)])
        a = pts[idx[:, 0]]
        b = pts[idx[:, 1]]
        c = pts[idx[:, 2]]
        areas = 0.5 * np.abs((b[:,0]-a[:,0])*(c[:,1]-a[:,1]) - (b[:,1]-a[:,1])*(c[:,0]-a[:,0]))
        return float(areas.min())

    def _score(pts):
        ha = _hull_area(pts)
        if ha <= 1e-12:
            return 0.0
        return _min_tri_area(pts) / ha

    def _soft_min(areas, p=40.0):
        m = areas.min()
        return float(m - np.log(np.sum(np.exp(-p*(areas - m)))) / p)

    def _candidates():
        cands = []
        # 3-fold symmetric: 3 inner, 3 mid, 3 mid2, 4 outer-ish pattern
        base = []
        angs = np.linspace(0, 2*np.pi, 12, endpoint=False)
        for r, cnt in [(0.30, 3), (0.62, 3), (0.92, 3)]:
            for i in range(cnt):
                a = angs[(i * 12 // cnt) % 12] + (r * 0.5)
                base.append((r*np.cos(a) + 0.5, r*np.sin(a) * 1.1547 * 0.866 + 0.5))
        base.append((0.5, 0.5))
        cands.append(np.array(base[:13] if len(base) >= 13 else base + [(0.5, 0.9)]))
        # Triangular-lattice-like
        lat = []
        for row in range(4):
            y = row / 3.0
            for col in range(4 - row % 2):
                x = (col + 0.5*(row % 2)) / 3.5
                lat.append((x, y))
        cands.append(np.array(lat[:13]))
        # quasi-random (deterministic Sobol via numpy)
        from scipy.stats import qmc
        cands.append(qmc.Sobol(d=2, scramble=True, seed=12345).random(13))
        # random
        cands.append(rng.random((13, 2)))
        return cands

    def _optimize(pts, stages=((0.02, 3000), (0.005, 4000), (0.0015, 3000))):
        pts = np.array(pts, dtype=float)
        # translate/scale into unit square roughly, then optimize raw score (scale invariant)
        best = pts.copy()
        best_s = _score(best)
        for radius, iters in stages:
            cur = best.copy()
            cur_s = best_s
            for it in range(iters):
                i = rng.integers(N)
                delta = rng.normal(0, radius, 2)
                trial = cur.copy()
                trial[i] += delta
                s = _score(trial)
                if s >= cur_s - 1e-12:
                    cur, cur_s = trial, s
                    if cur_s > best_s:
                        best, best_s = cur.copy(), cur_s
        return best, best_s

    best_pts, best_score = None, -1.0
    for cand in _candidates():
        # ensure exactly 13 distinct finite points
        cand = np.array(cand[:13], dtype=float)
        if cand.shape[0] < 13:
            cand = np.vstack([cand, rng.random((13 - cand.shape[0], 2))])
        if not np.all(np.isfinite(cand)):
            continue
        # dedupe jitter
        for i in range(13):
            for j in range(i+1, 13):
                if np.linalg.norm(cand[i]-cand[j]) < 1e-9:
                    cand[j] += rng.normal(0, 1e-4, 2)
        pts, s = _optimize(cand)
        if s > best_score:
            best_pts, best_score = pts, s

    # final normalization: scale so convex hull area = 1 (unit-area region)
    ha = _hull_area(best_pts)
    if ha > 0:
        best_pts = best_pts / np.sqrt(ha)
    else:
        best_pts = np.array([[0.0, 0.0]] * 13)
    return best_pts
