import time
import numpy as np

_N = 13
# All C(13,3)=286 index triples
_IDX = np.array([(i, j, k) for i in range(_N) for j in range(i + 1, _N)
                  for k in range(j + 1, _N)])
# For each point i: rows of _IDX containing i, and their positions in _IDX
_TRIANGLES_WITH = []
for i in range(_N):
    rows = np.where((_IDX == i).any(axis=1))[0]
    _TRIANGLES_WITH.append(rows)


def _full_areas(points):
    a = points[_IDX[:, 0]]
    b = points[_IDX[:, 1]]
    c = points[_IDX[:, 2]]
    return 0.5 * np.abs((b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) -
                        (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0]))


def _areas_for_point(points, i):
    """Areas of the 66 triangles containing point i (closed form, vectorized)."""
    others = np.arange(_N)
    others = others[others != i]
    jj, kk = np.triu_indices(len(others), 1)
    pj = points[others[jj]]
    pk = points[others[kk]]
    pi = points[i]
    return 0.5 * np.abs((pj[:, 0] - pi[0]) * (pk[:, 1] - pi[1]) -
                        (pj[:, 1] - pi[1]) * (pk[:, 0] - pi[0]))


def _hull_area(points):
    pts = sorted(map(tuple, points))
    if len(pts) <= 2:
        return 1e-9
    def cross(o, a, b):
        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    h = np.asarray(hull)
    if len(h) < 3:
        return 1e-9
    x, y = h[:, 0], h[:, 1]
    return 0.5 * abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1)))


def _is_hull_vertex(points, i, tol=1e-12):
    """Check via: point i is a hull vertex iff hull area shrinks without it? 
    Cheaper: point is inside hull of others iff removing it doesn't change hull."""
    rest = points[[j for j in range(_N) if j != i]]
    # quick check: is points[i] inside convex hull of rest?
    # Use cross-product test over hull of rest.
    return abs(_hull_area(rest) - _hull_area(points)) > tol


def _clip(p):
    return np.clip(p, 0.0, 1.0)


def heilbronn_convex13() -> np.ndarray:
    """
    Greedy bottleneck descent with incremental triangle-area updates:
    moving one point only changes its 66 triangles, so we cache all 286
    areas and update only the affected slice on acceptance. Hull area is
    recomputed only when a hull vertex moves.
    """
    n = _N
    rng = np.random.default_rng(seed=20240713)
    t0 = time.time()
    budget = 6.5

    def ring(nc, rad, phase=0.0, cx=0.5, cy=0.5, jitter=0.0):
        ang = phase + 2 * np.pi * np.arange(nc) / nc
        pts = np.stack([cx + rad * np.cos(ang) + rng.normal(0, jitter, nc),
                        cy + rad * np.sin(ang) + rng.normal(0, jitter, nc)],
                       axis=1)
        return _clip(pts)

    inits = []
    # ring of 12 + center
    inits.append(np.vstack([ring(12, 0.47, 0.13), [[0.5, 0.5]]]))
    # ring of 13
    inits.append(ring(13, 0.47, 0.05))
    # two rings: 7 outer, 6 inner
    inits.append(np.vstack([ring(7, 0.47, 0.0), ring(6, 0.25, 0.5)]))
    # outer 8 + inner 4 + center
    inits.append(np.vstack([ring(8, 0.47, 0.2), ring(4, 0.22, 0.9),
                            [[0.5, 0.5]]]))
    # jittered variants for diversity
    inits.append(_clip(np.vstack([ring(12, 0.46, 0.7, jitter=0.03),
                                  [[0.5, 0.5]]])))
    inits.append(ring(13, 0.46, 0.4, jitter=0.03))
    inits.append(_clip(np.vstack([ring(7, 0.47, 0.3, jitter=0.03),
                                  ring(6, 0.24, 0.8, jitter=0.03)])))
    # random disk
    ang = rng.uniform(0, 2 * np.pi, n)
    r = 0.47 * np.sqrt(rng.uniform(0.2, 1.0, n))
    inits.append(_clip(np.stack([0.5 + r * np.cos(ang),
                                 0.5 + r * np.sin(ang)], axis=1)))

    best_pts, best_val = None, -1.0
    restart_budget = budget / len(inits)

    for ri, x0 in enumerate(inits):
        tr = t0 + restart_budget * (ri + 1)
        x = _clip(np.asarray(x0, dtype=float).copy())
        areas = _full_areas(x)
        ha = _hull_area(x)
        cur = areas.min() / ha if ha > 1e-9 else 0.0
        local_best, local_val = x.copy(), cur

        sigma = 0.02
        stall = 0
        rrng = np.random.default_rng(seed=99000 + 137 * ri)

        while time.time() - t0 < budget:
            if time.time() > tr and sigma < 5e-3:
                break
            improved = False
            # try a batch of moves at current sigma
            for _ in range(300):
                if time.time() - t0 > budget:
                    break
                vi = rrng.integers(n)
                cand = x.copy()
                cand[vi] = _clip(cand[vi] + rrng.normal(0, sigma, 2))
                if np.allclose(cand[vi], x[vi]):
                    continue
                new_tri = _areas_for_point(cand, vi)
                rows = _TRIANGLES_WITH[vi]
                old_vals = areas[rows]
                # candidate min: min over changed and unchanged parts
                areas[rows] = new_tri
                cand_min = areas.min()
                # hull: if vi inside hull, hull unchanged; else recompute
                hull_v = _is_hull_vertex(x, vi)
                if hull_v:
                    cand_ha = _hull_area(cand)
                else:
                    cand_ha = ha
                cand_val = cand_min / cand_ha if cand_ha > 1e-9 else 0.0
                if cand_val > cur + 1e-14:
                    x = cand
                    cur = cand_val
                    ha = cand_ha
                    improved = True
                    if cand_val > local_val:
                        local_val = cand_val
                        local_best = x.copy()
                else:
                    areas[rows] = old_vals
            if not improved:
                sigma *= 0.6
                stall += 1
                if sigma < 3e-6:
                    break
            else:
                sigma = min(sigma * 1.15, 0.02)
        if local_val > best_val:
            best_val = local_val
            best_pts = local_best

    if best_pts is None:
        best_pts = np.vstack([ring(12, 0.47), [[0.5, 0.5]]])
    if not np.all(np.isfinite(best_pts)):
        best_pts = _clip(np.nan_to_num(best_pts))
    return best_pts
