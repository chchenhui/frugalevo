<DESCRIPTION>
The previous generations plateaued because all four existing seeds (regular 13-gon, ellipse, perturbed polygon, uniform random) fall into the same family of near-symmetric, boundary-dominated basins, and every local-search refinement converges to ~0.02 normalized min-area. This edit adds three structurally distinct seeds that are constructed directly in the valid domain (all points inside a unit-area convex hull, so no post-hoc rescaling is needed — the failure mode that zeroed Gen 5 and Gen 9): (1) a two-cluster configuration splitting points between two arcs of the boundary, which explores asymmetric boundary layouts; (2) a Reuleaux-like rounded-square boundary with points on the curved hull, exploring a different hull shape family than polygons/ellipses; (3) a spiral fill combining an interior spiral with a few hull anchors, exploring mixed interior/boundary distributions. Each seed is built with fixed parameterization (no RNG or seeded RNG only) for determinism, and the convex hull is scaled to unit area at construction time so the returned configuration is valid-by-construction. Widening the seed set to seven distinct basins gives the downstream multi-scale refinement a materially better chance of escaping the 0.02 plateau.

def _hull_area(pts):
    from scipy.spatial import ConvexHull
    return ConvexHull(pts).volume


def _scale_to_unit_area(pts):
    a = _hull_area(pts)
    return pts / np.sqrt(a)


def _regular_polygon_seeds(n, rng):
    seeds = []
    # regular n-gon on unit-area circle
    R = (1.0 / (0.5 * n * np.sin(2 * np.pi / n))) ** 0.5
    pts = np.array([[R * np.cos(2 * np.pi * i / n), R * np.sin(2 * np.pi * i / n)] for i in range(n)])
    seeds.append(("regular_%dgon" % n, pts.copy()))
    # ellipse-perturbed variant
    a, b = 1.3, 1.0 / 1.3
    pts2 = np.array([[R * a * np.cos(2 * np.pi * i / n), R * b * np.sin(2 * np.pi * i / n)] for i in range(n)])
    hull_area2 = 0.5 * n * R * R * a * b * np.sin(2 * np.pi / n)
    pts2 = pts2 / np.sqrt(hull_area2)
    seeds.append(("ellipse_%dgon" % n, pts2.copy()))

    # --- NEW SEED 1: two-cluster boundary arcs (asymmetric basin) ---
    # Split the n points between two arcs of a circle separated by a gap.
    # Built directly on the hull (boundary points), then scaled to unit hull area.
    nA = n // 2
    nB = n - nA
    arc_pts = []
    # Arc A: centered at angle 0, arc B: centered at angle pi
    halfA = np.pi / 3.0
    halfB = np.pi / 4.0
    for i in range(nA):
        t = -halfA + (2 * halfA) * (i / max(nA - 1, 1))
        arc_pts.append([np.cos(t), np.sin(t)])
    for i in range(nB):
        t = np.pi - halfB + (2 * halfB) * (i / max(nB - 1, 1))
        arc_pts.append([np.cos(t), np.sin(t)])
    arc_pts = np.asarray(arc_pts)
    # circle radius chosen so unit circle hull area ~= area of convex hull here,
    # then rescale exactly so hull area == 1 (construction-time scaling, not post-hoc fixup)
    seeds.append(("two_cluster_arcs", _scale_to_unit_area(arc_pts)))

    # --- NEW SEED 2: Reuleaux-like rounded-square boundary ---
    # Points placed on a superellipse |x|^4 + |y|^4 = r^4 (a "rounded square"),
    # a hull family distinct from polygons and ellipses.
    th = 2 * np.pi * (np.arange(n) + 0.5 * (n % 2)) / n
    se = np.stack([np.sign(np.cos(th)) * np.abs(np.cos(th)) ** 0.5,
                   np.sign(np.sin(th)) * np.abs(np.sin(th)) ** 0.5], axis=1)
    seeds.append(("rounded_square", _scale_to_unit_area(se)))

    # --- NEW SEED 3: interior spiral + hull anchors ---
    # Archimedean spiral interior points plus 3 anchor vertices on a triangle hull,
    # exploring a mixed interior/boundary distribution basin.
    k_anchor = 3
    k_spiral = n - k_anchor
    anchors = np.array([[np.cos(2 * np.pi * i / 3), np.sin(2 * np.pi * i / 3)] for i in range(3)])
    spiral = []
    for i in range(k_spiral):
        t = (i + 1) / float(k_spiral + 1)
        ang = 2 * np.pi * 2.2 * t
        rad = 0.85 * t
        spiral.append([rad * np.cos(ang), rad * np.sin(ang)])
    spiral = np.asarray(spiral)
    sp = np.vstack([anchors, spiral])
    seeds.append(("spiral_plus_anchors", _scale_to_unit_area(sp)))

    return seeds
