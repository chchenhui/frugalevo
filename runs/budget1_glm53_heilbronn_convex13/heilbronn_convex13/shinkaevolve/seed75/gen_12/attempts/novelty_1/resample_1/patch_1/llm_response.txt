Use a structured symmetric initialization (concentric rings with 3-fold symmetry plus center point) followed by a direct local optimization of the exact minimum-triangle-area objective using coordinate descent with shrinking perturbation steps. The exact min over all 286 triangles is computed vectorized; points are pushed toward the boundary where Heilbronn optima live. This deterministic approach exploits known structural properties (boundary-heavy, symmetric configurations) and converges to a strong local optimum of the true objective, vastly outperforming random points.

# EVOLVE-BLOCK-START
import numpy as np


def _min_triangle_area(points: np.ndarray) -> float:
    """Exact minimum area among all C(n,3) triangles, vectorized."""
    n = len(points)
    idx = np.array([(i, j, k) for i in range(n) for j in range(i + 1, n)
                    for k in range(j + 1, n)])
    a = points[idx[:, 0]]
    b = points[idx[:, 1]]
    c = points[idx[:, 2]]
    areas = 0.5 * np.abs((b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) -
                         (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0]))
    return areas.min()


def heilbronn_convex13() -> np.ndarray:
    """
    Construct 13 points in a convex region (unit square) maximizing the
    smallest triangle area, via symmetric initialization + perturbation
    refinement of the exact objective.

    Returns:
        points: np.ndarray of shape (13,2).
    """
    n = 13
    # Symmetric init: center point + inner ring (3 pts) + outer ring (9 pts),
    # inside unit square [0,1]^2, centered at (0.5, 0.5).
    cx, cy = 0.5, 0.5
    pts = [(cx, cy)]
    # inner triangle
    for i in range(3):
        ang = 2 * np.pi * i / 3 - np.pi / 2
        pts.append((cx + 0.16 * np.cos(ang), cy + 0.16 * np.sin(ang)))
    # outer 9-gon (triangle-ish placement near boundary)
    for i in range(9):
        ang = 2 * np.pi * i / 9
        # superellipse-like radius to hug square boundary
        r = 0.47 / max(abs(np.cos(ang)), abs(np.sin(ang)))
        pts.append((cx + r * np.cos(ang), cy + r * np.sin(ang)))
    points = np.array(pts)

    # Project into unit square with margin
    def clip(p):
        return np.clip(p, 0.0, 1.0)

    points = clip(points)

    rng = np.random.default_rng(seed=12345)
    best_val = _min_triangle_area(points)
    step = 0.05
    # Random-restart coordinate perturbation refinement (deterministic seed)
    while step > 1e-7:
        improved = False
        for trial in range(300):
            cand = points + rng.normal(0, step, points.shape)
            cand = clip(cand)
            v = _min_triangle_area(cand)
            if v > best_val + 1e-12:
                points = cand
                best_val = v
                improved = True
        if not improved:
            step *= 0.5

    return points


# EVOLVE-BLOCK-END
