        rng = np.random.default_rng(seed=42)
        N = 13

        TRI = np.array(list(itertools.combinations(range(N), 3)), dtype=int)

        def min_tri_area(P):
            a = P[TRI[:, 0]]
            b = P[TRI[:, 1]]
            c = P[TRI[:, 2]]
            areas = 0.5 * np.abs((b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1])
                                 - (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0]))
            return float(areas.min())

        def convex_hull(P):
            pts = sorted(map(tuple, P))
            uniq = []
            for p in pts:
                if not uniq or p != uniq[-1]:
                    uniq.append(p)

            def cross(o, a, b):
                return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])

            lower = []
            for p in uniq:
                while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
                    lower.pop()
                lower.append(p)
            upper = []
            for p in reversed(uniq):
                while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
                    upper.pop()
                upper.append(p)
            return np.array(lower[:-1] + upper[:-1], dtype=float)

        def hull_area(P):
            H = convex_hull(P)
            x, y = H[:, 0], H[:, 1]
            return 0.5 * abs(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1)))

        def ring(m, r, phase):
