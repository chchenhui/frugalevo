import time
import numpy as np

_N = 13
_IDX = np.array([(i, j, k) for i in range(_N) for j in range(i + 1, _N)
                  for k in range(j + 1, _N)])


def _tri_areas(points, idx=_IDX):
    a = points[idx[:, 0]]
    b = points[idx[:, 1]]
    c = points[idx[:, 2]]
    return 0.5 * np.abs((b[:, 0] - a[:, 0]) * (c[:, 1] - a[:, 1]) -
                        (b[:, 1] - a[:, 1]) * (c[:, 0] - a[:, 0]))


def _hull_area(points):
    # monotone chain convex hull, then shoelace
    pts = sorted(map(tuple, points))
    if len(pts) <= 2:
        return 1e-9
    def cross(o, a, b):
        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    h = np.asarray(hull)
    if len(h) < 3:
        return 1e-9
    x, y = h[:, 0], h[:, 1]
    return 0.5 * abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1)))


def _score(points):
    ha = _hull_area(points)
    if ha < 1e-9:
        return 0.0, None
    areas = _tri_areas(points)
    return areas.min() / ha, areas


def _soft_score(points, k):
    """Smoothed bottleneck: mean of smallest k triangle areas, normalized."""
    ha = _hull_area(points)
    if ha < 1e-9:
        return 0.0
    areas = _tri_areas(points)
    if k <= 1:
        return areas.min() / ha
    ks = np.partition(areas, k - 1)[:k]
    return ks.mean() / ha


def _clip(p):
    return np.clip(p, 0.0, 1.0)


def heilbronn_convex13() -> np.ndarray:
    """
    Optimize 13 points in the unit square for max (min triangle area)/(hull area)
    using multi-restart annealing with coordinated bottleneck-vertex moves.
    Deterministic (fixed seed) and time-budgeted.
    """
    n = _N
    rng = np.random.default_rng(seed=20240713)
    t0 = time.time()
    budget = 7.0  # seconds

    def init_circle(jitter):
        ang = 2 * np.pi * np.arange(n) / n + rng.uniform(0, 2 * np.pi)
        r = 0.45 + rng.normal(0, jitter, n)
        p = 0.5 + r * np.cos(ang), 0.5 + r * np.sin(ang)
        return _clip(np.stack(p, axis=1))

    def init_disk(jitter):
        ang = rng.uniform(0, 2 * np.pi, n)
        r = np.sqrt(rng.uniform(0.05, 0.5, n))
        return _clip(np.stack([0.5 + r*np.cos(ang) + rng.normal(0, jitter, n),
                               0.5 + r*np.sin(ang) + rng.normal(0, jitter, n)],
                              axis=1))

    inits = [init_circle(0.02), init_circle(0.08), init_disk(0.05),
             init_circle(0.15), init_disk(0.10), init_circle(0.30)]

    best_pts, best_val = None, -1.0

    for ri, x0 in enumerate(inits):
        if time.time() - t0 > budget:
            break
        x = x0.copy()
        cur, areas = _score(x)
        k = 12  # soft window
        cur_soft = _soft_score(x, k)
        sigma = 0.03
        temp = 1e-4
        it = 0
        while sigma > 5e-6 and time.time() - t0 < budget:
            it += 1
            # choose move type
            u = rng.random()
            cand = x.copy()
            if u < 0.20 and areas is not None:
                # coordinated move on the 3 vertices of the min triangle
                tri = _IDX[np.argmin(areas)]
                for vi in tri:
                    cand[vi] += rng.normal(0, sigma * 2, 2)
                cand = _clip(cand)
                accept_rule = "strict"
            elif u < 0.55:
                # single random point
                vi = rng.integers(n)
                cand[vi] += rng.normal(0, sigma * 2, 2)
                cand = _clip(cand)
                accept_rule = "soft"
            else:
                # global jitter
                cand = _clip(x + rng.normal(0, sigma, x.shape))
                accept_rule = "soft"

            cand_soft = _soft_score(cand, min(k, max(1, k - it // 4000)))
            cand_val, cand_areas = _score(cand)

            if cand_val > cur or (accept_rule == "soft" and
                                  cand_soft > cur_soft - temp * 0.0):
                if cand_soft > cur_soft or cand_val > cur:
                    x = cand
                    cur_soft = max(cur_soft, cand_soft)
                    cur = max(cur, cand_val)
                    areas = cand_areas
                else:
                    x = cand
                    cur_soft = cand_soft
                    cur = cand_val
                    areas = cand_areas
                if cand_val > best_val:
                    best_val = cand_val
                    best_pts = cand.copy()
            else:
                # occasional Metropolis uphill step on soft objective
                if accept_rule == "soft" and cand_soft > cur_soft - temp:
                    x = cand
                    cur_soft = cand_soft
                    cur, areas = cand_val, cand_areas
            if it % 1500 == 0:
                sigma *= 0.85
                temp *= 0.7
                k = max(3, k - 1)

        # final strict polish of best from this restart
        pol = x.copy()
        pv, pa = _score(pol)
        sig = 2e-3
        while sig > 1e-6 and time.time() - t0 < budget:
            improved = False
            for _ in range(120):
                tri = _IDX[np.argmin(pa)]
                mode = rng.random()
                cand = pol.copy()
                if mode < 0.5:
                    for vi in tri:
                        cand[vi] += rng.normal(0, sig, 2)
                else:
                    for _ in range(3):
                        cand[rng.integers(n)] += rng.normal(0, sig, 2)
                cand = _clip(cand)
                v, a = _score(cand)
                if v > pv + 1e-12:
                    pol, pv, pa = cand, v, a
                    improved = True
            if not improved:
                sig *= 0.5
        if pv > best_val:
            best_val = pv
            best_pts = pol.copy()

    if best_pts is None:
        # fallback: regular 13-gon-ish configuration
        best_pts = init_circle(0.0)
    # ensure finiteness
    if not np.all(np.isfinite(best_pts)):
        best_pts = _clip(np.nan_to_num(best_pts))
    return best_pts
