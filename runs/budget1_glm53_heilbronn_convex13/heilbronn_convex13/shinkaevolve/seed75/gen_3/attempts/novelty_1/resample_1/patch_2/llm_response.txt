def _min_triangle_area(P):
    """Vectorized computation of the minimum (and argmin) triangle area over all C(13,3) triples."""
    n = len(P)
    idx = np.array([(i, j, k) for i in range(n) for j in range(i + 1, n) for k in range(j + 1, n)])
    A, B, C = P[idx[:, 0]], P[idx[:, 1]], P[idx[:, 2]]
    areas = 0.5 * np.abs((B[:, 0] - A[:, 0]) * (C[:, 1] - A[:, 1]) -
                         (B[:, 1] - A[:, 1]) * (C[:, 0] - A[:, 0]))
    m = int(np.argmin(areas))
    return float(areas[m]), idx[m], areas


def _seed_configs():
    """Deterministic structured initial configurations."""
    seeds = []
    # 1) Regular 13-gon shrunk slightly toward center
    t = 2 * np.pi * np.arange(13) / 13
    poly = np.stack([0.45 + 0.42 * np.cos(t), 0.45 + 0.42 * np.sin(t)], axis=1)
    seeds.append(poly)
    # 2) Triangular-lattice-like layout
    pts = []
    for r in range(5):
        for c in range(r + 1):
            pts.append([0.08 + 0.84 * c / 4, 0.08 + 0.84 * r / 4])
    seeds.append(np.array(pts[:13]))
    # 3) 3-fold symmetric shells: 3 outer + 3 mid + 6 inner + center
    pts = [[0.5, 0.06], [0.97, 0.92], [0.03, 0.92]]
    for radius in (0.28, 0.15):
        for a in (np.pi / 2, np.pi / 2 + 2 * np.pi / 3, np.pi / 2 + 4 * np.pi / 3):
            pts.append([0.5 + radius * np.cos(a), 0.5 + radius * np.sin(a)])
    for a in np.linspace(0, 2 * np.pi, 7)[:-1]:
        pts.append([0.5 + 0.07 * np.cos(a), 0.5 + 0.07 * np.sin(a)])
    seeds.append(np.array(pts))
    # 4) Random (seeded)
    rng = np.random.default_rng(1234)
    seeds.append(0.1 + 0.8 * rng.random((13, 2)))
    return seeds


def _optimize(P, rng, iters=4000):
    """Seeded annealing: move the vertex of the smallest triangle along the
    exact gradient direction that increases its area."""
    P = P.copy()
    best_P = P.copy()
    best_val, _, _ = _min_triangle_area(P)
    cur_val = best_val
    step0 = 0.05
    for it in range(iters):
        step = step0 * (0.98 ** (it // 20)) + 1e-4
        val, tri, _ = _min_triangle_area(P)
        cur_val = val
        i, j, k = tri
        # choose the vertex to move: the one whose removal best helps
        for v in (k, j, i):
            a, b = P[tri[0]], P[tri[1]]
            if v == tri[0]:
                a, b = P[tri[1]], P[tri[2]]
            elif v == tri[1]:
                a, b = P[tri[0]], P[tri[2]]
            d = b - a
            L = np.hypot(*d)
            if L < 1e-9:
                continue
            perp = np.array([-d[1], d[0]]) / L
            # ensure direction increases area for vertex P[v]
            increase = np.sign(np.dot(np.cross(np.append(d, 0), np.append(P[v] - a, 0)), [0, 0, 1]))
            newv = P[v] + step * perp * increase + 0.3 * step * rng.standard_normal(2)
            newv = np.clip(newv, 0.0, 1.0)
            old = P[v].copy()
            P[v] = newv
            nv, _, _ = _min_triangle_area(P)
            # annealing acceptance
            if nv >= val - 0.5 * step * step or rng.random() < 0.02:
                if nv > best_val:
                    best_val = nv
                    best_P = P.copy()
                break
            P[v] = old
        # occasional random kick to escape local optima
        if it % 250 == 249:
            v = rng.integers(0, 13)
            P[v] = np.clip(P[v] + 0.03 * rng.standard_normal(2), 0.0, 1.0)
    return best_P, best_val


def solve():
    """Deterministic multi-seed annealing for the 13-point Heilbronn problem."""
    try:
        best_P, best_val = None, -1.0
        for si, seed in enumerate(_seed_configs()):
            rng = np.random.default_rng(1000 + si)
            P, val = _optimize(np.asarray(seed, dtype=float), rng)
            if val > best_val:
                best_val, best_P = val, P
        # final greedy polish: small gradient-only refinement
        rng = np.random.default_rng(7)
        P, val = _optimize(best_P, rng, iters=1200)
        if val > best_val:
            best_val, best_P = val, P
        # validate: finite, unique
        assert np.all(np.isfinite(best_P)), "non-finite coordinates"
        if np.min(np.linalg.norm(best_P[:, None] - best_P[None], axis=2) + np.eye(13)) < 1e-9:
            # nudge duplicates apart deterministically
            rng = np.random.default_rng(0)
            best_P = best_P + 1e-6 * rng.standard_normal(best_P.shape)
        return np.asarray(best_P, dtype=float)
    except Exception:
        # Fallback: regular polygon (always feasible and deterministic)
        t = 2 * np.pi * np.arange(13) / 13
        return np.stack([0.5 + 0.45 * np.cos(t), 0.5 + 0.45 * np.sin(t)], axis=1)
