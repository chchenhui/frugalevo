        from itertools import combinations

        m = n  # number of points
        # Precompute all C(13,3) = 286 triangle index triples once.
        combos = np.array(list(combinations(range(m), 3)), dtype=np.int64)

        def min_tri_area(P):
            v1 = P[combos[:, 1]] - P[combos[:, 0]]
            v2 = P[combos[:, 2]] - P[combos[:, 0]]
            cross = v1[:, 0] * v2[:, 1] - v1[:, 1] * v2[:, 0]
            return 0.5 * float(np.abs(cross).min())

        def climb(P, step_sizes, iters, rng):
            P = P.copy()
            best = min_tri_area(P)
            for it in range(iters):
                step = step_sizes[(it * len(step_sizes)) // iters]
                idx = int(rng.integers(m))
                old = P[idx].copy()
                P[idx] = np.clip(old + rng.normal(0.0, step, size=2), 0.0, 1.0)
                val = min_tri_area(P)
                if val > best + 1e-15:
                    best = val
                else:
                    P[idx] = old
            return P, best
