    from itertools import combinations

    n = 13
    # Precompute all index triples for the C(13,3) = 286 triangles once.
    _combos = np.array(list(combinations(range(n), 3)), dtype=np.int64)

    def _min_triangle_area(P):
        """Minimum area over all 286 triangles, vectorized."""
        v1 = P[_combos[:, 1]] - P[_combos[:, 0]]
        v2 = P[_combos[:, 2]] - P[_combos[:, 0]]
        cross = v1[:, 0] * v2[:, 1] - v1[:, 1] * v2[:, 0]
        return 0.5 * float(np.abs(cross).min())

    def _clip(P):
        return np.clip(P, 0.0, 1.0)

    def _hill_climb(P, steps, iters, rng):
        """Greedy single-point perturbation with decaying step size."""
        P = P.copy()
        best = _min_triangle_area(P)
        for it in range(iters):
            step = steps[max(0, len(steps) - 1 - (it * len(steps)) // iters)]
            idx = rng.integers(n)
            delta = rng.normal(0.0, step, size=2)
            old = P[idx].copy()
            P[idx] = _clip(old + delta)
            val = _min_triangle_area(P)
            if val > best + 1e-15:
                best = val
            else:
                P[idx] = old
        return P, best

    rng = np.random.default_rng(20240613)
    steps = np.array([0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.001], dtype=float)

    best_pts, best_val = None, -1.0

    # Multi-start: one jittered grid start + several random starts.
    starts = []
    # Jittered grid start (deterministic jitter from the same rng stream).
    side = 4
    g = np.stack(np.meshgrid(np.linspace(0.02, 0.98, side),
                             np.linspace(0.02, 0.98, side)), axis=-1).reshape(-1, 2)
    g = g[rng.permutation(len(g))[:n]]
    g = _clip(g + rng.normal(0.0, 0.01, g.shape))
    starts.append(g)
    for _ in range(11):
        starts.append(rng.random((n, 2)))

    for P0 in starts:
        P, val = _hill_climb(P0, steps, 3000, rng)
        if val > best_val:
            best_val, best_pts = val, P.copy()

    # Fine polish on the best configuration.
    best_pts, best_val = _hill_climb(best_pts, steps[-3:], 1500, rng)

    # Safety: ensure finite, unique, in-bounds points (should always hold).
    best_pts = _clip(np.nan_to_num(best_pts, nan=0.5))
    for i in range(n):
        for j in range(i + 1, n):
            if np.allclose(best_pts[i], best_pts[j], atol=1e-9):
                best_pts[j] = _clip(best_pts[j] + np.array([1e-6, 1e-6]))

    points = best_pts.tolist()
    return points
