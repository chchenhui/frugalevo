import math

def _triangle_area(p, q, r):
    return abs((q[0] - p[0]) * (r[1] - p[1]) - (q[1] - p[1]) * (r[0] - p[0])) / 2.0

def generate_points():
    # Deterministic 6-fold symmetric configuration for n = 13:
    #   - 1 point at the center
    #   - 6 points on a regular hexagon of circumradius R (outer ring)
    #   - 6 points on a regular hexagon of circumradius r = R/sqrt(3),
    #     rotated by 30 degrees (inner ring)
    # The convex hull (outer hexagon) has area (3*sqrt(3)/2)*R^2 = 1.
    sqrt3 = math.sqrt(3.0)
    R = math.sqrt(2.0 / (3.0 * sqrt3))   # makes hull area exactly 1
    r = R / sqrt3

    pts = [(0.0, 0.0)]
    for k in range(6):
        a = math.pi / 3.0 * k
        pts.append((R * math.cos(a), R * math.sin(a)))
    for k in range(6):
        a = math.pi / 3.0 * k + math.pi / 6.0
        pts.append((r * math.cos(a), r * math.sin(a)))

    pts = [(float(x), float(y)) for x, y in pts]

    # ---- Validation (cheap, deterministic sanity checks) ----
    assert len(pts) == 13, "must return exactly 13 points"
    assert all(math.isfinite(x) and math.isfinite(y) for x, y in pts), \
        "all coordinates must be finite"
    # uniqueness
    for i in range(len(pts)):
        for j in range(i + 1, len(pts)):
            assert math.hypot(pts[i][0] - pts[j][0], pts[i][1] - pts[j][1]) > 1e-12, \
                "duplicate points detected"
    # minimum triangle area over all C(13,3) = 286 triples
    min_area = min(
        _triangle_area(pts[i], pts[j], pts[k])
        for i in range(13) for j in range(i + 1, 13) for k in range(j + 1, 13)
    )
    # Theoretical minimum for this construction: R^2 / (4*sqrt(3))
    expected = (R * R) / (4.0 * sqrt3)
    assert abs(min_area - expected) < 1e-12, \
        "computed min triangle area deviates from analytic value"

    return pts
