def solve(n):
    """Heilbronn triangle solver: place n points in a convex region to
    maximize the smallest normalized triangle area.

    Deterministic multi-start + greedy coordinate hill-climbing.
    Returns a list of (x, y) tuples whose convex hull has unit area.
    """
    import math
    import numpy as np

    try:
        if n is None:
            n = 13
        n = int(n)
        if n < 3:
            raise ValueError("n must be at least 3")
    except Exception:
        n = 13

    # ---------- geometry helpers ----------

    def hull_area(pts):
        """Convex hull area via Andrew's monotone chain (robust, exact)."""
        P = sorted(set(map(tuple, np.round(np.asarray(pts, float), 12))))
        if len(P) < 3:
            return 0.0

        def cross(o, a, b):
            return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

        lower = []
        for p in P:
            while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
                lower.pop()
            lower.append(p)
        upper = []
        for p in reversed(P):
            while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
                upper.pop()
            upper.append(p)
        hull = lower[:-1] + upper[:-1]
        if len(hull) < 3:
            return 0.0
        s = 0.0
        m = len(hull)
        for i in range(m):
            x1, y1 = hull[i]
            x2, y2 = hull[(i + 1) % m]
            s += x1 * y2 - x2 * y1
        return abs(s) / 2.0

    def min_tri_area(p):
        """Smallest (absolute) triangle area over all triples, vectorized."""
        p = np.asarray(p, float)
        m = len(p)
        best = np.inf
        for i in range(m - 2):
            d = p[i + 1:] - p[i]
            x = d[:, 0]
            y = d[:, 1]
            C = np.abs(x[:, None] * y[None, :] - y[:, None] * x[None, :])
            iu = np.triu_indices(len(d), k=1)
            v = C[iu]
            if v.size:
                mv = v.min()
                if mv < best:
                    best = mv
        return best / 2.0 if np.isfinite(best) else 0.0

    def score(pts):
        ha = hull_area(pts)
        if ha <= 0:
            return 0.0
        return min_tri_area(pts) / ha

    # ---------- deterministic local search ----------

    DIRS = [(math.cos(2 * math.pi * k / 8), math.sin(2 * math.pi * k / 8))
            for k in range(8)]

    def hill_climb(p, radii=(0.30, 0.15, 0.06, 0.03, 0.012, 0.005)):
        p = np.array(p, float).copy()
        cur = score(p)
        for r in radii:
            for _sweep in range(3):  # fixed sweep count -> deterministic
                improved = False
                for i in range(len(p)):
                    for (dx, dy) in DIRS:
                        cand = p.copy()
                        cand[i, 0] += dx * r
                        cand[i, 1] += dy * r
                        s = score(cand)
                        if s > cur + 1e-12:
                            p, cur = cand, s
                            improved = True
                if not improved:
                    break
        return p, cur

    # ---------- deterministic initializations ----------

    seeds = []

    # single ring on a circle
    seeds.append([(math.cos(2 * math.pi * i / n),
                   math.sin(2 * math.pi * i / n)) for i in range(n)])

    # two-ring configurations: k outer points, n-k inner points
    for k in (7, 8, 9):
        outer = [(math.cos(2 * math.pi * i / k),
                  math.sin(2 * math.pi * i / k)) for i in range(k)]
        inner_r = 0.42
        inner = [(inner_r * math.cos(2 * math.pi * i / (n - k) + 0.3),
                  inner_r * math.sin(2 * math.pi * i / (n - k) + 0.3))
                 for i in range(n - k)]
        seeds.append(outer + inner)

    # seeded uniform random starts (fixed RNG -> fully reproducible)
    rng = np.random.default_rng(12345)
    for _ in range(6):
        seeds.append(rng.uniform(-1.0, 1.0, size=(n, 2)).tolist())

    # perturbed triangular lattice
    lat = []
    rows = 4
    idx = 0
    while len(lat) < n:
        for r_i in range(rows):
            for c_i in range(rows - r_i):
                if len(lat) >= n:
                    break
                lat.append(((-1.0 + 0.66 * c_i + 0.33 * r_i),
                            (-1.0 + 0.66 * r_i)))
        idx += 1
        if idx > 3:
            break
    if len(lat) >= n:
        seeds.append(lat[:n])

    # ---------- optimize ----------

    best_pts = None
    best_val = -1.0
    try:
        for s in seeds:
            p = np.asarray(s, float)
            if len(p) != n:
                continue
            p2, val = hill_climb(p)
            if val > best_val:
                best_val = val
                best_pts = p2.copy()
    except Exception:
        # graceful fallback: keep whatever we have, else ring baseline
        if best_pts is None:
            best_pts = np.array(seeds[0], float)
            best_val = score(best_pts)

    if best_pts is None:
        best_pts = np.array(seeds[0], float)

    # ---------- normalize to unit hull area ----------
    ha = hull_area(best_pts)
    if ha > 0:
        best_pts = best_pts / math.sqrt(ha)

    return [(float(x), float(y)) for (x, y) in best_pts]
