        def _hull_area(pts):
            P = sorted(set(map(tuple, np.round(np.asarray(pts, float), 12))))
            if len(P) < 3:
                return 0.0

            def cross(o, a, b):
                return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

            lower = []
            for p in P:
                while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
                    lower.pop()
                lower.append(p)
            upper = []
            for p in reversed(P):

                    upper.pop()
                upper.append(p)

        def _min_tri(p):
