soft_quantile_phase</NAME>
<DESCRIPTION>
The pure soft-min objective collapses all gradient mass onto the single bottleneck triangle, which strands other near-degenerate triangles at the 0.02 plateau. Adding a second gradient phase that minimizes a soft τ-quantile of the 286 areas (τ ≈ 0.05–0.10) spreads gradient weight across the whole weak tail, matching the true "maximize the minimum" structure better and letting the plateau break. I add a differentiable quantile surrogate via a soft indicator (sigmoid on area − threshold), take its exact gradient with respect to points, and run it as an Adam phase on the incumbent best configuration before the final polish. The surrogate uses a threshold annealed from the min area upward, so it focuses on exactly the bottom ~5–10% of triangles.

def _quantile_grad(pts, tau=0.07, beta=200.0):
    """Soft tau-quantile of triangle areas with exact gradient.
    Threshold: tau-quantile of areas (exact); objective
    Q = sum_i a_i * s_i / sum_i s_i, s_i = sigmoid(beta*(a_i - thr)).
    Spreads gradient over the weak tail instead of the single min."""
    c, u, v = _tri_signed2(pts)
    s = np.sign(c); s[s == 0] = 1.0
    a = 0.5 * np.abs(c)
    k = int(np.clip(round(tau * len(a)), 1, len(a) - 1))
    thr = np.partition(a, k - 1)[k - 1]
    wgt = 1.0 / (1.0 + np.exp(beta * (a - thr)))   # soft indicator a <= thr
    denom = wgt.sum()
    if denom < 1e-12:
        wgt = np.ones_like(a)
        denom = len(a)
    Q = float((a * wgt).sum() / denom)
    # dQ/da_i = wgt_i/denom * (1 - beta*(a_i - Q_ish) ... )
    # exact: dQ/da = w*(1 + beta*(Q - a))/denom  (from product rule, both a and w depend on a)
    dq = wgt * (1.0 + beta * (Q - a)) / denom
    q = 0.5 * s * dq
    gA_t = np.stack([q*(u[:, 1] - v[:, 1]), q*(v[:, 0] - u[:, 0])], axis=1)
    gB_t = np.stack([q*v[:, 1], -q*v[:, 0]], axis=1)
    gC_t = np.stack([-q*u[:, 1], q*u[:, 0]], axis=1)
    g = np.zeros((N, 2))
    np.add.at(g, I0, gA_t)
    np.add.at(g, I1, gB_t)
    np.add.at(g, I2, gC_t)
    return Q, g


def _adam_quantile_run(pts, deadline, lr=0.01, tau=0.07):
    """Adam ascent on the soft tau-quantile objective; tracks exact score."""
    pts = _rescale_unit_hull(pts.copy())
    m = np.zeros((N, 2)); vv = np.zeros((N, 2))
    b1, b2, eps = 0.9, 0.999, 1e-8
    beta = 200.0
    step = 0
    best_pts, best_sc = pts.copy(), _score(pts)
    while time.time() < deadline:
        for _ in range(20):
            step += 1
            Q, g = _quantile_grad(pts, tau, beta)
            m = b1*m + (1-b1)*g
            vv = b2*vv + (1-b2)*g*g
            mh = m / (1 - b1**step)
            vh = vv / (1 - b2**step)
            pts = pts + lr * mh / (np.sqrt(vh) + eps)
            pts = _rescale_unit_hull(pts)
            sc = _score(pts)
            if sc > best_sc + 1e-14:
                best_sc = sc
                best_pts = pts.copy()
        beta = min(beta * 1.3, 5000.0)
        tau = max(tau * 0.85, 0.02)
        lr *= 0.95
        if lr < 1e-5:
            break
    return best_pts, best_sc
