<DESCRIPTION>
The original implementation could corrupt row identity and column placement during recursive reconstruction. In particular, it wrote completed futures into `result_df.loc[i]` in completion order rather than source-row order, rebuilt frames from `.values.tolist()` without preserving labels or indexes, appended and later merged an `original_index` helper column, and finally returned an empty column-ordering list. Those behaviors violate the required API and can cause validation failures even when shapes happen to match.

The replacement keeps every original value, row, index, and column exactly once. It computes serialized values only for scoring heuristics, evaluates a bounded set of candidate column layouts using the exact sorted-string LCP objective, and returns valid column orderings. It supports global layouts, merge/dependency constraints, mixed values, missing values, and row-specific conditional layouts without mutating source cells.
</DESCRIPTION>

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List, Dict
from collections import defaultdict
import heapq


class Evolved(Algorithm):
    """
    Prefix-cache-oriented column reorderer.

    The dataframe is never modified in place.  Values are only converted to
    strings for candidate scoring; returned cells remain the source objects.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _serialized_value(value) -> str:
        """Match fillna('').astype(str) for ordinary scalar dataframe cells."""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool) and missing:
                return ""
        except Exception:
            pass
        try:
            return str(value)
        except Exception:
            return repr(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        if left == right:
            return len(left)
        upper = min(len(left), len(right))
        lower = 0
        # Slice comparison performs the character comparison in C and avoids
        # a Python loop over potentially long prompt strings.
        while lower < upper:
            middle = (lower + upper + 1) // 2
            if left[:middle] == right[:middle]:
                lower = middle
            else:
                upper = middle - 1
        return lower

    def _trie_score(self, strings: List[str]) -> int:
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        return sum(self._lcp(ordered[i - 1], ordered[i]) for i in range(1, len(ordered)))

    @staticmethod
    def _column_scores(text):
        """Length-weighted repeated-value score for each source column."""
        nrows = len(text)
        ncols = len(text[0]) if nrows else 0
        scores = [0] * ncols
        for col in range(ncols):
            counts = {}
            lengths = {}
            for row in range(nrows):
                value = text[row][col]
                counts[value] = counts.get(value, 0) + 1
                lengths[value] = len(value)
            scores[col] = sum(
                lengths[value] * count * (count - 1)
                for value, count in counts.items()
                if count > 1
            )
        return scores

    @staticmethod
    def _name_positions(columns):
        positions = defaultdict(list)
        for i, name in enumerate(columns):
            positions[name].append(i)
        return positions

    def _units_with_constraints(self, columns, scores, col_merge, one_way_dep):
        """
        Build indivisible merge units and a deterministic dependency-respecting
        global order.  Dependencies are interpreted as parent-before-child.
        """
        ncols = len(columns)
        name_positions = self._name_positions(columns)
        parent = list(range(ncols))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a, b):
            a, b = find(a), find(b)
            if a != b:
                parent[b] = a

        for merge_group in col_merge or []:
            found = []
            for name in merge_group:
                found.extend(name_positions.get(name, []))
            for pos in found[1:]:
                union(found[0], pos)

        grouped = defaultdict(list)
        for pos in range(ncols):
            grouped[find(pos)].append(pos)

        # Preserve requested member ordering inside an explicitly merged unit.
        units = list(grouped.values())
        for unit in units:
            unit.sort()

        unit_of = {}
        for unit_id, unit in enumerate(units):
            for pos in unit:
                unit_of[pos] = unit_id

        edges = defaultdict(set)
        indegree = [0] * len(units)

        def matching_position(token):
            # Existing API historically allowed substring dependency names.
            exact = name_positions.get(token)
            if exact:
                return exact[0]
            for pos, name in enumerate(columns):
                if str(token) in str(name):
                    return pos
            return None

        for pair in one_way_dep or []:
            if len(pair) != 2:
                continue
            first = matching_position(pair[0])
            second = matching_position(pair[1])
            if first is None or second is None:
                continue
            a, b = unit_of[first], unit_of[second]
            if a != b and b not in edges[a]:
                edges[a].add(b)
                indegree[b] += 1

        unit_score = [sum(scores[p] for p in unit) for unit in units]
        ready = []
        for unit_id, degree in enumerate(indegree):
            if degree == 0:
                heapq.heappush(ready, (-unit_score[unit_id], min(units[unit_id]), unit_id))

        ordered_units = []
        while ready:
            _, _, unit_id = heapq.heappop(ready)
            ordered_units.append(unit_id)
            for child in edges[unit_id]:
                indegree[child] -= 1
                if indegree[child] == 0:
                    heapq.heappush(ready, (-unit_score[child], min(units[child]), child))

        # Cyclic dependency input remains deterministic and preserves all data.
        if len(ordered_units) != len(units):
            remaining = [u for u in range(len(units)) if u not in set(ordered_units)]
            remaining.sort(key=lambda u: (-unit_score[u], min(units[u])))
            ordered_units.extend(remaining)

        return [pos for unit_id in ordered_units for pos in units[unit_id]]

    def _conditional_orders(self, text, global_order, max_depth):
        """
        Construct one bounded conditional-prefix candidate.  It is deliberately
        disabled by callers when explicit merge/dependency constraints require
        a single global ordering.
        """
        nrows = len(text)
        ncols = len(global_order)
        orders = [None] * nrows
        nodes_seen = [0]
        candidate_limit = min(32, ncols)
        node_limit = 512

        def finish(rows, prefix, remaining):
            tail = [col for col in global_order if col in remaining]
            final = prefix + tail
            for row in rows:
                orders[row] = final

        def visit(rows, remaining, prefix, depth):
            if (len(rows) < 2 or not remaining or depth >= max_depth or
                    nodes_seen[0] >= node_limit):
                finish(rows, prefix, remaining)
                return
            nodes_seen[0] += 1

            # Cheap candidate set biased toward globally useful fields.
            candidates = [c for c in global_order if c in remaining][:candidate_limit]
            best_col = None
            best_score = 0
            for col in candidates:
                counts = {}
                for row in rows:
                    value = text[row][col]
                    counts[value] = counts.get(value, 0) + 1
                score = sum(
                    len(value) * count * (count - 1)
                    for value, count in counts.items()
                    if count > 1
                )
                if score > best_score or (score == best_score and
                                          best_col is not None and col < best_col):
                    best_col, best_score = col, score

            if best_col is None or best_score <= 0:
                finish(rows, prefix, remaining)
                return

            buckets = defaultdict(list)
            for row in rows:
                buckets[text[row][best_col]].append(row)
            if len(buckets) <= 1:
                finish(rows, prefix, remaining)
                return

            next_remaining = [c for c in remaining if c != best_col]
            for key in sorted(buckets):
                visit(buckets[key], next_remaining, prefix + [best_col], depth + 1)

        visit(list(range(nrows)), list(range(ncols)), [], 0)
        for row in range(nrows):
            if orders[row] is None:
                orders[row] = list(global_order)
        return orders

    @staticmethod
    def _strings_for_orders(text, orders):
        return ["".join(text[row][col] for col in order) for row, order in enumerate(orders)]

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        # Empty frames still need a valid ordering result.
        nrows, ncols = df.shape
        columns = list(df.columns)
        if ncols == 0:
            return df.copy(), [[] for _ in range(nrows)]

        values = df.to_numpy(dtype=object, copy=True)
        text = [
            [self._serialized_value(values[row, col]) for col in range(ncols)]
            for row in range(nrows)
        ]
        scores = self._column_scores(text)

        baseline = list(range(ncols))
        global_order = self._units_with_constraints(
            columns, scores, col_merge or [], one_way_dep or []
        )

        candidates = [baseline]
        if global_order != baseline:
            candidates.append(global_order)

        # Conditional row-specific layouts cannot safely preserve a global
        # merge/dependency ordering, so use them only when no such constraint
        # was supplied.  Bound work on large/wide inputs.
        if (not col_merge and not one_way_dep and nrows > 1 and ncols > 1 and
                nrows <= 50000):
            depth = ncols if col_stop is None else max(1, min(ncols, int(col_stop)))
            conditional = self._conditional_orders(text, global_order, min(depth, 12))
            candidates.append(conditional)

        best_orders = None
        best_score = -1
        for candidate in candidates[:3]:
            if candidate and isinstance(candidate[0], list):
                orders = candidate
            else:
                orders = [list(candidate) for _ in range(nrows)]
            score = self._trie_score(self._strings_for_orders(text, orders))
            if score > best_score:
                best_score = score
                best_orders = orders

        # A uniform layout can retain its natural reordered dataframe columns.
        uniform = all(order == best_orders[0] for order in best_orders) if nrows else True
        if uniform:
            order = best_orders[0] if nrows else global_order
            output = pd.DataFrame(
                values[:, order],
                index=df.index.copy(),
                columns=[columns[pos] for pos in order],
                dtype=object,
            )
        else:
            # For row-specific layouts, each physical output position contains
            # the source column named by that row's ordering entry.
            out_values = [
                [values[row, pos] for pos in best_orders[row]]
                for row in range(nrows)
            ]
            output = pd.DataFrame(out_values, index=df.index.copy(), columns=columns, dtype=object)

        column_orderings = [
            [columns[pos] for pos in order]
            for order in best_orders
        ]
        return output, column_orderings
