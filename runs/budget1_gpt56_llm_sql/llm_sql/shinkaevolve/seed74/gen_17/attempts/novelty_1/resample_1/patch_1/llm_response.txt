<DESCRIPTION>
The original implementation failed validation because it could corrupt row identity and column/value alignment: it added and removed an `original_index` column, merged dataframes, rebuilt rows from `.values.tolist()`, returned empty column orderings, and finally sorted mixed-type data using all columns. It also relied on attribute access for column names, which is invalid for many dataframe column labels.

This replacement preserves every original value, row, and column. It uses normalized serialized cell values only for scoring candidate column orders, evaluates bounded alternatives with exact sorted-string LCP scoring, and returns valid per-row column orderings. It avoids destructive column merging, unsafe mixed-type sorting, recursive dataframe reconstruction, and evaluator-state assumptions.

# EVOLVE-BLOCK-START
import math
from collections import Counter, defaultdict
from typing import Dict, List, Tuple

import pandas as pd
from solver import Algorithm


class Evolved(Algorithm):
    """
    Bounded prompt-prefix-oriented dataframe reordering.

    The returned dataframe always contains exactly the source cells.  For a
    global ordering its columns are reordered normally.  For a conditional
    ordering, each output row is a permutation of its source row and the
    matching entry in ``column_orderings`` identifies the source column stored
    in each output position.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _serial_value(value) -> str:
        """Match the evaluator's fillna('').astype(str) behaviour for scalars."""
        if value is None or value is pd.NA:
            return ""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool) and missing:
                return ""
        except Exception:
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        if left == right:
            return len(left)
        upper = min(len(left), len(right))
        low = 0
        # Prefix slice comparisons execute the character comparison in C and
        # avoid a Python loop over potentially long prompt strings.
        while low < upper:
            middle = (low + upper + 1) // 2
            if left[:middle] == right[:middle]:
                low = middle
            else:
                upper = middle - 1
        return low

    def _trie_score(self, rows: List[str]) -> int:
        if len(rows) < 2:
            return 0
        ordered = sorted(rows)
        return sum(self._lcp(ordered[i - 1], ordered[i]) for i in range(1, len(ordered)))

    @staticmethod
    def _resolve_name(columns: List, requested):
        """Keep the historic API's exact-name-then-unique-substring behaviour."""
        if requested in columns:
            return requested
        matches = [name for name in columns if str(requested) in str(name)]
        return matches[0] if len(matches) == 1 else None

    def _merged_blocks(self, columns: List, col_merge: List[List[str]]) -> List[List]:
        """
        ``col_merge`` is treated as a required contiguous ordering group.
        Unlike the old implementation it never combines or removes cells.
        """
        used = set()
        blocks = []
        for requested_group in col_merge or []:
            group = []
            for requested in requested_group:
                name = self._resolve_name(columns, requested)
                if name is not None and name not in used:
                    group.append(name)
                    used.add(name)
            if group:
                blocks.append(group)
        for name in columns:
            if name not in used:
                blocks.append([name])
        return blocks

    def _apply_dependencies(
        self,
        blocks: List[List],
        scores: Dict,
        one_way_dep: List[Tuple[str, str]],
        columns: List,
    ) -> List[List]:
        """Topologically order merge blocks while preferring high-value blocks."""
        block_of = {name: index for index, block in enumerate(blocks) for name in block}
        edges = defaultdict(set)
        indegree = [0] * len(blocks)

        for before_requested, after_requested in one_way_dep or []:
            before = self._resolve_name(columns, before_requested)
            after = self._resolve_name(columns, after_requested)
            if before is None or after is None:
                continue
            source, target = block_of[before], block_of[after]
            if source != target and target not in edges[source]:
                edges[source].add(target)
                indegree[target] += 1

        ready = [i for i in range(len(blocks)) if indegree[i] == 0]
        result = []
        while ready:
            ready.sort(key=lambda i: (-sum(scores.get(c, 0) for c in blocks[i]), i))
            current = ready.pop(0)
            result.append(blocks[current])
            for target in sorted(edges[current]):
                indegree[target] -= 1
                if indegree[target] == 0:
                    ready.append(target)

        # A cyclic dependency cannot be simultaneously honoured.  Keep the
        # remaining blocks deterministically rather than dropping any data.
        if len(result) != len(blocks):
            present = {id(block) for block in result}
            result.extend(block for block in blocks if id(block) not in present)
        return result

    def _global_orders(
        self,
        columns: List,
        cell_strings: List[List[str]],
        col_merge: List[List[str]],
        one_way_dep: List[Tuple[str, str]],
    ) -> List[List]:
        counts = [Counter(cell_strings[r][c] for r in range(len(cell_strings)))
                  for c in range(len(columns))]

        pair_scores = {}
        simple_scores = {}
        for col_index, name in enumerate(columns):
            pair_scores[name] = sum(
                len(value) * count * (count - 1)
                for value, count in counts[col_index].items()
            )
            simple_scores[name] = sum(
                len(value) * max(0, count - 1)
                for value, count in counts[col_index].items()
            )

        blocks = self._merged_blocks(columns, col_merge)
        pair_blocks = self._apply_dependencies(blocks, pair_scores, one_way_dep, columns)
        simple_blocks = self._apply_dependencies(blocks, simple_scores, one_way_dep, columns)

        def flatten(ordered_blocks, score):
            answer = []
            for block in ordered_blocks:
                # Preserve the caller's merge sequence, except that independent
                # singleton blocks have already been selected by block score.
                if len(block) == 1:
                    answer.extend(block)
                else:
                    answer.extend(sorted(block, key=lambda c: (-score.get(c, 0), columns.index(c))))
            return answer

        pair_order = flatten(pair_blocks, pair_scores)
        simple_order = flatten(simple_blocks, simple_scores)
        original_order = [name for block in blocks for name in block]
        return [pair_order, simple_order, original_order]

    def _conditional_orders(
        self,
        columns: List,
        cell_strings: List[List[str]],
        fallback_order: List,
    ) -> List[List]:
        """
        Build a bounded conditional prefix partition tree.  It is deliberately
        used only on modest unconstrained tables; global alternatives remain
        available for very wide data or explicit ordering constraints.
        """
        nrows, ncols = len(cell_strings), len(columns)
        positions = {name: i for i, name in enumerate(columns)}
        fallback_positions = [positions[name] for name in fallback_order]
        result = [None] * nrows

        # Avoid an expensive tree on instances where its construction would
        # dominate the actual prompt-cache benefit.
        if nrows == 0 or ncols == 0 or nrows * ncols > 500000:
            return [list(fallback_order) for _ in range(nrows)]

        def assign(rows, remaining, depth):
            if not rows:
                return
            if not remaining:
                for row in rows:
                    result[row] = []
                return
            if len(rows) < 2 or depth >= min(12, ncols):
                tail = [p for p in fallback_positions if p in remaining]
                for row in rows:
                    result[row] = tail
                return

            # Bound consideration on wide groups using the globally useful tail.
            candidates = [p for p in fallback_positions if p in remaining][:32]
            best_col = None
            best_score = 0
            for col in candidates:
                frequency = Counter(cell_strings[row][col] for row in rows)
                score = sum(
                    len(value) * count * (count - 1)
                    for value, count in frequency.items()
                    if count > 1
                )
                if score > best_score:
                    best_score = score
                    best_col = col

            if best_col is None or best_score <= 0:
                tail = [p for p in fallback_positions if p in remaining]
                for row in rows:
                    result[row] = tail
                return

            next_remaining = tuple(p for p in remaining if p != best_col)
            partitions = defaultdict(list)
            for row in rows:
                partitions[cell_strings[row][best_col]].append(row)

            for group in partitions.values():
                assign(group, next_remaining, depth + 1)
                for row in group:
                    result[row] = [best_col] + result[row]

        assign(list(range(nrows)), tuple(range(ncols)), 0)
        return [[columns[p] for p in order] for order in result]

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        # No source dataframe mutation and no synthetic index column.
        source = df.copy(deep=True)
        columns = list(source.columns)
        nrows, ncols = source.shape

        if ncols == 0:
            return source, [[] for _ in range(nrows)]

        values = source.to_numpy(dtype=object, copy=False)
        cell_strings = [
            [self._serial_value(values[row, col]) for col in range(ncols)]
            for row in range(nrows)
        ]

        global_orders = self._global_orders(columns, cell_strings, col_merge, one_way_dep)
        candidates = []

        for order in global_orders:
            positions = [columns.index(name) for name in order]
            serialized = ["".join(cell_strings[row][col] for col in positions) for row in range(nrows)]
            candidates.append((self._trie_score(serialized), 0, order, None))

        # Conditional row-specific orders are incompatible with required merge
        # and dependency ordering constraints, so retain only valid globals then.
        if not col_merge and not one_way_dep:
            conditional = self._conditional_orders(columns, cell_strings, global_orders[0])
            serialized = []
            for row, order in enumerate(conditional):
                serialized.append("".join(cell_strings[row][columns.index(name)] for name in order))
            candidates.append((self._trie_score(serialized), 1, None, conditional))

        # Higher exact trie reuse wins.  Global is preferred for exact ties
        # because it retains ordinary dataframe column semantics.
        candidates.sort(key=lambda item: (-item[0], item[1]))
        _, kind, global_order, row_orders = candidates[0]

        if kind == 0:
            result = source.loc[:, global_order].copy()
            return result, [list(global_order) for _ in range(nrows)]

        # A row-specific permutation has no single truthful dataframe header.
        # Keep the original output slots and return the exact source-column
        # mapping for every row in column_orderings.
        output = []
        for row, order in enumerate(row_orders):
            output.append([values[row, columns.index(name)] for name in order])
        result = pd.DataFrame(output, columns=columns, index=source.index, dtype=object)
        return result, row_orders
