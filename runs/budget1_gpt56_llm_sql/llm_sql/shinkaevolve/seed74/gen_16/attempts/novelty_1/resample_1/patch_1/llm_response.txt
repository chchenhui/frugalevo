<DESCRIPTION>
The original implementation failed validation because it could corrupt row identity and values during recursive reconstruction: it wrote completed futures into unrelated row indices, rebuilt frames from raw `.values.tolist()`, changed column layouts, appended/dropped `original_index`, merged rows back through pandas, and finally returned an empty column-ordering list. It also optimized squared cell-length statistics and adjacent operations rather than the evaluator's serialized character-prefix objective.

This replacement keeps every original value, row, column, and shape intact. It builds a bounded set of candidate column-order strategies, measures each using the exact sorted serialized-row LCP objective, and returns the best valid row permutation plus matching per-row column orderings. Missing values are normalized only for scoring, never written back into the output data.
</DESCRIPTION>

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List, Dict
from collections import Counter
import math


class Evolved(Algorithm):
    """
    Prefix-cache-oriented row and column reorderer.

    Values are never normalized or altered in the returned DataFrame.  String
    normalization is used solely to estimate evaluator serialization reuse.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _score_text(value) -> str:
        """Match evaluator-style fillna('').astype(str) for scalar cells."""
        if value is None:
            return ""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool) and missing:
                return ""
        except Exception:
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        if left == right:
            return len(left)
        limit = min(len(left), len(right))
        lo, hi = 0, limit
        # Slice comparisons run in C and avoid a Python character loop.
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if left[:mid] == right[:mid]:
                lo = mid
            else:
                hi = mid - 1
        return lo

    def _trie_score(self, strings: List[str]) -> int:
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        return sum(self._lcp(ordered[i - 1], ordered[i]) for i in range(1, len(ordered)))

    @staticmethod
    def _resolve_column(name, columns):
        """Resolve API dependency names conservatively and deterministically."""
        if name in columns:
            return columns.index(name)
        matches = [i for i, col in enumerate(columns) if str(name) in str(col)]
        return matches[0] if len(matches) == 1 else None

    def _make_units(self, columns, col_merge):
        """
        Convert requested merged column sets into non-overlapping contiguous
        units.  Overlapping merge requests are combined rather than allowing
        an invalid duplicated column order.
        """
        ncols = len(columns)
        parent = list(range(ncols))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a, b):
            a, b = find(a), find(b)
            if a != b:
                parent[b] = a

        for requested in col_merge or []:
            positions = [i for i, col in enumerate(columns) if col in requested]
            for pos in positions[1:]:
                union(positions[0], pos)

        groups = {}
        for pos in range(ncols):
            groups.setdefault(find(pos), []).append(pos)

        # Each merge block has source-column order unless the global rank later
        # supplies another deterministic valid order.
        return sorted(groups.values(), key=lambda group: min(group))

    def _unit_dependencies(self, units, columns, one_way_dep):
        position_to_unit = {}
        for unit_index, unit in enumerate(units):
            for pos in unit:
                position_to_unit[pos] = unit_index

        predecessors = [set() for _ in units]
        for pair in one_way_dep or []:
            if len(pair) != 2:
                continue
            left = self._resolve_column(pair[0], columns)
            right = self._resolve_column(pair[1], columns)
            if left is None or right is None:
                continue
            source = position_to_unit[left]
            target = position_to_unit[right]
            if source != target:
                predecessors[target].add(source)
        return predecessors

    @staticmethod
    def _topological_units(units, predecessors, priority):
        """Stable topological ordering, with a deterministic cycle fallback."""
        remaining = set(range(len(units)))
        result = []
        while remaining:
            ready = [u for u in remaining if not (predecessors[u] & remaining)]
            if not ready:
                # Invalid cyclic dependencies cannot all be honored.  Preserve
                # a stable order rather than dropping or fabricating columns.
                ready = list(remaining)
            chosen = min(ready, key=lambda u: (priority[u], min(units[u])))
            result.append(chosen)
            remaining.remove(chosen)
        return result

    @staticmethod
    def _flatten_units(unit_order, units):
        return [position for unit in unit_order for position in units[unit]]

    def _build_strings(self, cell_text, orders):
        return [
            "".join(cell_text[row][col] for col in order)
            for row, order in enumerate(orders)
        ]

    def _conditional_orders(self, cell_text, units, predecessors, global_order):
        """
        A bounded conditional prefix partition tree.  It chooses a leading
        remaining unit using length-weighted repeated values within each group,
        then independently chooses suffixes for value partitions.
        """
        nrows = len(cell_text)
        nunits = len(units)
        unit_values = [
            ["".join(cell_text[row][col] for col in unit) for row in range(nrows)]
            for unit in units
        ]
        orders = [None] * nrows
        node_budget = 96
        candidate_budget = 24
        nodes = [0]

        def cheap_tail(remaining):
            local_predecessors = [
                predecessors[u] & set(remaining) for u in range(nunits)
            ]
            result = []
            left = set(remaining)
            while left:
                ready = [u for u in left if not (local_predecessors[u] & left)]
                if not ready:
                    ready = list(left)
                chosen = min(ready, key=lambda u: global_order.index(u))
                result.append(chosen)
                left.remove(chosen)
            return result

        def visit(rows, remaining, prefix):
            if not rows:
                return
            if not remaining:
                final = self._flatten_units(prefix, units)
                for row in rows:
                    orders[row] = final
                return

            nodes[0] += 1
            if nodes[0] > node_budget or len(rows) <= 1:
                suffix = cheap_tail(remaining)
                final = self._flatten_units(prefix + suffix, units)
                for row in rows:
                    orders[row] = final
                return

            available = [
                unit for unit in remaining
                if not (predecessors[unit] & set(remaining))
            ]
            if not available:
                available = list(remaining)

            # Bound wide-frame group work while still considering the most
            # globally promising units first.
            available.sort(key=lambda u: global_order.index(u))
            examined = available[:candidate_budget]
            best_unit = None
            best_score = 0

            for unit in examined:
                counts = Counter(unit_values[unit][row] for row in rows)
                score = sum(
                    len(value) * count * (count - 1)
                    for value, count in counts.items()
                    if count > 1
                )
                if score > best_score or (
                    score == best_score and best_unit is not None and
                    global_order.index(unit) < global_order.index(best_unit)
                ):
                    best_score = score
                    best_unit = unit

            if best_unit is None or best_score <= 0:
                suffix = cheap_tail(remaining)
                final = self._flatten_units(prefix + suffix, units)
                for row in rows:
                    orders[row] = final
                return

            partitions = {}
            for row in rows:
                partitions.setdefault(unit_values[best_unit][row], []).append(row)

            next_remaining = [u for u in remaining if u != best_unit]
            for key in sorted(partitions):
                visit(partitions[key], next_remaining, prefix + [best_unit])

        visit(list(range(nrows)), list(range(nunits)), [])
        fallback = self._flatten_units(global_order, units)
        return [order if order is not None else fallback for order in orders]

    def _materialize(self, df, orders, cell_text):
        """Create a value-preserving output frame and aligned row orderings."""
        strings = self._build_strings(cell_text, orders)
        row_order = sorted(range(len(orders)), key=lambda row: (strings[row], row))

        values = df.to_numpy(dtype=object, copy=False)
        output_values = [
            [values[row, col] for col in orders[row]]
            for row in row_order
        ]

        # Positional output columns are deliberately the original schema.  A
        # row-specific ordering describes which source field occupies each
        # position in that row.
        output = pd.DataFrame(
            output_values,
            columns=df.columns.copy(),
            index=df.index.take(row_order),
            dtype=object,
        )
        output_orders = [
            [df.columns[col] for col in orders[row]]
            for row in row_order
        ]
        return output, output_orders, strings

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        # Empty frames still require a valid ordering result.
        if df.shape[1] == 0:
            return df.copy(), [[] for _ in range(len(df))]

        columns = list(df.columns)
        nrows, ncols = df.shape
        raw_values = df.to_numpy(dtype=object, copy=False)
        cell_text = [
            [self._score_text(raw_values[row, col]) for col in range(ncols)]
            for row in range(nrows)
        ]

        units = self._make_units(columns, col_merge)
        predecessors = self._unit_dependencies(units, columns, one_way_dep)

        # Required global frequency-ranked candidate:
        # sum(len(v) * count(v) * (count(v)-1)).
        unit_weight = []
        for unit in units:
            values = ["".join(cell_text[row][col] for col in unit) for row in range(nrows)]
            counts = Counter(values)
            unit_weight.append(sum(
                len(value) * count * (count - 1)
                for value, count in counts.items()
                if count > 1
            ))

        original_priority = [min(unit) for unit in units]
        frequency_priority = [
            (-unit_weight[unit], min(units[unit]))
            for unit in range(len(units))
        ]
        original_units = self._topological_units(
            units, predecessors, original_priority
        )
        frequency_units = self._topological_units(
            units, predecessors, frequency_priority
        )

        original_order = self._flatten_units(original_units, units)
        frequency_order = self._flatten_units(frequency_units, units)

        candidates = [
            [original_order[:] for _ in range(nrows)],
            [frequency_order[:] for _ in range(nrows)],
        ]

        # A bounded conditional candidate is useful when different value groups
        # benefit from different leading fields.
        if nrows > 1 and len(units) > 1:
            candidates.append(
                self._conditional_orders(
                    cell_text, units, predecessors, frequency_units
                )
            )

        best_orders = candidates[0]
        best_score = -1
        for orders in candidates:
            score = self._trie_score(self._build_strings(cell_text, orders))
            if score > best_score:
                best_score = score
                best_orders = orders

        output, column_orderings, _ = self._materialize(
            df, best_orders, cell_text
        )

        # Shape and source values are preserved exactly, modulo permitted row
        # and per-row field permutations represented by column_orderings.
        assert output.shape == df.shape
        assert len(column_orderings) == len(output)
        return output, column_orderings
