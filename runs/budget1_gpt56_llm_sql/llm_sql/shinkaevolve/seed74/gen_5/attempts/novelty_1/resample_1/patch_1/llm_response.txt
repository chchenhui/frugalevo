<DESCRIPTION>
The validation failure was caused by structural corruption rather than a visible stdout exception. The previous implementation added and removed an `original_index` column, rebuilt frames from `.values`, merged rows back with pandas, sorted the final frame again, and returned an empty `column_orderings` list. It also wrote asynchronous results to `result_df.loc[i]` in completion order, which can attach reordered values to the wrong source row. These behaviors can alter row/value identity and make returned column metadata inconsistent with the returned data.

This fix replaces the unsafe recursive dataframe mutation with a bounded deterministic optimizer. It preserves every input cell and row identity, normalizes values only for scoring, returns valid per-row column orderings, honors merge/dependency ordering constraints, and selects between a few bounded column-order candidates using the actual serialized character-prefix Trie objective.

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List, Dict
from collections import Counter, defaultdict
import heapq


class Evolved(Algorithm):
    """
    Prefix-cache-aware row and column reordering.

    Values are never changed.  For row-specific column orders the returned
    dataframe uses the original column positions as output slots and
    ``column_orderings[row]`` describes which source column occupies each slot.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _score_value(value) -> str:
        """Match evaluator serialization without mutating the stored value."""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool):
                if missing:
                    return ""
            elif bool(missing):
                return ""
        except (TypeError, ValueError):
            # Non-scalar objects are not normally dataframe cells, but str()
            # remains the safest representation if pandas cannot test them.
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        if left == right:
            return len(left)
        high = min(len(left), len(right))
        low = 0
        # Slice comparisons execute in C and avoid a Python character loop.
        while low < high:
            middle = (low + high + 1) // 2
            if left[:middle] == right[:middle]:
                low = middle
            else:
                high = middle - 1
        return low

    def _trie_score(self, strings: List[str]) -> int:
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        return sum(self._lcp(ordered[i - 1], ordered[i])
                   for i in range(1, len(ordered)))

    @staticmethod
    def _positions_for_name(columns, name):
        exact = [i for i, col in enumerate(columns) if col == name]
        if exact:
            return exact
        # Retain the old API's useful substring interpretation for dependency
        # specifications while remaining deterministic.
        return [i for i, col in enumerate(columns) if str(name) in str(col)]

    def _constraint_units(self, columns, col_merge, one_way_dep, priority):
        """
        Construct contiguous merge units and topologically order those units.
        A merge list means its supplied columns must remain adjacent; no cell
        values are merged or fabricated.
        """
        count = len(columns)
        parent = list(range(count))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a, b):
            a, b = find(a), find(b)
            if a != b:
                parent[b] = a

        for group in col_merge or []:
            positions = []
            for name in group:
                positions.extend(self._positions_for_name(columns, name))
            positions = sorted(set(positions))
            for pos in positions[1:]:
                union(positions[0], pos)

        groups = defaultdict(list)
        for pos in range(count):
            groups[find(pos)].append(pos)

        units = list(groups.values())
        unit_of = {}
        for unit_id, unit in enumerate(units):
            for pos in unit:
                unit_of[pos] = unit_id

        edges = defaultdict(set)
        indegree = [0] * len(units)
        internal_edges = defaultdict(set)

        for pair in one_way_dep or []:
            if len(pair) != 2:
                continue
            lefts = self._positions_for_name(columns, pair[0])
            rights = self._positions_for_name(columns, pair[1])
            for left in lefts:
                for right in rights:
                    a, b = unit_of[left], unit_of[right]
                    if a == b:
                        internal_edges[a].add((left, right))
                    elif b not in edges[a]:
                        edges[a].add(b)
                        indegree[b] += 1

        # Order members of each merged unit by priority, respecting internal
        # dependencies where possible. Cycles fall back deterministically.
        for unit_id, unit in enumerate(units):
            if not internal_edges[unit_id]:
                unit.sort(key=lambda x: (priority[x], x))
                continue
            local_in = {x: 0 for x in unit}
            local_out = defaultdict(set)
            for a, b in internal_edges[unit_id]:
                if a != b and b not in local_out[a]:
                    local_out[a].add(b)
                    local_in[b] += 1
            ready = [(priority[x], x) for x in unit if local_in[x] == 0]
            heapq.heapify(ready)
            result = []
            while ready:
                _, x = heapq.heappop(ready)
                result.append(x)
                for nxt in local_out[x]:
                    local_in[nxt] -= 1
                    if local_in[nxt] == 0:
                        heapq.heappush(ready, (priority[nxt], nxt))
            if len(result) != len(unit):
                result.extend(sorted((x for x in unit if x not in result),
                                     key=lambda x: (priority[x], x)))
            units[unit_id] = result

        unit_priority = [
            min((priority[p] for p in unit), default=0) for unit in units
        ]
        ready = [(unit_priority[i], i) for i in range(len(units))
                 if indegree[i] == 0]
        heapq.heapify(ready)
        ordered_units = []
        while ready:
            _, unit_id = heapq.heappop(ready)
            ordered_units.append(unit_id)
            for nxt in edges[unit_id]:
                indegree[nxt] -= 1
                if indegree[nxt] == 0:
                    heapq.heappush(ready, (unit_priority[nxt], nxt))

        # Dependency cycles are handled predictably instead of dropping data.
        if len(ordered_units) != len(units):
            ordered_units.extend(sorted(
                (i for i in range(len(units)) if i not in set(ordered_units)),
                key=lambda i: (unit_priority[i], i)
            ))
        return [pos for unit_id in ordered_units for pos in units[unit_id]]

    def _conditional_orders(self, cells, primary_order):
        """
        Build a bounded conditional prefix partition tree.  It is intentionally
        used only on moderate unconstrained tables; each depth examines at most
        32 candidate fields and never performs dataframe operations in nodes.
        """
        rows = len(cells)
        cols = len(primary_order)
        orders = [None] * rows
        max_depth = min(cols, 8)
        candidates = primary_order[:min(cols, 32)]

        def visit(indices, remaining, prefix, depth):
            if not indices:
                return
            if depth >= max_depth or not remaining or len(indices) < 2:
                tail = [p for p in primary_order if p in remaining]
                for row in indices:
                    orders[row] = prefix + tail
                return

            best_col = None
            best_value = 0
            for col in candidates:
                if col not in remaining:
                    continue
                counts = Counter(cells[row][col] for row in indices)
                value = sum(len(text) * amount * (amount - 1)
                            for text, amount in counts.items() if amount > 1)
                if value > best_value:
                    best_col, best_value = col, value

            if best_col is None or best_value <= 0:
                tail = [p for p in primary_order if p in remaining]
                for row in indices:
                    orders[row] = prefix + tail
                return

            buckets = defaultdict(list)
            for row in indices:
                buckets[cells[row][best_col]].append(row)
            next_remaining = [p for p in remaining if p != best_col]
            for bucket in buckets.values():
                visit(bucket, next_remaining, prefix + [best_col], depth + 1)

        visit(list(range(rows)), list(primary_order), [], 0)
        return [order if order is not None else list(primary_order)
                for order in orders]

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        del early_stop, row_stop, col_stop, distinct_value_threshold, parallel

        row_count, col_count = df.shape
        if col_count == 0:
            return df.copy(), [[] for _ in range(row_count)]
        if row_count == 0:
            return df.copy(), []

        columns = list(df.columns)
        cells = [
            [self._score_value(df.iat[row, col]) for col in range(col_count)]
            for row in range(row_count)
        ]

        # Frequency ranking requested by the API guidance:
        # sum(len(v) * count(v) * (count(v)-1)).
        pair_weight = []
        frequency_weight = []
        for col in range(col_count):
            counts = Counter(cells[row][col] for row in range(row_count))
            pair_weight.append(sum(
                len(value) * amount * (amount - 1)
                for value, amount in counts.items()
            ))
            frequency_weight.append(sum(
                len(value) * amount for value, amount in counts.items()
            ))

        global_frequency = self._constraint_units(
            columns, col_merge, one_way_dep,
            [(-pair_weight[col], col) for col in range(col_count)]
        )
        global_length = self._constraint_units(
            columns, col_merge, one_way_dep,
            [(-frequency_weight[col], col) for col in range(col_count)]
        )

        candidate_orders = [
            [list(global_frequency) for _ in range(row_count)],
            [list(global_length) for _ in range(row_count)],
        ]

        # Conditional orders can violate merge adjacency/dependency constraints,
        # so use them only when there are no supplied ordering constraints.
        # The size bound prevents expensive per-group work on very wide input.
        if (not col_merge and not one_way_dep and
                row_count * col_count <= 1500000 and col_count > 1):
            candidate_orders.append(
                self._conditional_orders(cells, global_frequency)
            )

        best_orders = candidate_orders[0]
        best_strings = None
        best_score = -1
        for orders in candidate_orders:
            strings = [
                "".join(cells[row][col] for col in orders[row])
                for row in range(row_count)
            ]
            score = self._trie_score(strings)
            if score > best_score:
                best_score = score
                best_orders = orders
                best_strings = strings

        # Sorting output rows by their actual serialized form gives a stable
        # insertion order while retaining every original dataframe index/value.
        row_order = sorted(range(row_count), key=lambda row: (best_strings[row], row))
        output_values = [
            [df.iat[row, col] for col in best_orders[row]]
            for row in row_order
        ]
        output_index = [df.index[row] for row in row_order]
        output = pd.DataFrame(
            output_values, index=output_index, columns=df.columns, dtype=object
        )
        column_orderings = [
            [columns[col] for col in best_orders[row]]
            for row in row_order
        ]

        assert output.shape == df.shape
        assert len(column_orderings) == len(output)
        return output, column_orderings
