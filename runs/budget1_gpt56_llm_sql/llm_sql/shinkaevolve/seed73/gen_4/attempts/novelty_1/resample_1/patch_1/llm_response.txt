<DESCRIPTION>
The original implementation could not pass validation because it corrupted row identity and column placement during recursive/threaded processing. In particular, `as_completed()` results were written using completion order rather than the source row index, temporary `original_index` values were mixed into recursive column reordering, rows were reconstructed from `.values.tolist()` without preserving labels/types, and the method finally returned an empty column-ordering list even though rows may have been permuted. The recursive merge/split path could also alter schema semantics and did not optimize the evaluator's actual serialized character-prefix objective.

This replacement preserves every source row, cell, index, and shape. It constructs a bounded set of valid column-order candidates, including original order, global frequency-weighted order, and conditional prefix grouping. Candidates are selected using the exact lexicographically-adjacent serialized-row LCP objective. It returns row-specific column orderings consistent with the output values, supports mixed types and missing values, respects merged-column adjacency and one-way dependency precedence, and avoids evaluator-state access.

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List, Dict
from collections import Counter, defaultdict


class Evolved(Algorithm):
    """
    Prefix-cache-oriented column reordering.

    The returned dataframe contains the original values from each source row,
    potentially permuted within that row.  column_orderings[row][position]
    identifies the source column for the value stored at that position.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _serialized_value(value) -> str:
        """Match evaluator normalization without changing the stored value."""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool) and missing:
                return ""
        except (TypeError, ValueError):
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        if left == right:
            return len(left)
        limit = min(len(left), len(right))
        low, high = 0, limit
        # Slice comparisons are implemented in C and avoid Python character loops.
        while low < high:
            middle = (low + high + 1) // 2
            if left[:middle] == right[:middle]:
                low = middle
            else:
                high = middle - 1
        return low

    def _trie_score(self, strings: List[str]) -> int:
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        return sum(self._lcp(ordered[i - 1], ordered[i]) for i in range(1, len(ordered)))

    @staticmethod
    def _resolve_column(columns: List, requested):
        """Use exact names first, then retain the old API's unique substring behavior."""
        if requested in columns:
            return requested
        matches = [column for column in columns if str(requested) in str(column)]
        return matches[0] if len(matches) == 1 else None

    def _build_units(
        self,
        columns: List,
        col_merge: List[List[str]],
        one_way_dep: List[Tuple[str, str]],
    ):
        """
        Build inseparable merge units and predecessor relationships between units.
        Values are never concatenated: a merge only requires adjacency.
        """
        parent = list(range(len(columns)))

        def find(item):
            while parent[item] != item:
                parent[item] = parent[parent[item]]
                item = parent[item]
            return item

        def union(left, right):
            left, right = find(left), find(right)
            if left != right:
                parent[right] = left

        positions = {column: index for index, column in enumerate(columns)}
        for merge_group in col_merge or []:
            resolved = [self._resolve_column(columns, item) for item in merge_group]
            resolved = [item for item in resolved if item is not None]
            if len(resolved) > 1:
                first = positions[resolved[0]]
                for item in resolved[1:]:
                    union(first, positions[item])

        grouped = defaultdict(list)
        for index, column in enumerate(columns):
            grouped[find(index)].append((index, column))

        units = []
        for members in grouped.values():
            members.sort()
            units.append([column for _, column in members])
        units.sort(key=lambda unit: positions[unit[0]])

        unit_for_column = {}
        for unit_index, unit in enumerate(units):
            for column in unit:
                unit_for_column[column] = unit_index

        predecessors = defaultdict(set)
        for dependency in one_way_dep or []:
            if len(dependency) != 2:
                continue
            source = self._resolve_column(columns, dependency[0])
            target = self._resolve_column(columns, dependency[1])
            if source is None or target is None:
                continue
            source_unit = unit_for_column[source]
            target_unit = unit_for_column[target]
            if source_unit != target_unit:
                predecessors[target_unit].add(source_unit)

        return units, predecessors

    @staticmethod
    def _flatten_units(unit_order, units):
        result = []
        for unit_index in unit_order:
            result.extend(units[unit_index])
        return result

    def _make_strings(self, values, column_positions, order):
        pieces = []
        for row in values:
            pieces.append("".join(row[column_positions[column]] for column in order))
        return pieces

    def _global_order(self, units, predecessors, cell_strings, positions):
        """Topological global order ranked by length-weighted repeated values."""
        weights = []
        for unit_index, unit in enumerate(units):
            counts = Counter(
                "".join(cell_strings[row][positions[column]] for column in unit)
                for row in range(len(cell_strings))
            )
            weight = sum(len(value) * count * (count - 1) for value, count in counts.items())
            weights.append(weight)

        remaining = set(range(len(units)))
        selected = []
        while remaining:
            available = [
                unit for unit in remaining
                if not (predecessors.get(unit, set()) & remaining)
            ]
            if not available:
                # Cyclic dependencies cannot all be satisfied; retain deterministic order.
                available = list(remaining)
            chosen = min(available, key=lambda unit: (-weights[unit], unit))
            selected.append(chosen)
            remaining.remove(chosen)
        return selected, weights

    def _conditional_orders(
        self,
        units,
        predecessors,
        cell_strings,
        positions,
        global_order,
        global_weights,
    ):
        """
        A bounded conditional prefix partition tree.  Each leaf receives a valid
        unit order; only a small, deterministic set of leading candidates is
        evaluated per node.
        """
        row_count = len(cell_strings)
        orders = [None] * row_count
        max_depth = min(8, len(units))
        candidate_limit = min(24, len(units))
        preferred = sorted(range(len(units)), key=lambda unit: (-global_weights[unit], unit))
        preferred = preferred[:candidate_limit]

        def finish(rows, chosen, remaining):
            tail = [unit for unit in global_order if unit in remaining]
            order = self._flatten_units(chosen + tail, units)
            for row in rows:
                orders[row] = order

        def visit(rows, chosen, remaining, depth):
            if not rows or not remaining:
                finish(rows, chosen, remaining)
                return
            if depth >= max_depth or len(rows) < 2:
                finish(rows, chosen, remaining)
                return

            available = [
                unit for unit in preferred
                if unit in remaining and not (predecessors.get(unit, set()) & remaining)
            ]
            if not available:
                available = [
                    unit for unit in remaining
                    if not (predecessors.get(unit, set()) & remaining)
                ]
            if not available:
                finish(rows, chosen, remaining)
                return

            best_unit = None
            best_score = 0
            best_groups = None
            for unit in available:
                groups = defaultdict(list)
                for row in rows:
                    token = "".join(cell_strings[row][positions[column]] for column in units[unit])
                    groups[token].append(row)
                score = sum(len(token) * len(group) * (len(group) - 1) for token, group in groups.items())
                if score > best_score or (score == best_score and best_unit is not None and unit < best_unit):
                    best_unit = unit
                    best_score = score
                    best_groups = groups

            if best_unit is None or best_score <= 0:
                finish(rows, chosen, remaining)
                return

            next_remaining = set(remaining)
            next_remaining.remove(best_unit)
            for token in sorted(best_groups):
                visit(best_groups[token], chosen + [best_unit], next_remaining, depth + 1)

        visit(list(range(row_count)), [], set(range(len(units))), 0)
        return orders

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        # Empty frames still require a valid ordering result.
        if df.shape[1] == 0:
            return df.copy(), [[] for _ in range(len(df))]

        columns = list(df.columns)
        positions = {column: index for index, column in enumerate(columns)}
        source_values = df.to_numpy(dtype=object, copy=True)
        cell_strings = [
            [self._serialized_value(value) for value in row]
            for row in source_values
        ]

        units, predecessors = self._build_units(columns, col_merge, one_way_dep)
        original_units = list(range(len(units)))
        global_units, global_weights = self._global_order(
            units, predecessors, cell_strings, positions
        )

        # At most three whole candidates are measured by the exact serial Trie score.
        candidates = []
        candidates.append([self._flatten_units(original_units, units) for _ in range(len(df))])
        candidates.append([self._flatten_units(global_units, units) for _ in range(len(df))])

        # Conditional construction is useful on heterogeneous data but is bounded.
        if len(df) > 1 and len(units) > 1:
            candidates.append(
                self._conditional_orders(
                    units, predecessors, cell_strings, positions,
                    global_units, global_weights,
                )
            )

        best_orders = candidates[0]
        best_score = -1
        for candidate_orders in candidates:
            serialized_rows = [
                "".join(cell_strings[row][positions[column]] for column in order)
                for row, order in enumerate(candidate_orders)
            ]
            score = self._trie_score(serialized_rows)
            if score > best_score:
                best_score = score
                best_orders = candidate_orders

        output = []
        for row_index, order in enumerate(best_orders):
            output.append([
                source_values[row_index][positions[column]]
                for column in order
            ])

        # Object dtype preserves mixed cells, including pd.NA, timestamps, and numerics.
        reordered = pd.DataFrame(output, index=df.index, columns=df.columns, dtype=object)
        return reordered, best_orders
