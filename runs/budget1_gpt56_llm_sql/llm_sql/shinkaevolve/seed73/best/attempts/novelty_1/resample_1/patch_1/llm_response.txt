<DESCRIPTION>
The original implementation failed validation because it could alter the dataset structure and did not return valid column orderings. In particular, it added and removed an `original_index` column, used merge/reset operations that could change row/value alignment, optionally merged columns, and finally returned an empty ordering list even after moving values between columns. The threaded recursion also wrote completed futures using completion order rather than source-row order.

This replacement keeps every original cell, row, column, and index intact. It builds a small bounded set of column-order candidates, including a global frequency-weighted ordering and a conditional grouping ordering, scores candidates using serialized-row Trie reuse via sorted LCPs, and returns the best candidate. Each returned row has a matching per-row column ordering describing the source column of every output position.
</DESCRIPTION>

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List, Dict
from collections import Counter, defaultdict


class Evolved(Algorithm):
    """
    Prefix-cache-oriented row and column reorderer.

    Values are never changed, fabricated, merged, or discarded.  A row-specific
    ordering describes which original source column supplies each output cell.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _string_value(value) -> str:
        """Match evaluator-style missing-value normalization without changing data."""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool) and missing:
                return ""
        except (TypeError, ValueError):
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        """Bounded Python-level LCP using C-level slice comparisons."""
        if left == right:
            return len(left)
        high = min(len(left), len(right))
        low = 0
        while low < high:
            middle = (low + high + 1) // 2
            if left[:middle] == right[:middle]:
                low = middle
            else:
                high = middle - 1
        return low

    def _trie_score(self, strings: List[str]) -> int:
        """Exact serial Trie reuse for a fixed collection of complete strings."""
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        return sum(self._lcp(ordered[i - 1], ordered[i]) for i in range(1, len(ordered)))

    @staticmethod
    def _resolve_column(name, columns):
        """Use exact names first, then the legacy unique-substring convention."""
        exact = [i for i, col in enumerate(columns) if col == name]
        if exact:
            return exact[0]
        partial = [i for i, col in enumerate(columns) if str(name) in str(col)]
        return partial[0] if len(partial) == 1 else None

    def _column_units(self, columns, col_merge):
        """
        Construct atomic column units.  A requested merged group remains
        consecutive, but no physical DataFrame merge is ever performed.
        """
        count = len(columns)
        parent = list(range(count))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a, b):
            a, b = find(a), find(b)
            if a != b:
                parent[b] = a

        for group in col_merge or []:
            positions = [self._resolve_column(name, columns) for name in group]
            positions = [p for p in positions if p is not None]
            for position in positions[1:]:
                union(positions[0], position)

        grouped = defaultdict(list)
        for position in range(count):
            grouped[find(position)].append(position)
        return sorted((sorted(unit) for unit in grouped.values()), key=lambda unit: unit[0])

    def _dependency_adjust(self, unit_order, units, columns, one_way_dep):
        """Move dependency source units before destination units deterministically."""
        result = list(unit_order)
        position_to_unit = {}
        for unit_index, unit in enumerate(units):
            for position in unit:
                position_to_unit[position] = unit_index

        edges = []
        for source_name, target_name in one_way_dep or []:
            source = self._resolve_column(source_name, columns)
            target = self._resolve_column(target_name, columns)
            if source is not None and target is not None:
                source_unit = position_to_unit[source]
                target_unit = position_to_unit[target]
                if source_unit != target_unit:
                    edges.append((source_unit, target_unit))

        # A bounded stable repair also tolerates contradictory dependency input.
        for _ in range(max(1, len(result))):
            changed = False
            locations = {unit: i for i, unit in enumerate(result)}
            for source, target in edges:
                if locations[source] > locations[target]:
                    result.pop(locations[source])
                    locations = {unit: i for i, unit in enumerate(result)}
                    result.insert(locations[target], source)
                    changed = True
            if not changed:
                break
        return result

    def _global_order(self, units, cell_strings, one_way_dep, columns):
        """
        Rank columns by sum(len(v) * count(v) * (count(v)-1)).
        This emphasizes repeated leading text while retaining deterministic ties.
        """
        scores = []
        for unit_index, unit in enumerate(units):
            score = 0
            for column in unit:
                counts = Counter(row[column] for row in cell_strings)
                score += sum(len(value) * frequency * (frequency - 1)
                             for value, frequency in counts.items())
            scores.append((score, unit_index))
        order = [unit for _, unit in sorted(scores, key=lambda item: (-item[0], item[1]))]
        return self._dependency_adjust(order, units, columns, one_way_dep)

    def _conditional_orders(self, units, cell_strings, global_order,
                            one_way_dep, columns):
        """
        Build a bounded conditional prefix partition tree.  Only a small number
        of globally promising units are considered at each node.
        """
        row_count = len(cell_strings)
        unit_count = len(units)
        orders = [None] * row_count
        max_depth = min(unit_count, 12)
        candidate_units = global_order[:min(unit_count, 32)]
        nodes_seen = [0]
        max_nodes = 512

        def visit(rows, remaining, prefix, depth):
            if not rows:
                return
            if (len(rows) < 2 or depth >= max_depth or not remaining or
                    nodes_seen[0] >= max_nodes):
                tail = [unit for unit in global_order if unit in remaining]
                final_units = self._dependency_adjust(prefix + tail, units, columns, one_way_dep)
                flat = [column for unit in final_units for column in units[unit]]
                for row in rows:
                    orders[row] = flat
                return

            candidates = [unit for unit in candidate_units if unit in remaining]
            if not candidates:
                candidates = list(remaining[:min(len(remaining), 8)])

            best_unit = None
            best_score = 0
            for unit in candidates:
                score = 0
                for column in units[unit]:
                    counts = Counter(cell_strings[row][column] for row in rows)
                    score += sum(len(value) * count * (count - 1)
                                 for value, count in counts.items())
                if score > best_score or (score == best_score and
                                          best_unit is not None and unit < best_unit):
                    best_score = score
                    best_unit = unit

            if best_unit is None or best_score <= 0:
                tail = [unit for unit in global_order if unit in remaining]
                final_units = self._dependency_adjust(prefix + tail, units, columns, one_way_dep)
                flat = [column for unit in final_units for column in units[unit]]
                for row in rows:
                    orders[row] = flat
                return

            nodes_seen[0] += 1
            partitions = defaultdict(list)
            for row in rows:
                key = tuple(cell_strings[row][column] for column in units[best_unit])
                partitions[key].append(row)

            next_remaining = [unit for unit in remaining if unit != best_unit]
            for key in sorted(partitions):
                visit(partitions[key], next_remaining, prefix + [best_unit], depth + 1)

        visit(list(range(row_count)), list(range(unit_count)), [], 0)

        fallback = [column for unit in global_order for column in units[unit]]
        return [order if order is not None else fallback for order in orders]

    def _candidate_strings(self, cell_strings, orders):
        return [
            "".join(cell_strings[row][column] for column in orders[row])
            for row in range(len(cell_strings))
        ]

    def fixed_reorder(self, df: pd.DataFrame, row_sort: bool = True):
        """Compatibility helper using the reliable global ordering."""
        return self.reorder(df, parallel=False)

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        del early_stop, row_stop, col_stop, distinct_value_threshold, parallel

        row_count, column_count = df.shape
        columns = list(df.columns)
        if row_count == 0 or column_count == 0:
            return df.copy(), [[] for _ in range(row_count)]

        values = df.to_numpy(dtype=object, copy=True)
        cell_strings = [
            [self._string_value(values[row, column]) for column in range(column_count)]
            for row in range(row_count)
        ]

        units = self._column_units(columns, col_merge)
        global_units = self._global_order(units, cell_strings, one_way_dep, columns)
        global_positions = [column for unit in global_units for column in units[unit]]

        # At most three whole candidates: original, frequency-ranked, conditional.
        candidates = [
            [list(range(column_count)) for _ in range(row_count)],
            [list(global_positions) for _ in range(row_count)],
            self._conditional_orders(units, cell_strings, global_units, one_way_dep, columns),
        ]

        best_orders = candidates[0]
        best_strings = self._candidate_strings(cell_strings, best_orders)
        best_score = self._trie_score(best_strings)

        for candidate in candidates[1:]:
            candidate_strings = self._candidate_strings(cell_strings, candidate)
            score = self._trie_score(candidate_strings)
            if score > best_score:
                best_score = score
                best_orders = candidate
                best_strings = candidate_strings

        # Lexicographic row placement is deterministic and makes prefix-sharing
        # locality explicit, while preserving every original source row/index.
        row_order = sorted(range(row_count), key=lambda row: (best_strings[row], row))
        output = [
            [values[row, source_column] for source_column in best_orders[row]]
            for row in row_order
        ]

        reordered = pd.DataFrame(output, columns=columns, index=df.index.take(row_order), dtype=object)
        column_orderings = [
            [columns[source_column] for source_column in best_orders[row]]
            for row in row_order
        ]

        assert reordered.shape == df.shape
        assert len(column_orderings) == len(reordered)
        assert all(len(order) == column_count for order in column_orderings)
        return reordered, column_orderings
