<DESCRIPTION>
The original implementation failed validation because it could corrupt row/column alignment while recursively rebuilding DataFrames, used positional writes with `as_completed`, added and removed an `original_index` helper column, returned an empty `column_orderings` list, and relied on attribute access for arbitrary column names. It also optimized squared cell lengths and adjacent operations rather than the evaluator’s serialized character-prefix Trie objective.

This fix replaces the unsafe recursive reconstruction with bounded, deterministic global column-order candidates. Each candidate preserves every original cell, row, column, and index. Candidates are measured using the actual serialized-row Trie proxy: the sum of LCP lengths between lexicographically adjacent serialized rows. The selected output includes valid per-row column orderings and does not create helper columns or mutate values.
</DESCRIPTION>

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List, Dict
from collections import Counter, defaultdict
import math


class Evolved(Algorithm):
    """
    Safe prefix-cache-oriented reordering.

    The returned DataFrame always contains the original values and columns.
    Only a global column permutation and a harmless row ordering are applied.
    A global permutation is intentionally used here because it preserves the
    normal DataFrame meaning of every labelled cell while still allowing a
    valid column ordering for every output row.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _safe_strings(df: pd.DataFrame) -> List[List[str]]:
        """
        Match the evaluator's normalization without modifying the stored data.
        Object conversion avoids failures for mixed/categorical columns.
        """
        if df.empty:
            return [[] for _ in range(len(df))]

        obj = df.astype(object)
        normalized = obj.where(pd.notna(obj), "")
        return [
            [str(value) for value in row]
            for row in normalized.itertuples(index=False, name=None)
        ]

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        """Bounded Python-level LCP using slice comparisons implemented in C."""
        if left == right:
            return len(left)
        high = min(len(left), len(right))
        low = 0
        while low < high:
            mid = (low + high + 1) // 2
            if left[:mid] == right[:mid]:
                low = mid
            else:
                high = mid - 1
        return low

    @classmethod
    def _trie_score(cls, rows: List[List[str]], order: List[int]) -> Tuple[int, List[int]]:
        """
        Exact ideal serial-Trie reuse for a fixed collection of serialized rows.
        The returned sorted positions are also a useful deterministic row order.
        """
        serialized = ["".join(row[column] for column in order) for row in rows]
        positions = sorted(range(len(serialized)), key=lambda i: serialized[i])

        score = 0
        for previous, current in zip(positions, positions[1:]):
            score += cls._lcp(serialized[previous], serialized[current])
        return score, positions

    @staticmethod
    def _column_repeat_scores(rows: List[List[str]], ncols: int) -> List[int]:
        """
        Length-weighted pair repetition:
        sum(len(value) * count(value) * (count(value) - 1)).
        """
        scores = [0] * ncols
        for column in range(ncols):
            counts = Counter(row[column] for row in rows)
            scores[column] = sum(
                len(value) * count * (count - 1)
                for value, count in counts.items()
                if count > 1
            )
        return scores

    @staticmethod
    def _merge_blocks(columns: List[object], col_merge: List[List[str]]) -> List[List[int]]:
        """
        Construct contiguous blocks for requested merged columns.  The API has
        historically used col_merge as a column grouping hint; values are never
        concatenated or changed.
        """
        name_to_positions = defaultdict(list)
        for position, name in enumerate(columns):
            name_to_positions[str(name)].append(position)

        parent = list(range(len(columns)))

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a, b):
            a, b = find(a), find(b)
            if a != b:
                parent[b] = a

        for group in col_merge or []:
            positions = []
            for requested in group:
                matches = name_to_positions.get(str(requested), [])
                if matches:
                    positions.append(matches[0])
            for position in positions[1:]:
                union(positions[0], position)

        grouped = defaultdict(list)
        for position in range(len(columns)):
            grouped[find(position)].append(position)

        # Preserve source order within blocks unless later ranking changes block order.
        return sorted(grouped.values(), key=lambda block: min(block))

    @staticmethod
    def _dependency_edges(columns: List[object], one_way_dep: List[Tuple[str, str]]) -> List[Tuple[int, int]]:
        """
        Preserve the prior API's substring matching behaviour, but never assert
        or fail merely because a user supplied ambiguous optional hint.
        """
        edges = []
        names = [str(column) for column in columns]
        for before, after in one_way_dep or []:
            left = [i for i, name in enumerate(names) if str(before) in name]
            right = [i for i, name in enumerate(names) if str(after) in name]
            if len(left) == 1 and len(right) == 1 and left[0] != right[0]:
                edges.append((left[0], right[0]))
        return edges

    @staticmethod
    def _apply_constraints(
        proposed: List[int],
        blocks: List[List[int]],
        dependencies: List[Tuple[int, int]],
    ) -> List[int]:
        """
        Rank merge blocks by the proposed order, then perform a stable
        topological ordering between blocks for one-way dependency hints.
        """
        block_for = {}
        for block_id, block in enumerate(blocks):
            for column in block:
                block_for[column] = block_id

        rank = {column: i for i, column in enumerate(proposed)}
        block_rank = {
            block_id: min(rank.get(column, len(proposed)) for column in block)
            for block_id, block in enumerate(blocks)
        }

        edges = defaultdict(set)
        indegree = [0] * len(blocks)
        for left, right in dependencies:
            a, b = block_for[left], block_for[right]
            if a != b and b not in edges[a]:
                edges[a].add(b)
                indegree[b] += 1

        available = [i for i in range(len(blocks)) if indegree[i] == 0]
        available.sort(key=lambda i: (block_rank[i], i))
        ordered_blocks = []

        while available:
            current = available.pop(0)
            ordered_blocks.append(current)
            for child in sorted(edges[current], key=lambda i: (block_rank[i], i)):
                indegree[child] -= 1
                if indegree[child] == 0:
                    available.append(child)
                    available.sort(key=lambda i: (block_rank[i], i))

        # Cycles are optional hints: append unresolved blocks deterministically.
        if len(ordered_blocks) != len(blocks):
            used = set(ordered_blocks)
            ordered_blocks.extend(
                sorted(
                    (i for i in range(len(blocks)) if i not in used),
                    key=lambda i: (block_rank[i], i),
                )
            )

        result = []
        for block_id in ordered_blocks:
            result.extend(sorted(blocks[block_id], key=lambda c: rank.get(c, c)))
        return result

    @staticmethod
    def _conditional_proxy_order(rows: List[List[str]], ncols: int, base_scores: List[int]) -> List[int]:
        """
        A bounded conditional-prefix grouping proxy.  It does not rebuild
        row-specific DataFrames (which would lose labelled-cell identity).
        Instead it estimates which columns repeatedly become useful leaders in
        value partitions and uses that as a distinct global ordering proposal.
        """
        if ncols == 0 or len(rows) < 2:
            return list(range(ncols))

        remaining = set(range(ncols))
        groups = [list(range(len(rows)))]
        bonus = [0] * ncols
        max_depth = min(ncols, 6)
        max_groups = 64

        for _ in range(max_depth):
            candidates = list(remaining)
            if not candidates:
                break

            gains = [0] * ncols
            for group in groups:
                if len(group) < 2:
                    continue
                for column in candidates:
                    counts = Counter(rows[row][column] for row in group)
                    gains[column] += sum(
                        len(value) * count * (count - 1)
                        for value, count in counts.items()
                        if count > 1
                    )

            chosen = max(candidates, key=lambda c: (gains[c], base_scores[c], -c))
            if gains[chosen] <= 0:
                break

            bonus[chosen] += gains[chosen]
            remaining.remove(chosen)

            next_groups = []
            for group in groups:
                partitions = defaultdict(list)
                for row in group:
                    partitions[rows[row][chosen]].append(row)
                next_groups.extend(partitions.values())

            groups = [group for group in next_groups if len(group) > 1]
            groups.sort(key=len, reverse=True)
            groups = groups[:max_groups]
            if not groups:
                break

        return sorted(
            range(ncols),
            key=lambda c: (bonus[c], base_scores[c], -c),
            reverse=True,
        )

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        """
        Select one of at most three inexpensive global permutations by measuring
        the actual serialized character-prefix objective.  Parameters that
        historically controlled unsafe recursion are accepted for API
        compatibility; they do not alter data preservation semantics.
        """
        self.df = df

        nrows, ncols = df.shape
        columns = list(df.columns)
        if nrows == 0 or ncols == 0:
            output = df.copy().astype(object)
            return output, [[] for _ in range(nrows)]

        rows = self._safe_strings(df)
        repeat_scores = self._column_repeat_scores(rows, ncols)
        blocks = self._merge_blocks(columns, col_merge)
        dependencies = self._dependency_edges(columns, one_way_dep)

        original = list(range(ncols))
        frequency_ranked = sorted(
            range(ncols),
            key=lambda c: (repeat_scores[c], -c),
            reverse=True,
        )
        conditional_ranked = self._conditional_proxy_order(rows, ncols, repeat_scores)

        proposals = []
        seen = set()
        for proposal in (original, frequency_ranked, conditional_ranked):
            constrained = self._apply_constraints(proposal, blocks, dependencies)
            key = tuple(constrained)
            if key not in seen:
                seen.add(key)
                proposals.append(constrained)

        best_order = proposals[0]
        best_score, best_positions = self._trie_score(rows, best_order)

        for proposal in proposals[1:]:
            score, positions = self._trie_score(rows, proposal)
            if score > best_score:
                best_order, best_score, best_positions = proposal, score, positions

        # iloc preserves original row indexes, values, and duplicate column names.
        output = df.iloc[best_positions, best_order].copy().astype(object)
        ordering_names = [columns[column] for column in best_order]
        column_orderings = [list(ordering_names) for _ in range(len(output))]

        assert output.shape == df.shape
        return output, column_orderings
