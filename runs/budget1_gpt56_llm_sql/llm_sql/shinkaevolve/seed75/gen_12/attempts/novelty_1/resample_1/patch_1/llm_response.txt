<DESCRIPTION>
The original implementation could lose row identity and corrupt output structure during recursive reconstruction: `column_recursion` wrote completed futures using enumeration order rather than source-row order, recursive results rebuilt DataFrames from `.values.tolist()`, and `reorder` always returned an empty column-ordering list. It also added and removed an `original_index` column while recursively changing column layouts, making validation failures likely even when shapes appeared correct.

This replacement keeps every original row and cell value intact, uses object dtype for mixed values, returns valid per-row column orderings, and avoids recursive DataFrame reconstruction. It evaluates a small bounded set of column-order candidates using the actual serialized-string Trie/LCP objective, including a global frequency ordering and a conditional grouping ordering. Missing values are normalized only for scoring; stored values are never changed.
</DESCRIPTION>

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List, Dict
from collections import Counter
from functools import lru_cache


class Evolved(Algorithm):
    """
    Safe bounded row/column reordering for serialized prefix reuse.

    The returned dataframe uses its original columns as positional output slots.
    `column_orderings[i][j]` identifies the source column whose value is stored
    in output row i, slot j.  This permits row-specific column orders without
    dropping, fabricating, or modifying any source values.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df
        self.dep_graph = None
        self.num_rows = 0
        self.num_cols = 0
        self.column_stats = None
        self.val_len = None
        self.row_stop = None
        self.col_stop = None
        self.base = 2000

    @staticmethod
    def _score_string(value) -> str:
        """Match evaluator normalization without changing the stored value."""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool) and missing:
                return ""
        except Exception:
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        """Bounded Python-level LCP: slice comparisons execute in C."""
        if left == right:
            return len(left)
        limit = min(len(left), len(right))
        low, high = 0, limit
        while low < high:
            mid = (low + high + 1) // 2
            if left[:mid] == right[:mid]:
                low = mid
            else:
                high = mid - 1
        return low

    def _trie_score(self, strings: List[str]) -> int:
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        return sum(self._lcp(ordered[i - 1], ordered[i])
                   for i in range(1, len(ordered)))

    @lru_cache(maxsize=8192)
    def calculate_length(self, value):
        return len(self._score_string(value))

    def get_dependent_columns(self, col: str) -> List[str]:
        # Kept for API compatibility. Dependency handling is performed directly
        # on positional column units in reorder().
        return []

    @lru_cache(maxsize=8192)
    def get_cached_dependent_columns(self, col: str) -> List[str]:
        return self.get_dependent_columns(col)

    def find_max_group_value(self, df: pd.DataFrame, value_counts: Dict,
                             early_stop: int = 0):
        best_value = None
        best_score = early_stop
        for value, count in value_counts.items():
            score = self.calculate_length(value) * max(0, count - 1)
            if score >= best_score:
                best_value, best_score = value, score
        return best_value

    def fixed_reorder(self, df: pd.DataFrame,
                      row_sort: bool = True) -> Tuple[pd.DataFrame, List[List[str]]]:
        """Compatibility fallback that never changes values or row identity."""
        result = df.copy().astype(object)
        order = list(df.columns)
        if row_sort and len(result) > 1:
            serial = [
                "".join(self._score_string(result.iat[r, c])
                        for c in range(result.shape[1]))
                for r in range(result.shape[0])
            ]
            positions = sorted(range(len(result)), key=lambda r: serial[r])
            result = result.iloc[positions].copy()
        return result, [order[:] for _ in range(len(result))]

    def _units(self, columns: List[str], col_merge: List[List[str]]):
        """Build atomic adjacent column groups while retaining every position."""
        used = set()
        units = []
        name_to_positions = {}
        for pos, name in enumerate(columns):
            name_to_positions.setdefault(name, []).append(pos)

        for requested_group in col_merge or []:
            positions = []
            for name in requested_group:
                for pos in name_to_positions.get(name, []):
                    if pos not in used:
                        positions.append(pos)
                        used.add(pos)
            if positions:
                units.append(positions)

        for pos in range(len(columns)):
            if pos not in used:
                units.append([pos])

        # Keep a deterministic source-position order for otherwise unspecified
        # merged columns.
        return units

    def _dependency_edges(self, columns, units, one_way_dep):
        position_to_unit = {}
        for unit_id, unit in enumerate(units):
            for position in unit:
                position_to_unit[position] = unit_id

        edges = set()
        for before, after in one_way_dep or []:
            before_matches = [i for i, name in enumerate(columns)
                              if name == before or str(before) in str(name)]
            after_matches = [i for i, name in enumerate(columns)
                             if name == after or str(after) in str(name)]
            if len(before_matches) == 1 and len(after_matches) == 1:
                source = position_to_unit[before_matches[0]]
                target = position_to_unit[after_matches[0]]
                if source != target:
                    edges.add((source, target))
        return edges

    def _legal_global_order(self, ranked_units, unit_count, edges):
        predecessors = {u: set() for u in range(unit_count)}
        for source, target in edges:
            predecessors[target].add(source)

        remaining = set(range(unit_count))
        answer = []
        while remaining:
            available = [u for u in ranked_units
                         if u in remaining and not (predecessors[u] & remaining)]
            if not available:
                # Cyclic constraints cannot be satisfied simultaneously; use a
                # deterministic order rather than losing columns.
                available = sorted(remaining)
            selected = available[0]
            answer.append(selected)
            remaining.remove(selected)
        return answer

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        # Do not add bookkeeping columns: this is important for preserving both
        # user values and dataframe shape.
        source = df.copy().astype(object)
        rows, cols = source.shape
        self.num_rows, self.num_cols = rows, cols

        if rows == 0 or cols == 0:
            return source, [[] for _ in range(rows)]

        column_names = list(source.columns)
        cell_strings = [
            [self._score_string(source.iat[r, c]) for c in range(cols)]
            for r in range(rows)
        ]

        units = self._units(column_names, col_merge)
        unit_count = len(units)
        edges = self._dependency_edges(column_names, units, one_way_dep)

        # Precompute each unit's serialized value once.  It is also the correct
        # representation for a merged block because fields have no separator.
        unit_values = []
        unit_weight = []
        for unit in units:
            values = ["".join(cell_strings[r][c] for c in unit)
                      for r in range(rows)]
            counts = Counter(values)
            weight = sum(len(value) * count * (count - 1)
                         for value, count in counts.items())
            unit_values.append(values)
            unit_weight.append(weight)

        ranked_frequency = sorted(
            range(unit_count),
            key=lambda u: (-unit_weight[u], min(units[u]))
        )
        ranked_source = sorted(range(unit_count), key=lambda u: min(units[u]))

        # Candidate 1: global length-weighted repeated-value ranking.
        global_frequency_units = self._legal_global_order(
            ranked_frequency, unit_count, edges
        )

        # Candidate 2: deterministic source order, still respecting supplied
        # one-way dependency constraints.
        source_units = self._legal_global_order(ranked_source, unit_count, edges)

        candidates = [
            [sum((units[u] for u in global_frequency_units), []) for _ in range(rows)],
            [sum((units[u] for u in source_units), []) for _ in range(rows)],
        ]

        # Candidate 3: conditional leading-field partitioning.  It is bounded
        # by depth and node count so wide or highly distinct inputs remain safe.
        max_depth = min(unit_count, col_stop if col_stop is not None else 12, 12)
        if max_depth > 0 and rows > 1:
            predecessors = {u: set() for u in range(unit_count)}
            for source_unit, target_unit in edges:
                predecessors[target_unit].add(source_unit)

            conditional = [None] * rows
            node_budget = [256]

            def finish(group_rows, prefix, remaining):
                tail_rank = [u for u in ranked_frequency if u in remaining]
                tail = self._legal_global_order(
                    tail_rank, len(tail_rank),
                    {(tail_rank.index(a), tail_rank.index(b))
                     for a, b in []
                     if a in tail_rank and b in tail_rank}
                ) if False else tail_rank
                # Enforce only dependencies whose endpoints remain. Prefix
                # units have already been selected, so they no longer block.
                rem_edges = {(a, b) for a, b in edges
                             if a in remaining and b in remaining}
                tail = self._legal_global_order(tail, unit_count, rem_edges)
                tail = [u for u in tail if u in remaining]
                positions = sum((units[u] for u in prefix + tail), [])
                for r in group_rows:
                    conditional[r] = positions[:]

            def visit(group_rows, prefix, remaining):
                if (not remaining or len(prefix) >= max_depth or
                        len(group_rows) <= 1 or node_budget[0] <= 0):
                    finish(group_rows, prefix, remaining)
                    return
                node_budget[0] -= 1
                available = [
                    u for u in remaining
                    if not (predecessors[u] & remaining)
                ]
                if not available:
                    finish(group_rows, prefix, remaining)
                    return

                # Avoid expensive consideration of every field on very wide
                # datasets while preserving a deterministic useful tail.
                available = sorted(
                    available,
                    key=lambda u: (-unit_weight[u], min(units[u]))
                )[:24]
                best_unit = None
                best_score = early_stop
                for u in available:
                    counts = Counter(unit_values[u][r] for r in group_rows)
                    score = sum(len(value) * count * (count - 1)
                                for value, count in counts.items())
                    if score > best_score:
                        best_unit, best_score = u, score

                if best_unit is None:
                    finish(group_rows, prefix, remaining)
                    return

                partitions = {}
                for r in group_rows:
                    partitions.setdefault(unit_values[best_unit][r], []).append(r)
                next_remaining = set(remaining)
                next_remaining.remove(best_unit)
                for key in sorted(partitions):
                    visit(partitions[key], prefix + [best_unit], next_remaining)

            visit(list(range(rows)), [], set(range(unit_count)))
            if all(order is not None and len(order) == cols for order in conditional):
                candidates.append(conditional)

        best_orders = candidates[0]
        best_score = -1
        for orders in candidates:
            serialized = [
                "".join(cell_strings[r][c] for c in orders[r])
                for r in range(rows)
            ]
            score = self._trie_score(serialized)
            if score > best_score:
                best_score = score
                best_orders = orders

        serialized = [
            "".join(cell_strings[r][c] for c in best_orders[r])
            for r in range(rows)
        ]
        row_positions = sorted(range(rows), key=lambda r: (serialized[r], r))

        output_values = [
            [source.iat[r, c] for c in best_orders[r]]
            for r in row_positions
        ]
        result = pd.DataFrame(
            output_values,
            index=source.index.take(row_positions),
            columns=source.columns,
            dtype=object,
        )
        column_orderings = [
            [column_names[c] for c in best_orders[r]]
            for r in row_positions
        ]

        assert result.shape == source.shape
        assert len(column_orderings) == len(result)
        return result, column_orderings
