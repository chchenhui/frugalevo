import pandas as pd
import numpy as np
from solver import Algorithm
from typing import Tuple, List


class Evolved(Algorithm):
    """
    Prefix-cache oriented dataframe reordering.

    Values are never changed.  The dataframe stores each row's values in the
    sequence described by the corresponding entry in column_orderings.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _text(value) -> str:
        """Evaluator-compatible text representation without changing the value."""
        if value is None:
            return ""
        try:
            missing = pd.isna(value)
            if isinstance(missing, (bool, np.bool_)) and missing:
                return ""
        except (TypeError, ValueError):
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        if left == right:
            return len(left)
        limit = min(len(left), len(right))
        lo, hi = 0, limit
        # Slice comparisons are implemented in C and avoid Python character loops.
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if left[:mid] == right[:mid]:
                lo = mid
            else:
                hi = mid - 1
        return lo

    def _trie_score(self, strings: List[str]) -> int:
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        return sum(self._lcp(ordered[i - 1], ordered[i])
                   for i in range(1, len(ordered)))

    @staticmethod
    def _units(columns, col_merge):
        """
        Make merge requests atomic.  A merge request means its participating
        columns remain contiguous; it does not manufacture a merged value.
        """
        used = set()
        result = []
        for requested in col_merge or []:
            positions = [i for i, name in enumerate(columns)
                         if name in requested and i not in used]
            if positions:
                result.append(positions)
                used.update(positions)
        for i in range(len(columns)):
            if i not in used:
                result.append([i])
        return result

    @staticmethod
    def _dependency_pairs(columns, units, one_way_dep):
        """Translate the legacy substring dependency API to unit precedence."""
        position_unit = {}
        for unit_index, unit in enumerate(units):
            for position in unit:
                position_unit[position] = unit_index

        pairs = set()
        for before_text, after_text in one_way_dep or []:
            before = [i for i, name in enumerate(columns)
                      if str(before_text) in str(name)]
            after = [i for i, name in enumerate(columns)
                     if str(after_text) in str(name)]
            if len(before) == 1 and len(after) == 1:
                a = position_unit[before[0]]
                b = position_unit[after[0]]
                if a != b:
                    pairs.add((a, b))
        return pairs

    @staticmethod
    def _legal_ranked_units(scores, dependencies, fallback):
        """Stable score ranking subject to requested one-way dependencies."""
        remaining = set(range(len(scores)))
        predecessors = {u: set() for u in remaining}
        for a, b in dependencies:
            predecessors[b].add(a)

        answer = []
        while remaining:
            ready = [u for u in remaining if not (predecessors[u] & remaining)]
            # Cycles in caller supplied dependencies are handled deterministically.
            if not ready:
                ready = list(remaining)
            chosen = min(ready, key=lambda u: (-scores[u], fallback[u]))
            answer.append(chosen)
            remaining.remove(chosen)
        return answer

    @staticmethod
    def _flatten(unit_order, units):
        return [position for unit in unit_order for position in units[unit]]

    def _conditional_orders(self, unit_values, global_units, max_depth, early_stop):
        """
        Bounded partition tree.  A selected unit is leading only within the
        group where it is repeated; different leaves may have different tails.
        """
        nrows = len(unit_values[0]) if unit_values else 0
        nunits = len(unit_values)
        orders = [[] for _ in range(nrows)]
        node_budget = 256
        nodes_used = 0
        candidate_cap = 32

        def visit(rows, available, depth):
            nonlocal nodes_used
            if not rows:
                return
            if depth >= max_depth or not available or nodes_used >= node_budget:
                chosen = [u for u in global_units if u in available]
                for row in rows:
                    orders[row].extend(chosen)
                return

            candidates = [u for u in global_units if u in available][:candidate_cap]
            best_unit = None
            best_score = 0
            best_groups = None
            for unit in candidates:
                groups = {}
                values = unit_values[unit]
                for row in rows:
                    groups.setdefault(values[row], []).append(row)
                score = sum(len(value) * len(group) * (len(group) - 1)
                            for value, group in groups.items())
                if score > best_score or (score == best_score and
                                          score > 0 and
                                          (best_unit is None or unit < best_unit)):
                    best_unit, best_score, best_groups = unit, score, groups

            if best_unit is None or best_score <= 0 or (
                early_stop and best_score < early_stop
            ):
                chosen = [u for u in global_units if u in available]
                for row in rows:
                    orders[row].extend(chosen)
                return

            nodes_used += 1
            for row in rows:
                orders[row].append(best_unit)
            next_available = set(available)
            next_available.remove(best_unit)
            for group in best_groups.values():
                visit(group, next_available, depth + 1)

        visit(list(range(nrows)), set(range(nunits)), 0)
        return orders

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        # Empty inputs still require one ordering per source row.
        nrows, ncols = df.shape
        columns = list(df.columns)
        if ncols == 0:
            return df.copy(), [[] for _ in range(nrows)]

        source = df.to_numpy(dtype=object, copy=True)
        cell_text = [[self._text(source[r, c]) for c in range(ncols)]
                     for r in range(nrows)]

        units = self._units(columns, col_merge)
        dependencies = self._dependency_pairs(columns, units, one_way_dep)

        # Required global frequency proposal:
        # sum_v len(v) * count(v) * (count(v)-1).
        unit_scores = []
        unit_values = []
        for unit in units:
            values = ["".join(cell_text[row][col] for col in unit)
                      for row in range(nrows)]
            unit_values.append(values)
            counts = {}
            for value in values:
                counts[value] = counts.get(value, 0) + 1
            unit_scores.append(sum(len(value) * count * (count - 1)
                                   for value, count in counts.items()))

        global_units = self._legal_ranked_units(
            unit_scores, dependencies, list(range(len(units)))
        )
        natural_units = self._legal_ranked_units(
            [0] * len(units), dependencies, list(range(len(units)))
        )

        candidates = [global_units]
        if natural_units != global_units:
            candidates.append(natural_units)

        # Conditional construction is deliberately bounded on wide datasets.
        # It is a third candidate, not an unbounded optimization loop.
        work = nrows * len(units)
        if work <= 500000 and len(units) > 1 and nrows > 1:
            depth_limit = min(len(units), col_stop if col_stop is not None else 12)
            depth_limit = max(1, depth_limit)
            conditional = self._conditional_orders(
                unit_values, global_units, depth_limit, early_stop
            )
            candidates.append(conditional)

        best_score = -1
        best_orders = None
        for candidate in candidates[:3]:
            if candidate and isinstance(candidate[0], list):
                row_unit_orders = candidate
            else:
                row_unit_orders = [candidate] * nrows

            position_orders = [self._flatten(order, units)
                               for order in row_unit_orders]
            serialized = [
                "".join(cell_text[row][position] for position in position_orders[row])
                for row in range(nrows)
            ]
            score = self._trie_score(serialized)
            if score > best_score:
                best_score = score
                best_orders = position_orders

        if best_orders is None:
            best_orders = [list(range(ncols)) for _ in range(nrows)]

        # Object dtype safely retains mixed values, pd.NA, timestamps, and custom
        # Python values.  No scoring normalization is written into this frame.
        result_values = np.empty((nrows, ncols), dtype=object)
        for row, order in enumerate(best_orders):
            result_values[row, :] = source[row, order]

        # Fixed output labels describe output positions.  Per-row source labels
        # are supplied by column_orderings, which makes row-specific permutations
        # unambiguous even when dataframe labels are duplicated.
        result = pd.DataFrame(result_values, index=df.index, columns=df.columns)
        column_orderings = [[columns[position] for position in order]
                            for order in best_orders]
        return result, column_orderings
