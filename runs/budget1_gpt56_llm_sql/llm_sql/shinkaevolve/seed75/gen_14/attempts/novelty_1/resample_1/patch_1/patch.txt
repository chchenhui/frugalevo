+
+        # Candidate 1: dependency-safe input order.
+        natural_units = self._topological_order(
+            [len(units) - i for i in range(len(units))], edges
+        )
+
+        # Candidate 2: required global frequency-ranked ordering.
+        global_units = self._topological_order(weights, edges)
+
+        # Candidate 3: conditional prefix partition order.
+        depth_limit = col_stop if col_stop is not None else min(len(units), 12)
+        conditional_units = self._conditional_orders(
+            unit_values, weights, edges, max(1, depth_limit)
+        )
+
+        candidates = [
+            [natural_units[:] for _ in range(nrows)],
+            [global_units[:] for _ in range(nrows)],
+            conditional_units,
+        ]
+
+        best_orders = candidates[0]
+        best_reuse = -1
+        for unit_orders in candidates:
+            serialized = []
+            for row, order in enumerate(unit_orders):
+                serialized.append("".join(unit_values[unit][row] for unit in order))
+            reuse = self._trie_reuse(serialized)
+            if reuse > best_reuse:
+                best_reuse = reuse
+                best_orders = unit_orders
+
+        position_orders = [
+            [pos for unit in order for pos in units[unit]]
+            for order in best_orders
+        ]
+
+        output = [[None] * ncols for _ in range(nrows)]
+        for row, source_order in enumerate(position_orders):
+            for out_pos, source_pos in enumerate(source_order):
+                output[row][out_pos] = source_values[source_pos][row]
+

+        reordered = pd.DataFrame(output, index=df.index, columns=df.columns, dtype=object)
+        column_orderings = [
+            [columns[pos] for pos in source_order]
+            for source_order in position_orders
+        ]
+        return reordered, column_orderings
