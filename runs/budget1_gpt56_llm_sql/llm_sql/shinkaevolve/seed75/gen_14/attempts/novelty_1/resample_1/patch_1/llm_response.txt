<DESCRIPTION>
The validation failure is caused by the original implementation corrupting row identity and column/value alignment during recursive processing. In particular, `column_recursion` writes completed futures using enumeration order rather than the original row index, `original_index` is injected into the data and later used as a regular reorderable field, and the function finally returns an empty column-ordering list. The previous recursive merge/split path can therefore reorder values independently of their source rows and does not provide valid per-row column orders.

The replacement keeps every original cell, row, index, shape, and column label unchanged. It builds a small bounded set of column-order candidates, including a global repetition-weighted ordering and a conditional grouping ordering, measures candidates using lexicographically sorted serialized-row LCP reuse, and returns the best valid row-specific ordering. Output values are only permuted within their original row.

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List
from collections import Counter
import heapq


class Evolved(Algorithm):
    """
    Prefix-cache-oriented row-local column ordering.

    Every output row contains exactly the values from its corresponding input
    row.  The returned ordering for each row describes the source columns whose
    values occupy the output positions.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _score_value(value) -> str:
        """Match the evaluator's missing-value normalization without mutating data."""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool) and missing:
                return ""
        except Exception:
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        if left == right:
            return len(left)
        limit = min(len(left), len(right))
        lo, hi = 0, limit
        # Slice comparison is implemented in C and avoids Python character loops.
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if left[:mid] == right[:mid]:
                lo = mid
            else:
                hi = mid - 1
        return lo

    def _trie_reuse(self, strings: List[str]) -> int:
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        return sum(self._lcp(ordered[i - 1], ordered[i])
                   for i in range(1, len(ordered)))

    @staticmethod
    def _resolve_column(name, columns):
        """Use the historical API's exact-or-unique-substring column lookup."""
        if name in columns:
            return columns.index(name)
        matches = [i for i, col in enumerate(columns) if str(name) in str(col)]
        return matches[0] if len(matches) == 1 else None

    def _make_units(self, columns, col_merge, one_way_dep):
        """Create contiguous column units while preserving all source columns."""
        ncols = len(columns)
        assigned = set()
        units = []

        for requested_group in col_merge or []:
            positions = []
            for name in requested_group:
                pos = self._resolve_column(name, columns)
                if pos is not None and pos not in assigned:
                    positions.append(pos)
                    assigned.add(pos)
            if positions:
                units.append(positions)

        for pos in range(ncols):
            if pos not in assigned:
                units.append([pos])

        unit_of = {}
        for unit_id, members in enumerate(units):
            for pos in members:
                unit_of[pos] = unit_id

        edges = {i: set() for i in range(len(units))}
        for pair in one_way_dep or []:
            if len(pair) != 2:
                continue
            left = self._resolve_column(pair[0], columns)
            right = self._resolve_column(pair[1], columns)
            if left is None or right is None:
                continue
            src, dst = unit_of[left], unit_of[right]
            if src != dst:
                edges[src].add(dst)

        return units, edges

    @staticmethod
    def _topological_order(weights, edges):
        """High-weight deterministic topological order; cycles are safely broken."""
        count = len(weights)
        indegree = [0] * count
        for src in range(count):
            for dst in edges.get(src, ()):
                indegree[dst] += 1

        ready = [(-weights[i], i) for i in range(count) if indegree[i] == 0]
        heapq.heapify(ready)
        result = []
        used = set()

        while ready:
            _, node = heapq.heappop(ready)
            if node in used:
                continue
            used.add(node)
            result.append(node)
            for dst in edges.get(node, ()):
                indegree[dst] -= 1
                if indegree[dst] == 0:
                    heapq.heappush(ready, (-weights[dst], dst))

        # Invalid/cyclic dependency input must not drop columns.
        for node in sorted((i for i in range(count) if i not in used),
                           key=lambda i: (-weights[i], i)):
            result.append(node)
        return result

    def _conditional_orders(self, unit_values, weights, edges, depth_limit):
        """
        Select leading units independently inside value partitions.  This is
        deliberately bounded: it performs no exhaustive edit search.
        """
        rows = len(unit_values[0]) if unit_values else 0
        units = len(unit_values)
        tail = self._topological_order(weights, edges)
        orders = [None] * rows

        if not rows or not units:
            return [[] for _ in range(rows)]

        max_depth = max(1, min(units, depth_limit))
        stack = [(list(range(rows)), [], list(range(units)), 0)]

        while stack:
            row_ids, prefix, remaining, depth = stack.pop()
            if not remaining:
                for row in row_ids:
                    orders[row] = prefix
                continue

            if depth >= max_depth or len(row_ids) < 2:
                suffix = [u for u in tail if u in set(remaining)]
                for row in row_ids:
                    orders[row] = prefix + suffix
                continue

            # Only inspect a bounded number of promising units on very wide data.
            candidates = sorted(remaining, key=lambda u: (-weights[u], u))[:24]
            best_unit = None
            best_score = 0

            for unit in candidates:
                counts = Counter(unit_values[unit][row] for row in row_ids)
                score = sum(
                    len(value) * count * (count - 1)
                    for value, count in counts.items()
                    if count > 1
                )
                if score > best_score or (score == best_score and
                                          best_unit is not None and unit < best_unit):
                    best_score = score
                    best_unit = unit

            if best_unit is None or best_score <= 0:
                suffix = [u for u in tail if u in set(remaining)]
                for row in row_ids:
                    orders[row] = prefix + suffix
                continue

            following = [u for u in remaining if u != best_unit]
            groups = {}
            values = unit_values[best_unit]
            for row in row_ids:
                groups.setdefault(values[row], []).append(row)

            # A selected field must precede dependency descendants where possible.
            descendants = set(edges.get(best_unit, ()))
            preferred = [u for u in following if u in descendants]
            rest = [u for u in following if u not in descendants]
            next_remaining = preferred + rest

            for group_rows in groups.values():
                stack.append((group_rows, prefix + [best_unit],
                              next_remaining, depth + 1))

        return [order if order is not None else tail for order in orders]

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        del early_stop, row_stop, distinct_value_threshold, parallel

        nrows, ncols = df.shape
        columns = list(df.columns)
        if ncols == 0:
            return df.copy(), [[] for _ in range(nrows)]

        # Keep original Python objects for output; strings are scoring-only.
        source_values = [
            df.iloc[:, pos].to_numpy(dtype=object, copy=True)
            for pos in range(ncols)
        ]
        cell_strings = [
            [self._score_value(value) for value in source_values[pos]]
            for pos in range(ncols)
        ]

        units, edges = self._make_units(columns, col_merge, one_way_dep)
        unit_values = [
            ["".join(cell_strings[pos][row] for pos in members)
             for row in range(nrows)]
            for members in units
        ]

        weights = []
        for values in unit_values:
            counts = Counter(values)
            weights.append(sum(
                len(value) * count * (count - 1)
                for value, count in counts.items()
                if count > 1
            ))

        # Candidate 1: dependency-safe input order.
        natural_units = self._topological_order(
            [len(units) - i for i in range(len(units))], edges
        )

        # Candidate 2: required global frequency-ranked ordering.
        global_units = self._topological_order(weights, edges)

        # Candidate 3: conditional prefix partition order.
        depth_limit = col_stop if col_stop is not None else min(len(units), 12)
        conditional_units = self._conditional_orders(
            unit_values, weights, edges, max(1, depth_limit)
        )

        candidates = [
            [natural_units[:] for _ in range(nrows)],
            [global_units[:] for _ in range(nrows)],
            conditional_units,
        ]

        best_orders = candidates[0]
        best_reuse = -1
        for unit_orders in candidates:
            serialized = []
            for row, order in enumerate(unit_orders):
                serialized.append("".join(unit_values[unit][row] for unit in order))
            reuse = self._trie_reuse(serialized)
            if reuse > best_reuse:
                best_reuse = reuse
                best_orders = unit_orders

        position_orders = [
            [pos for unit in order for pos in units[unit]]
            for order in best_orders
        ]

        output = [[None] * ncols for _ in range(nrows)]
        for row, source_order in enumerate(position_orders):
            for out_pos, source_pos in enumerate(source_order):
                output[row][out_pos] = source_values[source_pos][row]

        # Object dtype prevents mixed-type coercion and preserves missing objects.
        reordered = pd.DataFrame(output, index=df.index, columns=df.columns, dtype=object)
        column_orderings = [
            [columns[pos] for pos in source_order]
            for source_order in position_orders
        ]
        return reordered, column_orderings

# EVOLVE-BLOCK-END
