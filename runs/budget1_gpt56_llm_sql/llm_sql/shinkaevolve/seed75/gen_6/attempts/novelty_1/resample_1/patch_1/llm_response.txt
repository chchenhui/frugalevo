<DESCRIPTION>
The previous implementation could corrupt row identity and cell placement: it added an `original_index` cell, rebuilt frames from raw `.values`, wrote threaded results using completion order rather than source row order, merged/re-sorted data, and finally returned an empty `column_orderings` list. These behaviors violate the required preservation and row-specific ordering contract. The stdout only shows the algorithm reached preprocessing; no useful traceback was emitted, so the validation failure is consistent with these structural correctness issues.

This replacement preserves every original cell and row index, never adds helper columns, handles missing and mixed values safely, and returns a valid column ordering for every row. It evaluates a bounded set of column-order candidates using exact serialized-row Trie reuse via sorted-string LCP scoring, selecting between frequency-based global orders and a bounded conditional grouping order.

# EVOLVE-BLOCK-START
import pandas as pd
from solver import Algorithm
from typing import Tuple, List, Dict, Any
from collections import defaultdict


class Evolved(Algorithm):
    """
    Prefix-cache-aware dataframe reordering.

    The dataframe values are never normalized or changed.  Normalized strings
    are used only for estimating evaluator serialization and Trie reuse.
    """

    def __init__(self, df: pd.DataFrame = None):
        self.df = df

    @staticmethod
    def _serialized_value(value: Any) -> str:
        """Match fillna('').astype(str) for ordinary scalar dataframe cells."""
        try:
            missing = pd.isna(value)
            if isinstance(missing, bool) and missing:
                return ""
        except (TypeError, ValueError):
            pass
        return str(value)

    @staticmethod
    def _lcp(left: str, right: str) -> int:
        """Bounded Python-level LCP using C-level slice comparisons."""
        limit = min(len(left), len(right))
        if limit == 0:
            return 0
        if left == right:
            return limit
        lo, hi = 0, limit
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if left[:mid] == right[:mid]:
                lo = mid
            else:
                hi = mid - 1
        return lo

    def _trie_score(self, strings: List[str]) -> int:
        """Exact serial Trie reuse for a fixed collection of strings."""
        if len(strings) < 2:
            return 0
        ordered = sorted(strings)
        score = 0
        previous = ordered[0]
        for current in ordered[1:]:
            score += self._lcp(previous, current)
            previous = current
        return score

    @staticmethod
    def _topological_rank(
        units: List[int],
        weights: List[int],
        predecessors: Dict[int, set],
    ) -> List[int]:
        """Deterministic highest-weight topological ordering."""
        remaining = set(units)
        result = []
        while remaining:
            ready = [
                unit for unit in remaining
                if not (predecessors.get(unit, set()) & remaining)
            ]
            # Malformed cyclic dependency input is made deterministic rather
            # than losing columns.
            if not ready:
                ready = list(remaining)
            ready.sort(key=lambda unit: (-weights[unit], unit))
            chosen = ready[0]
            result.append(chosen)
            remaining.remove(chosen)
        return result

    def _make_units(
        self,
        columns: List[Any],
        col_merge: List[List[str]],
        one_way_dep: List[Tuple[str, str]],
    ):
        """Create contiguous merge units and dependency edges between units."""
        positions_by_name = defaultdict(list)
        for position, name in enumerate(columns):
            positions_by_name[str(name)].append(position)

        used = set()
        units = []
        for group in col_merge or []:
            positions = []
            for requested in group:
                matches = positions_by_name.get(str(requested), [])
                if len(matches) == 1 and matches[0] not in used:
                    positions.append(matches[0])
            if positions:
                positions.sort()
                units.append(positions)
                used.update(positions)

        for position in range(len(columns)):
            if position not in used:
                units.append([position])

        # Preserve input order of unrelated units.
        units.sort(key=lambda group: min(group))
        position_to_unit = {}
        for unit, positions in enumerate(units):
            for position in positions:
                position_to_unit[position] = unit

        predecessors = defaultdict(set)
        for before_text, after_text in one_way_dep or []:
            before = [
                position for position, name in enumerate(columns)
                if str(before_text) in str(name)
            ]
            after = [
                position for position, name in enumerate(columns)
                if str(after_text) in str(name)
            ]
            # The legacy API expected exactly one matching column for each
            # endpoint. Ignore ambiguous constraints rather than corrupt data.
            if len(before) == 1 and len(after) == 1:
                source = position_to_unit[before[0]]
                target = position_to_unit[after[0]]
                if source != target:
                    predecessors[target].add(source)
        return units, predecessors

    def _build_strings(
        self,
        cell_strings: List[List[str]],
        row_orders: List[List[int]],
    ) -> List[str]:
        return [
            "".join(cell_strings[row][column] for column in order)
            for row, order in enumerate(row_orders)
        ]

    def _conditional_orders(
        self,
        cell_strings: List[List[str]],
        units: List[List[int]],
        unit_weights: List[int],
        predecessors: Dict[int, set],
        max_depth: int,
        max_candidates: int,
    ) -> List[List[int]]:
        """
        Bounded conditional partition tree.  Each branch chooses a leading
        unit based on length-weighted repeated serialized values.
        """
        row_count = len(cell_strings)
        unit_count = len(units)
        orders = [[] for _ in range(row_count)]
        global_unit_order = self._topological_rank(
            list(range(unit_count)), unit_weights, predecessors
        )

        queue = [(list(range(row_count)), set(range(unit_count)), 0)]
        while queue:
            rows, remaining, depth = queue.pop()
            if not rows or not remaining:
                continue
            if depth >= max_depth or len(rows) < 2:
                tail = self._topological_rank(
                    list(remaining), unit_weights, predecessors
                )
                for row in rows:
                    for unit in tail:
                        orders[row].extend(units[unit])
                continue

            eligible = [
                unit for unit in remaining
                if not (predecessors.get(unit, set()) & remaining)
            ]
            if not eligible:
                eligible = sorted(remaining)
            eligible.sort(key=lambda unit: (-unit_weights[unit], unit))
            eligible = eligible[:max_candidates]

            best_unit = None
            best_score = 0
            best_buckets = None
            for unit in eligible:
                buckets = defaultdict(list)
                for row in rows:
                    key = tuple(cell_strings[row][column] for column in units[unit])
                    buckets[key].append(row)

                score = 0
                for key, bucket_rows in buckets.items():
                    count = len(bucket_rows)
                    if count > 1:
                        score += sum(len(part) for part in key) * count * (count - 1)
                if score > best_score or (
                    score == best_score and best_unit is not None and unit < best_unit
                ):
                    best_unit = unit
                    best_score = score
                    best_buckets = buckets

            if best_unit is None or best_score <= 0:
                tail = self._topological_rank(
                    list(remaining), unit_weights, predecessors
                )
                for row in rows:
                    for unit in tail:
                        orders[row].extend(units[unit])
                continue

            for row in rows:
                orders[row].extend(units[best_unit])
            next_remaining = set(remaining)
            next_remaining.remove(best_unit)
            for bucket_rows in best_buckets.values():
                queue.append((bucket_rows, set(next_remaining), depth + 1))

        return orders

    def reorder(
        self,
        df: pd.DataFrame,
        early_stop: int = 0,
        row_stop: int = None,
        col_stop: int = None,
        col_merge: List[List[str]] = [],
        one_way_dep: List[Tuple[str, str]] = [],
        distinct_value_threshold: float = 0.8,
        parallel: bool = True,
    ) -> Tuple[pd.DataFrame, List[List[str]]]:
        # No helper columns, merges, dataframe sorting, or value conversions:
        # source row identity and source cells remain intact.
        row_count, column_count = df.shape
        columns = list(df.columns)

        if column_count == 0:
            return df.copy(), [[] for _ in range(row_count)]
        if row_count == 0:
            return df.copy(), []

        raw_values = df.astype(object).values.tolist()
        cell_strings = [
            [self._serialized_value(value) for value in row]
            for row in raw_values
        ]

        units, predecessors = self._make_units(
            columns, col_merge, one_way_dep
        )
        unit_count = len(units)

        # Required global proposal: sum(len(v) * count(v) * (count(v)-1)).
        pair_weights = []
        # A separate, less pair-heavy proposal provides a cheap alternate.
        repeat_weights = []
        for positions in units:
            pair_weight = 0
            repeat_weight = 0
            for column in positions:
                counts = defaultdict(int)
                for row in range(row_count):
                    counts[cell_strings[row][column]] += 1
                for value, count in counts.items():
                    if count > 1:
                        pair_weight += len(value) * count * (count - 1)
                        repeat_weight += len(value) * (count - 1)
            pair_weights.append(pair_weight)
            repeat_weights.append(repeat_weight)

        global_pair_units = self._topological_rank(
            list(range(unit_count)), pair_weights, predecessors
        )
        global_repeat_units = self._topological_rank(
            list(range(unit_count)), repeat_weights, predecessors
        )

        def flatten(unit_order: List[int]) -> List[int]:
            result = []
            for unit in unit_order:
                result.extend(units[unit])
            return result

        candidates = []
        global_pair_order = flatten(global_pair_units)
        candidates.append([list(global_pair_order) for _ in range(row_count)])

        if global_repeat_units != global_pair_units:
            global_repeat_order = flatten(global_repeat_units)
            candidates.append([list(global_repeat_order) for _ in range(row_count)])

        # Wide frames are deliberately bounded; this is a candidate generator,
        # not an exhaustive recursive search.
        depth_limit = min(8, unit_count)
        candidate_limit = min(32, unit_count)
        conditional = self._conditional_orders(
            cell_strings,
            units,
            pair_weights,
            predecessors,
            depth_limit,
            candidate_limit,
        )
        candidates.append(conditional)

        best_orders = candidates[0]
        best_score = self._trie_score(self._build_strings(cell_strings, best_orders))
        for candidate in candidates[1:3]:
            score = self._trie_score(self._build_strings(cell_strings, candidate))
            if score > best_score:
                best_score = score
                best_orders = candidate

        # A global permutation retains naturally matching labels. Row-specific
        # permutations use original labels as positional output slots, while
        # column_orderings records exactly which original cell occupies each slot.
        is_global = all(order == best_orders[0] for order in best_orders)
        if is_global:
            final_positions = best_orders[0]
            result_columns = [columns[position] for position in final_positions]
        else:
            result_columns = columns

        output_rows = [
            [raw_values[row][position] for position in order]
            for row, order in enumerate(best_orders)
        ]
        result = pd.DataFrame(
            output_rows, index=df.index.copy(), columns=result_columns, dtype=object
        )
        column_orderings = [
            [columns[position] for position in order] for order in best_orders
        ]
        return result, column_orderings
