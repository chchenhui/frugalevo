# EVOLVE-BLOCK-START
"""
Real-Time Adaptive Signal Processing for Non-Stationary Time Series

Architecture (staged, decoupled):
  1. RobustVol       -> sigma + local volatility percentile map
  2. SGKernelBank    -> 3-scale Savitzky-Golay passes (half-widths
                        [sg//2, sg, 2*sg]) with Gaussian percentile
                        weights (20/50/80th pct of volatility),
                        blended per-sample => aggressive smoothing
  3. SlopeLedKF      -> 2-state constant-velocity KF + RTS smoother;
                        smoothed slope drives a lead compensation that
                        recovers lag without amplifying reversals
  4. ReversalGuard   -> localized hysteresis on weak gradient flips
  5. Output contract -> end-aligned window (unchanged)
"""
import numpy as np


# ------------------------------------------------------------------
# Stage 1: robust statistics + volatility percentile map
# ------------------------------------------------------------------
class RobustVol:
    def __init__(self, k=7):
        self.k = k

    def compute(self, x):
        n = len(x)
        half = self.k // 2
        resid = np.zeros(n)
        for i in range(n):
            lo, hi = max(0, i - half), min(n, i + half + 1)
            resid[i] = x[i] - np.median(x[lo:hi])
        mad = np.median(np.abs(resid - np.median(resid)))
        sigma = max(1e-8, 1.4826 * mad)
        v = np.abs(resid)
        kw = 3
        if n > kw:
            cs = np.cumsum(v)
            vs = np.empty_like(v)
            vs[:kw] = cs[:kw] / np.arange(1, kw + 1)
            vs[kw:] = (cs[kw:] - cs[:-kw]) / kw
            v = vs
        return sigma, v


# ------------------------------------------------------------------
# Stage 2: percentile-blended multi-scale Savitzky-Golay bank
# ------------------------------------------------------------------
class SGKernelBank:
    """Zero-phase order-2 SG at 3 scales, blended with Gaussian
    weights centered at the 20/50/80th percentiles of local
    volatility. High volatility -> wide (aggressive) kernel;
    low volatility -> narrow kernel. Falls back smoothly."""

    def __init__(self, base_half, pcts=(0.2, 0.5, 0.8), widths=(1.5, 1.0, 1.0)):
        self.base_half = max(2, base_half)
        self.pcts = pcts
        self.widths = widths

    @staticmethod
    def _sg_coeffs(half):
        m = 2 * half + 1
        # least-squares quadratic fit coefficients (convolution form)
        t = np.arange(m) - half
        A = np.vstack([np.ones(m), t, t * t]).T
        c, *_ = np.linalg.lstsq(A, np.eye(m)[0], rcond=None) * 0
        # closed form: averaging weights for order-2 polyfit
        denom = m * (m * m - 4.0) / 3.0
        w = (3.0 * (3.0 * m * m - 7.0 - 15.0 * t * t) / (m * (m * m - 4.0)))
        return w  # sums to 1 for order-2 SG

    @staticmethod
    def _apply(x, half):
        n = len(x)
        c = SGKernelBank._sg_coeffs(half)
        m = len(c)
        if m > n:
            # reduce window to fit
            half = max(1, (n - 1) // 2)
            c = SGKernelBank._sg_coeffs(half)
            m = len(c)
        pad = m // 2
        padded = np.concatenate((
            x[0] + (x[0] - x[1:pad + 1][::-1]) if pad < n else np.array([x[0]] * pad),
            x,
            x[-1] + (x[-1] - x[n - 2:n - 2 - pad:-1]) if pad < n else np.array([x[-1]] * pad),
        ))
        return np.convolve(padded, c, mode='valid')[:n]

    def run(self, x, v):
        n = len(x)
        halfs = [max(2, self.base_half // 2),
                 self.base_half,
                 2 * self.base_half]
        y0 = self._apply(x, halfs[0])
        y1 = self._apply(x, halfs[1])
        y2 = self._apply(x, halfs[2])
        bank = np.vstack([y0, y1, y2])
        # percentile anchors of local volatility
        pv = np.percentile(v, [20, 50, 80])
        weights = np.zeros((3, n))
        for j in range(3):
            weights[j] = np.exp(-0.5 * ((v - pv[j]) / (self.widths[j] * (pv[2] - pv[0] + 1e-12))) ** 2)
        weights /= (weights.sum(axis=0, keepdims=True) + 1e-12)
        return (weights * bank).sum(axis=0)


# ------------------------------------------------------------------
# Stage 3: 2-state CV Kalman + RTS + slope-gated lead
# ------------------------------------------------------------------
class SlopeLedKF:
    def __init__(self, sigma, q_scale=1.0, r_scale=0.8):
        self.sigma = sigma
        self.q_scale = q_scale
        self.r_scale = r_scale

    def run(self, x, lead_cap=3.0):
        n = len(x)
        F = np.array([[1.0, 1.0], [0.0, 1.0]])
        R = max(1e-6, (self.sigma * self.r_scale) ** 2)
        q = 0.05 * self.q_scale ** 2 * self.sigma ** 2

        m = min(n, 5)
        state = np.array([np.mean(x[:m]),
                          (x[m - 1] - x[0]) / max(1, m - 1)])
        P = np.eye(2) * (self.sigma ** 2 * 10 + 1e-3)

        s_f = np.zeros((n, 2))
        P_f = np.zeros((n, 2, 2))
        P_pred = np.zeros((n, 2, 2))

        for i in range(n):
            state = F @ state
            p00, p01, p11 = P[0, 0], P[0, 1], P[1, 1]
            Pp = np.array([[p00 + 2 * p01 + p11 + q, p01 + p11 + q],
                           [p01 + p11 + q, p11 + q]])
            P_pred[i] = Pp
            S = Pp[0, 0] + R
            inno = x[i] - state[0]
            if abs(inno) > 4.0 * np.sqrt(S):
                Rbig = R * 25.0
                S = Pp[0, 0] + Rbig
                K = Pp[:, 0] / S
                inno = x[i] - state[0]
                state = state + K * inno
                P = Pp - np.outer(K, Pp[0, :])
            else:
                K = Pp[:, 0] / S
                state = state + K * inno
                P = Pp - np.outer(K, Pp[0, :])
            s_f[i] = state
            P_f[i] = P

        # RTS smoother
        sm = s_f.copy()
        for i in range(n - 2, -1, -1):
            Pp = P_pred[i + 1]
            G = np.linalg.solve(Pp.T, (P_f[i] @ F.T).T).T
            sm[i] = s_f[i] + G @ (sm[i + 1] - F @ s_f[i])

        level = sm[:, 0].copy()
        slope = sm[:, 1]
        # slope-gated lead: shift level forward along the trend where
        # slope is confident (large relative to local slope spread)
        g = np.abs(slope)
        med_g = np.median(g) + 1e-12
        gate = np.clip(g / (2.0 * med_g), 0.0, 1.0) ** 2
        level = level + lead_cap * gate * slope
        return level


# ------------------------------------------------------------------
# Stage 4: localized reversal suppression hysteresis
# ------------------------------------------------------------------
def _local_hysteresis(y, flip_span=3, k=0.5):
    n = len(y)
    if n < 7:
        return y
    g = np.gradient(y)
    med_g = np.median(np.abs(g)) + 1e-12
    weak = np.abs(g) < k * med_g
    s = np.sign(g)
    flip = np.zeros(n, dtype=bool)
    for lag in range(1, flip_span + 1):
        if lag < n:
            flip[lag:] |= (s[lag:] * s[:-lag]) < 0
    flag = weak & flip
    if not np.any(flag):
        return y
    dil = np.zeros(n, dtype=bool)
    idx = np.where(flag)[0]
    for i in idx:
        dil[max(0, i - 2):min(n, i + 3)] = True
    out = y.copy()
    c = np.array([-3.0, 12.0, 17.0, 12.0, -3.0]) / 35.0
    i = 0
    while i < n:
        if dil[i]:
            j = i
            while j < n and dil[j]:
                j += 1
            lo, hi = max(0, i - 2), min(n, j + 2)
            seg = y[lo:hi]
            if len(seg) >= 5:
                pad = 2
                left = 2 * seg[0] - seg[1:pad + 1][::-1]
                right = 2 * seg[-1] - seg[-2:-pad - 2:-1]
                padded = np.concatenate((left, seg, right))
                sm = np.convolve(padded, c, mode='valid')
                core_start = i - lo
                core_len = j - i
                avail = min(core_len, max(0, len(sm) - core_start))
                if avail > 0:
                    out[i:i + avail] = sm[core_start:core_start + avail]
            i = j
        else:
            i += 1
    return out

# Pipeline
# ------------------------------------------------------------------
def adaptive_filter(x, window_size=20):
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n < window_size:
        raise ValueError(
            f"Input signal length ({n}) must be >= window_size ({window_size})")

    # Stage 1
    stats = RobustVol()
    sigma, v = stats.compute(x)

    # Stage 2: percentile-blended SG bank
    sg_half = max(2, window_size // 2)
    bank = SGKernelBank(base_half=sg_half)
    y_smooth = bank.run(x, v)

    # Stage 3: slope-gated lead KF (works on smoothed signal; sigma of residual)
    resid = x - y_smooth
    sigma_r = max(1e-8, 1.4826 * np.median(np.abs(resid - np.median(resid))))
    led = SlopeLedKF(sigma=sigma_r, q_scale=1.0, r_scale=0.8)
    y_led = led.run(y_smooth, lead_cap=min(3.0, sg_half * 0.5))

    # Blend: smoothing dominates, lead corrects lag
    blend = 0.65
    y = blend * y_smooth + (1.0 - blend) * y_led

    # Stage 4: reversal suppression
    y = _local_hysteresis(y)

    output_length = n - window_size + 1
    return y[n - output_length:]


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """Alias for the composed filter."""
    return adaptive_filter(x, window_size)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Main signal processing function that applies the selected algorithm.

    Args:
        input_signal: Input time series data
        window_size: Window size for processing
        algorithm_type: Type of algorithm to use ("basic" or "enhanced")

    Returns:
        Filtered signal
    """
    x = np.asarray(input_signal, dtype=float)
    return adaptive_filter(x, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal
        noise_level: Standard deviation of noise to add
        seed: Random seed for reproducibility

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000, noise_level=0.3, window_size=20):
    """
    Run the signal processing algorithm on a test signal.

    Args:
        noisy_signal: Input signal to filter (if provided, use this; otherwise generate)
        signal_length: Length if generating signal (for backward compatibility)
        noise_level: Noise level if generating signal (for backward compatibility)
        window_size: Window size for processing

    Returns:
        Dictionary containing results and metrics
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = np.corrcoef(filtered_signal, aligned_clean)[0, 1] if min_length > 1 else 0

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (noise_before - noise_after) / noise_before if noise_before > 0 else 0

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }
    elif len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }
    else:
        return {
            "filtered_signal": [],
            "clean_signal": [],
            "noisy_signal": [],
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": 0,
        }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
