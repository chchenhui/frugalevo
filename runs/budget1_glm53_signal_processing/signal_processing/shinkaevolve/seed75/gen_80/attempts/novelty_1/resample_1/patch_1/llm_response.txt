    # [sg_half//2, sg_half, 2*sg_half] -> wider calm-region smoothing
    # (noise reduction, fewer slope changes) while the short kernel keeps
    # responsiveness (low lag) in volatile regions.
    sg_half = max(2, window_size // 4)
    cand = sorted({max(1, sg_half // 2), sg_half, 2 * sg_half})
    halves = [h for h in cand if 2 * h + 1 <= max(3, n)]
    if len(halves) == 0:
        halves = [1]

    if len(halves) == 1 or n < 2 * max(halves) + 1:
        smoothed = _sg_smooth(level, halves[0])
    else:
        # Soft Gaussian membership in volatility space, centered at the
        # 20th/50th/80th percentiles -> smooth blending, no seams.
        # Guard degenerate percentiles: if spread collapses (flat
