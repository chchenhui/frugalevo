A zero-phase S-G filter constructed via the pinv(AᵀA)·Aᵀ row-0 projection with reflection padding provides strong noise suppression without introducing phase delay (protecting lag_error and responsiveness), while its local polynomial fitting preserves genuine trend dynamics. def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using weighted moving average.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Create weights that emphasize recent samples
    weights = np.exp(np.linspace(-2, 0, window_size))
    weights = weights / np.sum(weights)

    for i in range(output_length):
        window = x[i : i + window_size]

        # Weighted moving average with exponential weights
        y[i] = np.sum(window * weights)

    return y

def _sg_smooth_1d(x, half, order=2):
    """
    Zero-phase Savitzky-Golay smoothing of a 1D array.

    Uses reflection padding and the row-0 projection of
    pinv(A^T A) A^T (local polynomial fit evaluated at the window center),
    which yields exactly zero phase delay.

    Args:
        x: Input 1D array
        half: Half-width of the smoothing window (window = 2*half+1)
        order: Polynomial order of the local fit

    Returns:
        Smoothed 1D array of the same length
    """
    n = len(x)
    if n == 0 or half < 1:
        return x.copy()

    # Reflect-pad edges to avoid boundary bias
    pad = min(half, n - 1)
    left = x[1 : pad + 1][::-1]
    right = x[n - pad - 1 : n - 1][::-1]
    xp = np.concatenate([left, x, right])

    m = 2 * half + 1
    if m <= order:
        order = max(1, m - 1)

    # Vandermonde matrix for local polynomial fitting
    idx = np.arange(m) - half
    A = np.vander(idx, order + 1, increasing=True)
    # Zero-phase projection: row of pinv(A^T A) A^T that evaluates fit at center
    proj = np.linalg.pinv(A.T @ A) @ A.T
    kernel = proj[0]

    y = np.convolve(xp, kernel[::-1], mode="valid")
    return y[:n]


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using dual-pass zero-phase
    Savitzky-Golay smoothing.

    Stage 1: wide quadratic S-G pass (half = window_size // 2) for strong
             noise suppression with zero phase delay.
    Stage 2: narrower quadratic S-G pass (half = stage-1 half // 2) that
             further attenuates residual noise while preserving dynamics.
    Blend:  small fraction of the centered raw sample is blended back in
            to maintain responsiveness / trend tracking.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal (same length as input)
    """
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n == 0:
        return x.copy()

    sg_half = max(2, window_size // 2)
    sg_half = min(sg_half, max(1, (n - 1) // 2))

    # Stage 1: wide zero-phase S-G pass (quadratic)
    y1 = _sg_smooth_1d(x, sg_half, order=2)

    # Stage 2: narrower zero-phase S-G pass (quadratic)
    h2 = max(2, sg_half // 2)
    h2 = min(h2, max(1, (n - 1) // 2))
    y2 = _sg_smooth_1d(y1, h2, order=2)

    # Blend a small fraction of the raw signal back in to preserve
    # responsiveness / amplitude of genuine dynamics
    blend = 0.1
    y = (1.0 - blend) * y2 + blend * x

    return y
