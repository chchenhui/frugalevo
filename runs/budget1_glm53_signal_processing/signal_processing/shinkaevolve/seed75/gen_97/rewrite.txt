"""
Real-Time Adaptive Signal Processing Algorithm for Non-Stationary Time Series

Modular pipeline architecture:
  Stage A: volatility model (robust local noise scale)
  Stage B: forward constant-velocity Kalman filter (adaptive R, robust gating)
  Stage C: RTS backward smoother on [level, slope]
  Stage D: multi-scale zero-phase Savitzky-Golay bank, blended by volatility
  Stage E: confidence-gated slope lead compensation (lag reduction)
  Stage F: slope-magnitude-conditional polish — narrow SG applied ONLY in
           quiet regions (|slope| below the statistical noise-floor gate),
           left untouched where the trend is genuine (responsiveness kept).
  Stage G: end-aligned trim to sliding-window output length.
"""
import numpy as np


# ----------------------------------------------------------------------
# Shared utilities
# ----------------------------------------------------------------------
def _sg_kernel(half, poly=2):
    """Zero-phase Savitzky-Golay coefficients (value at window center)."""
    half = max(1, int(half))
    t = np.arange(-half, half + 1, dtype=float)
    if len(t) - 1 < poly:
        poly = len(t) - 1
    A = np.vstack([t ** k for k in range(poly + 1)]).T
    return (np.linalg.pinv(A.T @ A) @ A.T)[0]


def _zero_phase_smooth(sig, half, kernel=None):
    """Zero-phase SG pass with mirrored edge padding. Length-preserving."""
    sig = np.asarray(sig, dtype=float)
    m = len(sig)
    if m < 2 * half + 1:
        return sig.copy()
    if kernel is None:
        kernel = _sg_kernel(half)
    k = kernel[::-1]
    left = 2 * sig[0] - sig[1:half + 1][::-1]
    right = 2 * sig[-1] - sig[-2:-half - 2:-1][::-1]
    padded = np.concatenate([left, sig, right])
    out = np.convolve(padded, k, mode="valid")
    return out[:m] if len(out) >= m else sig.copy()


def _volatility_model(x):
    """Robust local volatility: median |diff| scale + 3-tap local mean."""
    dx = np.diff(x)
    q_scale = np.median(np.abs(dx)) + 1e-12
    local_std = np.convolve(np.abs(dx), np.ones(3) / 3.0, mode="same")
    r_arr = np.concatenate([[q_scale], local_std]) + 1e-9
    return dx, q_scale, local_std, r_arr


# ----------------------------------------------------------------------
# Stage B+C: Kalman forward + RTS backward
# ----------------------------------------------------------------------
def _kalman_rts(x, r_arr, q_scale):
    n = len(x)
    F = np.array([[1.0, 1.0], [0.0, 1.0]])
    H = np.array([[1.0, 0.0]])
    Q = np.eye(2) * (q_scale ** 2) * 0.05

    s_filt = np.zeros((n, 2))
    P_filt = np.zeros((n, 2, 2))
    s_pred = np.zeros((n, 2))
    P_pred = np.zeros((n, 2, 2))

    P = np.eye(2) * (r_arr[0] ** 2)
    s = np.array([x[0], 0.0])
    s_filt[0] = s
    P_filt[0] = P

    for i in range(1, n):
        s = F @ s
        P = F @ P @ F.T + Q
        s_pred[i] = s
        P_pred[i] = P
        R = max(r_arr[i] ** 2, 1e-12)
        S = H @ P @ H.T + R
        K = (P @ H.T) / S
        innov = x[i] - (H @ s)[0]
        Sv = float(S[0, 0]) if np.ndim(S) else float(S)
        if abs(innov) > 3.0 * np.sqrt(Sv):  # robust 3-sigma gating
            R = R * 9.0
            S = H @ P @ H.T + R
            K = (P @ H.T) / S
            innov = x[i] - (H @ s)[0]
        s = s + (K.flatten() * innov)
        P = P - K @ H @ P
        s_filt[i] = s
        P_filt[i] = P

    s_smooth = s_filt.copy()
    for i in range(n - 2, -1, -1):
        Pm = P_pred[i + 1]
        G = np.linalg.solve(Pm.T, (P_filt[i] @ F.T).T).T
        s_smooth[i] = s_filt[i] + G @ (s_smooth[i + 1] - s_pred[i + 1])

    return s_smooth


# ----------------------------------------------------------------------
# Stage D: multi-scale SG bank blended by volatility
# ----------------------------------------------------------------------
def _sg_bank(kf_out, local_std, n, sg_half):
    cand = sorted({max(1, sg_half // 2), sg_half,
                   min(max(1, n // 2 - 1), 3 * sg_half // 2)})
    halves = [h for h in cand if 2 * h + 1 <= max(3, n)] or [1]

    v = np.concatenate([local_std[:1], local_std]) if n > 1 else np.zeros(1)
    v = np.maximum(v, 1e-12)

    if len(halves) == 1 or n < 2 * max(halves) + 1:
        return _zero_phase_smooth(kf_out, halves[0])

    centers = np.percentile(v, [20.0, 50.0, 80.0])[:len(halves)]
    spread = max(centers[-1] - centers[0], 1e-9)
    W = np.stack([np.exp(-0.5 * ((v - c) / spread) ** 2) for c in centers])
    W = W / (W.sum(axis=0, keepdims=True) + 1e-12)
    out = np.zeros(n)
    for j, h in enumerate(halves):
        out += W[j] * _zero_phase_smooth(kf_out, h)
    return out


# ----------------------------------------------------------------------
# Main enhanced filter
# ----------------------------------------------------------------------
def enhanced_filter_with_trend_preservation(x, window_size=20):
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n < window_size:
        raise ValueError(
            f"Input signal length ({n}) must be >= window_size ({window_size})")
    output_length = n - window_size + 1

    # Stage A: volatility model
    dx, q_scale, local_std, r_arr = _volatility_model(x)

    # Stages B+C: Kalman + RTS
    s_smooth = _kalman_rts(x, r_arr, q_scale)
    kf_out = s_smooth[:, 0]

    # Stage D: multi-scale SG bank
    sg_half = max(2, window_size // 4)
    smoothed = _sg_bank(kf_out, local_std, n, sg_half)

    # Stage E: confidence-gated slope lead compensation
    rts_slope = s_smooth[:, 1]
    gate = np.tanh(
        (np.abs(rts_slope) - 0.25 * q_scale) / (0.15 * q_scale + 1e-12))
    gate = np.clip(gate, 0.0, 1.0)
    lead = 0.5 * sg_half * 0.25
    smoothed = smoothed + lead * gate * np.sign(rts_slope) * np.maximum(
        np.abs(rts_slope) - 0.25 * q_scale, 0.0)

    # Stage F: slope-magnitude-conditional quiet-region polish.
    # Narrow zero-phase SG ONLY where the trend is not statistically
    # meaningful (|slope| below the existing gate threshold); genuine
    # transitions are left untouched to preserve responsiveness.
    h_polish = max(2, sg_half // 3)
    c_polish = _sg_kernel(h_polish, 2)
    polished = _zero_phase_smooth(smoothed, h_polish, c_polish)
    if len(polished) == n:
        quiet = np.abs(rts_slope) < 0.25 * q_scale
        smoothed = np.where(quiet, polished, smoothed)

    # Stage G: end-aligned trim with explicit length guard
    if len(smoothed) >= output_length:
        y = smoothed[-output_length:]
    else:
        y = kf_out[-output_length:]
    if len(y) != output_length:
        y = kf_out[-output_length:]
    return y


def adaptive_filter(x, window_size=20):
    """Basic sliding-window box filter (backward-compatible path)."""
    x = np.asarray(x, dtype=float)
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})")
    c = np.cumsum(np.insert(x, 0, 0.0))
    return (c[window_size:] - c[:-window_size]) / window_size


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    np.random.seed(seed)
    t = np.linspace(0, 10, length)
    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )
    clean_signal += 0.1 * t * np.sin(0.2 * t)
    clean_signal += np.cumsum(np.random.randn(length) * 0.05)
    noisy_signal = clean_signal + np.random.normal(0, noise_level, length)
    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000, noise_level=0.3, window_size=20):
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]
        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]
        correlation = np.corrcoef(filtered_signal, aligned_clean)[0, 1] if min_length > 1 else 0
        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (noise_before - noise_after) / noise_before if noise_before > 0 else 0
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }
    elif len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }
    else:
        return {
            "filtered_signal": [],
            "clean_signal": [],
            "noisy_signal": [],
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": 0,
        }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
