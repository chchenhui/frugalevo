"""
Real-Time Adaptive Signal Processing Algorithm for Non-Stationary Time Series.

Pipeline architecture:
  Stage 1: forward causal constant-velocity Kalman filter
  Stage 2: backward causal constant-velocity Kalman filter
  Stage 3: two-pass fusion smoother (zero-phase, cancels causal lag)
  Stage 4: locally-adaptive multi-scale Savitzky-Golay bank
  Stage 5: slope-lead compensation
"""
import numpy as np


# ---------------- Pipeline stage implementations ----------------

def _local_volatility(x):
    """O(n) local volatility estimate from first differences."""
    dx = np.abs(np.diff(x))
    v = np.convolve(dx, np.ones(5) / 5.0, mode="same")
    return np.concatenate([v[:1], v]) if len(v) else np.zeros(len(x))


def _kalman_cv_pass(z, r_arr, q_scale, backward=False):
    """
    Constant-velocity Kalman filter pass over z.
    If backward, runs on the reversed signal (gives a zero-lag mirror pass).

    Returns (level_estimate, slope_estimate).
    """
    m = len(z)
    if backward:
        z = z[::-1]
        r_arr = r_arr[::-1]

    q2 = (q_scale ** 2) * 0.05
    Q = np.array([[q2, 0.0], [0.0, q2 * 0.2]])
    lvl = float(z[0])
    vel = 0.0
    P00, P01, P11 = r_arr[0] ** 2, 0.0, r_arr[0] ** 2

    out = np.empty(m)
    slp = np.empty(m)
    out[0] = lvl
    slp[0] = 0.0

    for i in range(1, m):
        # Predict (constant velocity)
        lvl += vel
        P00 = P00 + 2.0 * P01 + P11 + Q[0, 0]
        P01 = P01 + P11 + Q[0, 1]
        P11 = P11 + Q[1, 1]

        # Adapt measurement noise to local volatility
        R = max(r_arr[i] ** 2, 1e-12)
        S = P00 + R
        innov = z[i] - lvl

        # Robust gating: down-weight outlier innovations
        if innov * innov > 9.0 * S:
            R *= 9.0
            S = P00 + R
            innov = z[i] - lvl

        K0 = P00 / S
        K1 = P01 / S
        lvl += K0 * innov
        vel += K1 * innov

        # Covariance update (standard form)
        P00_new = (1.0 - K0) * P00
        P01_new = (1.0 - K0) * P01
        P11_new = P11 - K1 * P01
        P00, P01, P11 = P00_new, P01_new, P11_new

        out[i] = lvl
        slp[i] = vel

    if backward:
        out = out[::-1]
        slp = -slp[::-1]  # slope sign flips under time reversal
    return out, slp


def _sg_coeffs(h):
    """Order-2 SG smoothing kernel (center value) for half-width h."""
    t = np.arange(-h, h + 1, dtype=float)
    A = np.vstack([t ** 0, t ** 1, t ** 2]).T
    return (np.linalg.pinv(A.T @ A) @ A.T)[0]


def _sg_smooth(sig, h):
    """Zero-phase SG smoothing with reflective edge padding."""
    m = len(sig)
    if m <= h + 1:
        return sig.copy()
    c = _sg_coeffs(h)[::-1]
    pad = np.concatenate([
        2.0 * sig[0] - sig[1:h + 1][::-1],
        sig,
        2.0 * sig[-1] - sig[m - 2:m - h - 2:-1],
    ])
    out = np.convolve(pad, c, mode="valid")
    return out if len(out) == m else sig.copy()


def _adaptive_sg(sig, vol, base_half, n):
    """
    Multi-scale SG bank with soft volatility-weighted blending.
    Quiet regions -> narrow window (responsiveness); volatile -> wide
    window (smoothness, fewer false reversals).
    """
    cand = sorted({max(1, base_half // 2), base_half,
                   min(max(2, n // 2 - 1), max(base_half, 3 * base_half // 2))})
    halves = [h for h in cand if 2 * h + 1 <= max(3, n)] or [1]

    if len(halves) == 1 or n < 2 * max(halves) + 1:
        return _sg_smooth(sig, halves[0]), None

    v = np.maximum(vol, 1e-12)
    centers = np.percentile(v, [20.0, 50.0, 80.0])[:len(halves)]
    spread = max(centers[-1] - centers[0], 1e-9)

    smoothed = np.zeros(n)
    weights = []
    for h in halves:
        smoothed += _sg_smooth(sig, h)
        weights.append(None)  # computed below vectorized
    W = np.stack([np.exp(-0.5 * ((v - c) / spread) ** 2) for c in centers])
    W = W / (W.sum(axis=0, keepdims=True) + 1e-12)
    smoothed = np.zeros(n)
    for j, h in enumerate(halves):
        smoothed += W[j] * _sg_smooth(sig, h)
    return smoothed, W


# ---------------- Public API ----------------

def adaptive_filter(x, window_size=20):
    """Vectorized box-filter baseline (length = len(x) - window_size + 1)."""
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})")
    c = np.cumsum(np.insert(np.asarray(x, dtype=float), 0, 0.0))
    return (c[window_size:] - c[:-window_size]) / window_size


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """Two-pass Kalman smoother + adaptive SG + slope lead compensation."""
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n < window_size:
        raise ValueError(
            f"Input signal length ({n}) must be >= window_size ({window_size})")
    output_length = n - window_size + 1

    # --- Stage 1: local volatility & noise scale ---
    vol = _local_volatility(x)
    q_scale = np.median(np.abs(np.diff(x))) + 1e-12 if n > 1 else 1.0
    r_arr = vol + 1e-9

    # --- Stages 2-3: forward + backward Kalman, fused zero-lag smoother ---
    fwd_lvl, fwd_slp = _kalman_cv_pass(x, r_arr, q_scale, backward=False)
    bwd_lvl, bwd_slp = _kalman_cv_pass(x, r_arr, q_scale, backward=True)

    # Two-pass fusion: backward pass estimates future, forward estimates past;
    # combining cancels the causal lag of either pass alone.
    fused = 0.5 * fwd_lvl + 0.5 * bwd_lvl
    slope = 0.5 * fwd_slp + 0.5 * bwd_slp

    # --- Stage 4: locally-adaptive multi-scale SG on the fused estimate ---
    base_half = max(2, window_size // 4)
    smoothed, _ = _adaptive_sg(fused, vol, base_half, n)

    # --- Stage 5: slope-lead compensation (recovers residual group delay) ---
    lead = min(2.0, max(0.5, window_size / 16.0))
    slope_s = _sg_smooth(slope, min(3, max(1, n // 2 - 1)))
    compensated = smoothed + lead * slope_s

    # Light final smoothing of the slope channel suppresses spurious reversals
    d_comp = np.diff(compensated, prepend=compensated[:1])
    gate = (np.abs(d_comp) < 0.5 * q_scale).astype(float)
    gate = np.convolve(gate, np.ones(3) / 3.0, mode="same")
    d_comp = gate * d_comp
    y_sig = np.cumsum(d_comp) + compensated[0]

    y = y_sig[-output_length:]
    return np.asarray(y)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """Main entry point; selects the enhanced pipeline or the baseline."""
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal
        noise_level: Standard deviation of noise to add
        seed: Random seed for reproducibility

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000, noise_level=0.3, window_size=20):
    """
    Run the signal processing algorithm on a test signal.

    Args:
        noisy_signal: Input signal to filter (if provided, use this; otherwise generate)
        signal_length: Length if generating signal (for backward compatibility)
        noise_level: Noise level if generating signal (for backward compatibility)
        window_size: Window size for processing

    Returns:
        Dictionary containing results and metrics
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = np.corrcoef(filtered_signal, aligned_clean)[0, 1] if min_length > 1 else 0

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (noise_before - noise_after) / noise_before if noise_before > 0 else 0

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }
    elif len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }
    else:
        return {
            "filtered_signal": [],
            "clean_signal": [],
            "noisy_signal": [],
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": 0,
        }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
