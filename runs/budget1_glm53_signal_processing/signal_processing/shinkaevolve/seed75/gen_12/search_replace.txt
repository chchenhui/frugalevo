<DESCRIPTION>
Replace the exponential weighted moving average with a 2nd-order (constant-velocity) Kalman filter combined with Rauch-Tung-Striebel (RTS) backward smoothing. This yields: (1) zero-phase smoothing within the signal eliminating lag bias, (2) an adaptive process/measurement noise ratio estimated from signal SNR so the filter stiffens in quiet regions (noise suppression, fewer spurious slope reversals) and loosens in fast transitions (low lag, good tracking), and (3) O(N) computation maintaining efficiency. Output alignment (window-end indexed) is preserved so downstream code is unchanged.
</DESCRIPTION>

    x = np.asarray(x, dtype=float)
    n = len(x)

    # --- Adaptive noise estimation ---------------------------------------
    # Measurement noise estimated from high-frequency content (first difference).
    dx = np.diff(x)
    r_meas = np.var(dx) / 2.0 if len(dx) > 1 else 1.0
    total_var = np.var(x)
    signal_var = max(total_var - r_meas, 1e-12)
    # Process noise scaled by SNR: noisy -> stiff (less q), clean/fast -> responsive.
    snr = signal_var / max(r_meas, 1e-12)
    q_proc = 1e-3 * (1.0 + snr)

    # --- Forward Kalman filter (state: [level, velocity]) -----------------
    dt = 1.0
    F = np.array([[1.0, dt], [0.0, 1.0]])
    H = np.array([[1.0, 0.0]])
    Q = np.array([[q_proc * dt ** 2, q_proc * dt],
                  [q_proc * dt, q_proc]]) * 0.5
    R = max(r_meas, 1e-12)

    x_prior = np.zeros((n, 2))
    X_post = np.zeros((n, 2))

    v0 = dx[0] if len(dx) > 0 else 0.0
    X = np.array([x[0], v0])
    P = np.eye(2) * max(total_var, 1.0)
    F_T = F.T

    for k in range(n):
        # Predict
        Xp = F @ X
        Pp = F @ P @ F_T + Q
        x_prior[k] = Xp
        # Update
        S = Pp[0, 0] + R
        K0 = Pp[0, 0] / S
        K1 = Pp[1, 0] / S
        innov = x[k] - Xp[0]
        X = np.array([Xp[0] + K0 * innov, Xp[1] + K1 * innov])
        # Joseph-free covariance update (P symmetric)
        P00 = Pp[0, 0] - K0 * Pp[0, 0]
        P01 = Pp[0, 1] - K0 * Pp[0, 1]
        P10 = Pp[1, 0] - K1 * Pp[0, 0]
        P11 = Pp[1, 1] - K1 * Pp[0, 1]
        P = np.array([[P00, P01], [P10, P11]])
        X_post[k] = X

    # --- RTS backward smoothing for zero-phase estimates -------------------
    x_smooth = X_post[:, 0].copy()
    for k in range(n - 2, -1, -1):
        # Re-run predict from posterior k (cheap, 2x2 matrices)
        Pp = F @ P @ F_T + Q
        # C = P F' Pp^{-1}
        det = Pp[0, 0] * Pp[1, 1] - Pp[0, 1] * Pp[1, 0]
        if abs(det) < 1e-300:
            continue
        Pinv = np.array([[Pp[1, 1], -Pp[0, 1]], [-Pp[1, 0], Pp[0, 0]]]) / det
        C = P @ F_T @ Pinv
        dX = X_post[k + 1] - x_prior[k + 1]
        x_smooth[k] = (X_post[k] + C @ dX)[0]

    # --- Output aligned to window-end convention --------------------------
    output_length = n - window_size + 1
    y = x_smooth[window_size - 1 : window_size - 1 + output_length].copy()

    return y
