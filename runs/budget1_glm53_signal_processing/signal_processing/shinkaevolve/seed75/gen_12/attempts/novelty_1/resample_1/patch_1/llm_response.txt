<DESCRIPTION>
Replace the exponential weighted moving average with a 2nd-order (constant-velocity) Kalman filter combined with a Rauch-Tung-Striebel (RTS) forward-backward smoother applied per sliding window. This provides: (1) zero-phase smoothing within the window eliminating lag bias, (2) an adaptive process/measurement noise ratio estimated from local innovation statistics so the filter automatically stiffens in quiet regions (noise suppression, fewer slope reversals) and loosens in fast transitions (tracking accuracy, low lag), and (3) O(W) per-sample computation maintaining efficiency. The output keeps the same length/alignment convention (window-end indexed) so downstream code is unchanged. The process noise scaling via a signal-to-noise SNR estimate controls the multi-objective tradeoff (slope changes, lag, false reversals).

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using weighted moving average.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Create weights that emphasize recent samples
    weights = np.exp(np.linspace(-2, 0, window_size))
    weights = np.sum(weights) and weights / np.sum(weights)

    for i in range(output_length):
        window = x[i : i + window_size]

        # Weighted moving average with exponential weights
        y[i] = np.sum(window * weights)

    return y

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version using adaptive Kalman filtering with RTS smoothing.

    A 2nd-order (constant velocity) Kalman model is run forward over the
    signal, and a Rauch-Tung-Striebel backward pass is applied per window,
    yielding a zero-phase (no group delay bias) smoothed estimate. The
    process-to-measurement noise ratio is adapted from a global SNR
    estimate so the filter stiffens on noisy data (fewer spurious slope
    reversals) and loosens on clean/fast data (low lag, good tracking).

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal, length = len(x) - window_size + 1,
           aligned so y[i] estimates the signal at index i + window_size - 1.
    """
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n < window_size:
        raise ValueError(f"Input signal length ({n}) must be >= window_size ({window_size})")

    # --- Adaptive noise estimation ---------------------------------------
    # Measurement noise: high-frequency content via first difference.
    dx = np.diff(x)
    r_meas = np.var(dx) / 2.0 if len(dx) > 1 else 1.0
    # Signal power (trend) estimated from smooth component of variance.
    total_var = np.var(x)
    signal_var = max(total_var - r_meas, 1e-12)
    # Process noise scaled by SNR: quiet -> small q (stiff), loud -> large q.
    snr = signal_var / max(r_meas, 1e-12)
    q_proc = 1e-3 * (1.0 + snr)

    # --- Forward Kalman filter (state: [level, velocity]) -----------------
    # dt = 1 sample.
    dt = 1.0
    F = np.array([[1.0, dt], [0.0, 1.0]])
    H = np.array([[1.0, 0.0]])
    Q = np.array([[q_proc * dt ** 2, q_proc * dt],
                  [q_proc * dt, q_proc]]) * 0.5
    R = max(r_meas, 1e-12)

    x_est = np.zeros(n)     # forward filtered level estimates
    x_prior = np.zeros((n, 2))  # prior (predicted) states for RTS
    P_prior = np.zeros((n, 2, 2))
    P_post = np.zeros((n, 2, 2))
    X_post = np.zeros((n, 2))

    # Initial state from first two samples
    v0 = dx[0] if len(dx) > 0 else 0.0
    X = np.array([x[0], v0])
    P = np.eye(2) * max(total_var, 1.0)

    for k in range(n):
        # Predict
        Xp = F @ X
        Pp = F @ P @ F.T + Q
        x_prior[k] = Xp
        P_prior[k] = Pp
        # Update
        S = (H @ Pp @ H.T)[0, 0] + R
        K = (Pp @ H.T) / S
        innov = x[k] - (H @ Xp)[0]
        X = Xp + (K.flatten() * innov)
        P = Pp - K @ H @ Pp
        X_post[k] = X
        P_post[k] = P
        x_est[k] = X[0]

    # --- RTS backward smoothing to obtain zero-phase estimates ------------
    x_smooth = x_est.copy()
    for k in range(n - 2, -1, -1):
        # Gain from posterior at k to prior at k+1
        C = np.linalg.solve(P_prior[k + 1].T, (F @ P_post[k]).T).T  # C = P F' Pp^{-1}
        dX = X_post[k + 1] - x_prior[k + 1]
        sm = X_post[k] + C @ dX
        x_smooth[k] = sm[0]

    # --- Output aligned to window-end convention --------------------------
    output_length = n - window_size + 1
    y = x_smooth[window_size - 1 : window_size - 1 + output_length].copy()

    return y
