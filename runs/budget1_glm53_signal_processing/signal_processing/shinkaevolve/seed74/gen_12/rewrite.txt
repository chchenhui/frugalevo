"""
Real-Time Adaptive Signal Processing Algorithm for Non-Stationary Time Series

Architecture:
    raw window --> [Stage 1] ZeroLagPolynomial (vectorized quadratic endpoint
                       fit with adaptive window shrinkage for fast regimes)
                --> [Stage 2] AdaptiveVelocityKalman (constant-velocity model,
                       dynamics-scaled process noise, robust MAD noise estimate)
                --> [Stage 3] HysteresisTrendGate (discrete trend state machine
                       that releases slope reversals only when confirmed or
                       when they exceed the robust noise floor)
                --> filtered output
"""
import numpy as np


def _endpoint_taps(n):
    """Precompute convolution taps for evaluating a quadratic fit at the
    newest sample (t = 1) over a window of length n."""
    t = np.linspace(-1.0, 1.0, n)
    V = np.vstack([np.ones(n), t, t * t]).T
    eval_row = np.array([1.0, 1.0, 1.0])
    try:
        VtV_inv = np.linalg.pinv(V.T @ V)
    except np.linalg.LinAlgError:
        VtV_inv = np.linalg.pinv(V.T @ V + 1e-9 * np.eye(3))
    return eval_row @ VtV_inv @ V.T


def _robust_noise_std(d):
    """MAD-based robust std of first differences."""
    if len(d) < 2:
        return 0.0
    med = np.median(d)
    return np.median(np.abs(d - med)) / 0.6745


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Zero-lag polynomial endpoint estimator + adaptive Kalman velocity
    fusion + hysteresis trend gate.

    Returns filtered signal of length len(x) - window_size + 1.
    """
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )

    x = np.asarray(x, dtype=float)
    n = window_size
    out_len = len(x) - n + 1

    # ---------------------------------------------------------------
    # Stage 1: zero-lag endpoint polynomial fit (vectorized)
    # ---------------------------------------------------------------
    taps_full = _endpoint_taps(n)
    y_full = np.correlate(x, taps_full[::-1], mode="valid")

    # Short-window variant for fast regimes (less smoothing, more response)
    half = max(5, n // 2)
    if half != n:
        taps_half = _endpoint_taps(half)
        y_half = np.correlate(x, taps_half[::-1], mode="valid")
    else:
        y_half = y_full

    # Regime detection per output sample: if local dispersion of the raw
    # differences is high (fast dynamics), prefer the shorter (snappier) fit.
    dx = np.abs(np.diff(x))
    noise_ref = _robust_noise_std(np.diff(x))
    # activity measure over each window of diffs (aligned to output samples)
    act = np.array([
        np.mean(dx[i:i + n - 1]) for i in range(out_len)
    ])
    thr = max(1.5 * noise_ref, 1e-9)
    use_short = act > thr
    y1 = np.where(use_short, y_half, y_full)

    # ---------------------------------------------------------------
    # Stage 2: adaptive constant-velocity Kalman on endpoint estimates
    # ---------------------------------------------------------------
    # Robust measurement-noise estimate from the fit sequence
    r_std = _robust_noise_std(np.diff(y1))
    R = max(r_std, 1e-6) ** 2

    # Process noise scaled by observed signal dynamics
    q_scale = np.var(np.diff(x)) + 1e-12
    q_pos = np.clip(2e-3 * q_scale / (q_scale + 1.0), 1e-4, 1e-1)
    Q = np.array([[q_pos, 0.0], [0.0, 8.0 * q_pos]])

    F = np.array([[1.0, 1.0], [0.0, 1.0]])
    H = np.array([[1.0, 0.0]])
    I2 = np.eye(2)

    level = y1[0]
    slope = y1[1] - y1[0] if out_len > 1 else 0.0
    P = np.array([[R, 0.0], [0.0, R]])
    y2 = np.empty(out_len)
    y2[0] = level
    for i in range(1, out_len):
        # Predict
        level = level + slope
        P = F @ P @ F.T + Q
        # Update
        S = P[0, 0] + R
        K = np.array([P[0, 0] / S, P[1, 0] / S])
        innov = y1[i] - level
        level = level + K[0] * innov
        slope = slope + K[1] * innov
        P = (I2 - np.outer(K, H)) @ P
        y2[i] = level

    # ---------------------------------------------------------------
    # Stage 3: hysteresis trend gate (reversal suppression)
    # ---------------------------------------------------------------
    # A slope sign change in the output is only "released" (committed)
    # when either (a) the reversal magnitude exceeds a robust noise floor,
    # or (b) it persists for K consecutive steps. Otherwise the output
    # follows the committed trend direction, damping spurious ripple.
    d2 = np.diff(y2)
    floor_std = _robust_noise_std(d2)
    thresh = max(2.0 * floor_std, 1e-9)
    K_confirm = 3

    y3 = np.empty(out_len)
    y3[0] = y2[0]
    if out_len == 1:
        return y3
    y3[1] = y2[1]

    trend = np.sign(y3[1] - y3[0])  # committed trend direction (+1/-1/0)
    if trend == 0:
        trend = 1.0
    pending_sign = 0.0
    pending_count = 0
    pending_mag = 0.0

    for i in range(2, out_len):
        raw_d = y2[i] - y3[i - 1]
        s = np.sign(raw_d)
        if s == 0:
            s = trend
        if s == trend:
            # Agreement: pass through, reset pending reversal state
            pending_sign = 0.0
            pending_count = 0
            pending_mag = 0.0
            y3[i] = y3[i - 1] + raw_d
        else:
            # Conflict with committed trend
            if abs(raw_d) >= thresh:
                # Genuine, large reversal: commit immediately
                trend = s
                pending_sign = 0.0
                pending_count = 0
                pending_mag = 0.0
                y3[i] = y3[i - 1] + raw_d
            else:
                # Sub-noise reversal candidate: accumulate evidence
                if s == pending_sign:
                    pending_count += 1
                    pending_mag = max(pending_mag, abs(raw_d))
                else:
                    pending_sign = s
                    pending_count = 1
                    pending_mag = abs(raw_d)
                if pending_count >= K_confirm:
                    # Sustained sub-noise reversal: treat as genuine
                    trend = s
                    pending_sign = 0.0
                    pending_count = 0
                    pending_mag = 0.0
                    y3[i] = y3[i - 1] + raw_d
                else:
                    # Hold the committed trend: apply the component of the
                    # update along the trend direction, damp the reversal.
                    proj = trend * raw_d
                    held = 0.25 * proj  # small allowance along trend
                    y3[i] = y3[i - 1] + held
                    # Prevent drift away from the internal (accurate) level:
                    # gently pull the output back toward the Kalman estimate.
                    y3[i] = 0.8 * y3[i] + 0.2 * y2[i]
                    # Update trend magnitude memory from output motion
                    if np.sign(y3[i] - y3[i - 1]) != 0 and np.sign(y3[i] - y3[i - 1]) == trend:
                        pass  # still on trend

    return y3


def adaptive_filter(x, window_size=20):
    """Basic mode: zero-lag polynomial endpoint estimator only (fast path)."""
    if len(x) < window_size:
        raise ValueError(
            f"Input signal length ({len(x)}) must be >= window_size ({window_size})"
        )
    x = np.asarray(x, dtype=float)
    taps = _endpoint_taps(window_size)
    return np.correlate(x, taps[::-1], mode="valid")


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Main signal processing function that applies the selected algorithm.

    Args:
        input_signal: Input time series data
        window_size: Window size for processing
        algorithm_type: Type of algorithm to use ("basic" or "enhanced")

    Returns:
        Filtered signal
    """
    if algorithm_type == "enhanced":
        return enhanced_filter_with_trend_preservation(input_signal, window_size)
    return adaptive_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal
        noise_level: Standard deviation of noise to add
        seed: Random seed for reproducibility

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000, noise_level=0.3, window_size=20):
    """
    Run the signal processing algorithm on a test signal.

    Args:
        noisy_signal: Input signal to filter (if provided, use this; otherwise generate)
        signal_length: Length if generating signal (for backward compatibility)
        noise_level: Noise level if generating signal (for backward compatibility)
        window_size: Window size for processing

    Returns:
        Dictionary containing results and metrics
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = np.corrcoef(filtered_signal, aligned_clean)[0, 1] if min_length > 1 else 0

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (noise_before - noise_after) / noise_before if noise_before > 0 else 0

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }
    elif len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }
    else:
        return {
            "filtered_signal": [],
            "clean_signal": [],
            "noisy_signal": [],
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": 0,
        }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
