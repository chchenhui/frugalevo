def _curvature_clamp(y, limit):
    """
    Zero-phase curvature (second-difference) clamp.
    The velocity clamp bounds |dy| but still permits alternating
    +/-budget steps — the exact signature of residual ripple that
    survives as spurious slope changes. Forward sweep then mirrored
    backward sweep; each limits |y[k+1] - 2y[k] + y[k-1]| to `limit`
    by moving the outer sample of the triple onto the locally linear
    continuation. Alternating ripple folds into straight segments;
    genuine curved dynamics (curvature within budget) pass untouched.
    """
    n = len(y)
    if n < 3 or limit <= 0.0:
        return y.copy()
    out = y.copy()
    # forward sweep: enforce on triple (k-1, k, k+1) by moving k+1
    for k in range(1, n - 1):
        c = out[k + 1] - 2.0 * out[k] + out[k - 1]
        if abs(c) > limit:
            out[k + 1] = 2.0 * out[k] - out[k - 1] + np.sign(c) * limit
    # mirrored backward sweep: enforce by moving k-1
    for k in range(n - 2, 0, -1):
        c = out[k - 1] - 2.0 * out[k] + out[k + 1]
        if abs(c) > limit:
            out[k - 1] = 2.0 * out[k] - out[k + 1] + np.sign(c) * limit
    return out
