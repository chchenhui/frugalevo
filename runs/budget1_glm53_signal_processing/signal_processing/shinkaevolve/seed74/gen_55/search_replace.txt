<DESCRIPTION>
Replace the final bilateral one-pole damper with a fixed-λ Whittaker–Henderson polish (second-order difference penalty) solved via a banded O(n) Cholesky solve. The one-pole damper's position-dependent effective smoothing (its exponential memory makes the penalty non-uniform) leaks residual ripple — the slope-change count of 20.4 is the weakest smoothness metric. A constant λ with a second-order difference penalty applies a uniform curvature penalty whose passband is flatter in the band of interest, cutting residual ripple without the lag a stronger one-pole cascade would add. λ is derived from the residual MAD already computed (r_std) and the robust noise scale (sigma_r): clip(25·(r_std/σ_r)², 1, 200). The Whittaker system is pentadiagonal (bandwidth 2 per side), solved in O(n) with a simple banded Cholesky implementation, keeping runtime ≈0.05s. Applied to the full-length signal before the final slice, so the `len(y) = n − window_size + 1` contract is preserved.

def _bilateral_one_pole(x, alpha):
    """Forward then reversed one-pole smoothing — net zero phase delay."""
    y = np.empty_like(x)
    acc = x[0]
    y[0] = acc
    for i in range(1, len(x)):
        acc = alpha * x[i] + (1.0 - alpha) * acc
        y[i] = acc
    acc = y[-1]
    out = y.copy()

def _whittaker_polish(y, lam):
    """
    Fixed-lambda Whittaker-Henderson smoother (second-order difference
    penalty): minimize ||y - z||^2 + lam * ||D2 z||^2.

    The normal equations form a symmetric positive-definite pentadiagonal
    system solved with a banded Cholesky factorization in O(n). A constant
    lambda applies a uniform curvature penalty with a flat passband, so it
    removes residual ripple (slope changes) without the position-dependent
    leakage of a one-pole damper and without adding phase lag.
    """
    n = len(y)
    if n < 5:
        return y.copy()

    # Second-order difference matrix D2 (n-2 x n), penalty A = D2^T D2.
    # Row k of A (0-based, k=0..n-1) has contributions from D2 rows
    # max(0, k-2)..min(n-3, k):
    #   D2 row j has entries [1, -2, 1] at columns j, j+1, j+2.
    # Offsets: A[k, k+d] for d in {-2,-1,0,1,2}:
    #   d=0: 6 (interior), edges: k=0 or k=n-1 -> 5, k=1 or k=n-2 -> 5, corners -> 1/1
    #   d=+-1: -4 (interior), edges reduced
    #   d=+-2: 1 (interior), 0 near edges
    # Build explicitly for clarity:
    main = np.zeros(n)
    off1 = np.zeros(n - 1)   # A[k, k+1]
    off2 = np.zeros(n - 2)   # A[k, k+2]
    for k in range(n):
        acc = 0.0
        for j in range(max(0, k - 2), min(n - 3, k) + 1):
            # contribution of A[k,k] = sum_j D[j,k]^2
            acc += 1.0
        main[k] = acc
    for k in range(n - 1):
        acc = 0.0
        for j in range(max(0, min(k, k + 1) - 2), min(n - 3, max(k, k + 1)) + 1):
            lo = j
            hi = j + 2
            if lo <= k < hi and lo <= k + 1 < hi:
                s = 1.0 if k == lo else (-2.0 if k == lo + 1 else 1.0)
                t = 1.0 if k + 1 == lo else (-2.0 if k + 1 == lo + 1 else 1.0)
                acc += s * t
        off1[k] = acc
    for k in range(n - 2):
        acc = 0.0
        for j in range(max(0, min(k, k + 2) - 2), min(n - 3, max(k, k + 2)) + 1):
            lo = j
            hi = j + 2
            if lo <= k < hi and lo <= k + 2 < hi:
                s = 1.0 if k == lo else (-2.0 if k == lo + 1 else 1.0)
                t = 1.0 if k + 2 == lo else (-2.0 if k + 2 == lo + 1 else 1.0)
                acc += s * t
        off2[k] = acc

    # Banded Cholesky of (I + lam*A): L L^T with bandwidth 2.
    b0 = 1.0 + lam * main      # diagonal
    b1 = lam * off1            # first superdiagonal
    b2 = lam * off2            # second superdiagonal

    d0 = np.zeros(n)
    l1 = np.zeros(n - 1)       # L[k+1, k]
    l2 = np.zeros(n - 2)       # L[k+2, k]
    for k in range(n):
        v = b0[k]
        if k >= 1:
            v -= l1[k - 1] * l1[k - 1]
        if k >= 2:
            v -= l2[k - 2] * l2[k - 2]
        if v <= 1e-12:
            v = 1e-12
        d0[k] = np.sqrt(v)
        if k < n - 1:
            l1[k] = b1[k]
            if k >= 1:
                l1[k] -= l1[k - 1] * d0[k]
            l1[k] /= d0[k]
        if k < n - 2:
            l2[k] = b2[k] / d0[k]

    # Solve (L L^T) z = y by forward/back substitution.
    rhs = y.astype(float).copy()
    z = np.zeros(n)
    for k in range(n):
        v = rhs[k]
        if k >= 1:
            v -= l1[k - 1] * z[k - 1]
        if k >= 2:
            v -= l2[k - 2] * z[k - 2]
        z[k] = v / d0[k]
    for k in range(n - 1, -1, -1):
        v = z[k]
        if k < n - 1:
            v -= l1[k] * z[k + 1]
        if k < n - 2:
            v -= l2[k] * z[k + 2]
        z[k] = v / d0[k]
    return z
