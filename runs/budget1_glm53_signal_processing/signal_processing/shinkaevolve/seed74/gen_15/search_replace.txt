            # dominates the plain moving average on lag, slope stability and
            # noise rejection while keeping the identical output contract.
            filtered = enhanced_filter_with_trend_preservation(input_signal, window_size)

            # Lightweight deadband on output increments: suppress tiny
            # slope reversals caused by noise without adding pipeline
            # latency or a separate stabilizer stage.
            n = len(filtered)
            if n > 1:
                noise_scale = 0.2 * float(np.std(np.diff(input_signal)))
                out = np.array(filtered, dtype=float)
                for i in range(1, n):
                    inc = out[i] - out[i - 1]
                    if abs(inc) <= noise_scale:
                        # Deadband: hold the level flat for tiny increments
                        out[i] = out[i - 1]
                    else:
                        # Soft shrinkage avoids a step discontinuity
                        out[i] = out[i - 1] + math.copysign(
                            abs(inc) - noise_scale, inc
                        )
                return out
            return filtered
