def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using weighted moving average.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Create weights that emphasize recent samples
    weights = np.exp(np.linspace(-2, 0, window_size))
    weights = weights / np.sum(weights)

    for i in range(output_length):
        window = x[i : i + window_size]

        # Weighted moving average with exponential weights
        y[i] = np.sum(window * weights)

    return y

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version using an adaptive Kalman filter with zero-phase smoothing.

    Combines:
      1. A causal constant-velocity Kalman filter whose measurement noise is
         adapted online from local residuals (handles non-stationary volatility).
      2. A forward-backward (RTS-style) smoothing pass that cancels phase delay,
         minimizing lag error while preserving trend dynamics and suppressing
         spurious slope reversals.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window (controls output length / initial
                     measurement noise scale)

    Returns:
        y: Filtered output signal of length len(x) - window_size + 1
    """
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n < window_size:
        raise ValueError(f"Input signal length ({n}) must be >= window_size ({window_size})")

    # --- Adaptive constant-velocity Kalman filter (causal pass) ---
    # State: [level, velocity]
    dt = 1.0
    q = 0.02  # process noise (level)
    qv = 0.005  # process noise (velocity)

    # Initial estimate of measurement noise from local differences
    if n >= 2:
        diff = np.diff(x)
        r0 = max(np.var(diff) / 2.0, 1e-8)
    else:
        r0 = 1.0

    # Forward pass
    x_f = np.zeros(n)
    xf_lvl = x[0]
    xf_vel = 0.0
    P = np.array([[r0, 0.0], [0.0, r0]])
    F = np.array([[1.0, dt], [0.0, 1.0]])
    Q = np.array([[q, 0.0], [0.0, qv]])
    H = np.array([[1.0, 0.0]])

    levels_f = np.zeros(n)
    vels_f = np.zeros(n)
    levels_b = np.zeros(n)
    vels_b = np.zeros(n)

    for k in range(n):
        if k == 0:
            pass  # initial state
        else:
            # Predict
            xf_lvl = xf_lvl + xf_vel * dt
            P = F @ P @ F.T + Q
        # Adapt measurement noise from recent innovation magnitude
        innov = x[k] - xf_lvl
        r = r0 * 0.5 + max(innov * innov, 0.0) * 0.5
        r = max(r, 1e-8)
        # Update
        S = P[0, 0] + r
        K = np.array([P[0, 0] / S, P[1, 0] / S])
        xf_lvl = xf_lvl + K[0] * innov
        xf_vel = xf_vel + K[1] * innov
        P = (np.eye(2) - np.outer(K, H)) @ P
        levels_f[k] = xf_lvl
        vels_f[k] = xf_vel

    # --- Backward pass (RTS smoother): zero-phase, removes lag ---
    # Run filter backwards over reversed signal
    xb_lvl = x[-1]
    xb_vel = 0.0
    Pb = np.array([[r0, 0.0], [0.0, r0]])
    for k in range(n - 1, -1, -1):
        if k != n - 1:
            xb_lvl = xb_lvl + xb_vel * dt
            Pb = F @ Pb @ F.T + Q
        innov = x[k] - xb_lvl
        r = r0 * 0.5 + max(innov * innov, 0.0) * 0.5
        r = max(r, 1e-8)
        S = Pb[0, 0] + r
        K = np.array([Pb[0, 0] / S, Pb[1, 0] / S])
        xb_lvl = xb_lvl + K[0] * innov
        xb_vel = xb_vel + K[1] * innov
        Pb = (np.eye(2) - np.outer(K, H)) @ Pb
        levels_b[k] = xb_lvl
        vels_b[k] = xb_vel

    # Combine forward and backward estimates (variance-weighted blend).
    # Use velocity confidence as blend weight for levels.
    w_f = 1.0 / (1.0 + np.abs(vels_f))
    w_b = 1.0 / (1.0 + np.abs(vels_b))
    smooth = (w_f * levels_f + w_b * levels_b) / (w_f + w_b)

    # --- Output with same length convention as windowed filters ---
    output_length = n - window_size + 1
    y = smooth[window_size - 1 :]

    return y
