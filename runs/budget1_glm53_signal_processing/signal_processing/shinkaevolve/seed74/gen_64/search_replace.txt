<DESCRIPTION>
The velocity clamp bounds per-step displacement but still permits alternating ±limit steps, which is how ~20 slope changes survive as bounded oscillation. Add a zero-phase curvature clamp on the second difference |y[k+1] - 2y[k] + y[k-1]| ≤ c * (robust d2 scale), applied in forward+backward sweeps with the same slack-doubling retry pattern (corr ≥ 0.97) used by the velocity clamp. This bounds the oscillation rate of the derivative without adding lag or shifting the signal level.</DESCRIPTION>

        # ---- Curvature clamp (zero-phase) ----
        # Bound second differences to suppress alternating ±limit steps that
        # pass the velocity clamp but still produce spurious slope changes.
        def _curvature_clamp(self, y, corr=0.97, c=1.5):
            y = np.asarray(y, dtype=float)
            n = len(y)
            if n < 3:
                return y
            slack = 1.0
            for _ in range(6):
                out = y.copy()
                # robust scale of curvature in the current output
                d2 = out[2:] - 2.0 * out[1:-1] + out[:-2]
                d2_scale = np.median(np.abs(d2 - np.median(d2))) * 1.4826
                if d2_scale <= 0:
                    d2_scale = np.std(d2) + 1e-12
                lim = slack * c * d2_scale
                # forward sweep
                for k in range(1, n - 1):
                    curv = out[k + 1] - 2.0 * out[k] + out[k - 1]
                    excess = abs(curv) - lim
                    if excess > 0:
                        adj = -np.sign(curv) * excess * 0.5
                        out[k] += adj
                # backward sweep (zero-phase: symmetric, no lag introduced)
                for k in range(n - 2, 0, -1):
                    curv = out[k + 1] - 2.0 * out[k] + out[k - 1]
                    excess = abs(curv) - lim
                    if excess > 0:
                        adj = -np.sign(curv) * excess * 0.5
                        out[k] += adj
                # accept if correlation with original stays high
                denom = (np.linalg.norm(y - y.mean()) * np.linalg.norm(out - out.mean()))
                if denom > 0 and np.dot(y - y.mean(), out - out.mean()) / denom >= corr:
                    return out
                slack *= 2.0
            return y

        return y
