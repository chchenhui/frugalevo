<DESCRIPTION>
Builds on the best prior program (two-pass Kalman/RTS with locally adaptive per-sample process noise + zero-phase Whittaker polish) and adds the recommended velocity-aware displacement clamp: after the second RTS refinement pass, a forward + mirrored backward clamp limits per-step displacement |y[k]−y[k−1]| ≤ 2·|v[k]| + slack·σ_r, where v[k] is the smoother's own velocity state and slack is expanded if a correlation guard (corr < 0.97 vs. unclamped) fires. This directly suppresses noise-induced slope reversals (the dominant composite levers: smoothness 0.50, false reversals 14.2) while genuine trend steps pass untouched because the velocity state is large there — the clamp is effectively zero-phase since it's applied symmetrically. The Whittaker banded polish with constant lambda replaces the cascaded one-pole damper for flatter passband / sharper ripple stopband.

# EVOLVE-BLOCK-START
"""
Real-Time Adaptive Signal Processing Algorithm for Non-Stationary Time Series

Two-pass Kalman (constant-velocity) forward filter + RTS backward smoother with
locally adaptive per-sample process noise, a velocity-aware zero-phase
displacement clamp, and a fixed-lambda Whittaker-Henderson polish.
Output contract: y[i] corresponds to input time i + window_size - 1,
len(y) = len(x) - window_size + 1.
"""
import numpy as np


def adaptive_filter(x, window_size=20):
    """Baseline entry — routes to the hybrid smoother."""
    return enhanced_filter_with_trend_preservation(x, window_size)


def _whittaker_polish(y, lam):
    """Fixed-lambda Whittaker-Henderson smoother (2nd-order penalty),
    solved via banded Cholesky when scipy is available (O(n))."""
    y = np.asarray(y, dtype=float)
    n = len(y)
    if n < 5 or lam <= 0:
        return y.copy()

    main = np.full(n, 6.0)
    main[0] = main[-1] = 1.0
    if n > 2:
        main[1] = main[-2] = 5.0
    main = main * lam + 1.0

    off1 = np.full(n - 1, -4.0)
    if n > 2:
        off1[0] = off1[-1] = -2.0
    off1 = off1 * lam

    off2 = np.full(n - 2, 1.0) * lam

    try:
        from scipy.linalg import solveh_banded
        ab = np.zeros((3, n))
        ab[2, :] = main
        ab[1, 1:] = off1
        ab[0, 2:] = off2
        return solveh_banded(ab, y, lower=False, check_finite=False)
    except Exception:
        pass

    if n <= 2500:
        A = np.eye(n)
        np.fill_diagonal(A, main)
        for i in range(n - 1):
            A[i, i + 1] += off1[i]
            A[i + 1, i] += off1[i]
        for i in range(n - 2):
            A[i, i + 2] += off2[i]
            A[i + 2, i] += off2[i]
        try:
            return np.linalg.solve(A, y)
        except np.linalg.LinAlgError:
            return y.copy()
    return y.copy()


def _velocity_clamp(y, vel, slack):
    """Zero-phase velocity-aware displacement clamp.
    Forward sweep then mirrored backward sweep; each limits per-step
    displacement to |v[k]| budget + slack. Genuine trends (large |v|)
    pass; noise ripple (small |v|) is clamped."""
    n = len(y)
    out = y.copy()
    # forward
    for k in range(1, n):
        budget = 2.0 * abs(vel[k]) + slack
        d = out[k] - out[k - 1]
        if abs(d) > budget:
            out[k] = out[k - 1] + np.sign(d) * budget
    # mirrored backward
    for k in range(n - 2, -1, -1):
        budget = 2.0 * abs(vel[k]) + slack
        d = out[k] - out[k + 1]
        if abs(d) > budget:
            out[k] = out[k + 1] + np.sign(d) * budget
    return out


def kalman_rts_filter(x, window_size=20):
    """Two-pass adaptive Kalman/RTS + velocity clamp + Whittaker polish."""
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n < window_size:
        raise ValueError(
            f"Input signal length ({n}) must be >= window_size ({window_size})"
        )
    if n == 1:
        return x.copy()

    # --- Robust measurement-noise estimate: MAD of first differences ---
    sig_range = float(np.std(x)) if n > 1 else 0.0
    scale = max(sig_range, 1e-9)
    d1 = np.diff(x)
    sigma_r = np.median(np.abs(d1 - np.median(d1))) / (0.6745 * np.sqrt(2.0))
    if not np.isfinite(sigma_r) or sigma_r <= 0.0:
        sigma_r = scale
    sigma_r = float(np.clip(sigma_r, 1e-4 * scale, scale))
    R = sigma_r * sigma_r

    # --- Adaptive process noise: local acceleration magnitude ---
    w_r = int(max(3, min(window_size, n // 3)))
    if w_r % 2 == 0:
        w_r += 1
    rough = np.convolve(x, np.ones(w_r) / w_r, mode="valid")
    if rough.size > 2:
        d2 = rough[2:] - 2.0 * rough[1:-1] + rough[:-2]
        mad2 = np.median(np.abs(d2 - np.median(d2))) / 0.6745
        noise_d2 = (sigma_r / np.sqrt(w_r)) * np.sqrt(6.0)
        var_acc = mad2 * mad2 - noise_d2 * noise_d2
    else:
        var_acc = 0.0
    floor_acc = 0.05 * sigma_r
    if var_acc < floor_acc * floor_acc:
        var_acc = floor_acc * floor_acc
    sigma_acc = float(min(np.sqrt(var_acc), 0.6 * sigma_r))
    qa = sigma_acc * sigma_acc

    # --- Constant-velocity state model ---
    F = np.array([[1.0, 1.0], [0.0, 1.0]])
    H = np.array([1.0, 0.0])
    Q = qa * np.array([[0.25, 0.5], [0.5, 1.0]])

    xp = np.zeros((n, 2))
    xf = np.zeros((n, 2))
    Pp = np.zeros((n, 2, 2))
    Pf = np.zeros((n, 2, 2))
    I2 = np.eye(2)

    x_est = np.array([x[0], 0.0])
    P = np.eye(2) * max(R, 1e-15)

    # --- Forward Kalman pass #1 ---
    for k in range(n):
        if k == 0:
            xp[k] = x_est
            Pp[k] = P
        else:
            xp[k] = F @ x_est
            Pp[k] = F @ P @ F.T + Q
        S = Pp[k][0, 0] + R
        K = Pp[k][:, 0] / S
        x_est = xp[k] + K * (x[k] - xp[k, 0])
        P = (I2 - np.outer(K, H)) @ Pp[k]
        P = 0.5 * (P + P.T)
        xf[k] = x_est
        Pf[k] = P

    # --- RTS backward pass #1 ---
    xs = xf.copy()
    for k in range(n - 2, -1, -1):
        Ppred = Pp[k + 1]
        try:
            Ppred_inv = np.linalg.inv(Ppred)
        except np.linalg.LinAlgError:
            Ppred_inv = np.linalg.pinv(Ppred)
        G = Pf[k] @ F.T @ Ppred_inv
        xs[k] = xf[k] + G @ (xs[k + 1] - xp[k + 1])

    y = xs[:, 0].copy()

    # --- Locally adaptive refinement pass #2 (per-sample Q) ---
    if n >= 2:
        blk = int(max(5, min(window_size, n)))
        step = max(1, blk // 2)
        centers = []
        sigmas = []
        start = 0
        noise_d2_floor = (sigma_r / np.sqrt(blk)) * np.sqrt(6.0)
        while start < n:
            stop = min(start + blk, n)
            seg = y[start:stop]
            if seg.size > 2:
                dd2 = seg[2:] - 2.0 * seg[1:-1] + seg[:-2]
                mad2 = np.median(np.abs(dd2 - np.median(dd2))) / 0.6745
                var = mad2 * mad2 - noise_d2_floor * noise_d2_floor
                if var < (0.05 * sigma_r) ** 2:
                    var = (0.05 * sigma_r) ** 2
                s = float(np.sqrt(var))
                s = min(s, 1.5 * sigma_r)
            else:
                s = 0.05 * sigma_r
            centers.append(0.5 * (start + stop - 1))
            sigmas.append(s)
            if stop >= n:
                break
            start += step
        centers = np.asarray(centers, dtype=float)
        sigmas = np.asarray(sigmas, dtype=float)
        qa_loc = np.interp(np.arange(n, dtype=float), centers, sigmas)
        q2k = 0.25 * qa_loc * qa_loc

        x_est = np.array([y[0], 0.0])
        P = np.eye(2) * max(R, 1e-15)
        Qbase = np.array([[0.25, 0.5], [0.5, 1.0]])
        for k in range(n):
            if k == 0:
                xp[k] = x_est
                Pp[k] = P
            else:
                xp[k] = F @ x_est
                Pp[k] = F @ P @ F.T + q2k[k] * Qbase
            S = Pp[k][0, 0] + R
            K = Pp[k][:, 0] / S
            x_est = xp[k] + K * (y[k] - xp[k, 0])
            P = (I2 - np.outer(K, H)) @ Pp[k]
            P = 0.5 * (P + P.T)
            xf[k] = x_est
            Pf[k] = P
        xs = xf.copy()
        for k in range(n - 2, -1, -1):
            Ppred = Pp[k + 1]
            try:
                Ppred_inv = np.linalg.inv(Ppred)
            except np.linalg.LinAlgError:
                Ppred_inv = np.linalg.pinv(Ppred)
            G = Pf[k] @ F.T @ Ppred_inv
            xs[k] = xf[k] + G @ (xs[k + 1] - xp[k + 1])
        y = xs[:, 0].copy()
        vel = xs[:, 1].copy()
    else:
        vel = np.zeros(n)

    # --- Velocity-aware displacement clamp (zero-phase) ---
    # Limit |y[k]-y[k-1]| <= 2|v[k]| + slack*sigma_r, with a correlation
    # retry that doubles slack if the clamp distorted the signal too much.
    if n >= 3:
        slack = 0.5 * sigma_r
        y0 = y.copy()
        yc = _velocity_clamp(y, vel, slack)
        c = np.corrcoef(yc, y0)[0, 1]
        if not np.isfinite(c) or c < 0.97:
            yc = _velocity_clamp(y0, vel, 1.0 * sigma_r)
            c2 = np.corrcoef(yc, y0)[0, 1]
            if np.isfinite(c2) and c2 >= 0.97:
                y = yc
            else:
                y = y0  # keep unclamped if even loose clamp distorts
        else:
            y = yc

    # --- Whittaker-Henderson polish (zero-phase, constant lambda) ---
    dy = np.diff(y)
    if dy.size > 2:
        r_std = np.median(np.abs(dy - np.median(dy))) / (0.6745 * np.sqrt(2.0))
    else:
        r_std = 0.0
    lam = float(np.clip(25.0 * (r_std / max(sigma_r, 1e-12)) ** 2, 1.0, 200.0))
    y = _whittaker_polish(y, lam)

    # --- Slice to output contract ---
    y = y[window_size - 1:].copy()
    expected = n - window_size + 1
    if y.shape[0] != expected:
        y = y[:expected]
    return y


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """Enhanced entry — adaptive RTS + velocity clamp + Whittaker polish."""
    return kalman_rts_filter(x, window_size)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Main signal processing function that applies the selected algorithm.

    Args:
        input_signal: Input time series data
        window_size: Window size for processing
        algorithm_type: Type of algorithm to use ("basic", "enhanced", or "rts")

    Returns:
        Filtered signal
    """
    return kalman_rts_filter(input_signal, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Args:
        length: Length of the signal
        noise_level: Standard deviation of noise to add
        seed: Random seed for reproducibility

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000, noise_level=0.3, window_size=20):
    """
    Run the signal processing algorithm on a test signal.
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = np.corrcoef(filtered_signal, aligned_clean)[0, 1] if min_length > 1 else 0

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (noise_before - noise_after) / noise_before if noise_before > 0 else 0

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }
    elif len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }
    else:
        return {
            "filtered_signal": [],
            "clean_signal": [],
            "noisy_signal": [],
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": 0,
        }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
