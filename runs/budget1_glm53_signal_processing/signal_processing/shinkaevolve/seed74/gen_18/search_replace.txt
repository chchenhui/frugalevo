<DESCRIPTION>
The current code fuses an independent forward and backward Kalman filter with heuristic velocity-confidence weights. Replace this with a proper Rauch-Tung-Striebel (RTS) smoother: store the filtered states and covariances from the forward pass, then run the true backward recursion using the smoother gain G = P_f F' P_pred^{-1}, where P_pred is the *predicted* (prior) covariance at each step. The RTS smoother is the minimum-variance fixed-interval estimate for the linear-Gaussian model, so it should reduce noise and false reversals further while keeping zero phase, at the same O(n) cost. Also store predicted covariances during the forward pass so the backward recursion is exact. The adaptive measurement-noise heuristic is kept, which makes this an adaptive RTS smoother. Output slicing and length contract remain unchanged.

def _kalman_fused(x):
    """
    Adaptive constant-velocity Kalman filter run forward and backward,
    fused with velocity-confidence weights. Returns the zero-phase
    smoothed level estimate.
    """
    n = len(x)
    dt = 1.0
    q = 0.02
    qv = 0.005

    if n >= 2:
        diff = np.diff(x)
        r0 = max(np.var(diff) / 2.0, 1e-8)
    else:
        r0 = 1.0

    F = np.array([[1.0, dt], [0.0, 1.0]])
    Q = np.array([[q, 0.0], [0.0, qv]])
    H = np.array([[1.0, 0.0]])
    I2 = np.eye(2)

    # Forward pass
    levels_f = np.zeros(n)
    vels_f = np.zeros(n)
    xf_lvl = x[0]
    xf_vel = 0.0
    P = np.array([[r0, 0.0], [0.0, r0]])
    for k in range(n):
        if k > 0:
            xf_lvl = xf_lvl + xf_vel * dt
            P = F @ P @ F.T + Q
        innov = x[k] - xf_lvl
        r = max(r0 * 0.5 + max(innov * innov, 0.0) * 0.5, 1e-8)
        S = P[0, 0] + r
        K = np.array([P[0, 0] / S, P[1, 0] / S])
        xf_lvl = xf_lvl + K[0] * innov
        xf_vel = xf_vel + K[1] * innov
        P = (I2 - np.outer(K, H)) @ P
        P = 0.5 * (P + P.T)
        levels_f[k] = xf_lvl
        vels_f[k] = xf_vel

    # Backward pass
    levels_b = np.zeros(n)
    vels_b = np.zeros(n)
    xb_lvl = x[-1]
    xb_vel = 0.0
    Pb = np.array([[r0, 0.0], [0.0, r0]])
    for k in range(n - 1, -1, -1):
        if k < n - 1:
            xb_lvl = xb_lvl + xb_vel * dt
            Pb = F @ Pb @ F.T + Q
        innov = x[k] - xb_lvl
        r = max(r0 * 0.5 + max(innov * innov, 0.0) * 0.5, 1e-8)
        S = Pb[0, 0] + r
        K = np.array([Pb[0, 0] / S, Pb[1, 0] / S])
        xb_lvl = xb_lvl + K[0] * innov
        xb_vel = xb_vel + K[1] * innov
        Pb = (I2 - np.outer(K, H)) @ Pb
        Pb = 0.5 * (Pb + Pb.T)
        levels_b[k] = xb_lvl
        vels_b[k] = xb_vel

    # Variance-weighted fusion of forward / backward levels
    w_f = 1.0 / (1.0 + np.abs(vels_f))
    w_b = 1.0 / (1.0 + np.abs(vels_b))
    smooth = (w_f * levels_f + w_b * levels_b) / (w_f + w_b)
    return smooth

def _kalman_fused(x):
    """
    Adaptive constant-velocity Kalman filter followed by a true
    Rauch-Tung-Striebel (RTS) backward smoothing pass. Returns the
    zero-phase, minimum-variance smoothed level estimate.

    The RTS recursion uses the stored filtered states/covariances and
    the predicted (prior) covariances from the forward pass:
        G_k = P_f[k] F' P_pred[k+1]^{-1}
        x_s[k] = x_f[k] + G_k (x_s[k+1] - x_pred[k+1])
    which is the exact fixed-interval optimal smoother for this model.
    """
    n = len(x)
    dt = 1.0
    q = 0.02
    qv = 0.005

    if n >= 2:
        diff = np.diff(x)
        r0 = max(np.var(diff) / 2.0, 1e-8)
    else:
        r0 = 1.0

    F = np.array([[1.0, dt], [0.0, 1.0]])
    Q = np.array([[q, 0.0], [0.0, qv]])
    H = np.array([[1.0, 0.0]])
    I2 = np.eye(2)

    # Forward pass: store filtered states, filtered covariances, and
    # predicted (prior) states/covariances for the RTS backward pass.
    levels_f = np.zeros(n)
    vels_f = np.zeros(n)
    P_f = np.zeros((n, 2, 2))      # filtered covariance per step
    x_pred = np.zeros((n, 2))      # predicted state per step
    P_pred = np.zeros((n, 2, 2))   # predicted covariance per step

    xf_lvl = x[0]
    xf_vel = 0.0
    P = np.array([[r0, 0.0], [0.0, r0]])
    for k in range(n):
        if k > 0:
            xf_lvl = xf_lvl + xf_vel * dt
            P = F @ P @ F.T + Q
        x_pred[k] = [xf_lvl, xf_vel]
        P_pred[k] = P
        innov = x[k] - xf_lvl
        r = max(r0 * 0.5 + max(innov * innov, 0.0) * 0.5, 1e-8)
        S = P[0, 0] + r
        K = np.array([P[0, 0] / S, P[1, 0] / S])
        xf_lvl = xf_lvl + K[0] * innov
        xf_vel = xf_vel + K[1] * innov
        P = (I2 - np.outer(K, H)) @ P
        P = 0.5 * (P + P.T)
        levels_f[k] = xf_lvl
        vels_f[k] = xf_vel
        P_f[k] = P

    # Backward RTS recursion (exact minimum-variance smoother)
    xs_lvl = levels_f[-1]
    xs_vel = vels_f[-1]
    for k in range(n - 2, -1, -1):
        Pp = P_pred[k + 1]
        # Regularize in case of numerical singularity
        det = Pp[0, 0] * Pp[1, 1] - Pp[0, 1] * Pp[1, 0]
        if abs(det) < 1e-12:
            Pp = Pp + 1e-9 * I2
            det = Pp[0, 0] * Pp[1, 1] - Pp[0, 1] * Pp[1, 0]
        Pp_inv = np.array([[Pp[1, 1], -Pp[0, 1]],
                           [-Pp[1, 0], Pp[0, 0]]]) / det
        G = P_f[k] @ F.T @ Pp_inv  # 2x2 smoother gain
        diff_s = np.array([xs_lvl, xs_vel]) - x_pred[k + 1]
        corr = G @ diff_s
        xs_lvl = levels_f[k] + corr[0]
        xs_vel = vels_f[k] + corr[1]
        levels_f[k] = xs_lvl
        vels_f[k] = xs_vel

    return levels_f
