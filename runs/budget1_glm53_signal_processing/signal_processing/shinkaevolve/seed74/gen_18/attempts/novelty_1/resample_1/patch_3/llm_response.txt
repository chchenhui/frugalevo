def rts_smoother_filter(input_signal, window_size):
    """
    Sliding-window Kalman filter + true Rauch-Tung-Striebel (RTS) smoother.

    Forward pass: constant-velocity Kalman filter over each window,
    storing filtered states/covariances and the predicted (prior)
    covariances. Backward pass: exact RTS recursion
        G_k      = P_f[k] F' P_pred[k+1]^{-1}
        x_s[k]   = x_f[k] + G_k (x_s[k+1] - F x_f[k])
    which is the minimum-variance fixed-interval estimate (no
    information double-counting as with an independent backward
    filter blend). Output length is explicitly validated as
    n - window_size + 1.
    """
    x = np.asarray(input_signal, dtype=float).ravel()
    n = x.size
    dim = 2  # [level, slope]

    dt = 1.0
    F = np.array([[1.0, dt], [0.0, 1.0]])
    H = np.array([[1.0, 0.0]])
    # Noise levels adapted to signal scale to remain robust across scales.
    sigma = float(np.std(x)) + 1e-12
    R = np.array([[(sigma * 0.10) ** 2]])
    q_a = (sigma * 0.05) ** 2
    Q = q_a * np.array([[dt ** 4 / 4.0, dt ** 3 / 2.0],
                         [dt ** 3 / 2.0, dt ** 2]])
    P0 = np.eye(dim) * (sigma ** 2)
    x0 = np.array([x[0], 0.0])

    out_len = n - window_size + 1
    out = np.zeros(out_len)
    P_f = np.zeros((window_size, dim, dim))
    P_pred = np.zeros((window_size, dim, dim))
    x_f = np.zeros((window_size, dim))
    x_smooth = np.zeros((window_size, dim))

    for start in range(out_len):
        y = x[start:start + window_size]
        # --- Forward Kalman pass (store filtered & predicted moments) ---
        x_f[0] = x0
        P_f[0] = P0
        for k in range(1, window_size):
            x_pred = F @ x_f[k - 1]
            P_pred[k] = F @ P_f[k - 1] @ F.T + Q
            S = H @ P_pred[k] @ H.T + R
            K = P_pred[k] @ H.T @ np.linalg.inv(S)
            x_f[k] = x_pred + (K @ (y[k] - H @ x_pred)).ravel()
            P_f[k] = (np.eye(dim) - K @ H) @ P_pred[k] @ (np.eye(dim) - K @ H).T + K @ R @ K.T

        # --- True RTS backward recursion (minimum-variance smoother) ---
        x_smooth[window_size - 1] = x_f[window_size - 1]
        for k in range(window_size - 2, -1, -1):
            P_pred_next = P_pred[k + 1]
            G = P_f[k] @ F.T @ np.linalg.inv(P_pred_next)
            x_smooth[k] = x_f[k] + G @ (x_smooth[k + 1] - F @ x_f[k])

        # Emit the smoothed estimate at the window's leading edge
        # (minimal latency, zero phase distortion from smoothing).
        out[start] = x_smooth[window_size - 1][0]

    if out.size != out_len:
        raise ValueError(
            "rts_smoother_filter: expected output length %d, got %d"
            % (out_len, out.size)
        )
    return out
