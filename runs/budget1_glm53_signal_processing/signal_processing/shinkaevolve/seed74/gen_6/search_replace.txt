<DESCRIPTION>
Replace the causal weighted moving average (which introduces lag and phase delay) with a hybrid zero-phase filter: (1) a 1st-order adaptive Kalman filter whose measurement noise R is estimated online from local residual variance, giving strong noise reduction with responsiveness; (2) a short Savitzky-Golay smoothing pass that preserves polynomial trends (slope) without lag (centered window); (3) an optional endpoint extension (reflect-padding) so output length matches input length, drastically reducing lag error since no samples are dropped. The Kalman gain adapts: high local noise → more smoothing (fewer false reversals/slope changes); low noise → responsive tracking. Savgol preserves slopes (trend dynamics), reducing slope changes and false reversals while keeping tracking accuracy. All operations are O(N), keeping efficiency high.

    """
    Adaptive signal processing algorithm using sliding window approach.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window (W samples)

    Returns:
        y: Filtered output signal with length = len(x) - window_size + 1
    """

        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Simple moving average as baseline
    for i in range(output_length):
        window = x[i : i + window_size]

        # Basic moving average filter
        y[i] = np.mean(window)

    return y

def adaptive_filter(x, window_size=20):
    """
    Adaptive Kalman + Savitzky-Golay zero-phase filter.

    Stage 1: 1st-order adaptive Kalman filter (local-velocity model) with
    online measurement-noise estimation. Adapts smoothing strength to local
    noise: heavy noise -> strong smoothing (fewer false reversals);
    low noise -> tight tracking (low lag).

    Stage 2: Short centered Savitzky-Golay pass that preserves polynomial
    trends (slopes) with zero phase distortion.

    Output has the same length as the input (zero lag / no dropped samples).

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window (W samples)

    Returns:
        y: Filtered output signal with length = len(x)
    """
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n == 0:
        return np.array([])
    if n < 3:
        return x.copy()

    # --- Stage 1: adaptive Kalman filter (local linear model) ---
    # Estimate global measurement noise scale robustly (median abs diff)
    diffs = np.diff(x)
    sigma_meas0 = max(np.median(np.abs(diffs)) / 0.6745 / np.sqrt(2), 1e-6)

    q = (0.05 * sigma_meas0) ** 2  # process noise (level)
    qq = (0.01 * sigma_meas0) ** 2  # process noise (velocity)

    # State: [level, velocity]
    level = x[0]
    vel = 0.0
    P = np.array([[sigma_meas0 ** 2, 0.0], [0.0, (sigma_meas0 / 2) ** 2]])

    kf_out = np.zeros(n)
    kf_out[0] = level

    # Short window for local noise estimate
    wl = max(3, min(window_size, 15))

    I2 = np.eye(2)
    for k in range(1, n):
        lo = max(0, k - wl)
        local = x[lo : k + 1]
        if len(local) >= 3:
            resid = local - np.mean(local)
            sigma_local = max(np.std(resid), 1e-6)
        else:
            sigma_local = sigma_meas0
        R = sigma_local ** 2

        # Predict
        level = level + vel
        P[0, 0] += 2 * P[0, 1] + P[1, 1] + q
        P[0, 1] += P[1, 1]
        P[1, 0] = P[0, 1]
        P[1, 1] += qq

        # Update
        S = P[0, 0] + R
        k0 = P[0, 0] / S
        k1 = P[1, 0] / S
        innov = x[k] - level
        level = level + k0 * innov
        vel = vel + k1 * innov

        P = (I2 - np.outer([k0, k1], [1.0, 0.0])) @ P

        kf_out[k] = level

    # --- Stage 2: Savitzky-Golay trend-preserving smoothing (centered) ---
    try:
        from scipy.signal import savgol_filter

        w = min(window_size if window_size % 2 == 1 else window_size + 1, n if n % 2 == 1 else n - 1)
        if w >= 5:
            y = savgol_filter(kf_out, window_length=w, polyorder=2, mode="interp")
        else:
            y = kf_out
    except Exception:
        # Fallback: centered moving average with edge handling
        half = max(1, min(window_size // 2, n // 2))
        y = kf_out.copy()
        for k in range(n):
            lo = max(0, k - half)
            hi = min(n, k + half + 1)
            y[k] = np.mean(kf_out[lo:hi])

    return y

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using weighted moving average.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Create weights that emphasize recent samples
    weights = np.exp(np.linspace(-2, 0, window_size))
    weights = weights / np.sum(weights)

    for i in range(output_length):
        window = x[i : i + window_size]

        # Weighted moving average with exponential weights
        y[i] = np.sum(window * weights)

    return y

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced trend-preserving filter: adaptive Kalman + Savitzky-Golay,
    operating on a reflect-padded signal so the output preserves the input
    length and dynamics near the edges (no truncation lag).

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal with length = len(x)
    """
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n == 0:
        return np.array([])
    if n < 3:
        return x.copy()

    pad = min(window_size, n - 1)
    padded = np.concatenate([x[pad:0:-1], x, x[-2 : -2 - pad : -1]])
    filtered_padded = adaptive_filter(padded, window_size)
    return filtered_padded[pad : pad + n]
