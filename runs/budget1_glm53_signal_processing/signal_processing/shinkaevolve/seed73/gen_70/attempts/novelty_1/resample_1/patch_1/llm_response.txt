    @staticmethod
    def _median_snap(x):
        """5-tap median snap: blend sub-noise samples toward the local median."""
        m = len(x)
        if m < 5:
            return x
        # edge-padded 5-tap running median, fully vectorized
        pad = np.concatenate((x[:2][::-1], x, x[-2:][::-1]))
        windows = np.lib.stride_tricks.sliding_window_view(pad, 5)
        med5 = np.median(windows, axis=1)

        # robust local noise scale from diffs of x
        d = np.diff(x)
        med_d = np.median(d)
        sigma = np.median(np.abs(d - med_d)) / 0.6745 / np.sqrt(2.0)
        sigma = max(sigma, 1e-12)

        # step sizes relative to the median reference (validated threshold)
        steps = np.zeros(m)
        steps[1:] = np.abs(np.diff(x))
        steps[:-1] = np.maximum(steps[:-1], steps[1:])
        small = steps < 0.5 * sigma
        return np.where(small, 0.5 * x + 0.5 * med5, x)
