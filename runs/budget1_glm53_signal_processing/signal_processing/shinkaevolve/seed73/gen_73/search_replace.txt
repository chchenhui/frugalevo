class ZeroPhasePolishStage:
    """
    Stage 6: small-window (7-tap) quadratic Savitzky-Golay polish applied to
    the Stage-5 output. SG is a symmetric zero-phase FIR smoother, so this
    adds no group delay (lag/responsiveness untouched) while suppressing the
    residual high-frequency noise that survives the hysteresis snap. The
    short window preserves genuine trend dynamics; clamped odd-length guard
    mirrors the validated sg_win defensive discipline.
    """

    def __init__(self, win=7):
        self.win = win

    def run(self, y):
        m = len(y)
        win = self.win
        if m < 5:
            return y
        # defensive clamp: odd window, >= 5, <= m (odd-bounded)
        if win > m:
            win = m if m % 2 == 1 else m - 1
        win = max(win, 5)
        if win % 2 == 0:
            win -= 1
            win = max(win, 5)
        try:
            from scipy.signal import savgol_filter
            out = savgol_filter(y, win, 2)
            if not np.all(np.isfinite(out)):
                return y
            return out
        except Exception:
            # fallback: 5-tap binomial [1,4,6,4,1]/16 zero-phase smoothing
            k = np.array([1.0, 4.0, 6.0, 4.0, 1.0]) / 16.0
            pad = np.concatenate((y[:2][::-1], y, y[-2:][::-1]))
            out = np.convolve(pad, k, mode="valid")
            return out
