def _bidirectional_kalman(x):
    """
    Forward adaptive-Q Kalman recursion fused 50/50 with a backward pass
    (the same filter applied to the time-reversed input).

    The forward pass is causal: it lags behind genuine turning points later
    in the signal. The backward pass sees those same turning points with
    "future" context, so its phase error is opposite in sign. Averaging the
    two estimates cancels much of the interior phase error, improving
    tracking accuracy and lag without any added smoothness cost.
    """
    x = np.asarray(x, dtype=float)
    if len(x) < 4:
        return _adaptive_kalman_filter(x)
    fwd = _adaptive_kalman_filter(x)
    bwd = _adaptive_kalman_filter(x[::-1])[::-1]
    return 0.5 * fwd + 0.5 * bwd
