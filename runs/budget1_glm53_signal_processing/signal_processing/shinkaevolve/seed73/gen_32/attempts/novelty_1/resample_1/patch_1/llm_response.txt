        slope_pred = slope

        # Innovation and adaptive R (EWMA of squared innovations)

        R = float(np.clip(0.98 * R + 0.02 * max(innov * innov, R_min), R_min, R_max))

    # ---- Recency-weighted Savitzky-Golay filter (single pass) ----
    # SG with poly_order=2 over a wide window gives strong noise suppression
    # while preserving trend curvature (no lag for polynomial trends).
    # Exponential recency weighting tilts the fit toward the newest samples,
    # compensating the SG group delay without unstable slope extrapolation.
    poly_order = 2
    sg_window = 2 * window_size - 1  # keep odd
    if sg_window <= poly_order:
        sg_window = poly_order + 1 + (poly_order % 2)

    half = sg_window // 2

    # Exponential recency weights (steep: e^{-1.5} at oldest sample)
    w = np.exp(np.linspace(-1.5, 0.0, sg_window))
    w = w / np.sum(w)

    # Build weighted design matrix: fit polynomial on normalized offsets
    offsets = np.linspace(-1.0, 1.0, sg_window)
    A = np.vander(offsets, poly_order + 1, increasing=True)  # (sg_window, p+1)
    W = np.diag(w)
    # Weighted least-squares projection onto the fitted value at the
    # rightmost sample (offset = +1 => last column position)
    coeffs_at_end = np.linalg.lstsq(W @ A, w, rcond=None)[0]
    # Filter kernel: applying to a window yields fitted value at the newest point
    kernel = coeffs_at_end[::-1]  # reverse for np.convolve correlation convention

    # Pad edges by reflection to avoid boundary artifacts
    x_pad = np.concatenate([x[half:0:-1], x, x[-2 : -2 - half : -1]])
    y_full = np.convolve(x_pad, kernel, mode="valid")[:n]

    # ---- Extract window-aligned output ----
    y = y_full[window_size - 1 :]

    return y
