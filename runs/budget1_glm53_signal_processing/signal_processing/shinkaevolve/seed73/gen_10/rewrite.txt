"""
Real-Time Adaptive Kalman Trend Filter for Non-Stationary Time Series

Causal adaptive Kalman filter with local linear trend model (position + slope).
Adaptive measurement/process noise estimation enables near-zero phase lag while
suppressing noise-induced slope reversals.
"""
import numpy as np


def adaptive_kalman_filter(x, window_size=20):
    """
    Adaptive Kalman filter with local linear trend state model.

    State: [level, slope]. Measurement: level + noise.
    Measurement noise R adapted from innovation statistics.
    Process noise Q scaled adaptively when a genuine trend change is detected
    (large normalized innovation), keeping responsiveness to real dynamics
    while rejecting noise-driven reversals.

    Args:
        x: Input signal (1D array)
        window_size: retained for API compatibility (also used as warmup span)

    Returns:
        y: Filtered output, length = len(x) - window_size + 1 (aligned like the
           original sliding-window filter, but with near-zero lag since the
           Kalman estimate is causal at the last sample of each window).
    """
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n < window_size:
        raise ValueError(
            f"Input signal length ({n}) must be >= window_size ({window_size})"
        )

    # ---- State-space model: constant-velocity (local linear trend) ----
    dt = 1.0
    F = np.array([[1.0, dt], [0.0, 1.0]])   # state transition
    H = np.array([[1.0, 0.0]])              # observe level only

    # ---- Initialize from a short warmup regression (reduces startup bias) ----
    w = min(window_size, n)
    tt = np.arange(w, dtype=float)
    if w >= 2:
        slope0, level0 = np.polyfit(tt, x[:w], 1)[0:2]
        level0 = level0 + slope0 * (w - 1)  # level at end of warmup
    else:
        level0, slope0 = x[0], 0.0
    x_state = np.array([level0, slope0])

    # Rough noise estimate from warmup residuals (robust MAD-based)
    resid = x[:w] - np.polyval(np.polyfit(tt, x[:w], 1), tt)
    sigma0 = max(np.median(np.abs(resid - np.median(resid))) * 1.4826, 1e-6)

    # Covariance initialization
    P = np.diag([sigma0 ** 2, (sigma0 * 0.5) ** 2])

    # Process noise (small baseline: level and slope mostly stable per-step)
    q_level = (0.05 * sigma0) ** 2
    q_slope = (0.01 * sigma0) ** 2
    Q_base = np.diag([q_level, q_slope])

    R_base = sigma0 ** 2

    levels = np.empty(n)
    levels[:] = np.nan

    # Running noise-level estimate (EWMA of squared innovations)
    innov_var = R_base
    alpha_r = 0.05

    I2 = np.eye(2)

    for k in range(n):
        # -------- Predict --------
        x_state = F @ x_state
        P = F @ P @ F.T + Q_base

        # -------- Innovation --------
        innov = x[k] - (H @ x_state)[0]
        S = (H @ P @ H.T)[0, 0] + innov_var  # innovation variance
        nsq = innov * innov / max(S, 1e-12)  # normalized squared innovation

        # -------- Adaptive process noise --------
        # If innovation is much larger than expected, signal may have a genuine
        # trend change -> temporarily boost Q to let the filter track quickly.
        if nsq > 4.0:
            boost = min(nsq / 4.0, 50.0)
            Q_extra = np.diag([q_level * boost, q_slope * boost * 5.0])
            P = P + Q_extra

        # -------- Update (Kalman gain) --------
        recompute = nsq > 4.0
        if recompute:
            S = (H @ P @ H.T)[0, 0] + innov_var
        K = (P @ H.T) / max(S, 1e-12)
        x_state = x_state + (K[:, 0] * innov)
        P = (I2 - K @ H) @ P @ I2

        # -------- Adaptive measurement noise --------
        # EWMA of squared innovations tracks local noise power.
        innov_var = (1.0 - alpha_r) * innov_var + alpha_r * min(innov * innov, 25 * R_base)
        innov_var = float(np.clip(innov_var, 0.05 * R_base, 20 * R_base))

        levels[k] = x_state[0]

    # -------- Output alignment (same convention as sliding window) --------
    # Original filter outputs y[i] from window x[i : i+W]. To compare fairly we
    # take the causal Kalman estimate at the END of each window (index i+W-1),
    # which is the best available estimate with essentially zero lag.
    y = levels[window_size - 1:]
    return y


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """Alias kept for API compatibility."""
    return adaptive_kalman_filter(x, window_size)


def process_signal(input_signal, window_size=20, algorithm_type="enhanced"):
    """
    Main signal processing function that applies the selected algorithm.

    Args:
        input_signal: Input time series data
        window_size: Window size for processing
        algorithm_type: Type of algorithm ("basic" or "enhanced")

    Returns:
        Filtered signal
    """
    x = np.asarray(input_signal, dtype=float)
    if len(x) == 0:
        return np.array([])
    if algorithm_type == "enhanced":
        return adaptive_kalman_filter(x, window_size)
    else:
        return adaptive_kalman_filter(x, window_size)

def generate_test_signal(length=1000, noise_level=0.3, seed=42):
    """
    Generate synthetic test signal with known characteristics.

    Returns:
        Tuple of (noisy_signal, clean_signal)
    """
    np.random.seed(seed)
    t = np.linspace(0, 10, length)

    clean_signal = (
        2 * np.sin(2 * np.pi * 0.5 * t)
        + 1.5 * np.sin(2 * np.pi * 2 * t)
        + 0.5 * np.sin(2 * np.pi * 5 * t)
        + 0.8 * np.exp(-t / 5) * np.sin(2 * np.pi * 1.5 * t)
    )

    trend = 0.1 * t * np.sin(0.2 * t)
    clean_signal += trend

    random_walk = np.cumsum(np.random.randn(length) * 0.05)
    clean_signal += random_walk

    noise = np.random.normal(0, noise_level, length)
    noisy_signal = clean_signal + noise

    return noisy_signal, clean_signal


def run_signal_processing(noisy_signal=None, signal_length=1000, noise_level=0.3, window_size=20):
    """
    Run the signal processing algorithm on a test signal.

    Returns:
        Dictionary containing results and metrics
    """
    if noisy_signal is not None:
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")
        clean_signal = None
    else:
        noisy_signal, clean_signal = generate_test_signal(signal_length, noise_level)
        filtered_signal = process_signal(noisy_signal, window_size, "enhanced")

    if len(filtered_signal) > 0 and clean_signal is not None:
        delay = window_size - 1
        aligned_clean = clean_signal[delay:]
        aligned_noisy = noisy_signal[delay:]

        min_length = min(len(filtered_signal), len(aligned_clean))
        filtered_signal = filtered_signal[:min_length]
        aligned_clean = aligned_clean[:min_length]
        aligned_noisy = aligned_noisy[:min_length]

        correlation = (
            np.corrcoef(filtered_signal, aligned_clean)[0, 1] if min_length > 1 else 0
        )

        noise_before = np.var(aligned_noisy - aligned_clean)
        noise_after = np.var(filtered_signal - aligned_clean)
        noise_reduction = (
            (noise_before - noise_after) / noise_before if noise_before > 0 else 0
        )

        return {
            "filtered_signal": filtered_signal,
            "clean_signal": aligned_clean,
            "noisy_signal": aligned_noisy,
            "correlation": correlation,
            "noise_reduction": noise_reduction,
            "signal_length": min_length,
        }
    elif len(filtered_signal) > 0:
        return {
            "filtered_signal": filtered_signal,
            "clean_signal": None,
            "noisy_signal": None,
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": len(filtered_signal),
        }
    else:
        return {
            "filtered_signal": [],
            "clean_signal": [],
            "noisy_signal": [],
            "correlation": 0,
            "noise_reduction": 0,
            "signal_length": 0,
        }


if __name__ == "__main__":
    results = run_signal_processing()
    print("Signal processing completed!")
    print(f"Correlation with clean signal: {results['correlation']:.3f}")
    print(f"Noise reduction: {results['noise_reduction']:.3f}")
    print(f"Processed signal length: {results['signal_length']}")
