<DESCRIPTION>
Replace the exponentially-weighted moving average (which introduces significant lag, lag_error=0.38) with a zero-lag dual-pass filter: forward EMA followed by a backward EMA cancels phase delay, and a Savitzky-Golay-style local polynomial fit evaluated at the window center preserves slope dynamics and reduces spurious slope changes/false reversals. The polynomial fit preserves genuine trends (high tracking accuracy) while the EMA passes suppress broadband noise, improving smoothness and noise reduction simultaneously with minimal added computation (O(W) per sample with precomputed SG coefficients via least-squares). The combination (zero-lag EMA cascade + centered polynomial fit, blended 50/50) attacks all four objectives: slope-change minimization (smooth polynomial fit), lag minimization (backward pass cancels phase), tracking accuracy (centered fit follows local trend), false-reversal penalty (noise is heavily attenuated before derivative estimation).

def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced version with trend preservation using weighted moving average.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    output_length = len(x) - window_size + 1
    y = np.zeros(output_length)

    # Create weights that emphasize recent samples
    weights = np.exp(np.linspace(-2, 0, window_size))
    weights = weights / np.sum(weights)

    for i in range(output_length):
        window = x[i : i + window_size]

        # Weighted moving average with exponential weights
        y[i] = np.sum(window * weights)

    return y

def _dual_pass_ema(x, alpha):
    """
    Zero-lag dual-pass exponential smoothing: forward EMA followed by
    backward EMA. The two passes cancel the phase lag introduced by a
    single causal EMA, keeping responsiveness high while suppressing noise.

    Args:
        x: Input signal
        alpha: EMA smoothing factor (0 < alpha <= 1)

    Returns:
        Zero-lag smoothed signal (same length as x)
    """
    n = len(x)
    # Forward pass
    fwd = np.empty(n)
    fwd[0] = x[0]
    for i in range(1, n):
        fwd[i] = alpha * x[i] + (1 - alpha) * fwd[i - 1]
    # Backward pass (cancels group delay of forward pass)
    bwd = np.empty(n)
    bwd[-1] = fwd[-1]
    for i in range(n - 2, -1, -1):
        bwd[i] = alpha * fwd[i] + (1 - alpha) * bwd[i + 1]
    return bwd


def enhanced_filter_with_trend_preservation(x, window_size=20):
    """
    Enhanced filter with trend preservation using a zero-lag dual-pass EMA
    cascade blended with a centered Savitzky-Golay polynomial fit.

    The dual-pass EMA provides low-lag noise suppression (phase delay of
    forward pass is cancelled by backward pass), while the Savitzky-Golay
    fit evaluated at the window center preserves local polynomial trends
    (slope) without attenuation, minimizing slope changes and false
    reversals.

    Args:
        x: Input signal (1D array of real-valued samples)
        window_size: Size of the sliding window

    Returns:
        y: Filtered output signal
    """
    if len(x) < window_size:
        raise ValueError(f"Input signal length ({len(x)}) must be >= window_size ({window_size})")

    x = np.asarray(x, dtype=float)
    n = len(x)

    # --- Stage 1: zero-lag dual-pass EMA cascade ---
    # Choose alpha so effective smoothing matches the window size
    alpha1 = 2.0 / (window_size + 1.0)
    s = _dual_pass_ema(x, alpha1)
    s = _dual_pass_ema(s, alpha1)  # second cascade: stronger noise rejection

    # --- Stage 2: centered Savitzky-Golay polynomial fit ---
    # Polynomial order (must be < window_size; 3 preserves cubic trends)
    poly_order = min(3, window_size - 1)
    half = window_size // 2
    # Precompute SG weights for evaluating the fit at the window center
    t = np.arange(window_size, dtype=float) - (window_size - 1) / 2.0
    V = np.vander(t, poly_order + 1, increasing=True)
    VtV_inv_Vt = np.linalg.pinv(V.T @ V) @ V.T
    center_idx = window_size // 2  # index of window center in local coords
    sg_weights = VtV_inv_Vt[:, center_idx]

    y = np.empty(n)
    # Edges: replicate nearest valid smoothed value (no extra delay)
    y[:half] = s[:half]
    y[n - half:] = s[n - half:]
    # Interior: apply SG filter to the EMA-smoothed signal
    if n >= window_size:
        # Vectorized convolution for interior points
        padded = np.convolve(s, sg_weights[::-1], mode="valid")
        # np.convolve 'valid' output index i corresponds to window [i, i+W-1]
        # centered at i + (W-1)/2; for odd W that is i + half
        if window_size % 2 == 1:
            interior = padded
            y[half : half + len(interior)] = interior
        else:
            interior = padded
            y[half : half + len(interior)] = interior

    # --- Stage 3: blend EMA (noise suppression) with SG (trend preservation) ---
    y = 0.5 * y + 0.5 * s

    # Keep the same output-length contract as the original sliding-window
    # filter: length = len(x) - window_size + 1, aligned to window ends.
    output_length = n - window_size + 1
    start = half  # window centered alignment
    return y[start : start + output_length]
