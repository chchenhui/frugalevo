"""
Expert parallelism load balancer (EPLB) for vLLM.
"""

# EVOLVE-BLOCK-START

import heapq

import numpy as np
import torch


# ---------------------------------------------------------------------------
# Stage 1: weighted balanced packing (heap-based LPT with capacity)
# ---------------------------------------------------------------------------

def balanced_packing(weight: torch.Tensor,
                     num_packs: int) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Pack n weighted objects into m packs, each holding exactly n/m objects,
    minimizing load imbalance (LPT with cardinality constraint).

    Returns pack_index [X, n] and rank_in_pack [X, n].
    """
    num_layers, num_groups = weight.shape
    assert num_groups % num_packs == 0
    groups_per_pack = num_groups // num_packs

    if groups_per_pack == 1:
        pack_index = torch.arange(num_groups,
                                  dtype=torch.int64).expand(num_layers, -1)
        rank_in_pack = torch.zeros(num_layers, num_groups, dtype=torch.int64)
        return pack_index, rank_in_pack

    w = weight.float().cpu().numpy()
    order = np.argsort(-w, axis=1, kind="stable")
    pack_index = np.empty((num_layers, num_groups), dtype=np.int64)
    rank_in_pack = np.empty((num_layers, num_groups), dtype=np.int64)

    for i in range(num_layers):
        w_i = w[i]
        heap = [(0.0, p, p, 0) for p in range(num_packs)]
        heapq.heapify(heap)
        counts = [0] * num_packs
        loads = [0.0] * num_packs
        for group in order[i]:
            while heap and heap[0][3] >= groups_per_pack:
                heapq.heappop(heap)
            _, _, pack, _ = heap[0]
            pack_index[i, group] = pack
            rank_in_pack[i, group] = counts[pack]
            counts[pack] += 1
            loads[pack] += w_i[group]
            if counts[pack] < groups_per_pack:
                heapq.heapreplace(
                    heap, (loads[pack], pack, pack, counts[pack]))
            else:
                heapq.heappop(heap)

    return torch.from_numpy(pack_index), torch.from_numpy(rank_in_pack)


# ---------------------------------------------------------------------------
# Stage 2: replication via marginal-gain water-filling greedy
# ---------------------------------------------------------------------------

def replicate_experts(
        weight: torch.Tensor,
        num_phy: int) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Replicate num_log experts to num_phy physical slots.

    Duplicating an expert with cnt replicas and load w reduces its
    per-replica load by w/(cnt*(cnt+1)); greedily pick the largest
    marginal gain each step.

    Returns phy2log [X, num_phy], rank [X, num_phy], logcnt [X, num_log].
    """
    n, num_log = weight.shape
    num_redundant = num_phy - num_log
    assert num_redundant >= 0
    w = weight.float().cpu().numpy()
    phy2log = np.tile(np.arange(num_phy, dtype=np.int64), (n, 1))
    rank = np.zeros((n, num_phy), dtype=np.int64)
    logcnt = np.ones((n, num_log), dtype=np.int64)

    if num_redundant > 0:
        rows = np.arange(n)
        # marginal gain of adding one replica = w / (cnt * (cnt+1))
        gain = w / 2.0
        for i in range(num_log, num_phy):
            best = gain.argmax(axis=1)
            r = logcnt[rows, best]
            phy2log[:, i] = best
            rank[:, i] = r
            logcnt[rows, best] = r + 1
            gain[rows, best] = w[rows, best] / ((r + 1) * (r + 2))

    return (torch.from_numpy(phy2log), torch.from_numpy(rank),
            torch.from_numpy(logcnt))


# ---------------------------------------------------------------------------
# Stage 3: vectorized top-K pairwise swap refinement
# ---------------------------------------------------------------------------

def refine_placement(pl: np.ndarray,
                     pr: np.ndarray,
                     per_load: np.ndarray,
                     epg: int,
                     num_gpus: int,
                     max_rounds: int,
                     topk: int) -> None:
    """
    In-place refinement. Each round, for the top-K heaviest GPUs, evaluate
    swapping the heaviest slot on the heavy GPU with the lightest slot on
    *every* other GPU (a batched [K, P] candidate set), and apply the swap
    that minimizes the resulting maximum GPU load.
    """
    n, num_phy = pl.shape
    gpu_of = (np.arange(num_phy) // epg)
    K = min(topk, num_gpus)
    idx_gpu = np.arange(num_gpus)
    # slot indices grouped by gpu: [num_gpus, epg]
    slots = (idx_gpu[:, None] * epg + np.arange(epg)[None, :])

    for i in range(n):
        L = per_load[i].copy()
        # within-gpu sorting is by construction (LPT packing), but re-sort
        # so the heaviest slot per gpu is first.
        order = np.argsort(-L[slots], axis=1)
        slots_sorted = np.take_along_axis(slots, order, axis=1)
        Lslot = L[slots_sorted]  # [P, epg] descending

        for _ in range(max_rounds):
            gloads = Lslot.sum(axis=1)
            cur_max = gloads.max()
            # heavy gpus = top-K by load
            heavy = np.argsort(-gloads)[:K]
            # heaviest slot on each heavy gpu, lightest slot on each gpu
            a = slots_sorted[heavy, 0]          # [K]
            la = Lslot[heavy, 0]                # [K]
            b = slots_sorted[:, -1]             # [P] lightest per gpu
            lb = Lslot[:, -1]                   # [P]

            # candidate new loads: [K, P]
            new_heavy = gloads[heavy, None] - la[:, None] + lb[None, :]
            new_other = gloads[None, :] + la[:, None] - lb[None, :]
            cand = np.where(idx_gpu[None, :] == heavy[:, None],
                            new_heavy, new_other)
            cand_max = cand.max(axis=1)         # [K] resulting max load

            # forbid swaps that move load the wrong way
            valid = (la[:, None] > lb[None, :]).all(axis=1) if False else None
            # pick best (k, g) pair
            best_k = int(cand_max.argmin())
            if cand_max[best_k] >= cur_max - 1e-12:
                break
            cand_row = cand[best_k]
            g = int(cand_row.argmin() == 0) if False else None
            # recompute best g for best_k
            row = cand[best_k]
            g = int(np.argmin(row)) if False else None
            # choose g minimizing the max, tie-break by lower other load
            g = int(np.lexsort((new_other[best_k], row))[0])
            if heavy[best_k] == g:
                # swap within same gpu: skip
                kg = heavy[best_k]
                # try second-best heavy if any
                break
            a_i = a[best_k]
            b_i = b[g]
            if la[best_k] <= lb[g]:
                break
            # apply swap
            ia = 0
            ib = epg - 1
            kg = heavy[best_k]
            Lslot[kg, ia], Lslot[g, ib] = Lslot[g, ib], Lslot[kg, ia]
            slots_sorted[kg, ia], slots_sorted[g, ib] = (
                slots_sorted[g, ib], slots_sorted[kg, ia])
            la_v = L[a_i].copy()
            L[a_i] = L[b_i]
            L[b_i] = la_v
            pl[i, a_i], pl[i, b_i] = pl[i, b_i], pl[i, a_i]
            pr[i, a_i], pr[i, b_i] = pr[i, b_i], pr[i, a_i]

        # write back final loads
        per_load[i] = L


# ---------------------------------------------------------------------------
# Stage 4: global assembly
# ---------------------------------------------------------------------------

def rebalance_experts_hierarchical(
    weight: torch.Tensor,
    num_physical_experts: int,
    num_groups: int,
    num_nodes: int,
    num_gpus: int,
):
    num_layers, num_logical_experts = weight.shape
    assert num_physical_experts % num_gpus == 0
    phy_experts_per_gpu = num_physical_experts // num_gpus

    # global replication across the full physical pool
    phy2log, phyrank, logcnt = replicate_experts(
        weight, num_physical_experts)

    # pack physical experts (with replica-shared loads) to GPUs globally
    tokens_per_phy = (weight / logcnt).gather(-1, phy2log)
    pack_index, rank_in_pack = balanced_packing(tokens_per_phy, num_gpus)
    phy2pphy = pack_index * phy_experts_per_gpu + rank_in_pack

    inv = torch.empty_like(phy2pphy)
    inv.scatter_(1, phy2pphy,
                 torch.arange(phy2pphy.size(1), dtype=torch.int64,
                              device=phy2pphy.device).expand(phy2pphy.shape))
    pphy2phy = inv

    pphy2log = phy2log.gather(-1, pphy2phy)
    pphyrank = phyrank.gather(-1, pphy2phy)

    # vectorized top-K pairwise swap refinement
    pl_np = pphy2log.numpy()
    pr_np = pphyrank.numpy()
    per_load_np = (weight / logcnt).gather(-1, phy2log).numpy()
    refine_placement(pl_np, pr_np, per_load_np,
                     phy_experts_per_gpu, num_gpus,
                     max_rounds=2 * num_physical_experts,
                     topk=4)

    return (torch.from_numpy(pl_np), torch.from_numpy(pr_np), logcnt)


def rebalance_experts(
    weight: torch.Tensor,
    num_replicas: int,
    num_groups: int,
    num_nodes: int,
    num_gpus: int,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Entry point for expert-parallelism load balancer.
    """
    num_layers, num_logical_experts = weight.shape
    weight = weight.float().cpu()
    phy2log, phyrank, logcnt = rebalance_experts_hierarchical(
        weight, num_replicas, num_groups, num_nodes, num_gpus)

    num_redundant_experts = num_replicas - num_logical_experts
    maxlogcnt = num_redundant_experts + 1
    log2phy: torch.Tensor = torch.full(
        (num_layers, num_logical_experts, maxlogcnt),
        -1, dtype=torch.int64, device=logcnt.device)
    log2phy.view(num_layers, -1).scatter_(
        -1,
        phy2log * maxlogcnt + phyrank,
        torch.arange(num_replicas, dtype=torch.int64).expand(num_layers, -1))
    return phy2log, log2phy, logcnt
