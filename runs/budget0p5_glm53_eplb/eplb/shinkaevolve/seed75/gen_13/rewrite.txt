"""
Expert parallelism load balancer (EPLB) for vLLM.

Global water-filling replication + LPT packing + swap local search.
"""

# EVOLVE-BLOCK-START

import torch


def _compute_replica_counts(w: torch.Tensor, num_replicas: int):
    """Water-filling: give each redundant replica to the expert whose
    per-replica load w/cnt is currently highest. Batched over layers."""
    num_layers, num_log = w.shape
    cnt = torch.ones(num_layers, num_log, dtype=torch.int64)
    rows = torch.arange(num_layers)
    for _ in range(num_replicas - num_log):
        idx = (w / cnt).argmax(-1)
        cnt[rows, idx] += 1
    return cnt


def _lpt_assign(rep_load: torch.Tensor, per_gpu: int):
    """LPT with capacity: pack replicas (sorted desc by load) onto the
    least-loaded GPU that still has capacity. Batched over layers."""
    num_layers, num_phy = rep_load.shape
    num_gpus = num_phy // per_gpu
    rows = torch.arange(num_layers)
    order = rep_load.sort(-1, descending=True).indices
    loads = torch.zeros(num_layers, num_gpus)
    counts = torch.zeros(num_layers, num_gpus, dtype=torch.int64)
    gpu_assign = torch.zeros(num_layers, num_phy, dtype=torch.int64)
    inf = float("inf")
    for j in range(num_phy):
        item = order[:, j]
        cand = torch.where(counts < per_gpu, loads,
                           torch.full_like(loads, inf))
        g = cand.argmin(-1)
        gpu_assign[rows, item] = g
        loads[rows, g] += rep_load[rows, item]
        counts[rows, g] += 1
    return gpu_assign, loads


def _swap_refine(rep_load: torch.Tensor, gpu_assign: torch.Tensor,
                 loads: torch.Tensor, rounds: int = 5):
    """Vectorized local search: swap the heaviest item on the most-loaded
    GPU with the lightest item on the least-loaded GPU when it shrinks the
    max-min gap. Swaps preserve per-GPU item counts (hence capacity)."""
    num_layers, num_phy = rep_load.shape
    rows = torch.arange(num_layers)
    inf = float("inf")
    for _ in range(rounds):
        gmax = loads.argmax(-1, keepdim=True)
        gmin = loads.argmin(-1, keepdim=True)
        gap = (loads.gather(1, gmax) - loads.gather(1, gmin)).squeeze(1)
        on_max = gpu_assign == gmax
        on_min = gpu_assign == gmin
        heavy = torch.where(on_max, rep_load,
                            torch.full_like(rep_load, -inf)).argmax(-1)
        light = torch.where(on_min, rep_load,
                            torch.full_like(rep_load, inf)).argmin(-1)
        wh = rep_load[rows, heavy]
        wl = rep_load[rows, light]
        delta = wh - wl
        good = (delta > 0) & (delta < gap)
        if not good.any():
            break
        r = rows[good]
        h = heavy[good]
        l = light[good]
        gm = gmax[good].squeeze(1)
        gn = gmin[good].squeeze(1)
        tmp = gpu_assign[r, h].clone()
        gpu_assign[r, h] = gpu_assign[r, l]
        gpu_assign[r, l] = tmp
        loads[r, gm] += wl[good] - wh[good]
        loads[r, gn] += wh[good] - wl[good]
    return gpu_assign


def rebalance_experts(
    weight: torch.Tensor,
    num_replicas: int,
    num_groups: int,
    num_nodes: int,
    num_gpus: int,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Entry point for expert-parallelism load balancer.

    Parameters:
        weight: [layers, num_logical_experts], the load statistics for all
            logical experts
        num_replicas: number of physical experts, must be a multiple of
            `num_gpus`
        num_groups: number of expert groups
        num_nodes: number of server nodes, where the intra-node network
        num_gpus: number of GPUs, must be a multiple of `num_nodes`

    Returns:
        physical_to_logical_map: [layers, num_replicas]
        logical_to_physical_map: [layers, num_logical_experts, X]
        expert_count: [layers, num_logical_experts]
    """
    num_layers, num_log = weight.shape
    w = weight.float().cpu()
    device = w.device
    rows = torch.arange(num_layers)

    # Phase 1: water-filling replica counts
    cnt = _compute_replica_counts(w, num_replicas)

    # Phase 2: expand replicas -> (expert id, replica rank, per-replica load)
    exp_e = torch.repeat_interleave(
        torch.arange(num_log, device=device).unsqueeze(0).expand(
            num_layers, -1),
        cnt,
        dim=1,
    )  # [L, num_replicas]
    starts = torch.cumsum(cnt, dim=1) - cnt  # first position of each expert
    rep_pos = torch.arange(num_replicas, device=device).unsqueeze(0).expand(
        num_layers, -1)
    rep_rank = rep_pos - starts.gather(1, exp_e)
    rep_load = w.gather(1, exp_e) / cnt.gather(1, exp_e).float()

    # Phase 3: LPT packing onto GPUs + swap local search
    per_gpu = num_replicas // num_gpus
    gpu_assign, loads = _lpt_assign(rep_load, per_gpu)
    gpu_assign = _swap_refine(rep_load, gpu_assign, loads)

    # Phase 4: build physical layout (gpu-major, stable slot order)
    # slot = rank of the replica among its GPU's items (by rep index)
    key = gpu_assign * num_replicas + rep_pos
    slot_order = key.argsort(dim=1)
    slot = torch.empty_like(slot_order)
    slot.scatter_(1, slot_order,
                  torch.arange(num_replicas, device=device).unsqueeze(0)
                  .expand(num_layers, -1) % per_gpu)
    phy = gpu_assign * per_gpu + slot

    phy2log = torch.empty(num_layers, num_replicas, dtype=torch.int64,
                          device=device)
    phy2log[rows, phy] = exp_e
    phyrank = torch.empty(num_layers, num_replicas, dtype=torch.int64,
                          device=device)
    phyrank[rows, phy] = rep_rank

    # Phase 5: build logical -> physical reverse map
    maxlogcnt = num_replicas - num_log + 1
    log2phy: torch.Tensor = torch.full(
        (num_layers, num_log, maxlogcnt), -1, dtype=torch.int64,
        device=device)
    log2phy.view(num_layers, -1).scatter_(
        -1,
        phy2log * maxlogcnt + phyrank,
        torch.arange(num_replicas, dtype=torch.int64,
                     device=device).expand(num_layers, -1),
    )
    return phy2log, log2phy, cnt


# EVOLVE-BLOCK-END

__all__ = ["rebalance_experts"]
