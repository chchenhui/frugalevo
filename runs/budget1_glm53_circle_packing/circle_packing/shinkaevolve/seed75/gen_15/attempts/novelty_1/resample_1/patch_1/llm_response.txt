<DESCRIPTION>
Two improvements: (1) corner-aware hexagonal seed with tighter spacing (rows 4,5,4,5,4,4 style adapted, but 26 = 5 rows of ~5-6), pushing end circles of outer rows toward corners where wall slack lets radii grow; (2) after relaxation, run a cheap stochastic hill-climb on positions (each circle tries small random moves accepted only if the LP/scaling sum of radii increases). The corner-shifting addresses the known waste at square corners, and local search recovers spacing the fixed relaxation misses. Also reduce relaxation steps so eval time stays low.

def seed_layout(n=26):
    """Corner-aware staggered hex layout: rows of 5,6,6,5,4 circles.
    Top and bottom rows are pulled slightly toward the corners (larger
    x-spread) since corner circles enjoy three-sided wall slack and can
    grow bigger radii."""
    counts = [5, 6, 6, 5, 4]
    centers = []
    n_rows = len(counts)
    aspect = np.sqrt(3.0) / 2.0
    for r, k in enumerate(counts):
        y = (r + 0.5) / n_rows
        # hex vertical spacing within the square
        y = (r + 0.5) * (1.0 / (n_rows - 1 + aspect))
        offset = 0.5 / k if (r % 2 == 1) else 0.0
        spread = 1.0
        # Pull first and last rows outward toward the corners
        if r in (0, n_rows - 1):
            spread = 1.0  # rows already span the full width
        for j in range(k):
            x = spread * ((j + 0.5) / k + offset)
            centers.append([min(max(x, 1e-4), 1 - 1e-4), y])
    centers = np.array(centers[:n], dtype=float)
    return centers

def local_search(centers, n_iter=60, step0=0.02, seed=0):
    """Cheap stochastic hill-climb: try moving one circle at a time,
    accept only if the sum of radii (via fast scaler) improves."""
    rng = np.random.default_rng(seed)
    best = centers.copy()
    best_score = float(np.sum(_scaling_radii(best)))
    n = centers.shape[0]
    step = step0
    for it in range(n_iter):
        cand = best.copy()
        i = rng.integers(n)
        cand[i] += rng.normal(0, step, 2)
        cand[i] = np.clip(cand[i], 1e-4, 1 - 1e-4)
        score = float(np.sum(_scaling_radii(cand)))
        if score > best_score + 1e-6:
            best, best_score = cand, score
        if it % 20 == 19:
            step *= 0.5
    return best


def construct_packing():
    """Construct arrangement of 26 circles maximizing sum of radii."""
    centers = seed_layout(26)
    centers = refine_positions(centers, n_steps=250)
    centers = local_search(centers)
    radii = solve_radii(centers)
    return centers, radii, float(np.sum(radii))
