Four large circles are placed near the corners (exploiting three-sided wall slack), with a staggered hex-like interior filling the middle. Radii are computed by an exact linear program (as in the 2.44-scoring program) rather than water-filling, and centers are refined by a deterministic constraint-repulsion hill-climb (as in the 2.45 program), re-solving the LP each round. Multiple structured seeds are tried and the best is returned, improving robustness without relying on a single fragile layout.

"""Constructor-based circle packing for n=26 circles.

Corner-anchored seeds + exact LP radii + constraint-repulsion refinement.
"""
import numpy as np


def construct_packing():
    n = 26
    seeds = build_seeds(n)
    best = None
    for centers0 in seeds:
        centers = refine(centers0, iters=300)
        radii = solve_radii_lp(centers)
        if radii is None:
            radii = compute_max_radii(centers)
        s = float(np.sum(radii))
        if best is None or s > best[2]:
            best = (centers.copy(), radii.copy(), s)

    centers, radii, sum_radii = best
    # tiny safety shrink to guarantee strict feasibility
    radii = radii * (1.0 - 1e-6)
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii


def build_seeds(n):
    seeds = []
    # --- Seed A: corner-anchored with hex interior ---
    c = []
    corner = 0.085
    for (cx, cy) in [(corner, corner), (1 - corner, corner),
                     (corner, 1 - corner), (1 - corner, 1 - corner)]:
        c.append((cx, cy))
    # interior staggered rows between the corner band
    rows = [4, 5, 4, 5, 4]  # 22 -> total 26
    y0, y1 = 0.17, 0.83
    dy = (y1 - y0) / (len(rows) - 1)
    for ri, cnt in enumerate(rows):
        y = y0 + ri * dy
        s = 0.72 / (cnt - 1) if cnt > 1 else 0.0
        for j in range(cnt):
            x = 0.14 + j * s
            c.append((x, y))
    seeds.append(np.array(c))

    # --- Seed B: wider staggered rows 5,4,5,4,5,3 (from prior best) ---
    c = []
    row_counts = [5, 4, 5, 4, 5, 3]
    sp = 0.2
    y0 = 0.1
    gap = (1.0 - 2 * y0) / 5.0
    for row, count in enumerate(row_counts):
        y = y0 + row * gap
        for j in range(count):
            c.append((0.5 + (j - (count - 1) / 2.0) * sp, y))
    seeds.append(np.array(c))

    # --- Seed C: corner anchors + 5x4 interior grid, perturbed diagonally ---
    c = []
    for (cx, cy) in [(0.07, 0.07), (0.93, 0.07), (0.07, 0.93), (0.93, 0.93)]:
        c.append((cx, cy))
    for i in range(4):
        for j in range(5):
            x = 0.16 + i * 0.17 + 0.04 * (j % 2)
            y = 0.16 + j * 0.17
            c.append((x, y))
    seeds.append(np.array(c[:26]))

    return seeds


def refine(centers, iters=300):
    centers = centers.copy()
    radii = compute_max_radii(centers)
    for it in range(iters):
        step = 0.02 * (1.0 - it / iters) + 0.0005
        n = centers.shape[0]
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            fx = fy = 0.0
            for j in range(n):
                if j == i:
                    continue
                dx = xi - centers[j, 0]
                dy = yi - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                ov = ri + radii[j] - dist
                if ov > 0.0 and dist > 1e-12:
                    fx += ov * dx / dist
                    fy += ov * dy / dist
            if xi - ri < 0.0:
                fx += (ri - xi)
            if xi + ri > 1.0:
                fx -= (xi + ri - 1.0)
            if yi - ri < 0.0:
                fy += (ri - yi)
            if yi + ri > 1.0:
                fy -= (yi + ri - 1.0)
            norm = np.sqrt(fx * fx + fy * fy)
            if norm > 1e-12:
                centers[i, 0] = min(max(xi + step * fx / norm, 1e-9), 1 - 1e-9)
                centers[i, 1] = min(max(yi + step * fy / norm, 1e-9), 1 - 1e-9)
        if it % 15 == 14:
            radii = compute_max_radii(centers)
    return centers


def solve_radii_lp(centers, eps=1e-9):
    """Exact LP: maximize sum(r_i) s.t. r_i+r_j <= d_ij, walls."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None
    n = centers.shape[0]
    c = -np.ones(n)
    rows, rhs = [], []
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if d < 1e-12:
                continue
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(d - 2.0 * eps)
    for i in range(n):
        x, y = centers[i]
        for lim in (x - eps, y - eps, 1.0 - x - eps, 1.0 - y - eps):
            row = np.zeros(n)
            row[i] = 1.0
            rows.append(row)
            rhs.append(max(lim, 0.0))
    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method="highs")
    except Exception:
        return None
    if not res.success:
        return None
    return np.maximum(res.x, 0.0)


def compute_max_radii(centers):
    """Gauss-Seidel water-fill + monotone repair (always feasible)."""
    n = centers.shape[0]
    D = np.zeros((n, n))
    wall = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        wall[i] = min(x, y, 1.0 - x, 1.0 - y)
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            D[i, j] = d
            D[j, i] = d
    r = np.zeros(n)
    for _ in range(300):
        changed = False
        for i in range(n):
            cap = wall[i]
            for j in range(n):
                if j != i and D[i, j] - r[j] < cap:
                    cap = D[i, j] - r[j]
            cap = max(cap, 0.0)
            if abs(cap - r[i]) > 1e-12:
                changed = True
            r[i] = cap
        if not changed:
            break
    r = np.minimum(r, wall)
    for i in range(n):
        for j in range(i + 1, n):
            if r[i] + r[j] > D[i, j]:
                scale = D[i, j] / (r[i] + r[j])
                r[i] *= scale
                r[j] *= scale
    r = np.minimum(r, wall)
    return r

def run_packing():
    """Run the circle packing constructor for n=26"""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """
    Visualize the circle packing

    Args:
        centers: np.array of shape (n, 2) with radius of each circle
    """
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))

    # Draw unit square
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    # Draw circles
    for i, (center, radius) in enumerate(zip(centers, radii)):
        circle = Circle(center, radius, alpha=0.5)
        ax.add_patch(circle)
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")

    # Uncomment to visualize:
    visualize(centers, radii)
