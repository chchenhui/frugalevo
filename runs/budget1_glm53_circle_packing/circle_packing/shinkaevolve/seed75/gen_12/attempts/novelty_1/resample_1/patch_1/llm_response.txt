<DESCRIPTION>
The current layout uses 5 staggered rows (6,5,6,5,6 with the top row truncated to 4), so the arrangement doesn't fill the square vertically — the rows only span about 0.744 of the unit height. Two improvements: (1) center the rows vertically so wall distances are larger and more balanced; (2) replace the overly conservative "half nearest-neighbor distance" radius rule with an iterative fixed-point relaxation: start each radius at its full wall distance (allowing circles to grow where there is free space, e.g. the truncated top row and near walls), then repeatedly cap r_i = min(wall_i, min_j (d_ij - r_j)). This converges to a maximal valid packing where each circle is as large as possible given its neighbors, exploiting the extra space left by the two removed circles and the vertical margins, which should push the sum of radii well above the current 2.17.

def compute_max_radii(centers):
    """
    Compute radii as the minimum of distance-to-wall and half the distance
    to the nearest neighbor. This guarantees a fully valid packing:
    r_i + r_j <= dist(i, j) for all pairs, and r_i <= wall distance.
    """
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        wall = min(x, y, 1.0 - x, 1.0 - y)
        min_half_dist = wall  # upper bound
        for j in range(n):
            if j == i:
                continue
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_half_dist = min(min_half_dist, 0.5 * d)

def compute_max_radii(centers):
    """
    Compute maximal radii via fixed-point relaxation:
    start with each radius at its full wall distance (so circles can grow
    into unused space such as the truncated top row and vertical margins),
    then repeatedly cap r_i = min(wall_i, min_j (d_ij - r_j)).
    This converges to a valid packing where each circle is as large as
    possible given its neighbors and the square boundary.
    """
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))

    # Precompute pairwise distances
    dist = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dist[i, j] = d
            dist[j, i] = d

    radii = wall.copy()
    for _ in range(500):
        max_change = 0.0
        for i in range(n):
            # Largest allowed radius given current neighbor radii
            cap = wall[i]
            for j in range(n):
                if j == i:
                    continue
                cap = min(cap, dist[i, j] - radii[j])
            cap = max(cap, 0.0)
            max_change = max(max_change, abs(radii[i] - cap))
            radii[i] = cap
        if max_change < 1e-12:
            break

    return radii
