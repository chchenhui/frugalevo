<DESCRIPTION>
The current program does random hill-climbing over centers with an exact LP radius solver, achieving 2.44. The key insight from the feedback is that center placement — not radius assignment — is the bottleneck. Instead of blind random jitter, I implement a physics-inspired "contact-force relaxation": after each LP solve, I identify the *active* constraints (circle-circle contacts and wall contacts), compute the net repulsion direction for each circle from its tight contacts (weighted by its radius), and move centers along those directions with an adaptive step size. This mimics the pressure-relaxation idea from Generation 7 but is fully deterministic and gradient-like: circles are pushed only where they are squeezed, so the packing "breathes" open in the constrained directions. I combine this with a decaying temperature of small random noise to escape symmetric traps, a restart from a second seed layout (a 6-row hex pattern with different row counts), and an "expand-and-shrink" globalization step (uniformly scale centers about the square center before relaxing). The exact LP (linprog highs) guarantees radii are always optimal for the given centers, so every accepted move is a true improvement in the objective.

# EVOLVE-BLOCK-START
"""Constructor-based circle packing for n=26 circles.

Approach: exact LP radius solver + contact-force center relaxation.
After solving optimal radii via LP, each circle is pushed away from its
tight contacts (other circles and walls) by a vector proportional to its
radius, then radii are re-solved. This is a deterministic, quasi-physical
descent on center positions, combined with mild annealed noise and
periodic expand/shrink globalization.
"""
import numpy as np


def construct_packing():
    n = 26

    candidates = []
    # Seed layout A: hex rows 5,4,5,4,5,3
    candidates.append(build_hex_rows([5, 4, 5, 4, 5, 3]))
    # Seed layout B: hex rows 4,5,4,5,4,4
    candidates.append(build_hex_rows([4, 5, 4, 5, 4, 4]))

    best_sum = -1.0
    best_centers = None
    best_radii = None

    for seed in candidates:
        c, s = _optimize_from(seed)
        if s > best_sum:
            best_sum = s
            best_centers = c[0]
            best_radii = c[1]

    centers = best_centers
    radii = best_radii * (1.0 - 1e-6)
    return centers, radii, float(np.sum(radii))


def build_hex_rows(row_counts):
    """Staggered hexagonal seed layout touching all four walls."""
    centers = []
    s = 1.0 / (1.0 + (len(row_counts) - 1) * np.sqrt(3.0) / 2.0)
    h = s * np.sqrt(3.0) / 2.0
    r = s / 2.0
    for row, count in enumerate(row_counts):
        y = r + row * h
        if row % 2 == 0:
            xs = [r + j * s for j in range(count)]
        else:
            xs = [r + s / 2.0 + j * s for j in range(count)]
        for x in xs:
            centers.append([x, y])
    arr = np.array(centers[:26], dtype=float)
    # Center the layout in the square if it doesn't span fully
    arr[:, 0] += (1.0 - (arr[:, 0].max() + arr[:, 0].min())) / 2.0
    arr[:, 1] += (1.0 - (arr[:, 1].max() + arr[:, 1].min())) / 2.0
    return arr


def _contact_push(centers, radii, tol=2e-3):
    """Compute net push direction for each circle from tight contacts."""
    n = centers.shape[0]
    push = np.zeros((n, 2))
    for i in range(n):
        x, y = centers[i]
        ri = radii[i]
        # Wall contacts
        slack = np.array([x - ri, 1.0 - x - ri, y - ri, 1.0 - y - ri])
        if slack[0] < tol:
            push[i, 0] -= 1.0
        if slack[1] < tol:
            push[i, 0] += 1.0
        if slack[2] < tol:
            push[i, 1] -= 1.0
        if slack[3] < tol:
            push[i, 1] += 1.0
        # Circle contacts
        for j in range(n):
            if i == j:
                continue
            dvec = centers[i] - centers[j]
            d = np.sqrt(np.sum(dvec ** 2)) + 1e-12
            gap = d - (ri + radii[j])
            if gap < tol:
                push[i] += dvec / d * (ri / (ri + radii[j]))
    # Normalize per-circle push so no circle is over-moved
    norms = np.linalg.norm(push, axis=1)
    scale = np.where(norms > 1e-12, 1.0 / np.maximum(norms, 1e-12), 0.0)
    return push * scale[:, None]


def _optimize_from(seed, n_iters=900):
    rng = np.random.default_rng(2024)
    centers = seed.copy()
    radii = solve_radii_lp(centers)
    if radii is None:
        radii = compute_max_radii(centers)
    best = (centers.copy(), radii.copy(), float(np.sum(radii)))
    cur_sum = best[2]

    step = 0.02
    since_improve = 0
    for it in range(n_iters):
        frac = 1.0 - it / n_iters
        push = _contact_push(best[0], best[1])
        cand = best[0] + push * step * (0.5 + frac)
        # mild annealed noise, larger for smaller circles
        sig = step * frac * 0.6
        w = 1.0 - best[1] / (best[1].max() + 1e-12)
        cand += rng.normal(0, 1, cand.shape) * (sig * (0.3 + 0.7 * w))[:, None]
        cand = np.clip(cand, 1e-4, 1 - 1e-4)

        # Occasionally try a uniform expansion about the center
        if it % 37 == 36:
            mid = np.array([0.5, 0.5])
            for scale in (1.03, 0.97):
                cand2 = mid + (best[0] - mid) * scale
                cand2 = np.clip(cand2, 1e-4, 1 - 1e-4)
                r2 = solve_radii_lp(cand2)
                if r2 is not None and np.sum(r2) > best[2]:
                    best = (cand2, r2, float(np.sum(r2)))

        r_new = solve_radii_lp(cand)
        if r_new is not None:
            s_new = float(np.sum(r_new))
            if s_new > best[2]:
                best = (cand, r_new, s_new)
                since_improve = 0
                step = min(step * 1.1, 0.03)
            else:
                since_improve += 1
                if since_improve > 60:
                    step *= 0.7
                    since_improve = 0
                    if step < 1e-3:
                        break

    return best[0], best[2]


def solve_radii_lp(centers, eps=1e-9):
    """Exact LP: maximize sum(r_i) s.t. pairwise and wall constraints."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None

    n = centers.shape[0]
    c = -np.ones(n)

    rows = []
    rhs = []
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(d - 2.0 * eps)

    for i in range(n):
        x, y = centers[i]
        for lim in (x - eps, y - eps, 1.0 - x - eps, 1.0 - y - eps):
            row = np.zeros(n)
            row[i] = 1.0
            rows.append(row)
            rhs.append(max(lim, 0.0))

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n,
                  method="highs")
    if not res.success:
        return None
    return np.maximum(res.x, 0.0)


def compute_max_radii(centers):
    """Fallback: uniform maximal radius."""
    n = centers.shape[0]
    r = np.inf
    for i in range(n):
        x, y = centers[i]
        r = min(r, x, y, 1.0 - x, 1.0 - y)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r = min(r, dist / 2.0)
    return np.full(n, r)

# This part remains fixed (not evolved)
def run_packing():
    """Run the circle packing constructor for n=26"""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """
    Visualize the circle packing

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle
    """
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    for i, (center, radius) in enumerate(zip(centers, radii)):
        circle = Circle(center, radius, alpha=0.5)
        ax.add_patch(circle)
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
    # AlphaEvive improved this to 2.635
    visualize(centers, radii)
