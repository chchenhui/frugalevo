<DESCRIPTION>
The best prior program (score 1.60, sum_radii 1.60) combined lattice enumeration with uniform-radius repulsion. The key remaining inefficiency is that uniform-radius relaxation treats all circles identically, while in an optimal packing of unequal circles, small circles should be pushed against walls while large circles occupy the interior. I propose a second refinement stage: a variable-radius relaxation where each circle's own grown radius (from compute_max_radii) determines its target spacing, plus an adaptive wall-pull term. Each iteration: (1) compute each circle's current radius; (2) push apart pairs whose distance is less than the sum of their desired radii (weighted per-pair, not uniform r0); (3) for each circle, compare its wall-limited radius with its neighbor-limited radius — if the neighbor constraint is tighter, the circle has room toward the wall, so pull it toward its nearest wall with a step proportional to its radius deficit. This per-circle adaptivity is exactly the mechanism that lets boundary circles shrink and hug walls while interior circles expand, moving toward the 2.635 target.</DESCRIPTION>

    # Initialize arrays for 26 circles
    n = 26
    sqrt3_2 = np.sqrt(3.0) / 2.0
    iu = np.triu_indices(n, k=1)

    def make_layout(row_counts, s, stagger):
        pts = np.zeros((n, 2))
        h = sqrt3_2 * s if stagger else s
        idx = 0
        for r_i, cnt in enumerate(row_counts):
            y = s / 2.0 + r_i * h
            x0 = 0.5 - (cnt - 1) * s / 2.0
            if stagger and r_i % 2 == 1:
                x0 += s / 2.0
            for c in range(cnt):
                pts[idx] = [x0 + c * s, y]
                idx += 1
        return pts

    def uniform_relax(pts, r0, n_iters=300):
        pts = pts.copy()
        for _ in range(n_iters):
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt((diff ** 2).sum(-1))
            d = dist[iu]
            ov = 2.0 * r0 - d
            mask = ov > 1e-12
            if not mask.any():
                break
            ii, jj = iu[0][mask], iu[1][mask]
            u = diff[ii, jj] / d[mask][:, None]
            push = 0.05 * ov[mask][:, None] * u
            move = np.zeros_like(pts)
            np.add.at(move, ii, push)
            np.add.at(move, jj, -push)
            pts += move
            pts[:, 0] = np.clip(pts[:, 0], r0, 1 - r0)
            pts[:, 1] = np.clip(pts[:, 1], r0, 1 - r0)
        return pts

    def adaptive_relax(pts, n_iters=400):
        """Variable-radius relaxation with adaptive wall-pull.

        Each circle uses its own grown radius as target spacing. A circle
        whose neighbor-limited radius is smaller than its wall-limited
        radius has spare room toward a wall, so it is pulled toward its
        nearest wall proportionally to that deficit (small/tight circles
        hug the walls; roomy interior circles stay put).
        """
        pts = pts.copy()
        for it in range(n_iters):
            radii = compute_max_radii(pts)
            # wall-limited radius for each circle
            r_wall = np.minimum(
                np.minimum(pts[:, 0], 1 - pts[:, 0]),
                np.minimum(pts[:, 1], 1 - pts[:, 1]),
            )
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt((diff ** 2).sum(-1))
            d = dist[iu]
            # desired spacing for each pair: sum of their current radii
            t = radii[iu[0]] + radii[iu[1]]
            ov = t - d
            mask = ov > 1e-9
            move = np.zeros_like(pts)
            if mask.any():
                ii, jj = iu[0][mask], iu[1][mask]
                u = diff[ii, jj] / d[mask][:, None]
                push = 0.02 * ov[mask][:, None] * u
                np.add.at(move, ii, push)
                np.add.at(move, jj, -push)
            # adaptive wall-pull: circles that are neighbor-limited but
            # have wall room move toward their nearest wall
            for i in range(n):
                deficit = r_wall[i] - radii[i]
                if deficit <= 1e-9:
                    continue
                # nearest wall direction and distance
                gaps = np.array([pts[i, 0], 1 - pts[i, 0],
                                 pts[i, 1], 1 - pts[i, 1]])
                w = int(np.argmin(gaps))
                wstep = 0.05 * deficit
                if w == 0:
                    pts[i, 0] -= wstep
                elif w == 1:
                    pts[i, 0] += wstep
                elif w == 2:
                    pts[i, 1] -= wstep
                else:
                    pts[i, 1] += wstep
            pts += move
            # keep inside square (soft: allow getting close to walls)
            pts[:, 0] = np.clip(pts[:, 0], 1e-4, 1 - 1e-4)
            pts[:, 1] = np.clip(pts[:, 1], 1e-4, 1 - 1e-4)
        return pts

    spacings = [0.14, 0.16, 0.175, 0.19, 1.0 / (5.0 + sqrt3_2)]
    layouts = [
        ([4, 5, 4, 5, 4, 4], True),
        ([5, 4, 5, 4, 5, 3], True),
        ([4, 5, 4, 5, 4, 4], False),
        ([5, 4, 5, 4, 4, 4], False),
        ([3, 5, 4, 5, 4, 5], True),
        ([5, 5, 4, 4, 4, 4], False),
    ]

    best = None
    for s in spacings:
        for row_counts, stagger in layouts:
            pts0 = make_layout(row_counts, s, stagger)
            r0 = s / 2.0
            pts_u = uniform_relax(pts0, r0)
            radii_u = compute_max_radii(pts_u)
            tot_u = radii_u.sum()
            # refine with adaptive variable-radius relaxation
            pts_a = adaptive_relax(pts_u, n_iters=200)
            radii_a = compute_max_radii(pts_a)
            tot_a = radii_a.sum()
            if tot_a > tot_u:
                pts_c, radii_c, tot = pts_a, radii_a, tot_a
            else:
                pts_c, radii_c, tot = pts_u, radii_u, tot_u
            if best is None or tot > best[2]:
