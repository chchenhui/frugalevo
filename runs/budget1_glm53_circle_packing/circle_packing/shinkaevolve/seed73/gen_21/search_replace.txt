    # Gaussian noise to break perfect symmetry; edge effects mean asymmetric

        ([4, 5, 4, 5, 4, 4], True),
        ([5, 4, 5, 4, 5, 3], True),
        ([4, 5, 4, 5, 4, 4], False),
        ([5, 4, 5, 4, 4, 4], False),

        p = make_layout(row_counts, s, stagger)
