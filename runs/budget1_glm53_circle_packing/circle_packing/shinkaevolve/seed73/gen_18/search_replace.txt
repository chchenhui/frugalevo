<DESCRIPTION>
Two improvements: (1) Enumerate several lattice topologies (row counts / stagger / spacing) and keep the best after relaxation, rather than one fixed layout. (2) After picking the best layout, run a "contact relaxation" pass: iteratively move each center to equalize the gap to its nearest constraints (neighbors and walls), allowing it to grow its own radius — this directly maximizes sum_radii at convergence rather than only enforcing a uniform target radius. The uniform-radius repulsion tends to leave circles with excess slack near walls and between rows; the local gap-equalization pass recovers that slack.

    def make_layout(row_counts, s, stagger):
        pts = np.zeros((n, 2))
        h = sqrt3_2 * s if stagger else s
        idx = 0
        for r_i, cnt in enumerate(row_counts):
            y = s / 2.0 + r_i * h
            x0 = 0.5 - (cnt - 1) * s / 2.0
            if stagger and r_i % 2 == 1:
                x0 += s / 2.0
            for c in range(cnt):
                pts[idx] = [x0 + c * s, y]
                idx += 1
        return pts

    def uniform_relax(pts, r0, n_iters=300):
        pts = pts.copy()
        for _ in range(n_iters):
            diff = pts[:, None, :] - pts[None, :, :]
            dist = np.sqrt((diff ** 2).sum(-1))
            d = dist[iu]
            ov = 2.0 * r0 - d
            mask = ov > 1e-12
            if not mask.any():
                break
            ii, jj = iu[0][mask], iu[1][mask]
            u = diff[ii, jj] / d[mask][:, None]
            push = 0.05 * ov[mask][:, None] * u
            move = np.zeros_like(pts)
            np.add.at(move, ii, push)
            np.add.at(move, jj, -push)
            pts += move
            pts[:, 0] = np.clip(pts[:, 0], r0, 1 - r0)
            pts[:, 1] = np.clip(pts[:, 1], r0, 1 - r0)
