def slsqp_polish(centers, radii, restarts=3):
    """
    Polish a feasible packing with SLSQP. Variables are z = [x, y, r]
    for each circle. Maximize sum(r) subject to:
      r_i + r_j - d_ij <= 0 for all pairs
      r_i - x_i <= 0, r_i - y_i <= 0,
      r_i - (1 - x_i) <= 0, r_i - (1 - y_i) <= 0
    Only results that are strictly better and feasible are kept.
    """
    try:
        from scipy.optimize import minimize
    except ImportError:
        return centers, radii, float(np.sum(radii))

    n = centers.shape[0]
    best_c, best_r = centers.copy(), radii.copy()
    best_s = float(np.sum(radii))

    idx_i, idx_j = np.triu_indices(n, k=1)

    def unpack(z):
        return z[:n], z[n:2 * n], z[2 * n:]

    def objective(z):
        return -np.sum(z[2 * n:])

    def obj_grad(z):
        g = np.zeros(3 * n)
        g[2 * n:] = -1.0
        return g

    def constraints(z):
        x, y, r = unpack(z)
        dx = x[idx_i] - x[idx_j]
        dy = y[idx_i] - y[idx_j]
        d = np.sqrt(dx * dx + dy * dy)
        pair = r[idx_i] + r[idx_j] - d
        wall = np.concatenate([
            r - x, r - y, r - (1.0 - x), r - (1.0 - y)
        ])
        return np.concatenate([pair, wall])

    def jacobian(z):
        x, y, r = unpack(z)
        dx = x[idx_i] - x[idx_j]
        dy = y[idx_i] - y[idx_j]
        d = np.sqrt(dx * dx + dy * dy) + 1e-12
        J = np.zeros((len(idx_i) + 4 * n, 3 * n))
        for k, (i, j) in enumerate(zip(idx_i, idx_j)):
            J[k, i] = -dx[k] / d[k]
            J[k, j] = dx[k] / d[k]
            J[k, n + i] = -dy[k] / d[k]
            J[k, n + j] = dy[k] / d[k]
            J[k, 2 * n + i] = 1.0
            J[k, 2 * n + j] = 1.0
        for i in range(n):
            row = len(idx_i) + i
            J[row, i] = -1.0
            J[row, 2 * n + i] = 1.0
            J[row + n, n + i] = -1.0
            J[row + n, 2 * n + i] = 1.0
            J[row + 2 * n, i] = 1.0
            J[row + 2 * n, 2 * n + i] = 1.0
            J[row + 3 * n, n + i] = 1.0
            J[row + 3 * n, 2 * n + i] = 1.0
        return J

    z0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    for attempt in range(restarts):
        maxiter = 200 + 150 * attempt
        res = minimize(objective, z0, jac=obj_grad,
                       constraints=[{'type': 'ineq',
                                     'fun': constraints,
                                     'jac': jacobian}],
                       bounds=bounds, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-10})
