The current program's key parameters are: 9 hex-row seeds, single SLSQP pass (maxiter=250, ftol=1e-12), a crude `_repair` that shrinks radii (losing radius mass), and no restarts. Key changes: (1) replace the shrinking repair with exact Gauss-Seidel `grow_radii` which maximally grows every radius given the centers, guaranteeing feasibility with no lost radius; (2) add 3–5 SLSQP restart ("polish") passes, feeding each repaired endpoint back as the new start with maxiter=500, ftol=1e-14, stopping when improvement < 1e-6; (3) supply analytic constraint Jacobians so SLSQP converges much further per iteration; (4) seed feasible radii from exact growth shrunk 2%, so SLSQP starts from a near-optimal point. Restarts from feasible near-optimal endpoints let SLSQP make fine tangency adjustments unreachable in one pass — the direct route toward closing the gap to 2.635.

"""Constructor-based circle packing for n=26 circles.

Approach: hexagonal-lattice seeds -> exact radius growth -> SLSQP joint
(center, radius) optimization with analytic Jacobians, repeated restart
("polish") passes from repaired endpoints until improvement stalls.
"""
import numpy as np
from scipy.optimize import minimize

SQRT3 = np.sqrt(3.0)
N = 26
EPS = 1e-9


def _lattice_candidates():
    cands = []
    row_sets = [
        [5, 4, 5, 4, 5, 3], [5, 4, 5, 4, 4, 4], [4, 5, 4, 5, 4, 4],
        [3, 5, 4, 5, 4, 5], [4, 4, 5, 5, 4, 4], [5, 5, 4, 4, 5, 3],
        [3, 4, 5, 5, 4, 5], [4, 5, 5, 4, 5, 3], [5, 4, 4, 5, 4, 4],
    ]
    r = min(1.0 / 10.0, 1.0 / (2.0 + 5.0 * SQRT3))
    for counts in row_sets:
        if sum(counts) != N:
            continue
        for sf in (1.0, 1.05):
            rr = r * sf
            dx, dy = 2.0 * rr, SQRT3 * rr
            pts = []
            for row, c in enumerate(counts):
                y = rr + row * dy
                span = 2.0 * rr + (c - 1) * dx
                start = (1.0 - span) / 2.0 + rr
                for k in range(c):
                    pts.append([start + k * dx, y])
            cands.append(np.clip(np.array(pts), 0.005, 0.995))
    rng = np.random.default_rng(12345)
    for base in list(cands[:3]):
        for _ in range(3):
            jit = rng.normal(0.0, 0.015, size=base.shape)
            cands.append(np.clip(base + jit, 0.005, 0.995))
    return cands


def grow_radii(centers, iters=400):
    """Exact Gauss-Seidel growth: r_i = min(wall_i, min_j (d_ij - r_j))."""
    n = centers.shape[0]
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    d = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2))
    np.fill_diagonal(d, np.inf)
    radii = np.zeros(n)
    for _ in range(iters):
        grew = False
        for i in range(n):
            cand = min(b[i], float(np.min(d[i, :] - radii)))
            cand = max(cand, 1e-9)
            if cand > radii[i] + 1e-12:
                radii[i] = cand
                grew = True
        if not grew:
            break
    return radii


def _slsqp_once(centers, max_iters):
    """One SLSQP pass with analytic Jacobians. Returns (z, ok)."""
    n = N
    centers = np.clip(centers, 0.002, 0.998)
    # Feasible start: exact growth shrunk slightly
    r = grow_radii(centers)
    d = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
    np.fill_diagonal(d, np.inf)
    wall = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                      np.minimum(centers[:, 1], 1 - centers[:, 1]))
    r = np.minimum(r * 0.98, np.maximum(wall * 0.98, 1e-9))
    for i in range(n):
        r[i] = max(min(r[i], float(np.min(d[i] - r)) - 1e-9), 1e-9)

    z0 = np.concatenate([centers[:, 0], centers[:, 1], r])
    m2 = 2 * n
    ii, jj = np.triu_indices(n, k=1)

    def obj(z):
        return -float(np.sum(z[m2:]))

    def obj_grad(z):
        g = np.zeros_like(z)
        g[m2:] = -1.0
        return g

    def pair_fun(z):
        x, y, rr = z[:n], z[n:m2], z[m2:]
        dvec = np.sqrt((x[ii] - x[jj]) ** 2 + (y[ii] - y[jj]) ** 2)
        return dvec - rr[ii] - rr[jj]

    def pair_jac(z):
        x, y, rr = z[:n], z[n:m2], z[m2:]
        dvec = np.sqrt((x[ii] - x[jj]) ** 2 + (y[ii] - y[jj]) ** 2)
        dvec = np.maximum(dvec, 1e-12)
        J = np.zeros((ii.size, z0.size))
        rows = np.arange(ii.size)
        J[rows, ii] = (x[ii] - x[jj]) / dvec
        J[rows, jj] = -(x[ii] - x[jj]) / dvec
        J[rows, n + ii] = (y[ii] - y[jj]) / dvec
        J[rows, n + jj] = -(y[ii] - y[jj]) / dvec
        J[rows, m2 + ii] = -1.0
        J[rows, m2 + jj] = -1.0
        return J

    def wall_fun(z):
        x, y, rr = z[:n], z[n:m2], z[m2:]
        return np.concatenate([x - rr, 1 - x - rr, y - rr, 1 - y - rr])

    def wall_jac(z):
        J = np.zeros((4 * n, z0.size))
        for i in range(n):
            J[i, i] = 1.0
            J[i, m2 + i] = -1.0
            J[n + i, i] = -1.0
            J[n + i, m2 + i] = -1.0
            J[2 * n + i, n + i] = 1.0
            J[2 * n + i, m2 + i] = -1.0
            J[3 * n + i, n + i] = -1.0
            J[3 * n + i, m2 + i] = -1.0
        return J

    cons = [{'type': 'ineq', 'fun': pair_fun, 'jac': pair_jac},
            {'type': 'ineq', 'fun': wall_fun, 'jac': wall_jac}]
    bounds = [(0.001, 0.999)] * m2 + [(1e-6, 0.5)] * n
    try:
        res = minimize(obj, z0, jac=obj_grad, bounds=bounds, constraints=cons,
                       method='SLSQP',
                       options={'maxiter': max_iters, 'ftol': 1e-14})
        return (res.x if res.x is not None else z0), True
    except Exception:
        return z0, False


def _repair(z):
    """Clip centers and exactly grow radii => always-valid packing."""
    n = N
    x = np.clip(z[:n], 0.001, 0.999)
    y = np.clip(z[n:2 * n], 0.001, 0.999)
    centers = np.stack([x, y], axis=1)
    r = grow_radii(centers)
    return centers, r, float(np.sum(r))


def _refine(centers, passes=4, max_iters=500):
    """SLSQP with iterative restart passes from repaired endpoints."""
    best = None
    cur = np.asarray(centers, dtype=float)
    prev_sum = -np.inf
    for _ in range(passes):
        z, ok = _slsqp_once(cur, max_iters)
        c, r, s = _repair(z)
        if best is None or s > best[2]:
            best = (c, r, s)
        if s < prev_sum + 1e-6:
            break
        prev_sum = s
        cur = c  # restart from repaired endpoint
    return best


def construct_packing():
    """
    Construct an arrangement of 26 circles in a unit square maximizing
    the sum of radii via multi-start SLSQP with restart polish passes.

    Returns:
        Tuple of (centers, radii, sum_of_radii)
    """
    best = None
    for cand in _lattice_candidates():
        try:
            centers, radii, s = _refine(cand)
        except Exception:
            continue
        if best is None or s > best[2]:
            best = (centers, radii, s)
    # Extra deep polish of the overall best configuration
    try:
        centers, radii, s = _refine(best[0], passes=5, max_iters=600)
        if s > best[2]:
            best = (centers, radii, s)
    except Exception:
        pass
    centers, radii, sum_radii = best
    return centers, radii, sum_radii


def compute_max_radii(centers):
    n = centers.shape[0]
    radii = np.array([min(c[0], c[1], 1 - c[0], 1 - c[1]) for c in centers])
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if radii[i] + radii[j] > dist:
                sc = dist / (radii[i] + radii[j])
                radii[i] *= sc
                radii[j] *= sc
    return radii

    centers, radii, sum_radii = construct_packing()
