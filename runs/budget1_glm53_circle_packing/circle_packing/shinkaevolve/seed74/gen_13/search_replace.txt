<DESCRIPTION>
The current layout uses uniform vertical spacing 1/7 ≈ 0.143, which is far too small for a hexagonal lattice: adjacent rows force pairs of radii to sum to ≤ 0.143, capping every circle near 0.07. The fix is to place circles on a true triangular (hexagonal) lattice with the correct spacing g = s·√3/2 between rows and s between columns, with margins r = s/2 to the square's borders. For 6 rows this gives s = 1/((5·√3/2)+1) ≈ 0.1875, so the base equal radius is ≈ 0.094, already yielding sum ≈ 2.4+ (vs 2.02 currently), and the LP for optimal radii at fixed centers squeezes out more by letting edge/corner circles and sparser rows (e.g., the 3-circle row) grow beyond the base radius.

I also generate several candidate row patterns (5-4-5-4-5-3, 4-5-4-5-4-4, and a variant with a widened 3-circle top/bottom row) and slight lattice scalings, solve the LP for each, and return the best — this makes the constructor robust to which staggered pattern fits the square best, at negligible cost (each LP is tiny: 26 variables, ~350 constraints). Staggered rows of 4 are centered so they nest into the gaps of the 5-rows at exactly the hex distance s, keeping all contacts at the ideal equilateral-triangle geometry that characterizes optimal packings.

    # Initialize arrays for 26 circles
    n = 26

    def hex_layout(rows, scale=1.0, wide_ends=False):
        """Build a true triangular-lattice layout.

        Row spacing g = s*sqrt(3)/2, column spacing s, margin r = s/2, so
        neighboring rows form equilateral triangles (optimal local geometry).
        s is limited by both the square width (max row count) and height.
        """
        R = len(rows)
        k = np.sqrt(3.0) / 2.0
        s_h = 1.0 / ((R - 1) * k + 1.0)   # height: (R-1)*g + 2r <= 1
        s_w = 1.0 / max(rows)             # width: count*s <= 1
        s = min(s_h, s_w) * scale
        r0 = s / 2.0
        g = s * k
        pts = []
        for ri, count in enumerate(rows):
            y = r0 + ri * g
            if count < 3 and wide_ends:
                # Sparser end rows: spread circles a bit wider for the LP
                sp = min(1.6 * s, (1.0 - s) / max(count - 1, 1))
            else:
                sp = s
            for c in range(count):
                x = 0.5 + (c - (count - 1) / 2.0) * sp
                x = min(max(x, 0.005), 0.995)
                pts.append([x, y])
        return np.array(pts)

    def lp_radii(centers):
        """Exact optimal radii at fixed centers via LP:
        maximize sum(r) s.t. r_i + r_j <= d_ij, r_i <= border distance."""
        radii = compute_max_radii(centers)
        try:
            from scipy.optimize import linprog

            A_ub = []
            b_ub = []
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if d < 1e-9:
                        continue
                    row = np.zeros(n)
                    row[i] = 1.0
                    row[j] = 1.0
                    A_ub.append(row)
                    b_ub.append(d)
            for i in range(n):
                x, y = centers[i]
                row = np.zeros(n)
                row[i] = 1.0
                A_ub.append(row)
                b_ub.append(min(x, y, 1 - x, 1 - y))
            res = linprog(c=-np.ones(n), A_ub=np.array(A_ub),
                          b_ub=np.array(b_ub), bounds=[(0, None)] * n,
                          method="highs")
            if res.success and np.sum(res.x) > np.sum(radii):
                radii = res.x
        except Exception:
            pass
        return radii

    # Candidate staggered row patterns totaling 26 circles
    candidates = [
        ([5, 4, 5, 4, 5, 3], False),
        ([4, 5, 4, 5, 4, 4], False),
        ([5, 4, 5, 4, 5, 3], True),   # widened sparse end row
        ([4, 5, 4, 5, 4, 4], True),
    ]

    best = None
    for rows, wide in candidates:
        for scale in (0.98, 1.0):
            centers_c = hex_layout(rows, scale=scale, wide_ends=wide)
            radii_c = lp_radii(centers_c)
            total = np.sum(radii_c)
            if best is None or total > best[2]:
                best = (centers_c, radii_c, total)

    centers, radii, sum_radii = best
    return centers, radii, sum_radii
