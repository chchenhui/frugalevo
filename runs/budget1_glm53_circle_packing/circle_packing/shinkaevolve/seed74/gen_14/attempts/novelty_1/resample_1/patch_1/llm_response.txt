<DESCRIPTION>
The previous best program (2.36) used a true triangular-lattice layout with row spacing g = s*sqrt(3)/2 (equilateral local geometry) and an exact LP for radii. The current program regressed (2.02) by using uniform row/column spacing, which creates non-equilateral gaps. I restore the triangular-lattice constructor, vectorize the LP constraint assembly with numpy broadcasting + scipy sparse matrices (per the identified bottleneck), and add a short randomized hill-climb on center positions: each step perturbs a few centers slightly, re-solves the LP, and keeps improvements. Vectorizing the LP makes each step cheap enough to fit many refinement iterations in the runtime budget, pushing sum of radii toward the 2.635 target.

    # Initialize arrays for 26 circles
    n = 26

    def hex_layout(rows, scale=1.0):
        """True triangular-lattice layout.

        Row spacing g = s*sqrt(3)/2, column spacing s, margin r = s/2, so
        neighboring rows form equilateral triangles (optimal local geometry).
        """
        R = len(rows)
        k = np.sqrt(3.0) / 2.0
        s_h = 1.0 / ((R - 1) * k + 1.0)   # height: (R-1)*g + 2r <= 1
        s_w = 1.0 / max(rows)             # width: count*s <= 1
        s = min(s_h, s_w) * scale
        r0 = s / 2.0
        g = s * k
        pts = []
        for ri, count in enumerate(rows):
            y = r0 + ri * g
            for c in range(count):
                x = 0.5 + (c - (count - 1) / 2.0) * s
                x = min(max(x, 0.005), 0.995)
                pts.append([x, y])
        return np.array(pts)

    def lp_radii(centers):
        """Exact optimal radii at fixed centers via vectorized sparse LP:
        maximize sum(r) s.t. r_i + r_j <= d_ij, r_i <= border distance."""
        radii = compute_max_radii(centers)
        try:
            from scipy.optimize import linprog
            from scipy.sparse import csr_matrix

            ii, jj = np.triu_indices(n, 1)
            d = np.linalg.norm(centers[ii] - centers[jj], axis=1)
            keep = d > 1e-9
            ii, jj, d = ii[keep], jj[keep], d[keep]
            # Pairwise constraints r_i + r_j <= d_ij
            data = np.concatenate([np.ones(len(d)), np.ones(len(d))])
            rows_ = np.concatenate([np.arange(len(d)), np.arange(len(d))])
            cols_ = np.concatenate([ii, jj])
            # Border constraints r_i <= min(x, y, 1-x, 1-y)
            bd = np.minimum.reduce([centers[:, 0], centers[:, 1],
                                    1.0 - centers[:, 0], 1.0 - centers[:, 1]])
            rows_b = len(d) + np.arange(n)
            data = np.concatenate([data, np.ones(n)])
            rows_ = np.concatenate([rows_, rows_b])
            cols_ = np.concatenate([cols_, np.arange(n)])
            A_ub = csr_matrix((data, (rows_, cols_)),
                              shape=(len(d) + n, n))
            b_ub = np.concatenate([d, bd])
            res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method="highs")
            if res.success and np.sum(res.x) > np.sum(radii):
                radii = res.x
        except Exception:
            pass
        return radii

    # Candidate staggered row patterns totaling 26 circles
    candidates = [
        [5, 4, 5, 4, 5, 3],
        [4, 5, 4, 5, 4, 4],
        [5, 4, 5, 4, 5, 3],
        [3, 5, 4, 5, 4, 5],
    ]

    best = None
    for rows in candidates:
        for scale in (0.98, 1.0):
            centers_c = hex_layout(rows, scale=scale)
            radii_c = lp_radii(centers_c)
            total = np.sum(radii_c)
            if best is None or total > best[2]:
                best = (centers_c, radii_c, total)

    centers, radii, sum_radii = best

    # Randomized hill-climb on center positions with LP re-solve.
    # Each step perturbs a few circles; keep the move if the LP total improves.
    try:
        import time
        rng = np.random.default_rng(0)
        deadline = time.time() + 0.9
        step = 0.02
        while time.time() < deadline:
            trial = centers.copy()
            k_move = rng.integers(1, 4)
            for idx in rng.choice(n, size=k_move, replace=False):
                trial[idx] += rng.normal(0.0, step, size=2)
            # Keep centers inside the square
            trial = np.clip(trial, 0.01, 0.99)
            radii_t = lp_radii(trial)
            total_t = np.sum(radii_t)
            if total_t > sum_radii + 1e-9:
                centers, radii, sum_radii = trial, radii_t, total_t
            else:
                # Adaptive step size: shrink when stuck, grow occasionally
                if rng.random() < 0.7:
                    step = max(step * 0.92, 0.002)
                else:
                    step = min(step * 1.4, 0.05)
    except Exception:
        pass

    return centers, radii, sum_radii
