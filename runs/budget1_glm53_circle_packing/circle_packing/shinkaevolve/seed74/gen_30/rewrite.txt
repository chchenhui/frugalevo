"""Constructor-based circle packing for n=26 circles."""
import time
import numpy as np

N = 26
TIME_BUDGET = 2.0  # seconds total for construction


def _lp_radii(centers):
    """Exact max-sum radii for fixed centers via linear programming."""
    n = centers.shape[0]
    try:
        from scipy.optimize import linprog
    except Exception:
        return _greedy_radii(centers)
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    d = np.sqrt(dx * dx + dy * dy)
    iu, ju = np.triu_indices(n, k=1)
    m_pairs = len(iu)
    A_ub = np.zeros((m_pairs + n, n))
    A_ub[:m_pairs, iu] = 1.0
    A_ub[:m_pairs, ju] = 1.0
    b_pairs = d[iu, ju]
    wall = np.minimum.reduce([centers[:, 0], centers[:, 1],
                              1 - centers[:, 0], 1 - centers[:, 1]])
    A_ub[m_pairs:, :] = np.eye(n)
    b_ub = np.concatenate([b_pairs, wall])
    try:
        res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method="highs")
        if res.success:
            return res.x
    except Exception:
        pass
    return _greedy_radii(centers)


def _greedy_radii(centers):
    """Fallback: iterative shrink-to-constraint radii (always valid)."""
    n = centers.shape[0]
    d = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
    radii = np.minimum.reduce([centers[:, 0], centers[:, 1],
                               1 - centers[:, 0], 1 - centers[:, 1]]).copy()
    for _ in range(200):
        new_r = radii.copy()
        for i in range(n):
            lim = min(centers[i, 0], centers[i, 1],
                      1 - centers[i, 0], 1 - centers[i, 1])
            if n > 1:
                lim = min(lim, (d[i] - radii).min())
            new_r[i] = max(lim, 0.0)
        if np.max(np.abs(new_r - radii)) < 1e-12:
            radii = new_r
            break
        radii = new_r
    return radii


def _hex_layout(rows, s=None, ymargin=None):
    """Build a staggered hexagonal row layout with the given row counts."""
    centers = []
    k = len(rows)
    dy = np.sqrt(3) / 2.0
    if s is None:
        s = 1.0 / max(rows)
    if ymargin is None:
        # vertical spacing consistent with s, centered
        total_h = (k - 1) * s * dy
        y0 = (1.0 - total_h) / 2.0
    else:
        y0 = ymargin
    for r, count in enumerate(rows):
        y = y0 + r * s * dy
        for c in range(count):
            x = 0.5 + (c - (count - 1) / 2.0) * s
            centers.append([x, y])
    return np.array(centers[:N])


def _grid_layout(nrows, ncols_list):
    centers = []
    dy = 1.0 / (nrows + 1)
    for r in range(nrows):
        y = (r + 1) * dy
        cnt = ncols_list[r] if r < len(ncols_list) else ncols_list[-1]
        dx = 1.0 / (cnt + 1)
        for c in range(cnt):
            centers.append([(c + 1) * dx, y])
    return np.array(centers[:N])


def construct_packing():
    """
    Construct an arrangement of 26 circles in a unit square maximizing the
    sum of radii. Two-phase: cheap multistart over base layouts scored by
    an exact LP radii solve, then time-budgeted hill climbing on centers.

    Returns:
        (centers, radii, sum_radii)
    """
    t0 = time.time()
    rng = np.random.default_rng(12345)

    # ---- Phase 1: candidate base layouts, cheap scoring ----
    candidates = [
        _hex_layout([5, 6, 5, 6, 5]),
        _hex_layout([5, 6, 5, 6, 4]),
        _hex_layout([6, 5, 6, 5, 6]),
        _hex_layout([4, 5, 4, 5, 4, 4]),
        _hex_layout([5, 4, 5, 4, 5, 3]),
        _grid_layout(5, [5, 6, 5, 6, 5]),
        _grid_layout(6, [4, 5, 4, 5, 4, 4]),
        _hex_layout([6, 6, 5, 6, 4]),
        _hex_layout([5, 6, 4, 5, 6]),
    ]
    # jittered variants of a few strong seeds
    base_seeds = [_hex_layout([5, 6, 5, 6, 5]),
                  _hex_layout([4, 5, 4, 5, 4, 4])]
    while time.time() - t0 < 0.4 and len(candidates) < 20:
        seed = base_seeds[len(candidates) % len(base_seeds)].copy()
        seed += rng.normal(0, 0.02, seed.shape)
        candidates.append(seed)

    best_centers = None
    best_sum = -1.0
    for c in candidates:
        c = np.clip(c, 0.01, 0.99)
        r = _lp_radii(c)
        s = r.sum()
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()

    # ---- Phase 2: hill climbing on centers with adaptive step ----
    centers = best_centers.copy()
    best_r = _lp_radii(centers)
    best_sum = best_r.sum()
    step = 0.02
    since_improve = 0
    while time.time() - t0 < TIME_BUDGET * 0.92:
        trial = centers.copy()
        k = rng.integers(0, N)
        trial[k] += rng.normal(0, step, 2)
        trial = np.clip(trial, 0.005, 0.995)
        r = _lp_radii(trial)
        s = r.sum()
        if s > best_sum + 1e-9:
            best_sum = s
            best_r = r
            centers = trial
            since_improve = 0
            step = max(step * 0.9, 0.002)
        else:
            since_improve += 1
            if since_improve > 300:
                # anneal: shrink step, occasionally kick
                step = max(step * 0.5, 0.0005)
                k2 = rng.integers(0, N)
                centers[k2] += rng.normal(0, 0.01, 2)
                centers = np.clip(centers, 0.005, 0.995)
                r = _lp_radii(centers)
                if r.sum() > best_sum:
                    best_sum = r.sum()
                    best_r = r
                else:
                    since_improve = 0

    radii = best_r
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii

# This part remains fixed (not evolved)
def run_packing():
    """Run the circle packing constructor for n=26"""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """
    Visualize the circle packing

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle
    """
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))

    # Draw unit square
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    for i, (center, radius) in enumerate(zip(centers, radii)):
        circle = Circle(center, radius, alpha=0.5)
        ax.add_patch(circle)
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
    # AlphaEvolve improved this to 2.635
    visualize(centers, radii)
