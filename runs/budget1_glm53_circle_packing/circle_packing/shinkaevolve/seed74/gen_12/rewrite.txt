"""Constructor-based circle packing for n=26 circles.

Architecture:
  1. multi-start layouts: staggered hex-style row variants
  2. fast radius evaluation: iterative constraint shrinking (no LP in the loop)
  3. hybrid local search: contact-gradient pushes + random perturbations
  4. final polish: exact LP for radii on the best center set
"""
import numpy as np

try:
    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:
    _HAS_SCIPY = False


# ------------------------------------------------------------- layouts ----
def build_layout(counts, ys):
    centers = []
    for k, (cnt, y) in enumerate(zip(counts, ys)):
        x0 = 0.5 / cnt if (k % 2 == 1) else 0.0
        xs = x0 + (np.arange(cnt) + 0.5) / cnt
        for x in xs:
            centers.append([x, y])
    return np.array(centers)


def initial_layouts():
    return [
        build_layout([5, 6, 5, 6, 4], [0.10, 0.30, 0.50, 0.70, 0.90]),
        build_layout([6, 5, 6, 5, 4], [0.10, 0.30, 0.50, 0.70, 0.90]),
        build_layout([4, 6, 6, 6, 4], [0.09, 0.30, 0.50, 0.70, 0.91]),
        build_layout([5, 6, 6, 5, 4], [0.10, 0.30, 0.50, 0.70, 0.90]),
        build_layout([6, 4, 6, 4, 6], [0.09, 0.30, 0.50, 0.70, 0.91]),
    ]


# ------------------------------------------------- fast radius solving ----
def fast_radii(centers, iters=60):
    """Iterative shrinking: r_i = min(walls, min_j (d_ij - r_j)). Monotone, safe."""
    n = len(centers)
    x = centers[:, 0]
    y = centers[:, 1]
    walls = np.minimum(np.minimum(x, y), np.minimum(1 - x, 1 - y))
    d = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
    np.fill_diagonal(d, np.inf)
    r = walls.copy()
    for _ in range(iters):
        # candidate radius for each i: min over j of d_ij - r_j
        cand = np.min(d - r[None, :], axis=1)
        new_r = np.minimum(walls, np.maximum(cand, 0.0))
        new_r = np.minimum(new_r, r)  # monotone shrink -> always valid
        if np.max(r - new_r) < 1e-12:
            r = new_r
            break
        r = new_r
    return r


def lp_radii(centers):
    """Exact LP: maximize sum r s.t. r_i+r_j<=d_ij, r_i<=walls. Fallback: fast_radii."""
    n = len(centers)
    if _HAS_SCIPY:
        A, b = [], []
        for i in range(n):
            for bnd in (centers[i][0], centers[i][1],
                        1 - centers[i][0], 1 - centers[i][1]):
                row = np.zeros(n); row[i] = 1.0
                A.append(row); b.append(bnd)
        for i in range(n):
            for j in range(i + 1, n):
                dd = np.hypot(*(centers[i] - centers[j]))
                row = np.zeros(n); row[i] = 1.0; row[j] = 1.0
                A.append(row); b.append(dd)
        res = linprog(c=-np.ones(n), A_ub=np.array(A), b_ub=np.array(b),
                      bounds=[(0, None)] * n, method="highs")
        if res.success:
            return np.maximum(res.x, 0.0)
    return fast_radii(centers, iters=200)


def ensure_valid(centers, radii):
    """Hard-clip radii so all constraints hold exactly."""
    x = centers[:, 0]; y = centers[:, 1]
    r = np.minimum(radii, np.minimum(np.minimum(x, y), np.minimum(1 - x, 1 - y)))
    n = len(r)
    d = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
    for _ in range(4):
        viol = False
        for i in range(n):
            for j in range(i + 1, n):
                s = r[i] + r[j]
                if s > d[i, j] and s > 0:
                    f = d[i, j] / s
                    r[i] *= f; r[j] *= f
                    viol = True
        if not viol:
            break
    return r


# ------------------------------------------------- hybrid local search ----
def contact_gradient(centers, radii, eps=1e-3):
    """Push each circle away from its touching neighbors/walls (quasi-gradient)."""
    n = len(centers)
    d = np.sqrt(((centers[:, None, :] - centers[None, :, :]) ** 2).sum(-1))
    np.fill_diagonal(d, np.inf)
    moves = np.zeros((n, 2))
    for i in range(n):
        # touching neighbors (within eps of contact)
        idx = np.where(d[i] - (radii[i] + radii) < eps)[0]
        if len(idx):
            dirs = centers[i] - centers[idx]
            norms = np.linalg.norm(dirs, axis=1, keepdims=True)
            dirs = dirs / np.maximum(norms, 1e-12)
            moves[i] += dirs.sum(axis=0)
        # walls that are tight
        x, y = centers[i]
        for wall, vec in ((x - radii[i], [-1, 0]), (1 - x - radii[i], [1, 0]),
                          (y - radii[i], [0, -1]), (1 - y - radii[i], [0, 1])):
            if abs(wall) < eps:
                moves[i] += np.array(vec)
        nn = np.linalg.norm(moves[i])
        if nn > 0:
            moves[i] /= nn
    return moves


def local_search(centers, steps=4000, step0=0.02, seed=0):
    rng = np.random.default_rng(seed)
    cur = centers.copy()
    cur_r = fast_radii(cur)
    cur_sum = cur_r.sum()
    best, best_r, best_sum = cur.copy(), cur_r, cur_sum
    grad_mode = True
    since_gain = 0
    for t in range(steps):
        step = step0 * (0.3 + 0.7 * (1 - t / steps)) + 5e-5
        cand = cur.copy()
        if grad_mode:
            g = contact_gradient(cur, cur_r)
            # move a random subset of circles along their gradient
            for i in range(len(cand)):
                if rng.random() < 0.25 and np.linalg.norm(g[i]) > 0:
                    cand[i] = cand[i] + g[i] * step * rng.uniform(0.5, 1.5)
        # random jitter on a couple of circles always
        for _ in range(2):
            i = rng.integers(len(cand))
            cand[i] = cand[i] + rng.normal(size=2) * step
        np.clip(cand, 0.02, 0.98, out=cand)
        r = fast_radii(cand)
        s = r.sum()
        if s > cur_sum + 1e-9:
            cur, cur_r, cur_sum = cand, r, s
            since_gain = 0
            if s > best_sum:
                best, best_r, best_sum = cand.copy(), r, s
        else:
            since_gain += 1
            if since_gain > 150:  # restart from best, switch mode
                cur, cur_r, cur_sum = best.copy(), best_r.copy(), best_sum
                grad_mode = not grad_mode
                since_gain = 0
    return best, best_r


# ------------------------------------------------------------- interface ----
def construct_packing():
    """
    Construct an arrangement of 26 circles in a unit square maximizing the
    sum of radii. Returns (centers, radii, sum_of_radii).
    """
    best_c, best_r, best_sum = None, None, -1.0
    for li, layout in enumerate(initial_layouts()):
        c, r = local_search(layout, seed=li)
        if r.sum() > best_sum:
            best_c, best_r, best_sum = c, r, r.sum()
    # final exact LP polish on the best centers
    r = ensure_valid(best_c, lp_radii(best_c))
    if r.sum() > best_r.sum():
        best_r = r
    return best_c, best_r, float(best_r.sum())


def compute_max_radii(centers):
    """Backward-compatible helper: radii for given centers."""
    return ensure_valid(centers, lp_radii(centers))

# This part remains fixed (not evolved)
def run_packing():
    """Run the circle packing constructor for n=26"""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """
    Visualize the circle packing

    Args:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
    """
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))

    # Draw unit square
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    # Draw circles
    for i, (center, radius) in enumerate(zip(centers, radii)):
        circle = Circle(center, radius, alpha=0.5)
        ax.add_patch(circle)
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
    # AlphaEvolve improved this to 2.635

    # Uncomment to visualize:
    visualize(centers, radii)
