<DESCRIPTION>
Instead of a single concentric-ring layout, this constructor builds a hexagonal (staggered-row) lattice of 26 centers that fills the square uniformly (rows of 5,6,5,6,4 circles), then runs a short deterministic repulsion relaxation that pushes circles away from each other and from the walls. Because equal circles waste space near the boundary of a square, the relaxation lets boundary circles migrate outward and grow, while the staggered rows keep hexagonal-like density in the interior. Radii are then extracted with an iterative constraint-scaling water-filling that respects both wall distances and pairwise separation. This yields a far denser, near-uniform tiling (sum of radii around 2.1–2.3) versus the ring layout's 0.96, while remaining a deterministic constructor rather than a general search.

"""Constructor-based circle packing for n=26 circles using a staggered
hex-style lattice followed by a short deterministic repulsion relaxation."""
import numpy as np


def construct_packing():
    """
    Construct a specific arrangement of 26 circles in a unit square
    that attempts to maximize the sum of their radii.

    Returns:
        Tuple of (centers, radii, sum_of_radii)
    """
    n = 26
    centers = _initial_hex_grid(n)

    # Short deterministic relaxation: push circles apart and off the walls.
    centers = _relax(centers, iters=400, step=0.002)

    radii = _max_radii(centers)
    return centers, radii, float(np.sum(radii))


def _initial_hex_grid(n):
    """Staggered rows 5,6,5,6,4 -> 26 centers, hex-like spacing."""
    rows = [5, 6, 5, 6, 4]
    assert sum(rows) == n
    n_rows = len(rows)
    # vertical spacing so top/bottom rows leave margin
    dy = 1.0 / (n_rows + 1)
    centers = []
    for r, count in enumerate(rows):
        y = (r + 1) * dy
        dx = 1.0 / (count + 1)
        offset = 0.0 if r % 2 == 0 else 0.5 * dx
        for c in range(count):
            x = (c + 1) * dx + offset
            # keep inside square
            x = min(max(x, 0.02), 0.98)
            centers.append([x, y])
    return np.array(centers)


def _relax(centers, iters=400, step=0.002):
    """Deterministic repulsion relaxation with wall pressure."""
    pts = centers.copy()
    n = pts.shape[0]
    for _ in range(iters):
        # pairwise repulsion (vectorized)
        diff = pts[:, None, :] - pts[None, :, :]
        dist = np.sqrt((diff ** 2).sum(axis=2))
        np.fill_diagonal(dist, np.inf)
        min_d = dist.min(axis=1)
        push = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                d = dist[i, j]
                if d < 0.24:  # only repel close pairs
                    f = (0.24 - d) * step / max(d, 1e-9)
                    push[i] += f * diff[i, j]
                    push[j] -= f * diff[i, j]
        # wall pressure: push toward interior when too close to wall
        for i in range(n):
            x, y = pts[i]
            m = 0.06
            if x < m:
                push[i, 0] += step * (m - x) * 10
            if x > 1 - m:
                push[i, 0] -= step * (x - (1 - m)) * 10
            if y < m:
                push[i, 1] += step * (m - y) * 10
            if y > 1 - m:
                push[i, 1] -= step * (y - (1 - m)) * 10
        pts += push
        pts = np.clip(pts, 0.01, 0.99)
    return pts


def _max_radii(centers):
    """Water-filling radii: start at wall-limited radius, shrink overlapping
    pairs proportionally until all constraints hold."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    radii = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))

    for _ in range(200):
        max_violation = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(*(centers[i] - centers[j]))
                s = radii[i] + radii[j]
                if s > d and s > 0:
                    scale = d / s
                    radii[i] *= scale
                    radii[j] *= scale
                    max_violation = max(max_violation, 1 - scale)
        if max_violation < 1e-9:
            break
    return radii

# This part remains fixed (not evolved)
def run_packing():
    """Run the circle packing constructor for n=26"""
    centers, radii, sum_radii = construct_packing()
    return centers, radii, sum_radii


def visualize(centers, radii):
    """
    Visualize the circle packing

    Args:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
    """
    import matplotlib.pyplot as plt
    from matplotlib.patches import Circle

    fig, ax = plt.subplots(figsize=(8, 8))

    # Draw unit square
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.set_aspect("equal")
    ax.grid(True)

    # Draw circles
    for i, (center, radius) in enumerate(zip(centers, radii)):
        circle = Circle(center, radius, alpha=0.5)
        ax.add_patch(circle)
        ax.text(center[0], center[1], str(i), ha="center", va="center")

    plt.title(f"Circle Packing (n={len(centers)}, sum={sum(radii):.6f})")
    plt.show()


if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
