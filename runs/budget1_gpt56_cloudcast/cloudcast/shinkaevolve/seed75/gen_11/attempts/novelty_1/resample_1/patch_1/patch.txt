+def _reverse_relax(graph, initial_costs, initial_choices):
+    """
+    Compute min_v(initial_cost[v] + shortest_path(v -> node)) for every node.
+
+    Relaxing on the reversed graph is the directed-Steiner shortest-path
+    closure operation.  Returned choices permit reconstruction of the actual
+    forwarding tree rather than merely its cost.
+    """
+    dist = dict(initial_costs)
+    choice = dict(initial_choices)
+    heap = []
+
+    for node, value in dist.items():
+        if math.isfinite(value):
+            heapq.heappush(heap, (value, str(node), node))
+
+    while heap:
+        current, _, node = heapq.heappop(heap)
+        if current > dist.get(node, math.inf) + EPSILON:
+            continue
+
+        for predecessor in graph.predecessors(node):
+            edge_cost = _safe_cost(graph[predecessor][node])
+            candidate = current + edge_cost
+            old = dist.get(predecessor, math.inf)
+
+            if candidate + EPSILON < old:
+                dist[predecessor] = candidate
+                # From predecessor, forward to node, then use node's plan.
+                choice[predecessor] = ("edge", node)
+                heapq.heappush(heap, (candidate, str(predecessor), predecessor))
+
+    return dist, choice

+def _exact_directed_steiner(graph, src, terminals):
+    """
+    Exact directed Steiner tree DP.
+
+    dp[mask][v] is the minimum cost of a directed broadcast subtree rooted at
+    v which reaches all terminals represented by mask.  At a relay, two
+    disjoint terminal subsets may be merged; reverse shortest-path relaxation
+    then finds the cheapest way to reach that relay from every upstream cloud.
+    """
+    nodes = list(graph.nodes())
+    terminal_count = len(terminals)
+    full_mask = (1 << terminal_count) - 1
+

+    decisions = [None] * (full_mask + 1)
+
+    for index, terminal in enumerate(terminals):
+        mask = 1 << index

+        initial = {node: math.inf for node in nodes}

+

+        return None

+
+    def collect(mask, node):
+        state = (mask, node)
+        if state in visiting:
+            return
+        visiting.add(state)
+
+        decision = decisions[mask].get(node)
+        if decision is None:
+            visiting.discard(state)
